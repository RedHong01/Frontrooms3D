def font(path, size):
    k = (path, size)
    if k not in _f:
        _f[k] = ImageFont.truetype(path, size)
    return _f[k]


def label(dr, x, y, text, size=28, fill=INK):
    """Bayon label; runs of digits in Courier Prime Bold at 1.25 em on the same baseline."""
    fb, fc = font(BAYON, size), font(COUR, int(round(size * 1.25)))
    asc_b = fb.getmetrics()[0]
    asc_c = fc.getmetrics()[0]
    import re
    for part in re.split(r"(\d+)", text):
        if not part:
            continue
        f, asc = (fc, asc_c) if part.isdigit() else (fb, asc_b)
        dr.text((x, y + asc_b - asc), part, font=f, fill=fill)
        x += dr.textlength(part, font=f)
    return x


def rgb(p):
    return Image.open(p).convert("RGB")


def on(bg, p, scale=1):
    im = Image.open(p).convert("RGBA")
    b = Image.new("RGBA", im.size, bg + (255,))
    out = Image.alpha_composite(b, im).convert("RGB")
    if scale != 1:
        out = out.resize((out.width * scale, out.height * scale), Image.NEAREST)
    return out


def heat(a, b, gain):
    d = np.abs(np.asarray(a).astype(int) - np.asarray(b).astype(int)).max(axis=2).astype(float) * gain
    d = np.clip(d, 0, 255).astype(np.uint8)
    out = np.zeros(d.shape + (3,), np.uint8)
    out[..., 0] = d
    out[..., 1] = (d * 0.85).astype(np.uint8)
    out[..., 2] = (d * 0.2).astype(np.uint8)
    return Image.fromarray(out)


def save(im, name):
    im.save(os.path.join(OUTD, name), optimize=True)
    print("[verif2]", name, im.size)


# ------------------------------------------------------------------ 08 main FBX vs trace (VL075), 2 x 2
def img08():
    sys.path.insert(0, os.path.join(W, "keyicon_design"))
    ref_rgba = Image.open(os.path.join(W, "keyicon_design/model/key_side.png")).convert("RGBA")
    a = np.asarray(ref_rgba).astype(float) / 255.0
    ref = a[..., 3] > 0.5
    mask = np.load(os.path.join(W, "keyicon_resume/mask_Kit_Key_Zone.npy"))
    H, Wd = mask.shape
    bg = np.array([0x1B, 0x1A, 0x17]) / 255.0
    p1 = a[..., :3] * a[..., 3:4] + bg * (1 - a[..., 3:4])
    p2 = np.where(mask[..., None], np.array(PAPER) / 255.0, bg)
    p3 = np.where((mask & ref)[..., None], np.array([0x8A, 0x87, 0x7E]) / 255.0, bg)
    diff = np.argwhere(mask ^ ref)
    k = 0.45
    pw, ph = int(Wd * k), int(H * k)

    def down(x):
        return Image.fromarray(np.round(np.clip(x, 0, 1) * 255).astype(np.uint8)).resize((pw, ph), Image.BILINEAR)
    P = [down(p1), down(p2), down(p3)]
    d3 = ImageDraw.Draw(P[2])
    for y, x in diff:
        d3.ellipse((x * k - 4, y * k - 4, x * k + 4, y * k + 4), fill=(0xE0, 0x44, 0x2E))
    # zoom: the 1/6-panel window (full res) that holds the most differing pixels, x6 nearest
    zw, zh = pw // 6, ph // 6
    best = None
    for cy, cx in diff:
        y0 = int(np.clip(cy - zh / 2, 0, H - zh)); x0 = int(np.clip(cx - zw / 2, 0, Wd - zw))
        n = int(((diff[:, 0] >= y0) & (diff[:, 0] < y0 + zh) & (diff[:, 1] >= x0) & (diff[:, 1] < x0 + zw)).sum())
        if best is None or n > best[0]:
            best = (n, x0, y0)
    _, x0, y0 = best
    z = p3[y0:y0 + zh, x0:x0 + zw].copy()
    for y, x in diff:
        if y0 <= y < y0 + zh and x0 <= x < x0 + zw:
            z[y - y0, x - x0] = np.array([0xE0, 0x44, 0x2E]) / 255.0
    Z = Image.fromarray(np.round(z * 255).astype(np.uint8)).resize((zw * 6, zh * 6), Image.NEAREST)
    G, LH = 24, 48
    sheet = Image.new("RGB", (pw * 2 + G, (ph + LH) * 2 + G), WHITE)
    dr = ImageDraw.Draw(sheet)
    labs = ["TRACE SOURCE, BLENDER RENDER", "MAIN, KIT_KEY_ZONE.FBX", "BOTH, DIFFERING PX IN RED", "ZOOM X6 ON %d OF THEM" % best[0]]
    for i, p in enumerate(P + [Z]):
        x, y = (i % 2) * (pw + G), (i // 2) * (ph + LH + G)
        label(dr, x, y + 6, labs[i], 28)
        sheet.paste(p, (x, y + LH))
    save(sheet, "08_main_fbx_vs_trace.png")
    return {"diff_px": int(len(diff)), "zoom_window": [x0, y0, zw, zh], "in_zoom": best[0]}
