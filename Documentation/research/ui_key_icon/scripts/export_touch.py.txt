"""Fix stage 2026-10-07 (critic issues 2 and 6): drop-in sprites with names Unity's Resources can load.

The game loads ONE key sprite (UI/HUD_KeyGlyph) and ONE crosshair (UI/HUD_Crosshair) on every
platform, so the @3x files of the Figma stage could never load. These files carry their own names:
  dropin/slot_40x22/HUD_KeyGlyph_Touch.png   120 x 66  (= png/A_S_held@3x.png; today's 40 x 22 pt rect)
  dropin/slot_49x22/HUD_KeyGlyph_Touch.png   147 x 66  (= png/A_L_held@3x.png; the 49 x 22 target)
  dropin/crosshair/HUD_Crosshair_Touch.png   144 x 144, white dot 72 px: 2 : 1 in the 24 pt phone rect
                                             (72 px at @3x), so the dot is 12 pt, like the desktop sprite
  dropin/prompt/HUD_KeyOutline.png           102 x 48  (= png/A_L_missing@2x.png; desktop, 51 x 24 rect)
  dropin/prompt/HUD_KeyOutline_Touch.png     153 x 72  (= png/A_L_missing@3x.png; phone, 51 x 24 pt rect)
  dropin/prompt/HUD_KeyOutline.svg           the vector master (= svg/A_L_missing.svg)
The outline is the locked-prompt and 'other zone' glyph; its box is the 49 x 22 key + 1 px all round
(design.py emit_outline), so place it 1 px left and up of the key slot. Checks -> keyicon_figma/export_touch_checks.json.
Run with Blender's Python 3.11 (numpy)."""
import json, os, shutil, sys
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import numpy as np
import iconlib as L

HERE = os.path.dirname(os.path.abspath(__file__))
PROJ = "/Users/redwang/Desktop/ArtCenter/Fall26T7/EGAM-401A-01 Individual Game Project/Frontrooms3D"
DES = os.environ.get("KEYICON_DES", os.path.join(PROJ, "Documentation/research/ui_key_icon/design"))
PNG, SVG, DROP = (os.path.join(DES, d) for d in ("png", "svg", "dropin"))
OUTJ = os.path.join(os.path.dirname(HERE), "keyicon_figma", "export_touch_checks.json")
os.makedirs(os.path.dirname(OUTJ), exist_ok=True)
checks = {}


def check(path, one_colour=True):
    rgb, a = L.read_png(path)
    u = np.unique(np.round(rgb * 255).astype(int).reshape(-1, 3), axis=0)
    c = {"size": [a.shape[1], a.shape[0]], "corners_alpha": [float(a[y, x]) for y, x in ((0, 0), (0, -1), (-1, 0), (-1, -1))],
         "rgb_values": len(u), "alpha_max": round(float(a.max()), 3)}
    c["pass"] = all(v == 0 for v in c["corners_alpha"]) and (len(u) == 1 or not one_colour)
    return c


for slot, n in (("slot_40x22", "A_S_held"), ("slot_49x22", "A_L_held")):
    dst = os.path.join(DROP, slot, "HUD_KeyGlyph_Touch.png")
    shutil.copyfile(os.path.join(PNG, n + "@3x.png"), dst)
    checks["dropin/%s/HUD_KeyGlyph_Touch.png" % slot] = check(dst)
P = os.path.join(DROP, "prompt")
os.makedirs(P, exist_ok=True)
for dst, src in (("HUD_KeyOutline.png", "A_L_missing@2x.png"), ("HUD_KeyOutline_Touch.png", "A_L_missing@3x.png")):
    shutil.copyfile(os.path.join(PNG, src), os.path.join(P, dst))
    checks["dropin/prompt/" + dst] = check(os.path.join(P, dst))
shutil.copyfile(os.path.join(SVG, "A_L_missing.svg"), os.path.join(P, "HUD_KeyOutline.svg"))
# touch crosshair: 144 x 144, dot 72 (2 : 1 for the 24 pt rect at @3x)
N, D, ss = 144, 72.0, 16
g = (np.arange(N * ss) + 0.5) / ss
X, Y = np.meshgrid(g, g)
a = ((((X - N / 2) ** 2 + (Y - N / 2) ** 2) <= (D / 2) ** 2).reshape(N, ss, N, ss).mean(axis=(1, 3)))
CH = os.path.join(DROP, "crosshair", "HUD_Crosshair_Touch.png")
L.write_png(CH, np.ones((N, N, 3)), a)
c = check(CH)
c["area_diameter_px"] = round(float(2 * np.sqrt(a.sum() / np.pi)), 3)
checks["dropin/crosshair/HUD_Crosshair_Touch.png"] = c
json.dump(checks, open(OUTJ, "w"), indent=1)
for k, v in checks.items():
    print(k, v)
print("[touch] all pass:", all(v["pass"] for v in checks.values()))
