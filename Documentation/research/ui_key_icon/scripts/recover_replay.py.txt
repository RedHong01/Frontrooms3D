def run_py(n, k=0, expand=None):
    b = blocks(dump(n))[k]
    body = b['body']
    if not b['quoted']:
        for var, val in (expand or {}).items():
            body = body.replace(var, val)
    p = subprocess.run(['/usr/bin/python3', '-'], input=body, text=True, cwd=KD, capture_output=True)
    assert p.returncode == 0, (n, p.stderr[-2000:])
    log.append(('patch', '#%d py: %s' % (n, p.stdout.strip()[:80])))
def run_cat(n, k, name):
    b = blocks(dump(n))[k]
    write(name, b['body'].rstrip('\n')); log.append((name, 'cat #%d' % n))
def run_sed(n, target):
    cmd = dump(n)
    m = [x for x in re.findall(r"sed -i '' (?:(?!&&|;\s*(?:cat|grep|cd|echo|ls)).)*?" + re.escape(target), cmd, re.S)]
    assert m, (n, target)
    for s in m:
        p = subprocess.run(s, shell=True, cwd=KD, capture_output=True, text=True, executable='/bin/zsh')
        assert p.returncode == 0, (n, s, p.stderr)
        log.append(('sed', '#%d %s' % (n, s[:100])))
