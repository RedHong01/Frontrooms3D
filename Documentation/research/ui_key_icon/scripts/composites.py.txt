_frames = {}
def frame(name, s=1):
    key = (name, s)
    if key not in _frames:
        img = L.load_frame(BGS.get(name) or BG_MEASURE_ONLY[name], CACHE)
        if s != 1:
            # only the bottom-left HUD region is ever cropped at 2x: upsample that part
            y0, x0 = 840, 0
            reg = img[y0:1080, x0:640]
            up = L.resample_bilinear(reg, reg.shape[1] * s, reg.shape[0] * s)
            big = np.zeros((1080 * s, 1920 * s, 3))
            big[y0 * s:1080 * s, x0 * s:640 * s] = up
            img = big
        _frames[key] = img
    return _frames[key].copy()


def text(img, font, s_text, size, x, baseline, colour, s=1, opacity=1.0, tracking=0.0):
    """Draw text (font px size at 1x) with its baseline at `baseline` (1x coords)."""
    polys, adv = font.text_polys(s_text, size * s, 0, 0, tracking * s)
    if not polys:
        return adv / s
    bx0, by0, bx1, by1 = L.bbox(polys)
    ox, oy = int(math.floor(x * s + bx0)) - 1, int(math.floor(baseline * s + by0)) - 1
    w, h = int(math.ceil(bx1 - bx0)) + 3, int(math.ceil(by1 - by0)) + 3
    local = L.xform(polys, 1.0, 1.0, x * s - ox, baseline * s - oy)
    cov = L.coverage(local, w, h, 1.0, "nonzero") * opacity
    if MEASURE is not None:
        _measure("text " + s_text, img, ox + 1, oy + 1, w - 3, h - 3, L.hexrgb(colour), opacity)
    sprite = np.tile(L.hexrgb(colour), (h, w, 1))
    img[:] = L.composite_over(img, sprite, cov, ox, oy)
    return adv / s


def sprite(img, name, x, y, s=1, opacity=1.0):
    rgb, a = L.read_png(os.path.join(PNG, "%s@%dx.png" % (name, s)))
    if MEASURE is not None:
        iy, ix = np.unravel_index(np.argmax(a), a.shape)          # the sprite's fullest pixel (its paper)
        _measure("sprite " + name, img, int(round(x * s)), int(round(y * s)), a.shape[1], a.shape[0], rgb[iy, ix], opacity * float(a.max()))
    img[:] = L.composite_over(img, rgb, a * opacity, int(round(x * s)), int(round(y * s)))
    return a.shape[1] / s


# ------------------------------------------------------------------ contrast measurement (fix stage 2026-10-07)
# When MEASURE is a list, every text and sprite drawn records the WCAG contrast of its colour
# (blended at its opacity, in linear light) against the ground pixels under its box, i.e. the
# frame plus anything drawn before it (the prompt card). Same method as the critic's contrast.py.
MEASURE = None


def _lum(rgb_lin):
    return 0.2126 * rgb_lin[..., 0] + 0.7152 * rgb_lin[..., 1] + 0.0722 * rgb_lin[..., 2]


def _measure(what, img, x, y, w, h, colour, opacity):
    g = img[max(0, y):y + h, max(0, x):x + w]
    if g.size == 0:
        return
    gl = L.to_lin(g)
    fl = gl * (1 - opacity) + L.to_lin(np.asarray(colour))[None, None, :] * opacity
    a, b = _lum(fl), _lum(gl)
    c = (np.maximum(a, b) + 0.05) / (np.minimum(a, b) + 0.05)
    MEASURE.append({"what": what, "min": float(c.min()), "p10": float(np.percentile(c, 10)), "median": float(np.median(c)),
                    "box": [x, y, w, h], "colour": "".join("%02X" % int(round(v * 255)) for v in np.asarray(colour)), "opacity": round(float(opacity), 3)})


def rect(img, x, y, w, h, colour, s=1, opacity=1.0):
    ww, hh = int(round(w * s)), int(round(h * s))
    img[:] = L.composite_over(img, np.tile(L.hexrgb(colour), (hh, ww, 1)), np.full((hh, ww), opacity), int(round(x * s)), int(round(y * s)))

COURIER_B = L.Font(os.path.join(PROJ, "Assets/Fonts/Period1990/CourierPrime/CourierPrime-Bold.ttf"))
# Courier Prime Bold sized so its cap height = Bayon 20's cap height (the tag's own type in the label).
# The HUD uses Unity legacy Text, whose fontSize is an int: every text size here is the whole
# number the game can render (2026-10-03, 平面视觉's type review). Exact cap matches kept for reference:
# Courier Prime Bold 24.63 -> 25 (cap 14.49 px vs Bayon 14.28), IBM Plex Mono 20.45 -> 20 (cap 13.96).
COUR_SZ_EXACT = 20 * (BAYON.cap / BAYON.upm) / (COURIER_B.cap / COURIER_B.upm)
MONO_SZ_EXACT = 20 * (BAYON.cap / BAYON.upm) / (MONO.cap / MONO.upm)
COUR_SZ = int(round(COUR_SZ_EXACT))      # 25
MONO_SZ = int(round(MONO_SZ_EXACT))      # 20


def draw_label(img, label, x, baseline, col, s=1, op=1.0):
    """'KEY 14' -> Bayon KEY + Courier Prime Bold 14 (Bayon draws 1 as I). Anything else: Bayon."""
    m = re.match(r"^KEY (\d+)$", label)
    if not m:
        return text(img, BAYON, label, 20, x, baseline, col, s, op)
    a = text(img, BAYON, "KEY", 20, x, baseline, col, s, op) + 6
    return a + text(img, COURIER_B, m.group(1), COUR_SZ, x + a, baseline, col, s, op)


def today_prompt(img, cx, baseline, size, s=1, col=PAPER):
    """The prompt the code draws today: "LOCKED  ·  NEEDS THIS ZONE'S KEY" (FrontRoomsMapWorld.Describe),
    Bayon, paper, centred. Bayon has no U+00B7: Unity falls back to another font for the dot
    (audit frame 48); drawn here with IBM Plex Mono."""
    parts = [(BAYON, "LOCKED  "), (MONO, "\u00b7"), (BAYON, "  NEEDS THIS ZONE'S KEY")]
    total = sum(f.width(t, size) for f, t in parts)
    x = cx - total / 2
    for f, t in parts:
        x += text(img, f, t, size, x, baseline, col, s)

# The prompt ground (critic issue 2): the HUD's own hint card, #141414 at 90 % ('media' in
# FrontRooms3DGame.cs BuildHud), padding 12 x 8 px around the glyph row and the meta line.
CARD, CARD_OP = "141414", 0.9
CARD_PAD_X, CARD_PAD_Y = 12, 8


def prompt(img, glyph, gw, gh, s=1, word="LOCKED", meta="NEEDS KEY 14", card=True, word_col=PAPER, meta_col=MUTED,
           cx=960, cy=584, label_sz=20, meta_sz=13, gap=10, meta_dy=None, pad=(CARD_PAD_X, CARD_PAD_Y)):
    """Locked-door prompt under the crosshair (Key prompt: centre x 960, row centre y 584, Bayon 20):
    the outline key + "LOCKED" (paper) on one row, the meta line (Plex Mono 13) centred under it,
    both on a #141414 90 % card. Returns the card box (x, y, w, h) in 1x units."""
    wword = BAYON.width(word, label_sz)
    total = gw + gap + wword
    x0 = cx - total / 2
    gx, gy = round(x0), cy - gh / 2
    wm = MONO.width(meta, meta_sz)
    mb = cy + (meta_dy if meta_dy is not None else max(30, gh / 2 + 16))
    box = None
    if card:
        bx0 = math.floor(min(gx, cx - wm / 2) - pad[0])
        bx1 = math.ceil(max(x0 + total, cx + wm / 2) + pad[0])
        by0 = math.floor(gy - pad[1])
        by1 = math.ceil(mb + pad[1])            # caps and figures only: the baseline is the ink bottom
        rect(img, bx0, by0, bx1 - bx0, by1 - by0, CARD, s, CARD_OP)
        box = (bx0, by0, bx1 - bx0, by1 - by0)
    px, py = OUTLINE_PAD.get(glyph, (0, 0))
    sprite(img, glyph, gx - px, gy - py, s)
    # cap-centred on the row: Bayon caps 0.714 em, baseline = row centre + cap / 2
    text(img, BAYON, word, label_sz, x0 + gw + gap, cy + round(BAYON.cap / BAYON.upm * label_sz / 2), word_col, s)
    text(img, MONO, meta, meta_sz, cx - wm / 2, mb, meta_col, s)
    return box


def label_parts(img, parts, x, baseline, s=1):
    """parts: [(font, text, size, colour, gap_after)] set on one baseline."""
    for font, t, size, col, gap in parts:
        x += text(img, font, t, size, x, baseline, col, s) + gap
    return x

def phone_frame(name):
    """The 16:9 frame cropped to 874:402 (rows 98-981 of 1080) and resampled to 2622 x 1206."""
    if name not in _phone:
        img = frame(name, 1)

def disc(img, cx, cy, d, s, colour="FFFFFF"):
    """An anti-aliased dot (diameter d pt) centred on (cx, cy) pt, at scale s."""
    n = int(math.ceil(d * s)) + 4
    x0, y0 = int(math.floor(cx * s - n / 2)), int(math.floor(cy * s - n / 2))
    ss = 16
    g = (np.arange(n * ss) + 0.5) / ss
    X, Y = np.meshgrid(g + x0, g + y0)
    a = ((((X - cx * s) ** 2 + (Y - cy * s) ** 2) <= (d * s / 2) ** 2).reshape(n, ss, n, ss).mean(axis=(1, 3)))
    img[:] = L.composite_over(img, np.tile(L.hexrgb(colour), (n, n, 1)), a, x0, y0)

    # 2d. phone composites (874 x 402 pt @3x): today's TestFlight HUD vs the proposal, held and locked

        import subprocess
        for version, st, bg in (("today", "held", "wallpaper"), ("target", "held", "wallpaper"), ("target", "held", "lit"),
