"""Check that main's Kit_Key_Zone.fbx (Codex-promoted, 8ef5b64) has the same
side silhouette the icon was traced from (keyicon_design/model/key_side.png,
rendered 12:50 at 20 px/mm, 70 x 35 mm frame centred on Unity (0, 0.5 mm, -4 mm))."""
import bpy, os, json, numpy as np
MAIN = "/Users/redwang/Desktop/ArtCenter/Fall26T7/EGAM-401A-01 Individual Game Project/Frontrooms3D/Assets/Resources/Props/Models"
SCR = "/Users/redwang/FrontRoomsVisualWork"
out = {}
for name in ("Kit_Key_Zone", "Kit_Key_Zone_Nickel"):
    bpy.ops.wm.read_factory_settings(use_empty=True)
    bpy.ops.import_scene.fbx(filepath=os.path.join(MAIN, name + ".fbx"))
    meshes = [o for o in bpy.data.objects if o.type == "MESH"]
    W, H, PX = 1400, 700, 20.0
    mask = np.zeros((H, W), bool)
    allv = []
    ntri = 0
    for o in meshes:
        dg = bpy.context.evaluated_depsgraph_get()
        me = o.evaluated_get(dg).to_mesh()
        me.calc_loop_triangles()
        M = o.matrix_world
        co = np.array([(M @ v.co)[:] for v in me.vertices]) * 1000.0  # mm, Blender world
        allv.append(co)
        # Blender (x,y,z) -> Unity Z = -y, Unity Y = z
        uz, uy = -co[:, 1], co[:, 2]
        px = (uz + 4.0) * PX + W / 2
        py = H / 2 - (uy - 0.5) * PX
        tris = np.array([t.vertices[:] for t in me.loop_triangles])
        ntri += len(tris)
        yy, xx = np.mgrid[0:H, 0:W]
        for a, b, c in tris:
            x0 = int(max(0, np.floor(min(px[a], px[b], px[c])))); x1 = int(min(W - 1, np.ceil(max(px[a], px[b], px[c]))))
            y0 = int(max(0, np.floor(min(py[a], py[b], py[c])))); y1 = int(min(H - 1, np.ceil(max(py[a], py[b], py[c]))))
            if x1 < x0 or y1 < y0:
                continue
            X = xx[y0:y1 + 1, x0:x1 + 1] + .5; Y = yy[y0:y1 + 1, x0:x1 + 1] + .5
            d = (py[b] - py[c]) * (px[a] - px[c]) + (px[c] - px[b]) * (py[a] - py[c])
            if abs(d) < 1e-9:
                continue
            l1 = ((py[b] - py[c]) * (X - px[c]) + (px[c] - px[b]) * (Y - py[c])) / d
            l2 = ((py[c] - py[a]) * (X - px[c]) + (px[a] - px[c]) * (Y - py[c])) / d
            l3 = 1 - l1 - l2
            mask[y0:y1 + 1, x0:x1 + 1] |= (l1 >= -1e-6) & (l2 >= -1e-6) & (l3 >= -1e-6)
    allv = np.concatenate(allv)
    lo, hi = allv.min(0), allv.max(0)
    ref = "key_side.png" if name == "Kit_Key_Zone" else "key_side_nickel.png"
    img = bpy.data.images.load(os.path.join(SCR, "keyicon_design/model", ref))
    a = np.array(img.pixels[:]).reshape(img.size[1], img.size[0], 4)[::-1, :, 3]
    refm = a > 0.5
    inter = (mask & refm).sum(); uni = (mask | refm).sum()
    diff = mask ^ refm
    # pixels of disagreement that are not on a 1-px boundary band
    from numpy import roll
    edge = np.zeros_like(refm)
    for dy in (-1, 0, 1):
        for dx in (-1, 0, 1):
            edge |= roll(roll(refm, dy, 0), dx, 1) != refm
    out[name] = dict(tris=int(ntri), bbox_mm_blender=[[round(x, 3) for x in lo], [round(x, 3) for x in hi]],
                     size_unityZ_mm=round(float(-lo[1] + hi[1]), 3), size_unityY_mm=round(float(hi[2] - lo[2]), 3),
                     thickness_mm=round(float(hi[0] - lo[0]), 3),
                     iou=round(float(inter / uni), 5), diff_px=int(diff.sum()), diff_off_edge_px=int((diff & ~edge).sum()),
                     ref_px=int(refm.sum()), main_px=int(mask.sum()))
    np.save(os.path.join(SCR, "keyicon_resume/mask_%s.npy" % name), mask)
    print("[verify]", name, out[name])
json.dump(out, open(os.path.join(SCR, "keyicon_resume/verify_main_fbx.json"), "w"), indent=1)
print("[verify] DONE")
