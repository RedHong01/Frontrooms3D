LABEL_RE = re.compile(r'(<div class="(?:lab|ovl|ovd|chip)"[^>]*>)(.*?)(</div>)', re.S)


def num_spans(html):
    """Digits inside Bayon labels -> Courier Prime Bold spans (the HUD label rule of board 04)."""
    def fix(m):
        inner = re.sub(r"(?<![&#\w])(\d+)", r'<span class="n">\1</span>', m.group(2))
        return m.group(1) + inner + m.group(3)
    return LABEL_RE.sub(fix, html)

    return ('<!doctype html><html><head><meta charset="utf-8"><style>%s</style></head><body>%s'
            '<div class="title">%s</div><div class="intro">%s</div>%s</body></html>' % (CSS, hdr, title, intro, num_spans(body)))


def img(path, x, y, w=None, h=None, px=False, extra=""):
    st = "left:%dpx;top:%dpx;" % (x, y)
    if w: st += "width:%dpx;" % w
    if h: st += "height:%dpx;" % h
    return '<img class="a%s" src="%s" style="%s%s">' % (" px" if px else "", url(path), st, extra)


def crop(path, x, y, w, h, ox=0, oy=0, label=None, dark_label=False, scale=1, px=False):
    """A window (w x h on the page) onto an image at `scale`, offset (ox, oy) in image px."""
    lab = ""
    if label:
        lab = '<div class="%s">%s</div>' % ("ovd" if dark_label else "ovl", label)
    sz = ""
    if scale != 1:
        sz = "transform:scale(%g);transform-origin:0 0;" % scale
    return ('<div class="panel" style="left:%dpx;top:%dpx;width:%dpx;height:%dpx">%s'
            '<img class="a%s" src="%s" style="left:%dpx;top:%dpx;%s"></div>'
            % (x, y, w, h, lab, " px" if px else "", url(path), -ox * scale, -oy * scale, sz))


def svg_inline(name, scale):
    s = open(os.path.join(SVG, name + ".svg")).read()
    s = re.sub(r"<\?xml[^>]*>", "", s)
    m = re.search(r'width="([\d.]+)" height="([\d.]+)"', s)
    w, h = float(m.group(1)), float(m.group(2))
    s = s.replace(m.group(0), 'width="%g" height="%g"' % (w * scale, h * scale), 1)
    return s, w * scale, h * scale


def panel(x, y, w, h, bg, inner="", label=None, dark_label=False):
    lab = '<div class="%s">%s</div>' % ("ovd" if dark_label else "ovl", label) if label else ""
    return '<div class="panel" style="left:%dpx;top:%dpx;width:%dpx;height:%dpx;background:%s">%s%s</div>' % (x, y, w, h, bg, lab, inner)


def centred_svg(name, scale, w, h, dy=0):
    s, sw, sh = svg_inline(name, scale)
    return '<div class="a" style="left:%dpx;top:%dpx">%s</div>' % ((w - sw) / 2, (h - sh) / 2 + dy, s)


def tile(png, x, y, bg, label=None, w=None, h=None, pad=12):
    """A true-size PNG on a flat swatch (lit #79693C / dark #25200F)."""
    import struct
    d = open(png, "rb").read(32)
    pw, ph = struct.unpack(">II", d[16:24])
    W = w or pw + 2 * pad; H = h or ph + 2 * pad
    lab = '<div class="lab" style="left:%dpx;top:%dpx">%s</div>' % (x, y - 28, label) if label else ""
    return lab + '<div class="panel" style="left:%dpx;top:%dpx;width:%dpx;height:%dpx;background:%s">%s</div>' % (
        x, y, W, H, bg, img(png, (W - pw) // 2, (H - ph) // 2))
