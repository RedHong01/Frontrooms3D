# Coordinates: x east, z north, y up, metres from the room's south-west clear corner (wall faces).
# URP 17 punctual falloff (RealtimeLights.hlsl DistanceAttenuation): E = I * cos(incidence) * spot * (1/d^2) * (1 - (d/r)^4)^2
# URP spot cone: spot = saturate(cos(a) * k + b)^2, k = 1/(cos(inner/2) - cos(outer/2)), b = -cos(outer/2) * k
import math

def sub(a, b): return [b[i] - a[i] for i in range(3)]
def norm(a):
    l = math.sqrt(sum(x * x for x in a)); return [x / l for x in a], l
def dot(a, b): return sum(a[i] * b[i] for i in range(3))
def att(d, r):
    f = d * d / (r * r); s = max(0.0, 1 - f * f); return s * s / (d * d)
def spot(c, inner=100, outer=160):
    ci = math.cos(math.radians(inner / 2)); co = math.cos(math.radians(outer / 2))
    k = 1 / (ci - co); b = -co * k; x = min(1.0, max(0.0, c * k + b)); return x * x
def axis(yaw_deg, tilt_deg):
    # yaw: plan direction in degrees from +x (east) toward +z (north); tilt: down
    y = math.radians(yaw_deg); t = math.radians(tilt_deg)
    return [math.cos(y) * math.cos(t), math.sin(y) * math.cos(t), -math.sin(t)]  # (x, z, y)
def board(name, scr, ax, brd, bnorm, I=4.0, r=9.0):
    d, l = norm(sub(scr, brd)); off = math.degrees(math.acos(max(-1, min(1, dot(d, ax)))))
    inc = dot([-x for x in d], bnorm); E = I * max(0, inc) * spot(dot(d, ax)) * att(l, r)
    print(f"{name}: {l:.2f} m, {off:.1f} deg off the screen axis, incidence {math.degrees(math.acos(max(-1, min(1, inc)))):.1f} deg, E {E:.3f} (mean card), {E * 0.515:.3f} (darkest card)")

# Points are (x, z, y). Kit_CRTTV: screen anchor 0.284 up, screen face ~0.54-0.57 m off the wall it hangs or stands against.
