    // ------------------------------------------------------------------ batch
    public static void RunBatch()
    {
        var args = Environment.GetCommandLineArgs();
        var plans = "W,Wp,Wp2,Wp3,D,Dp,Dp3,H";
        for (var i = 0; i < args.Length; i++)
        {
            if (args[i] == "-stationPlans" && i + 1 < args.Length) plans = args[i + 1];
            if (args[i] == "-stationOut" && i + 1 < args.Length) outRoot = args[i + 1];
            if (args[i] == "-stationNoE") ePass = false;
        }
        if (string.IsNullOrEmpty(outRoot)) outRoot = Path.Combine(Directory.GetParent(Application.dataPath).FullName, "Verification", "station_lookdev");
        Directory.CreateDirectory(outRoot);
        FrontRoomsKitLibrary.ClearCache();
        foreach (var k in plans.Split(','))
        {
            var key = k.Trim();
            if (key.Length == 0) continue;
            var t0 = DateTime.Now;
            try { RunPlan(key); }
            catch (Exception e)
            {
                Debug.LogError("[StationLookdev] plan " + key + " failed: " + e);
                File.AppendAllText(Path.Combine(outRoot, "errors.txt"), key + ": " + e + "\n");
            }
            File.AppendAllText(Path.Combine(outRoot, "timing.txt"), key + "\t" + (DateTime.Now - t0).TotalSeconds.ToString("0", CultureInfo.InvariantCulture) + " s\n");
        }
        Debug.Log("[StationLookdev] ALL DONE -> " + outRoot);
    }

    static void Log(string s) { log.Add(s); Debug.Log("[StationLookdev] " + (plan != null ? plan.key : "-") + " " + s); }
    static string F(float v) { return v.ToString("0.###", CultureInfo.InvariantCulture); }
    static void Mx(string k, float v) { metrics[k] = F(v); }
    static void Mx(string k, int v) { metrics[k] = v.ToString(CultureInfo.InvariantCulture); }
    static void Ms(string k, string v) { metrics[k] = "\"" + (v ?? "").Replace("\\", "/").Replace("\"", "'") + "\""; }
    static void Mv(string k, Vector3 v) { metrics[k] = "[" + F(v.x) + ", " + F(v.y) + ", " + F(v.z) + "]"; }

    static void RunPlan(string key)
    {
        plan = MakePlan(key);
        outDir = Path.Combine(outRoot, key);
        Directory.CreateDirectory(outDir);
        kinds.Clear(); mats.Clear(); log.Clear(); virtualProps.Clear(); virtualFoot.Clear(); metrics.Clear();
        lensRs.Clear(); lampLights.Clear();
        tv = null; tvLight = null; screenR = null;

        var m = BuildModule(plan);
        var seeds = CandidateSeeds(m, 300);
        Log("candidate seeds (arch widths fit G1/G2): " + seeds.Count);
        // Pick by the map's own data before building any geometry: exactly two openings (G2; the map may reopen a
        // wall to keep the chunk connected), and a straight corridor 3 cells out on E1's axis (the approach).
        var pick = -1; var fallback = -1; var tried = 0;
        foreach (var seed in seeds)
        {
            tried++;
            int openings; bool far, far2; string desc;
            SeedData(m, seed, out openings, out far, out far2, out desc);
            if (openings != 2) continue;
            if (fallback < 0) { fallback = seed; Log("first two-opening seed " + seed + ": " + desc); }
            if (far && far2) { pick = seed; Log("picked seed " + seed + " after " + tried + ": " + desc); break; }
        }
        if (pick < 0) { pick = fallback >= 0 ? fallback : seeds[0]; Log("WARNING no seed with two openings and a straight approach; using " + pick); Ms("seed_warning", "no straight approach"); }
        seedUsed = pick;
        NewScene();
        BuildWorld(m, seedUsed);
        Log("seed " + seedUsed + ": perimeter openings " + CountOpenings());
        Mx("seed", seedUsed);
        Mx("seeds_tried", tried);
        Ms("plan", plan.title);

        DisableZoneKeys();
        CollectLamps();
        Dress();
        Physics.SyncTransforms();
        ArchCheck(e1, "e1"); ArchCheck(e2, "e2");
        Measure();
        Shots();
        WalkGrid();
        ValidateModule(m);
        WriteMetrics();
        File.WriteAllLines(Path.Combine(outDir, "log.txt"), log.ToArray());
        Teardown();
        Debug.Log("[StationLookdev] " + key + " done -> " + outDir);
    }

    static void NewScene()
    {
        var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
        EditorSceneManager.SetActiveScene(scene);
        root = new GameObject("STATION LOOKDEV " + plan.key).transform;
        root.position = RootWorld;
        FrontRoomsPostStack.Ensure(root);
        SetupReflections();
        SceneFill();
    }

    static void Teardown()
    {
        if (world != null) world.Release();
        if (root != null) UnityEngine.Object.DestroyImmediate(root.gameObject);
        if (previewProfile != null) UnityEngine.Object.DestroyImmediate(previewProfile);
        world = null; root = null; previewProfile = null; mapT = null; extras = null;
        kinds.Clear(); lensRs.Clear(); lampLights.Clear();
    }

    // ------------------------------------------------------------------ hash (FrontRoomsMap.cs MapHash, internal: replicated)
    static uint Mix(uint h) { unchecked { h ^= h >> 16; h *= 0x7feb352du; h ^= h >> 15; h *= 0x846ca68bu; h ^= h >> 16; return h; } }
    static uint Hash(int seed, int a, int b, int salt)
    {
        unchecked
        {
            var h = Mix((uint)seed ^ 0x9e3779b9u);
            h = Mix(h ^ (uint)a * 0x85ebca6bu);
            h = Mix(h ^ (uint)b * 0xc2b2ae35u);
            h = Mix(h ^ (uint)salt * 0x27d4eb2fu);
            h = Mix(h ^ (uint)0 * 0x165667b1u);
            return h;
        }
    }
    static float Unit(uint h) { return (h >> 8) * (1f / 16777216f); }

    /// <summary>FrontRoomsMapWorld.ArchOpening: width and centre (metres from the edge's west or south end).</summary>
    static void ArchOpening(int seed, GridCoord a, bool alongX, out float width, out float centre)
    {
        var edgeHash = Hash(seed, a.x * 2 + (alongX ? 1 : 0), a.y, 97);
        width = ModuleUnits.ArchMinWidth + ModuleUnits.ArchWidthSpread * Unit(edgeHash);
        var margin = ModuleUnits.ArchCornerMargin + width * .5f;
        centre = Mathf.Lerp(margin, MapGrid.CellSize - margin, Unit(Hash((int)edgeHash, 3, 7, 11)));
    }

    static Opening MakeOpening(int seed, char side, int index, RoomModuleData m)
    {
        var o = new Opening { side = side, index = index };
        var cs = MapGrid.CellSize;
        GridCoord a; bool alongX;
        if (side == 'S') { o.inside = new GridCoord(x0 + index, y0); o.outside = new GridCoord(x0 + index, y0 - 1); a = o.outside; alongX = true; }
        else if (side == 'N') { o.inside = new GridCoord(x0 + index, y0 + m.depth - 1); o.outside = new GridCoord(x0 + index, y0 + m.depth); a = o.inside; alongX = true; }
        else if (side == 'W') { o.inside = new GridCoord(x0, y0 + index); o.outside = new GridCoord(x0 - 1, y0 + index); a = o.outside; alongX = false; }
        else { o.inside = new GridCoord(x0 + m.width - 1, y0 + index); o.outside = new GridCoord(x0 + m.width, y0 + index); a = o.inside; alongX = false; }
        ArchOpening(seed, a, alongX, out o.width, out o.centre);
        if (side == 'S') { o.mid = new Vector3(index * cs + o.centre, 0f, 0f); o.inward = Vector3.forward; o.along = Vector3.right; }
        else if (side == 'N') { o.mid = new Vector3(index * cs + o.centre, 0f, m.depth * cs); o.inward = Vector3.back; o.along = Vector3.right; }
        else if (side == 'W') { o.mid = new Vector3(0f, 0f, index * cs + o.centre); o.inward = Vector3.right; o.along = Vector3.forward; }
        else { o.mid = new Vector3(m.width * cs, 0f, index * cs + o.centre); o.inward = Vector3.left; o.along = Vector3.forward; }
        return o;
    }

    static List<int> CandidateSeeds(RoomModuleData m, int count)
    {
        FrontRoomsModulePreview.Placement(m, out x0, out y0);
        var list = new List<int>();
        for (var seed = 1; seed < 400000 && list.Count < count; seed++)
        {
            var a = MakeOpening(seed, plan.e1Side, plan.e1Index, m);
            var b = MakeOpening(seed, plan.e2Side, plan.e2Index, m);
            if (a.width < 1.6f) continue;                                   // G1: E1 the widest, 1.6-1.8 m
            if (b.width > 1.4f) continue;                                   // G2: E2 1.1-1.4 m
            if (plan.e1Want >= 0f && Mathf.Abs(a.centre - plan.e1Want) > plan.e1Tol) continue;
            list.Add(seed);
        }
        return list;
    }

    /// <summary>The generator's data for a seed (no geometry): perimeter openings and the cells on E1's axis outside.</summary>
    static void SeedData(RoomModuleData m, int seed, out int openings, out bool far, out bool far2, out string desc)
    {
        FrontRoomsModulePreview.Placement(m, out x0, out y0);
        var source = FrontRoomsLevelProfiles.Resolve();
        var prof = source != null ? source : FrontRoomsLevelProfile.Default;
        var g = prof.Generation(seed);
        g.lowShare = 0f; g.standardShare = 1f; g.tallShare = 0f;
        g.officeShare = m.theme == ZoneTheme.Office ? 1f : 0f;
        g.moduleChance = 0f;
        var cache = new FrontRoomsMapCache(g, prof.ModuleData());
        cache.Place(new GridCoord(0, 0), m.Clone(), x0, y0);
        openings = 0;
        var list = new List<string>();
        for (var i = 0; i < m.width; i++)
        {
            if (cache.Edge(new GridCoord(x0 + i, y0), new GridCoord(x0 + i, y0 - 1)) != EdgeKind.Wall) { openings++; list.Add("S" + i); }
            if (cache.Edge(new GridCoord(x0 + i, y0 + m.depth - 1), new GridCoord(x0 + i, y0 + m.depth)) != EdgeKind.Wall) { openings++; list.Add("N" + i); }
        }
        for (var j = 0; j < m.depth; j++)
        {
            if (cache.Edge(new GridCoord(x0, y0 + j), new GridCoord(x0 - 1, y0 + j)) != EdgeKind.Wall) { openings++; list.Add("W" + j); }
            if (cache.Edge(new GridCoord(x0 + m.width - 1, y0 + j), new GridCoord(x0 + m.width, y0 + j)) != EdgeKind.Wall) { openings++; list.Add("E" + j); }
        }
        var o = MakeOpening(seed, plan.e1Side, plan.e1Index, m);
        var c1 = o.outside; var c2 = Step(c1, -1, o); var c3 = Step(c2, -1, o);
        far = cache.Edge(c1, c2) != EdgeKind.Wall;
        far2 = far && cache.Edge(c2, c3) != EdgeKind.Wall;
        desc = "openings " + string.Join(" ", list.ToArray()) + "; approach " + (far ? (far2 ? "3 cells straight" : "2 cells") : "1 cell") + "; E1 " + o.width.ToString("0.00") + " m at " + o.centre.ToString("0.00");
    }

    /// <summary>Rays at 1.0 m through each opening's middle (expect clear) and just past its jambs (expect wall).</summary>
    static void ArchCheck(Opening o, string name)
    {
        var up = Vector3.up * 1.0f;
        var a = W(o.mid - o.inward * .6f + up); var b = W(o.mid + o.inward * .6f + up);
        var mid = Physics.Linecast(a, b);
        var edge = o.along * (o.width * .5f + .12f);
        var l = Physics.Linecast(W(o.mid + edge - o.inward * .6f + up), W(o.mid + edge + o.inward * .6f + up));
        var r = Physics.Linecast(W(o.mid - edge - o.inward * .6f + up), W(o.mid - edge + o.inward * .6f + up));
        var ok = !mid && l && r;
        Ms(name + "_arch_check", (ok ? "OK" : "MISMATCH") + " (middle " + (mid ? "blocked" : "clear") + ", jambs " + (l ? "wall" : "open") + "/" + (r ? "wall" : "open") + ")");
        Log(name + " arch check " + (ok ? "OK" : "MISMATCH"));
    }

    static GridCoord Step(GridCoord c, int dir, Opening o)
    {
        // one cell further along the opening's inward direction (dir +1) or outward (dir -1)
        var dx = Mathf.RoundToInt(o.inward.x) * dir; var dy = Mathf.RoundToInt(o.inward.z) * dir;
        return new GridCoord(c.x + dx, c.y + dy);
    }

    static bool cache_EdgeOk(GridCoord a, GridCoord b)
    {
        if (world == null || world.Cache == null) return false;
        return world.Cache.Edge(a, b) != EdgeKind.Wall;
    }

    // ------------------------------------------------------------------ module and map
    static RoomModuleData BuildModule(Plan p)
    {
        var m = new RoomModuleData { width = p.w, depth = p.d, height = ZoneHeight.Standard, theme = p.theme, fill = ModuleFill.None, columns = ModuleColumns.None };
        m.Normalize();
        SetEdge(m, p.e1Side, p.e1Index, ModuleEdge.Arch);
        SetEdge(m, p.e2Side, p.e2Index, ModuleEdge.Arch);
        for (var k = 0; k < m.lamps.Length && k < p.lamps.Length; k++)
        {
            var c = p.lamps[k];
            m.lamps[k] = c == 'S' ? ModuleLamp.Steady : c == 'D' ? ModuleLamp.Dim : c == 'O' ? ModuleLamp.Off : ModuleLamp.Auto;
        }
        m.props = p.props.ToArray();
        m.allowRotate = false;
        return m;
    }

    static void SetEdge(RoomModuleData m, char side, int index, ModuleEdge e)
    {
        if (side == 'S') m.south[index] = e; else if (side == 'N') m.north[index] = e; else if (side == 'W') m.west[index] = e; else m.east[index] = e;
    }

    static void BuildWorld(RoomModuleData m, int seed)
    {
        FrontRoomsModulePreview.Placement(m, out x0, out y0);
        ox = x0 * MapGrid.CellSize; oz = y0 * MapGrid.CellSize;
        e1 = MakeOpening(seed, plan.e1Side, plan.e1Index, m);
        e2 = MakeOpening(seed, plan.e2Side, plan.e2Index, m);
        var source = FrontRoomsLevelProfiles.Resolve();
        previewProfile = UnityEngine.Object.Instantiate(source != null ? source : FrontRoomsLevelProfile.Default);
        previewProfile.hideFlags = HideFlags.DontSave;
        var g = previewProfile.Generation(seed);
        g.lowShare = 0f; g.standardShare = 1f; g.tallShare = 0f;
        g.officeShare = m.theme == ZoneTheme.Office ? 1f : 0f;
        g.moduleChance = 0f;
        previewProfile.generation = g;
        previewProfile.buildRadius = 1;
        var go = new GameObject("STATION / map");
        go.SetActive(false);
        go.transform.SetParent(root, false);
        world = go.AddComponent<FrontRoomsMapWorld>();
        world.Profile = previewProfile;
        world.TagModuleProps = true;
        stamped = m.Clone();
        world.PlaceModule(stamped, new GridCoord(0, 0), x0, y0);
        // The capture eye (the map's player for lamp radii) starts just inside E1.
        var spawn = new Vector3(ox + e1.mid.x + e1.inward.x * 1.0f, .05f, oz + e1.mid.z + e1.inward.z * 1.0f);
        world.OverrideSpawn(spawn, Mathf.Atan2(e1.inward.x, e1.inward.z) * Mathf.Rad2Deg);
        go.SetActive(true);
        world.BuildForCapture();
        mapT = world.transform;
        extras = new GameObject("STATION / blockout extras").transform;
        extras.SetParent(root, false);
    }

    static int CountOpenings()
    {
        var n = 0; var list = new List<string>();
        for (var i = 0; i < plan.w; i++)
        {
            if (world.Cache.Edge(new GridCoord(x0 + i, y0), new GridCoord(x0 + i, y0 - 1)) != EdgeKind.Wall) { n++; list.Add("S" + i + ":" + world.Cache.Edge(new GridCoord(x0 + i, y0), new GridCoord(x0 + i, y0 - 1))); }
            if (world.Cache.Edge(new GridCoord(x0 + i, y0 + plan.d - 1), new GridCoord(x0 + i, y0 + plan.d)) != EdgeKind.Wall) { n++; list.Add("N" + i + ":" + world.Cache.Edge(new GridCoord(x0 + i, y0 + plan.d - 1), new GridCoord(x0 + i, y0 + plan.d))); }
        }
        for (var j = 0; j < plan.d; j++)
        {
            if (world.Cache.Edge(new GridCoord(x0, y0 + j), new GridCoord(x0 - 1, y0 + j)) != EdgeKind.Wall) { n++; list.Add("W" + j + ":" + world.Cache.Edge(new GridCoord(x0, y0 + j), new GridCoord(x0 - 1, y0 + j))); }
            if (world.Cache.Edge(new GridCoord(x0 + plan.w - 1, y0 + j), new GridCoord(x0 + plan.w, y0 + j)) != EdgeKind.Wall) { n++; list.Add("E" + j + ":" + world.Cache.Edge(new GridCoord(x0 + plan.w - 1, y0 + j), new GridCoord(x0 + plan.w, y0 + j))); }
        }
        Ms("openings", string.Join(" ", list.ToArray()));
        return n;
    }

    static void DisableZoneKeys()
    {
        foreach (var t in world.GetComponentsInChildren<Transform>(true))
            if (t.name.StartsWith("Key \u00b7 zone") || t.name.StartsWith("Key . zone"))
            {
                Log("zone key hidden (the floor key is the station's, T1 = A): " + t.name + " at " + Mod(t.position));
                t.gameObject.SetActive(false);
            }
    }

    static void CollectLamps()
    {
        foreach (var r in world.GetComponentsInChildren<Renderer>(true))
        {
            var k = Kind.Shell;
            if (r.name == "Lens" && r.transform.parent != null && r.transform.parent.name.StartsWith("Fixture")) { k = Kind.Lens; lensRs.Add(r); }
            else
            {
                for (var t = r.transform; t != null && t != mapT; t = t.parent)
                    if (t.name.StartsWith("Kit_")) { k = Kind.Furniture; break; }
            }
            kinds[r] = k;
        }
        foreach (var l in world.GetComponentsInChildren<Light>(true))
            if (l.transform.parent != null && l.transform.parent.name.StartsWith("Fixture")) lampLights.Add(l);
        Log("map lamps " + lampLights.Count + ", lenses " + lensRs.Count);
    }

    // ------------------------------------------------------------------ coordinates
    static Vector3 W(Vector3 m) { return mapT.TransformPoint(new Vector3(ox + m.x, m.y, oz + m.z)); }
    static Vector3 W(float x, float y, float z) { return W(new Vector3(x, y, z)); }
    static Vector3 Mod(Vector3 world) { var l = mapT.InverseTransformPoint(world); return new Vector3(l.x - ox, l.y, l.z - oz); }
    static Quaternion R(float yaw) { return mapT.rotation * Quaternion.Euler(0f, yaw, 0f); }

    static Renderer Box(Transform parent, string name, Vector3 centre, Vector3 size, float yaw, Material m, Kind k, bool solid)
    {
        var go = GameObject.CreatePrimitive(PrimitiveType.Cube);
        go.name = name;
        go.transform.SetParent(parent, false);
        go.transform.position = W(centre);
        go.transform.rotation = R(yaw);
        go.transform.localScale = size;
        if (!solid) UnityEngine.Object.DestroyImmediate(go.GetComponent<Collider>());
        var r = go.GetComponent<MeshRenderer>();
        r.sharedMaterial = m;
        kinds[r] = k;
        return r;
    }

    /// <summary>A box in a frame (origin o, yaw): lc and size in the frame's axes.</summary>
    static Renderer FBox(Transform p, string name, Vector3 o, float yaw, Vector3 lc, Vector3 size, Material m, Kind k, bool solid)
    {
        return Box(p, name, o + Rot(yaw, lc), size, yaw, m, k, solid);
    }

    static Renderer Seg(Transform parent, string name, Vector3 a, Vector3 b, float w, Material m, Kind k, bool solid)
    {
        var go = GameObject.CreatePrimitive(PrimitiveType.Cube);
        go.name = name;
        go.transform.SetParent(parent, false);
        var wa = W(a); var wb = W(b);
        var d = wb - wa;
        go.transform.position = (wa + wb) * .5f;
        var up = Mathf.Abs(Vector3.Dot(d.normalized, Vector3.up)) > .98f ? Vector3.forward : Vector3.up;
        go.transform.rotation = Quaternion.LookRotation(d, up);
        go.transform.localScale = new Vector3(w, w, d.magnitude);
        if (!solid) UnityEngine.Object.DestroyImmediate(go.GetComponent<Collider>());
        var r = go.GetComponent<MeshRenderer>();
        r.sharedMaterial = m;
        kinds[r] = k;
        return r;
    }

    static Renderer Cyl(Transform parent, string name, Vector3 baseCentre, float radius, float height, Material m, Kind k, bool solid)
    {
        var go = GameObject.CreatePrimitive(PrimitiveType.Cylinder);
        go.name = name;
        go.transform.SetParent(parent, false);
        go.transform.position = W(baseCentre + Vector3.up * (height * .5f));
        go.transform.rotation = R(0f);
        go.transform.localScale = new Vector3(radius * 2f, height * .5f, radius * 2f);
        UnityEngine.Object.DestroyImmediate(go.GetComponent<Collider>());
        if (solid) { var bc = go.AddComponent<BoxCollider>(); bc.size = new Vector3(1f, 2f, 1f); }
        var r = go.GetComponent<MeshRenderer>();
        r.sharedMaterial = m;
        kinds[r] = k;
        return r;
    }

    static Renderer Ball(Transform parent, string name, Vector3 c, float d, Material m, Kind k)
    {
        var go = GameObject.CreatePrimitive(PrimitiveType.Sphere);
        go.name = name;
        go.transform.SetParent(parent, false);
        go.transform.position = W(c);
        go.transform.localScale = Vector3.one * d;
        UnityEngine.Object.DestroyImmediate(go.GetComponent<Collider>());
        var r = go.GetComponent<MeshRenderer>();
        r.sharedMaterial = m;
        r.shadowCastingMode = ShadowCastingMode.Off;
        kinds[r] = k;
        return r;
    }

    static GameObject KitAt(string asset, Transform parent, Vector3 pos, Quaternion worldRot, Kind k)
    {
        var go = FrontRoomsKitLibrary.Spawn(asset, parent, Vector3.zero, Quaternion.identity, null, false);
        if (go == null) { Log("MISSING kit asset " + asset); return null; }
        go.transform.position = W(pos);
        go.transform.rotation = worldRot;
        foreach (var r in go.GetComponentsInChildren<Renderer>(true)) kinds[r] = k;
        return go;
    }

    }

    static float WallFace(Vector3 n)
    {
        // n into the room: the wall's face coordinate along -n
        if (n.x > .5f) return -WF;                      // west wall: face x = 0.08 -> dot(q, -n) = -x = -0.08
        if (n.x < -.5f) return plan.w * 3f - WF;        // east wall: dot(q, -n) = x = 8.92
        if (n.z > .5f) return -WF;                      // south wall
        return plan.d * 3f - WF;                        // north wall
    }

    // ------------------------------------------------------------------ the TV (S01) and its light (V3)
    static void BuildTv()
    {
        var rot = R(plan.tvYaw) * Quaternion.Euler(plan.tvTilt, 0f, 0f);
        var anchor = new Vector3(0f, .284f, .245f);
        var go = FrontRoomsKitLibrary.Spawn("Kit_CRTTV", extras, Vector3.zero, Quaternion.identity, null, false);
        go.name = "Kit_CRTTV (the beacon set)";
        go.transform.rotation = rot;
        go.transform.position = W(plan.screen) - rot * anchor;
        foreach (var r in go.GetComponentsInChildren<Renderer>(true)) kinds[r] = Kind.SetBody;
        var bc = go.AddComponent<BoxCollider>(); bc.center = new Vector3(0f, .25f, 0f); bc.size = new Vector3(.62f, .50f, .49f);
        tv = go.transform;
        var q = GameObject.CreatePrimitive(PrimitiveType.Quad);
        q.name = "screen surface (ad card, CRT_AdSurface)";
        UnityEngine.Object.DestroyImmediate(q.GetComponent<Collider>());
        q.transform.SetParent(tv, false);
        // size and place the picture on the kit's own Prop_ScreenCRT faces (LOD0 submesh bounds, set-local)
        var sc = anchor; var sz = new Vector2(.445f, .336f); var front = anchor.z; var found = false;
        foreach (var mr in go.GetComponentsInChildren<MeshRenderer>(true))
        {
            var mf = mr.GetComponent<MeshFilter>();
            if (mf == null || mf.sharedMesh == null) continue;
            var ms = mr.sharedMaterials;
            for (var i = 0; i < ms.Length && i < mf.sharedMesh.subMeshCount; i++)
            {
                if (ms[i] == null || !ms[i].name.StartsWith("Prop_ScreenCRT")) continue;
                var sb = mf.sharedMesh.GetSubMesh(i).bounds;
                var c = tv.InverseTransformPoint(mr.transform.TransformPoint(sb.center));
                var ext = tv.InverseTransformVector(mr.transform.TransformVector(sb.size));
                sc = c; sz = new Vector2(Mathf.Abs(ext.x), Mathf.Abs(ext.y)); front = c.z + Mathf.Abs(ext.z) * .5f;
                found = true;
                Log("screen faces (" + mr.name + " submesh " + i + "): centre " + c.ToString("F3") + " size " + ext.ToString("F3"));
                break;
            }
            if (found) break;
        }
        if (!found) Log("WARNING no Prop_ScreenCRT submesh found: picture placed at the screen anchor");
        q.transform.localPosition = new Vector3(sc.x, sc.y, front + .003f);
        q.transform.localRotation = Quaternion.Euler(0f, 180f, 0f);
        q.transform.localScale = new Vector3(sz.x * .96f, sz.y * .96f, 1f);
        Mx("screen_picture_w_m", sz.x * .96f); Mx("screen_picture_h_m", sz.y * .96f);
        screenR = q.GetComponent<MeshRenderer>();
        screenR.shadowCastingMode = ShadowCastingMode.Off;
        kinds[screenR] = Kind.Screen;
        var template = Resources.Load<Material>("Media/CRT_AdSurface");
        for (var c = 1; c <= 5; c++)
        {
            var m = template != null ? new Material(template) : new Material(Shader.Find("Universal Render Pipeline/Unlit"));
            m.name = "CRT ad card " + c;
            var tex = AssetDatabase.LoadAssetAtPath<Texture2D>(CardDir + "card_" + c + ".png");
            if (tex == null) Log("WARNING card texture missing " + c);
            if (m.HasProperty("_BaseMap")) m.SetTexture("_BaseMap", tex);
            if (m.HasProperty("_MainTex")) m.SetTexture("_MainTex", tex);
            if (m.HasProperty("_BaseColor")) m.SetColor("_BaseColor", Color.white);
            cardM[c] = m;
        }
        cardM[0] = cardM[1];
        screenOffM = Slot("Prop_ScreenCRT", "1E2422", .9f);
        var lg = new GameObject("TV light (V3, station: spot 100/160, range 9)");
        lg.transform.SetParent(tv, false);
        lg.transform.localPosition = new Vector3(sc.x, sc.y, Mathf.Max(front, anchor.z) + .02f);
        lg.transform.localRotation = Quaternion.identity;
        tvLight = lg.AddComponent<Light>();
        tvLight.type = LightType.Spot;
        tvLight.spotAngle = 160f; tvLight.innerSpotAngle = 100f;
        tvLight.range = 9f;
        tvLight.shadows = LightShadows.Soft;
        tvLight.shadowStrength = 1f;
        tvLight.shadowNearPlane = .05f;
        // shadow resolution tier: URP's default for additional lights is already High (UniversalAdditionalLightData)
        SetTv(1);
        if (plan.mount == "wall" || plan.mount == "corner") Bracket();
        Mv("screen_world_module", Mod(tvLight.transform.position));
        // the set's extent in module metres (the body must stay inside the room and under the ceiling)
        var tb = new Bounds(tv.position, Vector3.zero);
        foreach (var r in tv.GetComponentsInChildren<Renderer>(true)) if (r != screenR) tb.Encapsulate(r.bounds);
        Mv("set_bounds_min", Mod(tb.min)); Mv("set_bounds_max", Mod(tb.max));
        if (plan.mount == "wall" || plan.mount == "corner")
        {
            // set + bracket as one virtual prop for the module check: the measured extent, clamped to the clear floor
            var mn = Mod(tb.min); var mx = Mod(tb.max);
            float cx0 = Mathf.Max(mn.x, WF), cz0 = Mathf.Max(mn.z, WF), cx1 = Mathf.Min(mx.x, plan.w * 3f - WF), cz1 = Mathf.Min(mx.z, plan.d * 3f - WF);
            var cc = new Vector3((cx0 + cx1) * .5f, mn.y - .02f, (cz0 + cz1) * .5f);
            AddVirtual(plan.mount == "corner" ? "Kit_TVWallBracket_Corner" : "Kit_TVWallBracket", cc, 0f,
                new[] { cx0 - cc.x, cz0 - cc.z, cx1 - cc.x, cz1 - cc.z, Mathf.Min(mx.y - mn.y + .02f, 2.85f - cc.y) });
            Mx("set_poke_into_wall_m", Mathf.Max(0f, Mathf.Max(Mathf.Max(WF - mn.x, mx.x - (plan.w * 3f - WF)), Mathf.Max(WF - mn.z, mx.z - (plan.d * 3f - WF)))));
        }
    }

    }

    // ------------------------------------------------------------------ measurement (geometry and analytic light)
    static float Etv(Vector3 pModule, Vector3 nModule, float intensity)
    {
        var lp = Mod(tvLight.transform.position);
        var ax = (Quaternion.Inverse(mapT.rotation) * tvLight.transform.forward).normalized;
        var d = pModule - lp; var dist = d.magnitude; var dir = d / dist;
        var f = dist * dist / (tvLight.range * tvLight.range); var s = Mathf.Max(0f, 1f - f * f); var att = s * s / (dist * dist);
        var ci = Mathf.Cos(tvLight.innerSpotAngle * .5f * Mathf.Deg2Rad); var co = Mathf.Cos(tvLight.spotAngle * .5f * Mathf.Deg2Rad);
        var k = 1f / Mathf.Max(ci - co, .001f); var b = -co * k;
        var sp = Mathf.Clamp01(Vector3.Dot(dir, ax) * k + b); sp *= sp;
        var inc = Mathf.Max(0f, Vector3.Dot(-dir, nModule));
        return intensity * inc * sp * att;
    }

    static void Measure()
    {
        var lp = Mod(tvLight.transform.position);
        var ax = (Quaternion.Inverse(mapT.rotation) * tvLight.transform.forward).normalized;
        var axFlat = new Vector3(ax.x, 0f, ax.z).normalized;
        Mv("screen_centre", lp);
        Mv("screen_axis", ax);
        Mx("e1_width", e1.width); Mx("e1_centre_along", e1.centre); Mv("e1_mid", e1.mid);
        Mx("e2_width", e2.width); Mx("e2_centre_along", e2.centre); Mv("e2_mid", e2.mid);
        Mv("e1_inward", e1.inward); Mv("e2_inward", e2.inward);
        // E1 sightlines (plan angles, from the threshold centre at eye height)
        var e1Eye = e1.mid + Vector3.up * EyeH;
        var toScreen = lp - e1Eye;
        var toScreenFlat = new Vector3(toScreen.x, 0f, toScreen.z);
        Mx("e1_to_screen_m", toScreen.magnitude);
        Mx("set_off_e1_axis_deg", Vector3.Angle(toScreenFlat, e1.inward));
        Mx("e1_off_screen_axis_deg", Vector3.Angle(-toScreenFlat, axFlat));
        var toKey = keyLook - e1Eye; var toKeyFlat = new Vector3(toKey.x, 0f, toKey.z);
        Mx("key_off_e1_axis_deg", Vector3.Angle(toKeyFlat, e1.inward));
        var toE2 = e2.mid - e1.mid;
        Mx("e2_off_e1_axis_deg", Vector3.Angle(new Vector3(toE2.x, 0f, toE2.z), e1.inward));
        Mx("e1_to_e2_m", new Vector3(toE2.x, 0f, toE2.z).magnitude);
        // the key against the screen (G4)
        var sk = keyLook - lp;
        Mx("key_dist_3d_m", sk.magnitude);
        Mx("key_off_screen_axis_3d_deg", Vector3.Angle(sk, ax));
        Mx("key_off_screen_axis_plan_deg", Vector3.Angle(new Vector3(sk.x, 0f, sk.z), axFlat));
        Mx("key_ahead_of_screen_plane_m", Vector3.Dot(sk, axFlat));
        Mx("key_angle_to_screen_plane_deg", 90f - Vector3.Angle(new Vector3(sk.x, 0f, sk.z), axFlat));
        Mx("key_incidence_deg", Vector3.Angle(-sk, plan.keyNormal));
        var plateP = new Vector3(plan.keyWall.x, plan.keyWall.y + .01f, plan.keyWall.z) + plan.keyNormal * .056f;
        Mx("plate_E_analytic_mean", Etv(plateP, plan.keyNormal, CardI[0]));
        Mx("plate_E_analytic_darkest", Etv(plateP, plan.keyNormal, CardI[5]));
        Mx("plate_Y_analytic_mean", Etv(plateP, plan.keyNormal, CardI[0]) * Lum(CardC[0]));
        Mx("plate_Y_analytic_darkest", Etv(plateP, plan.keyNormal, CardI[5]) * Lum(CardC[5]));
        // light at E1 from the TV alone (analytic, no shadows): threshold floor, 1.0 m up facing the set, 1.5 m outside
        Mx("e1_E_tv_floor_threshold", Etv(e1.mid + Vector3.up * .002f, Vector3.up, CardI[0]));
        var faceSet = (lp - (e1.mid + Vector3.up)).normalized;
        Mx("e1_E_tv_1m_facing_set", Etv(e1.mid + Vector3.up, faceSet, CardI[0]));
        Mx("e1_E_tv_floor_outside_1p5", Etv(e1.mid - e1.inward * 1.5f + Vector3.up * .002f, Vector3.up, CardI[0]));
        Mx("e1_E_tv_floor_inside_1p0", Etv(e1.mid + e1.inward * 1.0f + Vector3.up * .002f, Vector3.up, CardI[0]));
        // the take
        var takeEye = new Vector3(plan.take.x, EyeH, plan.take.y);
        var view = (keyLook - takeEye).normalized;
        Mx("take_to_key_m", (keyLook - takeEye).magnitude);
        Mx("take_set_off_view_deg", Vector3.Angle(lp - takeEye, view));
        Mx("take_e2_dist_m", Vector3.Distance(new Vector3(plan.take.x, 0f, plan.take.y), e2.mid));
        Mx("take_e1_dist_m", Vector3.Distance(new Vector3(plan.take.x, 0f, plan.take.y), e1.mid));
        // G6: anything over 1.1 m on the straight floor lines from the take spot to each exit

    static float Lum(Color c) { return .2126f * c.r + .7152f * c.g + .0722f * c.b; }

    static string TallOnLine(Vector3 a, Vector3 b)
    {
        var hits = new List<string>();
        foreach (var kv in kinds)
        {
            var r = kv.Key;
            if (r == null || !r.enabled) continue;
            if (kv.Value != Kind.Furniture && kv.Value != Kind.SetBody) continue;
            var bb = r.bounds;
            var bmin = Mod(bb.min); var bmax = Mod(bb.max);
            var lo = Mathf.Min(bmin.y, bmax.y); var hi = Mathf.Max(bmin.y, bmax.y);
            if (hi <= 1.1f || lo >= 1.75f) continue;     // shorter than 1.1 m, or above head height
            var minx = Mathf.Min(bmin.x, bmax.x); var maxx = Mathf.Max(bmin.x, bmax.x);
            var minz = Mathf.Min(bmin.z, bmax.z); var maxz = Mathf.Max(bmin.z, bmax.z);
            if (maxx < 0f || minx > plan.w * 3f || maxz < 0f || minz > plan.d * 3f) continue;
            for (var t = 0f; t <= 1f; t += .02f)
            {
                var p = Vector3.Lerp(a, b, t);
                if (p.x > minx && p.x < maxx && p.z > minz && p.z < maxz) { var nm = r.transform.parent != null ? r.transform.parent.name + "/" + r.name : r.name; if (!hits.Contains(nm)) hits.Add(nm); break; }
            }
        }
        return hits.Count == 0 ? "none" : string.Join("; ", hits.ToArray());
    }

    static void SwapMaterials(Func<Renderer, Kind, Material> pick, Dictionary<Renderer, Material[]> saved, List<Renderer> hidden, List<Renderer> shown)
    {
        foreach (var kv in kinds)
        {
            var r = kv.Key;
            if (r == null) continue;
            var m = pick(r, kv.Value);
            var isPortal = kv.Value == Kind.Portal1 || kv.Value == Kind.Portal2 || kv.Value == Kind.ProbeIn || kv.Value == Kind.ProbeOut;
            if (!r.enabled && !isPortal) continue;
            if (!r.gameObject.activeInHierarchy) continue;
            saved[r] = r.sharedMaterials;
            if (m == null) { if (r.enabled) { hidden.Add(r); r.enabled = false; } continue; }
            if (!r.enabled) { r.enabled = true; shown.Add(r); }
            var arr = new Material[r.sharedMaterials.Length];
            for (var i = 0; i < arr.Length; i++) arr[i] = m;
            r.sharedMaterials = arr;
        }
    }

    static void Restore(Dictionary<Renderer, Material[]> saved, List<Renderer> hidden, List<Renderer> shown)
    {
        foreach (var kv in saved) if (kv.Key != null) kv.Key.sharedMaterials = kv.Value;
        foreach (var r in hidden) if (r != null) r.enabled = true;
        foreach (var r in shown) if (r != null) r.enabled = false;
    }

    static void RenderFlat(string file, Func<Renderer, Kind, Material> pick)
    {
        var saved = new Dictionary<Renderer, Material[]>(); var hidden = new List<Renderer>(); var shown = new List<Renderer>();
        SwapMaterials(pick, saved, hidden, shown);
        var fixtures = FrontRoomsWallFixtureRenderer.Enabled; FrontRoomsWallFixtureRenderer.Enabled = false;   // instanced outlets are not Renderers
        var fog = RenderSettings.fog; RenderSettings.fog = false;
        var data = cam.GetUniversalAdditionalCameraData();
        data.renderPostProcessing = false;
        var prev = cam.targetTexture;
        var rt = new RenderTexture(1920, 1080, 24, RenderTextureFormat.ARGB32) { antiAliasing = 1 };
        cam.targetTexture = rt; cam.allowMSAA = false;
        cam.Render();
        Save(rt, Path.Combine(outDir, file + ".png"));
        cam.targetTexture = prev; cam.allowMSAA = true; rt.Release();
        data.renderPostProcessing = true;
        RenderSettings.fog = fog;
        FrontRoomsWallFixtureRenderer.Enabled = fixtures;
        Restore(saved, hidden, shown);
    }

    static readonly Dictionary<string, Material> unlitCache = new Dictionary<string, Material>();
    static Material Unlit(Color c)
    {
        var key = c.ToString();
        Material m;
        if (unlitCache.TryGetValue(key, out m) && m != null) return m;
        m = new Material(Shader.Find("Universal Render Pipeline/Unlit"));
        m.SetColor("_BaseColor", c);
        unlitCache[key] = m;
        return m;
    }

    // ------------------------------------------------------------------ collision: capsule walk grids
    static readonly float[] Radii = { .30f, .35f, .40f, .45f, .50f, .55f, .60f, .65f, .70f };

    static void WalkGrid()
    {
        Physics.SyncTransforms();
        const float step = .05f;
        float minX = -3f, minZ = -3f, maxX = plan.w * 3f + 3f, maxZ = plan.d * 3f + 3f;
        var nx = Mathf.RoundToInt((maxX - minX) / step); var nz = Mathf.RoundToInt((maxZ - minZ) / step);
        var player = new byte[nx * nz]; var relay = new byte[nx * nz];
        var mask = ~0;
        for (var j = 0; j < nz; j++)
            for (var i = 0; i < nx; i++)
            {
                var x = minX + (i + .5f) * step; var z = minZ + (j + .5f) * step;
                byte lv = 0;
                for (var k = 0; k < Radii.Length; k++)
                {
                    var r = Radii[k];
                    if (Physics.CheckCapsule(W(x, .31f + r, z), W(x, 1.74f - r, z), r, mask, QueryTriggerInteraction.Ignore)) break;
                    lv = (byte)(k + 1);
                }
                player[i + j * nx] = lv;
                relay[i + j * nx] = (byte)(Physics.CheckCapsule(W(x, .70f, z), W(x, 1.65f, z), .30f, mask, QueryTriggerInteraction.Ignore) ? 0 : 1);
            }
        File.WriteAllBytes(Path.Combine(outDir, "walk_player.u8"), player);
        File.WriteAllBytes(Path.Combine(outDir, "walk_relay.u8"), relay);
        var sb = new StringBuilder();
        sb.Append("{\"step\": " + F(step) + ", \"minX\": " + F(minX) + ", \"minZ\": " + F(minZ) + ", \"nx\": " + nx + ", \"nz\": " + nz + ", \"radii\": [");
        sb.Append(string.Join(", ", Radii.Select(r => F(r)).ToArray()));
        sb.Append("], \"player\": \"capsule y 0.31-1.74, radius per level\", \"relay\": \"capsule r 0.30, y 0.40-1.95\"}");
        File.WriteAllText(Path.Combine(outDir, "walk_meta.json"), sb.ToString());
        Log("walk grid " + nx + " x " + nz);
    }

    static Cubemap flatCube;
    static void SetupReflections()
    {
        flatCube = new Cubemap(16, TextureFormat.RGBAHalf, false) { name = "Station_FlatAmbientCube" };
        var sky = FrontRoomsLook.AmbientSky.linear; var eq = FrontRoomsLook.AmbientEquator.linear; var gr = FrontRoomsLook.AmbientGround.linear;
        foreach (CubemapFace f in Enum.GetValues(typeof(CubemapFace)))
        {
            if (f == CubemapFace.Unknown) continue;
            var col = f == CubemapFace.PositiveY ? sky : f == CubemapFace.NegativeY ? gr : eq;
            var px = new Color[16 * 16];
            for (var i = 0; i < px.Length; i++)
            {
                var row = i / 16;
                var c = col;
                if (f != CubemapFace.PositiveY && f != CubemapFace.NegativeY) c = row < 8 ? Color.Lerp(sky, eq, row / 8f) : Color.Lerp(eq, gr, (row - 8) / 8f);
                px[i] = c;
            }
            flatCube.SetPixels(px, f);
        }
        flatCube.Apply();
        RenderSettings.defaultReflectionMode = DefaultReflectionMode.Custom;
        RenderSettings.customReflectionTexture = flatCube;
    }
