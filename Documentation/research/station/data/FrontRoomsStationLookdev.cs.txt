    static readonly float[] CardI = { 4.000f, 3.643f, 5.512f, 3.248f, 3.669f, 2.052f };
    static readonly float[] CardY = { .140f, .116f, .266f, .092f, .118f, .037f };
    static readonly Color SetWhite = new Color(.86f, .91f, 1f);
    static readonly Color[] CardC =
    {
        new Color(.922f, .822f, 1f),   // 0 = the loop mean ("mean card", I 4)
        new Color(.824f, .724f, 1f),   // 1 Front Rooms logo, violet (0-3 s)
        new Color(.951f, .850f, .884f), // 2 sofa set, mauve-pink (3-7 s)
        new Color(.951f, .818f, .672f), // 3 CALL 555-0147, dark navy with yellow type (7-10 s)
        new Color(.802f, .707f, 1f),   // 4 logo + address, violet (10-14 s)

    class V3Cfg
    {
        public string name = "r7";
        public bool r6Exact;
        public float k = 5f, floorFrac = .81f, ceilFrac = 1.4f, blend = .6f;
        public bool lumKeep = true, cookie = true;
        public string Describe()
        {
            if (r6Exact) return "V3 as specified (r6): I = 4 sqrt(Y/0.14) in 2.0-5.6, colour 35 % to white, no cookie";
            return "V3 r7: I = " + F(k) + " sqrt(Y/0.14) in " + F(floorFrac * k) + "-" + F(ceilFrac * k) + ", colour " + Mathf.RoundToInt(blend * 100f) + " % to white"
                + (lumKeep ? ", luminance kept" : "") + (cookie ? ", cosine cookie with up-clip" : "");
        }
    }

    static V3Cfg v3 = new V3Cfg();
    static V3Cfg SpecCfg() { return new V3Cfg { name = "spec", r6Exact = true, cookie = false, lumKeep = false, k = 4f, floorFrac = .5f, blend = .35f }; }

    static Color CardNorm(int c)
    {
        var b = CardC[c];
        return new Color((b.r - .35f * SetWhite.r) / .65f, (b.g - .35f * SetWhite.g) / .65f, (b.b - .35f * SetWhite.b) / .65f);
    }

    static Color BlendWhite(Color n, float b)
    {
        var c = new Color(n.r * (1f - b) + SetWhite.r * b, n.g * (1f - b) + SetWhite.g * b, n.b * (1f - b) + SetWhite.b * b);
        var m = Mathf.Max(c.r, Mathf.Max(c.g, c.b));
        return new Color(c.r / m, c.g / m, c.b / m);
    }

    static float LumLin(Color c) { var l = c.linear; return .2126f * l.r + .7152f * l.g + .0722f * l.b; }

    static void CardLight(int card, V3Cfg cfg, out float intensity, out Color colour)
    {
        if (cfg.r6Exact) { intensity = CardI[card]; colour = CardC[card]; return; }
        var raw = cfg.k * Mathf.Sqrt(CardY[card] / .140f);
        intensity = Mathf.Clamp(raw, cfg.floorFrac * cfg.k, cfg.ceilFrac * cfg.k);
        colour = BlendWhite(CardNorm(card), cfg.blend);
        if (cfg.lumKeep)
        {
            var reference = BlendWhite(CardNorm(0), cfg.blend);
            intensity *= LumLin(reference) / Mathf.Max(LumLin(colour), 1e-4f);
        }
    }

    static V3Cfg ParseV3(string s)
    {
        if (string.IsNullOrEmpty(s)) return new V3Cfg();
        if (s == "spec") return SpecCfg();
        var c = new V3Cfg();
        var parts = s.Split(':');
        if (parts.Length > 1)
            foreach (var kv in parts[1].Split(','))
            {
                var p = kv.Split('=');
                if (p.Length != 2) continue;
                var v = float.Parse(p[1], CultureInfo.InvariantCulture);
                if (p[0] == "k") c.k = v; else if (p[0] == "floor") c.floorFrac = v; else if (p[0] == "blend") c.blend = v;
                else if (p[0] == "cookie") c.cookie = v > .5f; else if (p[0] == "lum") c.lumKeep = v > .5f; else if (p[0] == "ceil") c.ceilFrac = v;
            }
        return c;
    }

    // ------------------------------------------------------------------ batch
    public static void RunBatch()
    {
        var args = Environment.GetCommandLineArgs();
        var plans = "Wp4,Wp4o,Dp4,H";
        var sweepN = 0; var sweepPlans = "Wp4,Dp4";
        for (var i = 0; i < args.Length; i++)
        {
            if (args[i] == "-stationPlans" && i + 1 < args.Length) plans = args[i + 1];
            if (args[i] == "-stationOut" && i + 1 < args.Length) outRoot = args[i + 1];
            if (args[i] == "-stationNoE") ePass = false;
            if (args[i] == "-stationV3" && i + 1 < args.Length) v3 = ParseV3(args[i + 1]);
            if (args[i] == "-stationGridOnly") gridOnly = true;
            if (args[i] == "-stationSweep" && i + 1 < args.Length) sweepN = int.Parse(args[i + 1], CultureInfo.InvariantCulture);
            if (args[i] == "-stationSweepPlans" && i + 1 < args.Length) sweepPlans = args[i + 1];
        }
        if (string.IsNullOrEmpty(outRoot)) outRoot = Path.Combine(Directory.GetParent(Application.dataPath).FullName, "Verification", "station_lookdev");
        Directory.CreateDirectory(outRoot);
        File.WriteAllText(Path.Combine(outRoot, "v3.txt"), v3.Describe() + "\n");
        foreach (var a0 in args) if (a0 == "-stationShadowTest") { ShadowTest(); return; }
        FrontRoomsKitLibrary.ClearCache();
        if (!gridOnly && plans.Length > 0) { try { RelayClipTest(); } catch (Exception e) { Debug.LogError("[StationLookdev] relay clip test failed: " + e); } }
        foreach (var k in plans.Split(','))
        {
            var key = k.Trim();
            if (key.Length == 0) continue;
            var t0 = DateTime.Now;
            try { RunPlan(key, -1, Path.Combine(outRoot, key)); }
            catch (Exception e)
            {
                Debug.LogError("[StationLookdev] plan " + key + " failed: " + e);
                File.AppendAllText(Path.Combine(outRoot, "errors.txt"), key + ": " + e + "\n");
                SafeTeardown();
            }
            File.AppendAllText(Path.Combine(outRoot, "timing.txt"), key + "\t" + (DateTime.Now - t0).TotalSeconds.ToString("0", CultureInfo.InvariantCulture) + " s\n");
        }
        if (sweepN > 0 && !gridOnly)
            foreach (var k in sweepPlans.Split(','))
            {
                var key = k.Trim();
                if (key.Length == 0) continue;
                var t0 = DateTime.Now;
                try { Sweep(key, sweepN); }
                catch (Exception e)
                {
                    Debug.LogError("[StationLookdev] sweep " + key + " failed: " + e);
                    File.AppendAllText(Path.Combine(outRoot, "errors.txt"), "sweep " + key + ": " + e + "\n");
                    SafeTeardown();
                }
                File.AppendAllText(Path.Combine(outRoot, "timing.txt"), "sweep " + key + "\t" + (DateTime.Now - t0).TotalSeconds.ToString("0", CultureInfo.InvariantCulture) + " s\n");
            }
        Debug.Log("[StationLookdev] ALL DONE -> " + outRoot);
    }

    static void SafeTeardown() { try { Teardown(); } catch (Exception) { } realGen = false; sweepMode = false; }

    static void Log(string s) { log.Add(s); Debug.Log("[StationLookdev] " + (plan != null ? plan.key : "-") + " " + s); }
    static string F(float v) { return v.ToString("0.###", CultureInfo.InvariantCulture); }
    static void Mx(string k, float v) { metrics[k] = F(v); }
    static void Mx(string k, int v) { metrics[k] = v.ToString(CultureInfo.InvariantCulture); }
    static void Ms(string k, string v) { metrics[k] = "\"" + (v ?? "").Replace("\\", "/").Replace("\"", "'") + "\""; }
    static void Mv(string k, Vector3 v) { metrics[k] = "[" + F(v.x) + ", " + F(v.y) + ", " + F(v.z) + "]"; }

    static void ResetState()
    {
        kinds.Clear(); mats.Clear(); log.Clear(); virtualProps.Clear(); virtualFoot.Clear(); metrics.Clear();
        lensRs.Clear(); lampLights.Clear(); keyParts.Clear();
        tv = null; tvLight = null; screenR = null; e2DoorHinge = null; tvScale = 1f;
        if (cookieTex != null) { UnityEngine.Object.DestroyImmediate(cookieTex); cookieTex = null; }
    }

    /// <summary>One plan. seed less than 0: pick by the r6 rule (arch widths as archWidth will set them, straight approach) in the forced
    /// generator (all Standard, no other modules); seed 0 or more with realGen: the sweep's seed in the real generator.</summary>
    static void RunPlan(string key, int seed, string dir)
    {
        plan = MakePlan(key);
        outDir = dir;
        Directory.CreateDirectory(outDir);
        ResetState();

        var m = BuildModule(plan);
        if (seed < 0)
        {
            var seeds = CandidateSeeds(m, 300);
            Log("candidate seeds (arch widths fit G1/G2): " + seeds.Count);
            var pick = -1; var fallback = -1; var tried = 0;
            foreach (var s in seeds)
            {
                tried++;
                int openings; bool far, far2; string desc;
                SeedData(m, s, out openings, out far, out far2, out desc);
                if (openings != 2) continue;
                if (fallback < 0) { fallback = s; Log("first two-opening seed " + s + ": " + desc); }
                if (far && far2) { pick = s; Log("picked seed " + s + " after " + tried + ": " + desc); break; }
            }
            if (pick < 0) { pick = fallback >= 0 ? fallback : seeds[0]; Log("WARNING no seed with two openings and a straight approach; using " + pick); Ms("seed_warning", "no straight approach"); }
            seedUsed = pick;
            Mx("seeds_tried", tried);
        }
        else seedUsed = seed;
        NewScene();
        BuildWorld(m, seedUsed);
        Log("seed " + seedUsed + ": perimeter openings " + CountOpenings());
        Mx("seed", seedUsed);
        Ms("plan", plan.title);
        Ms("generator", realGen ? "real (profile defaults: heights, themes, modules as the game)" : "forced (all Standard, no other modules)");
        Ms("v3", v3.Describe());

        DisableZoneKeys();
        CollectLamps();
        CorridorLamps();
        Dress();
        Physics.SyncTransforms();
        ArchCheck(e1, "e1"); ArchCheck(e2, "e2");
        Measure();
        if (gridOnly) { V3Grid(); }
        else if (sweepMode) { SweepShots(); WalkGrid(); }
        else { Shots(); WalkGrid(); ExtraShots(); Counts(); }
        ValidateModule(m);
        WriteMetrics();
        File.WriteAllLines(Path.Combine(outDir, "log.txt"), log.ToArray());
        Teardown();
        Debug.Log("[StationLookdev] " + key + " done -> " + outDir);
    }

    static void NewScene()
    {
        var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
        EditorSceneManager.SetActiveScene(scene);
        root = new GameObject("STATION LOOKDEV " + plan.key).transform;
        root.position = RootWorld;
        FrontRoomsPostStack.Ensure(root);
        SetupReflections();
        SceneFill();
    }

    static void Teardown()
    {
        if (world != null) world.Release();
        if (root != null) UnityEngine.Object.DestroyImmediate(root.gameObject);
        if (previewProfile != null) UnityEngine.Object.DestroyImmediate(previewProfile);
        world = null; root = null; previewProfile = null; mapT = null; extras = null;
        kinds.Clear(); lensRs.Clear(); lampLights.Clear(); keyParts.Clear();
    }

    // ------------------------------------------------------------------ hash (FrontRoomsMap.cs MapHash, internal: replicated)
    static uint Mix(uint h) { unchecked { h ^= h >> 16; h *= 0x7feb352du; h ^= h >> 15; h *= 0x846ca68bu; h ^= h >> 16; return h; } }
    static uint Hash(int seed, int a, int b, int salt)
    {
        unchecked
        {
            var h = Mix((uint)seed ^ 0x9e3779b9u);
            h = Mix(h ^ (uint)a * 0x85ebca6bu);
            h = Mix(h ^ (uint)b * 0xc2b2ae35u);
            h = Mix(h ^ (uint)salt * 0x27d4eb2fu);
            h = Mix(h ^ (uint)0 * 0x165667b1u);
            return h;
        }
    }
    static float Unit(uint h) { return (h >> 8) * (1f / 16777216f); }

    /// <summary>FrontRoomsMapWorld.ArchOpening: width and centre (metres from the edge's west or south end).</summary>
    static void ArchOpening(int seed, GridCoord a, bool alongX, out float width, out float centre)
    {
        var edgeHash = Hash(seed, a.x * 2 + (alongX ? 1 : 0), a.y, 97);
        width = ModuleUnits.ArchMinWidth + ModuleUnits.ArchWidthSpread * Unit(edgeHash);
        var margin = ModuleUnits.ArchCornerMargin + width * .5f;
        centre = Mathf.Lerp(margin, MapGrid.CellSize - margin, Unit(Hash((int)edgeHash, 3, 7, 11)));
    }

    static Opening MakeOpening(int seed, char side, int index, RoomModuleData m)
    {
        var o = new Opening { side = side, index = index };
        var cs = MapGrid.CellSize;
        GridCoord a; bool alongX;
        if (side == 'S') { o.inside = new GridCoord(x0 + index, y0); o.outside = new GridCoord(x0 + index, y0 - 1); a = o.outside; alongX = true; }
        else if (side == 'N') { o.inside = new GridCoord(x0 + index, y0 + m.depth - 1); o.outside = new GridCoord(x0 + index, y0 + m.depth); a = o.inside; alongX = true; }
        else if (side == 'W') { o.inside = new GridCoord(x0, y0 + index); o.outside = new GridCoord(x0 - 1, y0 + index); a = o.outside; alongX = false; }
        else { o.inside = new GridCoord(x0 + m.width - 1, y0 + index); o.outside = new GridCoord(x0 + m.width, y0 + index); a = o.inside; alongX = false; }
        ArchOpening(seed, a, alongX, out o.width, out o.centre);
        if (side == 'S') { o.mid = new Vector3(index * cs + o.centre, 0f, 0f); o.inward = Vector3.forward; o.along = Vector3.right; }
        else if (side == 'N') { o.mid = new Vector3(index * cs + o.centre, 0f, m.depth * cs); o.inward = Vector3.back; o.along = Vector3.right; }
        else if (side == 'W') { o.mid = new Vector3(0f, 0f, index * cs + o.centre); o.inward = Vector3.right; o.along = Vector3.forward; }
        else { o.mid = new Vector3(m.width * cs, 0f, index * cs + o.centre); o.inward = Vector3.left; o.along = Vector3.forward; }
        return o;
    }

    static List<int> CandidateSeeds(RoomModuleData m, int count)
    {
        FrontRoomsModulePreview.Placement(m, out x0, out y0);
        var list = new List<int>();
        for (var seed = 1; seed < 400000 && list.Count < count; seed++)
        {
            var a = MakeOpening(seed, plan.e1Side, plan.e1Index, m);
            var b = MakeOpening(seed, plan.e2Side, plan.e2Index, m);
            if (a.width < 1.6f) continue;                                   // G1: E1 the widest, 1.6-1.8 m (archWidth, M4 iii)
            if (b.width > 1.4f) continue;                                   // G2: E2 1.1-1.4 m
            if (plan.e1Want >= 0f && Mathf.Abs(a.centre - plan.e1Want) > plan.e1Tol) continue;
            list.Add(seed);
        }
        return list;
    }

    /// <summary>The generator's data for a seed (no geometry): perimeter openings and the cells on E1's axis outside.</summary>
    static void SeedData(RoomModuleData m, int seed, out int openings, out bool far, out bool far2, out string desc)
    {
        FrontRoomsModulePreview.Placement(m, out x0, out y0);
        var source = FrontRoomsLevelProfiles.Resolve();
        var prof = source != null ? source : FrontRoomsLevelProfile.Default;
        var g = prof.Generation(seed);
        g.lowShare = 0f; g.standardShare = 1f; g.tallShare = 0f;
        g.officeShare = m.theme == ZoneTheme.Office ? 1f : 0f;
        g.moduleChance = 0f;
        var cache = new FrontRoomsMapCache(g, prof.ModuleData());
        cache.Place(new GridCoord(0, 0), m.Clone(), x0, y0);
        openings = 0;
        var list = new List<string>();
        for (var i = 0; i < m.width; i++)
        {
            if (cache.Edge(new GridCoord(x0 + i, y0), new GridCoord(x0 + i, y0 - 1)) != EdgeKind.Wall) { openings++; list.Add("S" + i); }
            if (cache.Edge(new GridCoord(x0 + i, y0 + m.depth - 1), new GridCoord(x0 + i, y0 + m.depth)) != EdgeKind.Wall) { openings++; list.Add("N" + i); }
        }
        for (var j = 0; j < m.depth; j++)
        {
            if (cache.Edge(new GridCoord(x0, y0 + j), new GridCoord(x0 - 1, y0 + j)) != EdgeKind.Wall) { openings++; list.Add("W" + j); }
            if (cache.Edge(new GridCoord(x0 + m.width - 1, y0 + j), new GridCoord(x0 + m.width, y0 + j)) != EdgeKind.Wall) { openings++; list.Add("E" + j); }
        }
        var o = MakeOpening(seed, plan.e1Side, plan.e1Index, m);
        var c1 = o.outside; var c2 = Step(c1, -1, o); var c3 = Step(c2, -1, o);
        far = cache.Edge(c1, c2) != EdgeKind.Wall;
        far2 = far && cache.Edge(c2, c3) != EdgeKind.Wall;
        desc = "openings " + string.Join(" ", list.ToArray()) + "; approach " + (far ? (far2 ? "3 cells straight" : "2 cells") : "1 cell") + "; E1 " + o.width.ToString("0.00") + " m at " + o.centre.ToString("0.00");
    }

    /// <summary>Rays at 1.0 m through each opening's middle (expect clear) and just past its jambs (expect wall).</summary>
    static void ArchCheck(Opening o, string name)
    {
        var up = Vector3.up * 1.0f;
        var a = W(o.mid - o.inward * .6f + up); var b = W(o.mid + o.inward * .6f + up);
        var mid = Physics.Linecast(a, b);
        var edge = o.along * (o.width * .5f + .12f);
        var l = Physics.Linecast(W(o.mid + edge - o.inward * .6f + up), W(o.mid + edge + o.inward * .6f + up));
        var r = Physics.Linecast(W(o.mid - edge - o.inward * .6f + up), W(o.mid - edge + o.inward * .6f + up));
        var ok = !mid && l && r;
        Ms(name + "_arch_check", (ok ? "OK" : "MISMATCH") + " (middle " + (mid ? "blocked" : "clear") + ", jambs " + (l ? "wall" : "open") + "/" + (r ? "wall" : "open") + ")");
        Log(name + " arch check " + (ok ? "OK" : "MISMATCH"));
    }

    static GridCoord Step(GridCoord c, int dir, Opening o)
    {
        // one cell further along the opening's inward direction (dir +1) or outward (dir -1)
        var dx = Mathf.RoundToInt(o.inward.x) * dir; var dy = Mathf.RoundToInt(o.inward.z) * dir;
        return new GridCoord(c.x + dx, c.y + dy);
    }

    // ------------------------------------------------------------------ module and map
    static RoomModuleData BuildModule(Plan p)
    {
        var m = new RoomModuleData { width = p.w, depth = p.d, height = ZoneHeight.Standard, theme = p.theme, fill = ModuleFill.None, columns = ModuleColumns.None };
        m.Normalize();
        SetEdge(m, p.e1Side, p.e1Index, ModuleEdge.Arch);
        SetEdge(m, p.e2Side, p.e2Index, ModuleEdge.Arch);
        for (var k = 0; k < m.lamps.Length && k < p.lamps.Length; k++)
        {
            var c = p.lamps[k];
            m.lamps[k] = c == 'S' ? ModuleLamp.Steady : c == 'D' ? ModuleLamp.Dim : c == 'O' ? ModuleLamp.Off : ModuleLamp.Auto;
        }
        m.props = p.props.ToArray();
        m.allowRotate = false;
        return m;
    }

    static void SetEdge(RoomModuleData m, char side, int index, ModuleEdge e)
    {
        if (side == 'S') m.south[index] = e; else if (side == 'N') m.north[index] = e; else if (side == 'W') m.west[index] = e; else m.east[index] = e;
    }

    static void BuildWorld(RoomModuleData m, int seed)
    {
        FrontRoomsModulePreview.Placement(m, out x0, out y0);
        ox = x0 * MapGrid.CellSize; oz = y0 * MapGrid.CellSize;
        e1 = MakeOpening(seed, plan.e1Side, plan.e1Index, m);
        e2 = MakeOpening(seed, plan.e2Side, plan.e2Index, m);
        var source = FrontRoomsLevelProfiles.Resolve();
        previewProfile = UnityEngine.Object.Instantiate(source != null ? source : FrontRoomsLevelProfile.Default);
        previewProfile.hideFlags = HideFlags.DontSave;
        var g = previewProfile.Generation(seed);
        if (!realGen)
        {
            g.lowShare = 0f; g.standardShare = 1f; g.tallShare = 0f;
            g.officeShare = m.theme == ZoneTheme.Office ? 1f : 0f;
            g.moduleChance = 0f;
        }
        previewProfile.generation = g;
        previewProfile.buildRadius = 1;
        var go = new GameObject("STATION / map");
        go.SetActive(false);
        go.transform.SetParent(root, false);
        world = go.AddComponent<FrontRoomsMapWorld>();
        world.Profile = previewProfile;
        world.TagModuleProps = true;
        stamped = m.Clone();
        world.PlaceModule(stamped, new GridCoord(0, 0), x0, y0);
        // The capture eye (the map's player for lamp radii) starts just inside E1.
        var spawn = new Vector3(ox + e1.mid.x + e1.inward.x * 1.0f, .05f, oz + e1.mid.z + e1.inward.z * 1.0f);
        world.OverrideSpawn(spawn, Mathf.Atan2(e1.inward.x, e1.inward.z) * Mathf.Rad2Deg);
        go.SetActive(true);
        world.BuildForCapture();
        mapT = world.transform;
        extras = new GameObject("STATION / blockout extras").transform;
        extras.SetParent(root, false);
        if (world.Cache.Edge(e2.inside, e2.outside) == EdgeKind.Door)
        {
            foreach (var t in world.GetComponentsInChildren<Transform>(true))
                if (t.name.StartsWith("Door hinge") && Vector3.Distance(Mod(t.position), e2.mid) < 2.2f) { e2DoorHinge = t; break; }
            Log("E2 is a map door: hinge " + (e2DoorHinge != null ? e2DoorHinge.name : "not found"));
        }
    }

    static int CountOpenings()
    {
        var n = 0; var list = new List<string>();
        for (var i = 0; i < plan.w; i++)
        {
            if (world.Cache.Edge(new GridCoord(x0 + i, y0), new GridCoord(x0 + i, y0 - 1)) != EdgeKind.Wall) { n++; list.Add("S" + i + ":" + world.Cache.Edge(new GridCoord(x0 + i, y0), new GridCoord(x0 + i, y0 - 1))); }
            if (world.Cache.Edge(new GridCoord(x0 + i, y0 + plan.d - 1), new GridCoord(x0 + i, y0 + plan.d)) != EdgeKind.Wall) { n++; list.Add("N" + i + ":" + world.Cache.Edge(new GridCoord(x0 + i, y0 + plan.d - 1), new GridCoord(x0 + i, y0 + plan.d))); }
        }
        for (var j = 0; j < plan.d; j++)
        {
            if (world.Cache.Edge(new GridCoord(x0, y0 + j), new GridCoord(x0 - 1, y0 + j)) != EdgeKind.Wall) { n++; list.Add("W" + j + ":" + world.Cache.Edge(new GridCoord(x0, y0 + j), new GridCoord(x0 - 1, y0 + j))); }
            if (world.Cache.Edge(new GridCoord(x0 + plan.w - 1, y0 + j), new GridCoord(x0 + plan.w, y0 + j)) != EdgeKind.Wall) { n++; list.Add("E" + j + ":" + world.Cache.Edge(new GridCoord(x0 + plan.w - 1, y0 + j), new GridCoord(x0 + plan.w, y0 + j))); }
        }
        Ms("openings", string.Join(" ", list.ToArray()));
        Ms("e2_edge_kind", world.Cache.Edge(e2.inside, e2.outside).ToString());
        return n;
    }

    static Renderer Box(Transform parent, string name, Vector3 centre, Vector3 size, float yaw, Material m, Kind k, bool solid)
    {
        var go = GameObject.CreatePrimitive(PrimitiveType.Cube);
        go.name = name;

    static Renderer Seg(Transform parent, string name, Vector3 a, Vector3 b, float w, Material m, Kind k, bool solid)
    {
        var go = GameObject.CreatePrimitive(PrimitiveType.Cube);
        go.name = name;
        go.transform.SetParent(parent, false);
        var wa = W(a); var wb = W(b);
        var d = wb - wa;
        go.transform.position = (wa + wb) * .5f;
        var up = Mathf.Abs(Vector3.Dot(d.normalized, Vector3.up)) > .98f ? Vector3.forward : Vector3.up;
        go.transform.rotation = Quaternion.LookRotation(d, up);
        go.transform.localScale = new Vector3(w, w, d.magnitude);
        if (!solid) UnityEngine.Object.DestroyImmediate(go.GetComponent<Collider>());
        var r = go.GetComponent<MeshRenderer>();
        r.sharedMaterial = m;
        kinds[r] = k;
        return r;
    }

    static Renderer Cyl(Transform parent, string name, Vector3 baseCentre, float radius, float height, Material m, Kind k, bool solid)

    static float WallFace(Vector3 n)
    {
        // n into the room: the wall's face coordinate along -n
        if (n.x > .5f) return -WF;                      // west wall: face x = 0.08 -> dot(q, -n) = -x = -0.08
        if (n.x < -.5f) return plan.w * 3f - WF;        // east wall: dot(q, -n) = x = 8.92
        if (n.z > .5f) return -WF;                      // south wall
        return plan.d * 3f - WF;                        // north wall
    }

    // ------------------------------------------------------------------ the TV (S01) and its light (V3)
    /// <summary>The cosine-falloff cookie with an up-clip (era critic 12): value = cos(angle off the axis) x a clip that fades the light
    /// out between 10 and 30 deg above the horizontal (the cabinet top shades steep up-light). Spot cookie UV spans the outer cone.</summary>
    static Texture2D BuildCookie(float tiltDownDeg, float outerDeg, bool flipV)
    {
        const int n = 256;
        var t = new Texture2D(n, n, TextureFormat.RGBA32, false, true) { name = "TV light cookie (cosine + up-clip)", wrapMode = TextureWrapMode.Clamp, filterMode = FilterMode.Bilinear };
        var T = Mathf.Tan(outerDeg * .5f * Mathf.Deg2Rad);
        var ct = Mathf.Cos(tiltDownDeg * Mathf.Deg2Rad); var st = Mathf.Sin(tiltDownDeg * Mathf.Deg2Rad);
        var px = new Color[n * n];
        for (var j = 0; j < n; j++)
            for (var i = 0; i < n; i++)
            {
                var u = (i + .5f) / n * 2f - 1f; var v = (j + .5f) / n * 2f - 1f;
                if (flipV) v = -v;
                var d = new Vector3(u * T, v * T, 1f).normalized;
                var worldY = d.y * ct - d.z * st;
                var elev = Mathf.Asin(Mathf.Clamp(worldY, -1f, 1f)) * Mathf.Rad2Deg;
                var clip = 1f - Mathf.SmoothStep(0f, 1f, Mathf.InverseLerp(10f, 30f, elev));
                var val = Mathf.Max(0f, d.z) * clip;
                px[i + j * n] = new Color(val, val, val, val);
            }
        t.SetPixels(px);
        t.Apply(false, false);
        return t;
    }

    static float CookieValue(Vector3 dirLight)
    {
        // the cookie's value for a direction in the light's own frame (analytic twin of BuildCookie)
        if (!v3.cookie) return 1f;
        var d = dirLight.normalized;
        if (d.z <= 0f) return 0f;
        var ct = Mathf.Cos(plan.tvTilt * Mathf.Deg2Rad); var st = Mathf.Sin(plan.tvTilt * Mathf.Deg2Rad);
        var elev = Mathf.Asin(Mathf.Clamp(d.y * ct - d.z * st, -1f, 1f)) * Mathf.Rad2Deg;
        return d.z * (1f - Mathf.SmoothStep(0f, 1f, Mathf.InverseLerp(10f, 30f, elev)));
    }

    /// <summary>Pick the cookie's V orientation by test: the variant that dims the ceiling side more than the floor side is the right one.</summary>
    static void ChooseCookie()
    {
        // a ceiling point 0.7 m in front of the screen and a floor point 1.5 m in front: E with each cookie variant, read back
        var lp = Mod(tvLight.transform.position);
        var ax = (Quaternion.Inverse(mapT.rotation) * tvLight.transform.forward).normalized;
        var axFlat = new Vector3(ax.x, 0f, ax.z).normalized;
        var ceilP = lp + axFlat * .7f; ceilP.y = 2.88f;
        var floorP = lp + axFlat * 1.5f; floorP.y = .01f;
        float[] ce = new float[2]; float[] fe = new float[2];
        for (var v = 0; v < 2; v++)
        {
            if (cookieTex != null) UnityEngine.Object.DestroyImmediate(cookieTex);
            cookieTex = BuildCookie(plan.tvTilt, tvLight.spotAngle, v == 1);
            tvLight.cookie = cookieTex;
            ce[v] = PointE(ceilP, Vector3.down);
            fe[v] = PointE(floorP, Vector3.up);
        }
        tvLight.cookie = null;
        var cn = PointE(ceilP, Vector3.down); var fn = PointE(floorP, Vector3.up);
        cookieFlipV = (ce[1] / Mathf.Max(fe[1], 1e-5f)) < (ce[0] / Mathf.Max(fe[0], 1e-5f));
        if (cookieTex != null) UnityEngine.Object.DestroyImmediate(cookieTex);
        cookieTex = BuildCookie(plan.tvTilt, tvLight.spotAngle, cookieFlipV);
        tvLight.cookie = cookieTex;
        Ms("cookie_check", "ceiling E none/v0/v1 " + F(cn) + "/" + F(ce[0]) + "/" + F(ce[1]) + "; floor E " + F(fn) + "/" + F(fe[0]) + "/" + F(fe[1]) + "; flipV " + cookieFlipV);
        Log("cookie orientation flipV=" + cookieFlipV + " (ceiling none/v0/v1 " + F(cn) + "/" + F(ce[0]) + "/" + F(ce[1]) + ", floor " + F(fn) + "/" + F(fe[0]) + "/" + F(fe[1]) + ")");
    }

    /// <summary>Rendered illuminance at one point (white Lambert E pass of a 64 px camera looking at it from 0.3 m along its normal).</summary>
    static float PointE(Vector3 pModule, Vector3 nModule)
    {
        var go = new GameObject("point E camera");
        var c = go.AddComponent<Camera>();
        c.nearClipPlane = .01f; c.farClipPlane = 2f; c.fieldOfView = 10f;
        c.clearFlags = CameraClearFlags.SolidColor; c.backgroundColor = Color.black;
        var e = W(pModule + nModule * .3f);
        go.transform.SetPositionAndRotation(e, Quaternion.LookRotation(W(pModule) - e, Mathf.Abs(nModule.y) > .9f ? mapT.forward : Vector3.up));
        var prev = cam; cam = c;
        var v = RenderEValue();
        cam = prev;
        UnityEngine.Object.DestroyImmediate(go);
        return v;
    }

    /// <summary>0 = off (picture dead, light off); 1-5 = that card on screen and in the light; 9 = card 1 on screen, the loop-mean light;
    /// 15 = card 1 on screen, card 5's light; 20 + c = card c on screen with the r6 "as specified" light (comparison).</summary>
    static void SetTv(int state) { SetTvCfg(state, v3); }

    static void SetTvCfg(int state, V3Cfg cfg)
    {
        if (state == 0) { screenR.sharedMaterial = screenOffM; tvLight.enabled = false; return; }
        if (state >= 20) { var cc = state - 20; screenR.sharedMaterial = cardM[cc]; tvLight.enabled = true; float ii; Color col; CardLight(cc, SpecCfg(), out ii, out col); tvLight.intensity = ii * tvScale; tvLight.color = col; tvLight.cookie = null; return; }
        var screenCard = state >= 1 && state <= 5 ? state : 1;
        var lightCard = state == 9 ? 0 : state == 15 ? 5 : state;
        screenR.sharedMaterial = cardM[screenCard];
        tvLight.enabled = true;
        float i; Color c;
        CardLight(lightCard, cfg, out i, out c);
        tvLight.intensity = i * tvScale;
        tvLight.color = c;
        tvLight.cookie = cfg.cookie ? cookieTex : null;
    }

    }

    /// <summary>restore == null: hold off every built lamp outside the station whose mode is Stutter, Failing or Dead (returns them);
    /// else give those cells back to the map (Auto).</summary>
    static List<GridCoord> FreezeOtherLamps(List<GridCoord> restore)
    {
        if (restore != null)
        {
            foreach (var c in restore) world.SetLampMode(c, ModuleLamp.Auto);
            world.TickFixturesForTools(0f);
            return restore;
        }
        var list = new List<GridCoord>();
        for (var j = y0 - 7; j < y0 + plan.d + 7; j++)
            for (var i = x0 - 7; i < x0 + plan.w + 7; i++)
            {
                if (i >= x0 && i < x0 + plan.w && j >= y0 && j < y0 + plan.d) continue;
                var c = new GridCoord(i, j);
                if (world.LampLevel(c) == FrontRoomsMapWorld.NoLamp) continue;
                var mode = world.LampModeOf(c);
                if (mode == ModuleLamp.Stutter || mode == ModuleLamp.Failing || mode == ModuleLamp.Dead) { world.SetLampMode(c, ModuleLamp.Off); list.Add(c); }
            }
        world.TickFixturesForTools(0f);
        return list;
    }

    // ------------------------------------------------------------------ measurement (geometry and analytic light)
    static float Etv(Vector3 pModule, Vector3 nModule, float intensity)
    {
        var lp = Mod(tvLight.transform.position);
        var ax = (Quaternion.Inverse(mapT.rotation) * tvLight.transform.forward).normalized;
        var d = pModule - lp; var dist = d.magnitude; var dir = d / dist;
        var f = dist * dist / (tvLight.range * tvLight.range); var s = Mathf.Max(0f, 1f - f * f); var att = s * s / (dist * dist);
        var ci = Mathf.Cos(tvLight.innerSpotAngle * .5f * Mathf.Deg2Rad); var co = Mathf.Cos(tvLight.spotAngle * .5f * Mathf.Deg2Rad);
        var k = 1f / Mathf.Max(ci - co, .001f); var b = -co * k;
        var sp = Mathf.Clamp01(Vector3.Dot(dir, ax) * k + b); sp *= sp;
        var inc = Mathf.Max(0f, Vector3.Dot(-dir, nModule));
        var dl = Quaternion.Inverse(Quaternion.Inverse(mapT.rotation) * tvLight.transform.rotation) * dir;
        return intensity * inc * sp * att * CookieValue(dl);
    }

    /// <summary>Walking cells between two outside cells, around the station (station cells removed), over Open / Arch / Door edges (no glass).</summary>
    static int OutsideWalk(GridCoord a, GridCoord b, int max)
    {
        var seen = new Dictionary<GridCoord, int>();
        var q = new Queue<GridCoord>();
        q.Enqueue(a); seen[a] = 0;
        var dirs = new[] { new GridCoord(1, 0), new GridCoord(-1, 0), new GridCoord(0, 1), new GridCoord(0, -1) };
        while (q.Count > 0)
        {
            var c = q.Dequeue();
            var dc = seen[c];
            if (c.x == b.x && c.y == b.y) return dc;
            if (dc >= max) continue;
            foreach (var d in dirs)
            {
                var nb = new GridCoord(c.x + d.x, c.y + d.y);
                if (nb.x >= x0 && nb.x < x0 + plan.w && nb.y >= y0 && nb.y < y0 + plan.d) continue;
                if (seen.ContainsKey(nb)) continue;
                var k = world.Cache.Edge(c, nb);
                if (k == EdgeKind.Wall || k == EdgeKind.Window) continue;
                seen[nb] = dc + 1;
                q.Enqueue(nb);
            }
        }
        return -1;
    }

    static float Lum(Color c) { return .2126f * c.r + .7152f * c.g + .0722f * c.b; }

    static string TallOnLine(Vector3 a, Vector3 b)
    {
        var hits = new List<string>();
        foreach (var kv in kinds)
        {
            var r = kv.Key;
            if (r == null || !r.enabled) continue;
            if (kv.Value != Kind.Furniture && kv.Value != Kind.SetBody) continue;
            var bb = r.bounds;
            var bmin = Mod(bb.min); var bmax = Mod(bb.max);
            var lo = Mathf.Min(bmin.y, bmax.y); var hi = Mathf.Max(bmin.y, bmax.y);
            if (hi <= 1.1f || lo >= 1.75f) continue;
            var minx = Mathf.Min(bmin.x, bmax.x); var maxx = Mathf.Max(bmin.x, bmax.x);
            var minz = Mathf.Min(bmin.z, bmax.z); var maxz = Mathf.Max(bmin.z, bmax.z);
            if (maxx < 0f || minx > plan.w * 3f || maxz < 0f || minz > plan.d * 3f) continue;
            for (var t = 0f; t <= 1f; t += .02f)
            {
                var p = Vector3.Lerp(a, b, t);
                if (p.x > minx && p.x < maxx && p.z > minz && p.z < maxz) { var nm = r.transform.parent != null ? r.transform.parent.name + "/" + r.name : r.name; if (!hits.Contains(nm)) hits.Add(nm); break; }
            }
        }
        return hits.Count == 0 ? "none" : string.Join("; ", hits.ToArray());
    }

    static string DoorSwingClear(float swingSign)
    {
        var hits = new List<string>();
        var hingeM = Mod(e2DoorHinge.position);
        var width = 0f;
        foreach (var r in e2DoorHinge.GetComponentsInChildren<Renderer>(true)) width = Mathf.Max(width, r.bounds.size.x, r.bounds.size.z);
        foreach (var kv in kinds)
        {
            var r = kv.Key;

        }

    }

    /// <summary>The take's light scale: 1 until KeyTaken (0.25 s), then e^(-(t-0.25)/0.12) to zero by 0.6 s.</summary>
    static float LightCurve(float t)
    {
        if (t < .25f) return 1f;
        if (t >= .6f) return 0f;
        return Mathf.Exp(-(t - .25f) / .12f) * (1f - Mathf.InverseLerp(.5f, .6f, t));
    }

    /// <summary>Frames through the loop wrap with a W1 burst; the light track low-passed at tau 0.12 s (V3), the screen switching at the cut.</summary>
    static void PhotoSequence(string name, int frames, float dt)
    {
        var dir = Path.Combine(outDir, name);
        Directory.CreateDirectory(dir);
        var rt = new RenderTexture(960, 540, 24, RenderTextureFormat.ARGB32) { antiAliasing = 2 };
        var prev = cam.targetTexture; cam.targetTexture = rt;
        const float cut = 1.0f;                         // the wrap at t = 1.0 s (card 5 until then, card 1 after)
        float i5; Color c5; CardLight(5, v3, out i5, out c5);
        float i1; Color c1; CardLight(1, v3, out i1, out c1);
        var dips = new[] { cut - .55f, cut, cut + .55f };
        var cells = StationLampCells();
        var handles = new List<int>();
        var tlog = new StringBuilder();
        for (var f = 0; f < frames; f++)
        {
            var t = f * dt;
            foreach (var dd in dips)
                if (t >= dd && t - dt < dd)
                    foreach (var c in cells) handles.Add(world.SetLampOverride(c, FrontRoomsMapWorld.LampFx.Warn, .55f, .10f, new FrontRoomsMapWorld.LampEnvelope(.10f, .22f)));
            world.TickFixturesForTools(dt);
            var a = t < cut ? 0f : 1f - Mathf.Exp(-(t - cut) / .12f);
            var inten = Mathf.Lerp(i5, i1, a); var col = Color.Lerp(c5, c1, a);
            screenR.sharedMaterial = cardM[t < cut ? 5 : 1];
            tvLight.enabled = true; tvLight.intensity = inten; tvLight.color = col; tvLight.cookie = v3.cookie ? cookieTex : null;
            cam.Render();
            Save(rt, Path.Combine(dir, "f" + f.ToString("000") + ".png"));
            tlog.Append(F(t) + "\t" + F(inten) + "\n");
        }
        File.WriteAllText(Path.Combine(dir, "track.tsv"), tlog.ToString());
        foreach (var h in handles) world.RemoveLampOverride(h);
        for (var k = 0; k < 20; k++) world.TickFixturesForTools(.05f);
        cam.targetTexture = prev; rt.Release();
    }

    static void SwapMaterials(Func<Renderer, Kind, Material> pick, Dictionary<Renderer, Material[]> saved, List<Renderer> hidden, List<Renderer> shown)
    {
        foreach (var kv in kinds)
        {
            var r = kv.Key;
            if (r == null) continue;
            var m = pick(r, kv.Value);
            var isProbe = kv.Value == Kind.Portal1 || kv.Value == Kind.Portal2 || kv.Value == Kind.ProbeIn || kv.Value == Kind.ProbeOut || kv.Value == Kind.ProbeLeak;
            if (!r.enabled && !isProbe) continue;
            if (!r.gameObject.activeInHierarchy) continue;
            saved[r] = r.sharedMaterials;
            if (m == null) { if (r.enabled) { hidden.Add(r); r.enabled = false; } continue; }
            if (!r.enabled) { r.enabled = true; shown.Add(r); }
            var arr = new Material[r.sharedMaterials.Length];
            for (var i = 0; i < arr.Length; i++) arr[i] = m;
            r.sharedMaterials = arr;
        }
    }

    static void Restore(Dictionary<Renderer, Material[]> saved, List<Renderer> hidden, List<Renderer> shown)
    {
        foreach (var kv in saved) if (kv.Key != null) kv.Key.sharedMaterials = kv.Value;
        foreach (var r in hidden) if (r != null) r.enabled = true;
        foreach (var r in shown) if (r != null) r.enabled = false;
    }

    static readonly Dictionary<string, Material> unlitCache = new Dictionary<string, Material>();

    /// <summary>The same white pass on a tiny camera, returning the mean luminance (ChooseCookie's point test).</summary>
    static float RenderEValue()
    {
        var white = EWhite();
        var saved = new Dictionary<Renderer, Material[]>(); var hidden = new List<Renderer>(); var shown = new List<Renderer>();
        SwapMaterials((r, k) => k == Kind.Lens || k == Kind.Screen || k == Kind.Portal1 || k == Kind.Portal2 || k == Kind.ProbeIn || k == Kind.ProbeOut || k == Kind.ProbeLeak || k == Kind.Jewel ? null : white, saved, hidden, shown);
        var fog = RenderSettings.fog; RenderSettings.fog = false;
        var data = cam.GetUniversalAdditionalCameraData();
        data.renderPostProcessing = false;
        var rt = new RenderTexture(32, 32, 24, RenderTextureFormat.ARGBFloat) { antiAliasing = 1 };
        cam.targetTexture = rt; cam.allowMSAA = false;
        cam.Render();
        var prevActive = RenderTexture.active;
        RenderTexture.active = rt;
        var tex = new Texture2D(32, 32, TextureFormat.RGBAFloat, false, true);
        tex.ReadPixels(new Rect(0, 0, 32, 32), 0, 0);
        tex.Apply();
        RenderTexture.active = prevActive;
        var px = tex.GetPixels(); var s = 0f;
        foreach (var p in px) s += .2126f * p.r + .7152f * p.g + .0722f * p.b;
        UnityEngine.Object.DestroyImmediate(tex);
        cam.targetTexture = null; rt.Release();
        RenderSettings.fog = fog;
        Restore(saved, hidden, shown);

    static readonly float[] Radii = { .30f, .35f, .40f, .45f, .50f, .55f, .60f, .65f, .70f };

    static void WalkGrid()
    {
        Physics.SyncTransforms();
        const float step = .05f;
        float minX = -3f, minZ = -3f, maxX = plan.w * 3f + 3f, maxZ = plan.d * 3f + 3f;
        var nx = Mathf.RoundToInt((maxX - minX) / step); var nz = Mathf.RoundToInt((maxZ - minZ) / step);
        var player = new byte[nx * nz]; var relay = new byte[nx * nz];
        var mask = ~0;
        for (var j = 0; j < nz; j++)
            for (var i = 0; i < nx; i++)
            {
                var x = minX + (i + .5f) * step; var z = minZ + (j + .5f) * step;
                byte lv = 0;
                for (var k = 0; k < Radii.Length; k++)
                {
                    var r = Radii[k];
                    if (Physics.CheckCapsule(W(x, .31f + r, z), W(x, 1.74f - r, z), r, mask, QueryTriggerInteraction.Ignore)) break;
                    lv = (byte)(k + 1);
                }
                player[i + j * nx] = lv;
                relay[i + j * nx] = (byte)(Physics.CheckCapsule(W(x, .70f, z), W(x, 1.65f, z), .30f, mask, QueryTriggerInteraction.Ignore) ? 0 : 1);
            }
        File.WriteAllBytes(Path.Combine(outDir, "walk_player.u8"), player);
        File.WriteAllBytes(Path.Combine(outDir, "walk_relay.u8"), relay);
        var sb = new StringBuilder();
        sb.Append("{\"step\": " + F(step) + ", \"minX\": " + F(minX) + ", \"minZ\": " + F(minZ) + ", \"nx\": " + nx + ", \"nz\": " + nz + ", \"radii\": [");
        sb.Append(string.Join(", ", Radii.Select(r => F(r)).ToArray()));
        sb.Append("], \"player\": \"capsule y 0.31-1.74, radius per level\", \"relay\": \"capsule r 0.30, y 0.40-1.95\"}");
        File.WriteAllText(Path.Combine(outDir, "walk_meta.json"), sb.ToString());
    }

    static Cubemap flatCube;
    static void SetupReflections()
    {
        flatCube = new Cubemap(16, TextureFormat.RGBAHalf, false) { name = "Station_FlatAmbientCube" };
        var sky = FrontRoomsLook.AmbientSky.linear; var eq = FrontRoomsLook.AmbientEquator.linear; var gr = FrontRoomsLook.AmbientGround.linear;
        foreach (CubemapFace f in Enum.GetValues(typeof(CubemapFace)))
        {
            if (f == CubemapFace.Unknown) continue;
            var col = f == CubemapFace.PositiveY ? sky : f == CubemapFace.NegativeY ? gr : eq;
            var px = new Color[16 * 16];
            for (var i = 0; i < px.Length; i++)
            {
                var row = i / 16;
                var c = col;
                if (f != CubemapFace.PositiveY && f != CubemapFace.NegativeY) c = row < 8 ? Color.Lerp(sky, eq, row / 8f) : Color.Lerp(eq, gr, (row - 8) / 8f);
                px[i] = c;
            }
            flatCube.SetPixels(px, f);
        }
        flatCube.Apply();
        RenderSettings.defaultReflectionMode = DefaultReflectionMode.Custom;
        RenderSettings.customReflectionTexture = flatCube;
    }

    // ------------------------------------------------------------------ the seed sweep (gameplay critic 2 and 3)
    /// <summary>Random seeds in the real generator that pass the M4 rules proposed in 50_critic_fixes (i: E1's outside cell and the next
    /// cell out on its axis open, same height; ii: every other perimeter edge is a wall with a same-height neighbour, no map reopening;
    /// E2 an arch or a map door, never glass). Widths as hashed (archWidth, rule iii, does not exist yet). Up to n accepted seeds plus one
    /// where E2 is a map door.</summary>
    static void Sweep(string key, int n)
    {
        plan = MakePlan(key);
        var m = BuildModule(plan);
        FrontRoomsModulePreview.Placement(m, out x0, out y0);
        var source = FrontRoomsLevelProfiles.Resolve();
        var prof = source != null ? source : FrontRoomsLevelProfile.Default;
        var reasons = new SortedDictionary<string, int>();
        Action<string> bump = r => { int c; reasons.TryGetValue(r, out c); reasons[r] = c + 1; };
        var accepted = new List<int>(); var doorSeed = -1; var scanned = 0;
        var widths = new StringBuilder();
        for (var seed = 1; seed < 60000 && (accepted.Count < n || doorSeed < 0); seed++)
        {
            scanned++;
            var g = prof.Generation(seed);
            var cache = new FrontRoomsMapCache(g, prof.ModuleData());
            var fits = true;
            for (var j = 0; j < m.depth && fits; j++)
                for (var i = 0; i < m.width && fits; i++)
                {
                    var z = cache.ZoneOf(new GridCoord(x0 + i, y0 + j));
                    if (z.height != ZoneHeight.Standard || z.theme != m.theme) fits = false;
                }
            if (!fits) { bump("footprint not one Standard " + m.theme + " zone"); continue; }
            cache.Place(new GridCoord(0, 0), m.Clone(), x0, y0);
            var a = MakeOpening(seed, plan.e1Side, plan.e1Index, m);
            var b = MakeOpening(seed, plan.e2Side, plan.e2Index, m);
            if (cache.Edge(a.inside, a.outside) != EdgeKind.Arch) { bump("E1 not the module's arch (outside cell another height)"); continue; }
            var c1 = a.outside; var c2 = Step(c1, -1, a);
            var k12 = cache.Edge(c1, c2);
            if (k12 != EdgeKind.Open && k12 != EdgeKind.Arch) { bump("rule i: E1's next cell out walled, a door or glass"); continue; }
            if (cache.ZoneOf(c2).height != ZoneHeight.Standard) { bump("rule i: E1's next cell out another height"); continue; }
            var e2k = cache.Edge(b.inside, b.outside);
            if (e2k == EdgeKind.Window) { bump("E2 glass (Tall neighbour): no glass on a station wall"); continue; }
            if (e2k == EdgeKind.Wall) { bump("E2 walled (height border, no door rolled)"); continue; }
            var bad = ""; var glass = false;
            for (var i = 0; i < m.width; i++)
            {
                CheckEdge(cache, new GridCoord(x0 + i, y0), new GridCoord(x0 + i, y0 - 1), a, b, ref bad, ref glass);
                CheckEdge(cache, new GridCoord(x0 + i, y0 + m.depth - 1), new GridCoord(x0 + i, y0 + m.depth), a, b, ref bad, ref glass);
            }
            for (var j = 0; j < m.depth; j++)
            {
                CheckEdge(cache, new GridCoord(x0, y0 + j), new GridCoord(x0 - 1, y0 + j), a, b, ref bad, ref glass);
                CheckEdge(cache, new GridCoord(x0 + m.width - 1, y0 + j), new GridCoord(x0 + m.width, y0 + j), a, b, ref bad, ref glass);
            }
            if (glass) { bump("rule ii: glass on a station wall (Tall neighbour)"); continue; }
            if (bad.Length > 0) { bump("rule ii: " + bad); continue; }
            var isDoor = e2k == EdgeKind.Door;
            if (isDoor && doorSeed < 0) { doorSeed = seed; bump("accepted (E2 a map door)"); widths.Append(seed + "\tdoor\t" + F(a.width) + "\t" + F(b.width) + "\n"); continue; }
            if (isDoor) { bump("accepted, E2 a map door (beyond the first)"); continue; }
            if (accepted.Count < n) { accepted.Add(seed); bump("accepted"); widths.Append(seed + "\tarch\t" + F(a.width) + "\t" + F(b.width) + "\n"); }
        }
        var sdir = Path.Combine(outRoot, "sweep_" + key);
        Directory.CreateDirectory(sdir);
        var rs = new StringBuilder("{\n  \"scanned\": " + scanned + ",\n  \"accepted\": [" + string.Join(", ", accepted.Select(s => s.ToString()).ToArray()) + "],\n  \"door_seed\": " + doorSeed + ",\n  \"reasons\": {");
        var firstR = true;
        foreach (var kv in reasons) { rs.Append((firstR ? "\n" : ",\n") + "    \"" + kv.Key.Replace("\"", "'") + "\": " + kv.Value); firstR = false; }
        rs.Append("\n  }\n}\n");
        File.WriteAllText(Path.Combine(sdir, "sweep.json"), rs.ToString());
        File.WriteAllText(Path.Combine(sdir, "widths.tsv"), "seed\tE2\tE1_w\tE2_w\n" + widths.ToString());
        var list = new List<int>(accepted);
        if (doorSeed >= 0) list.Add(doorSeed);
        realGen = true; sweepMode = true;
        foreach (var s in list)
        {
            var t0 = DateTime.Now;
            try { RunPlan(key, s, Path.Combine(sdir, "s" + s)); }
            catch (Exception e)
            {
                Debug.LogError("[StationLookdev] sweep " + key + " seed " + s + " failed: " + e);
                File.AppendAllText(Path.Combine(sdir, "errors.txt"), s + ": " + e + "\n");
                try { Teardown(); } catch (Exception) { }
            }
            File.AppendAllText(Path.Combine(sdir, "timing.txt"), s + "\t" + (DateTime.Now - t0).TotalSeconds.ToString("0.0", CultureInfo.InvariantCulture) + " s\n");
        }
        realGen = false; sweepMode = false;
    }

    static void CheckEdge(FrontRoomsMapCache cache, GridCoord inside, GridCoord outside, Opening a, Opening b, ref string bad, ref bool glass)
    {
        var isA = inside.x == a.inside.x && inside.y == a.inside.y && outside.x == a.outside.x && outside.y == a.outside.y;
        var isB = inside.x == b.inside.x && inside.y == b.inside.y && outside.x == b.outside.x && outside.y == b.outside.y;
        if (isA || isB) return;
        var k = cache.Edge(inside, outside);
        if (k == EdgeKind.Window) glass = true;
        var hOut = cache.ZoneOf(outside).height;
        if (hOut != ZoneHeight.Standard) { if (bad.Length == 0) bad = "a station wall on a height border (" + hOut + " neighbour)"; return; }
        if (k != EdgeKind.Wall && bad.Length == 0) bad = "the map reopened a station wall (" + k + ")";
    }

    static ModuleProp Against(string kit, float along, FrontRoomsModuleEditing.WallSide side, int w, int d, float y = 0f)
    {
        var m = new RoomModuleData { width = w, depth = d };
        m.Normalize();
        return FrontRoomsModuleEditing.AgainstRoomWall(kit, along, side, m, y);
    }
