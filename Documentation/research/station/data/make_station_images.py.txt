LUM = np.array([.2126, .7152, .0722], np.float32)


def P(plan, f):
    return os.path.join(R, plan, f)


def met(plan):
    return json.load(open(P(plan, 'metrics.json')))


def lin(c):
    return np.where(c <= 0.04045, c / 12.92, ((c + 0.055) / 1.055) ** 2.4)


def enc(c):
    c = np.clip(c, 0, 1)
    return np.where(c <= 0.0031308, c * 12.92, 1.055 * np.power(c, 1 / 2.4) - 0.055)


def rgb(p):
    return np.asarray(Image.open(p).convert('RGB')).astype(np.float32) / 255.0


def Emap(p):
    a = np.fromfile(p, dtype='<f4').reshape(540, 960, 4)[::-1, :, :3]
    return a @ LUM


def idmask(p, c):
    q = (rgb(p) > .5).astype(np.int32)
    return (q[..., 0] == c[0]) & (q[..., 1] == c[1]) & (q[..., 2] == c[2])


def ramp(t):
    t = np.clip(t, 0, 1)
    stops = np.array([[0.05, 0.05, 0.25], [0.10, 0.30, 0.80], [0.10, 0.75, 0.70], [0.95, 0.85, 0.15], [1.0, 0.95, 0.85]])
    x = t * (len(stops) - 1)
    i = np.minimum(np.floor(x).astype(int), len(stops) - 2)
    f = (x - i)[..., None]
    return stops[i] * (1 - f) + stops[i + 1] * f


def pct(v):
    return '%d %%' % round(100 * v) if v is not None else '-'


def save(img, name, title, sentence, vl, status):
    out = S.frame(img, title, sentence, [vl, DATE, 'GAME VISUAL', REL + name, META_TAIL], status)
    path = os.path.join(OUT, name)
    S.save(out, path)
    print('wrote', path, out.size)
    return path


def crop_box(mask, aspect=16 / 9., grow=1.7, minh=200):
    ys, xs = np.nonzero(mask)
    if len(ys) == 0:
        return (0, 0, mask.shape[1], mask.shape[0])
    cy, cx = (ys.min() + ys.max()) / 2, (xs.min() + xs.max()) / 2
    h = max((ys.max() - ys.min()) * grow, minh)
    w = h * aspect
    x0, y0 = int(max(0, cx - w / 2)), int(max(0, cy - h / 2))
    x1, y1 = int(min(mask.shape[1], x0 + w)), int(min(mask.shape[0], y0 + h))
    return (x0, y0, x1, y1)

# ------------------------------------------------------------------ check 1a: the junction (VL355)
tiles = []
for p, name in PLANS:
    r = M['plans'][p]
    portal = idmask(P(p, 'approach_far_id.png'), (0, 0, 1))
    box = crop_box(portal)
    imgs = [rgb(P(p, 'approach_far_' + c + '.png')) for c in ('c1', 'c2', 'c5')]
    ys = [lin(i) @ LUM for i in imgs]
    hi, lo = np.maximum(np.maximum(ys[0], ys[1]), ys[2]), np.minimum(np.minimum(ys[0], ys[1]), ys[2])
    mv = ((hi - lo) / np.maximum(hi, 1e-4) >= .20) & portal & (hi > .01)
    base = imgs[1]
    grey = np.repeat((lin(base) @ LUM)[..., None], 3, -1)
    ov = enc(grey * .8)
    ov[mv] = np.array([1.0, .85, .1])
    ov[portal & ~mv] = ov[portal & ~mv] * .7 + np.array([.15, .25, .55]) * .3
    a = Image.fromarray((base * 255).astype(np.uint8)).crop(box)
    b = Image.fromarray((ov * 255).astype(np.uint8)).crop(box)
    tiles.append((a, name, 'E1 from 4.5 m, mean-bright card'))
    tiles.append((b, 'moves', 'screen %d px, %s of E1 swings >= 20 %%' % (r['c1a_far_screen_px_through_E1'], pct(r['c1a_far_moving_frac_of_E1']))))
ev = S.grid(tiles, cols=4, width=1920)
save(ev, 'station_c1_junction.jpg', 'Light leaving the room, junction',
     'From 4.5 m out the set adds 0.000 at E1; the screen shows through E1 only when it sits on E1\'s axis (rev 3, H); at 19-23 deg the jamb hides it.',
     'VL355', 'PARTIAL')

# ------------------------------------------------------------------ check 5: walk grids (VL357)
G = np.load(os.path.splitext(MJ)[0] + '_grids.npz')
tiles = []
LV = [(1.0, .15, .15), (1.0, .5, .1), (1.0, .8, .1), (.9, .9, .2), (.55, .85, .25), (.25, .75, .35), (.15, .65, .55), (.15, .5, .75), (.2, .35, .85)]
for p, name in PLANS:
    m, r = met(p), M['plans'][p]
    meta = json.load(open(P(p, 'walk_meta.json')))
    pl = G[p + '_pl']
    base = rgb(P(p, 'plan.png'))
    s_ = m['plan_ortho_size']
    k = 540.0 / s_
    W_, D_ = m['room_w_m'], m['room_d_m']
    H_, Wd = base.shape[0], base.shape[1]
    yy, xx = np.mgrid[0:H_, 0:Wd]
    X = W_ / 2 + (xx + .5 - Wd / 2) / k
    Z = D_ / 2 - (yy + .5 - H_ / 2) / k
    gi = np.floor((X - meta['minX']) / meta['step']).astype(int)
    gj = np.floor((Z - meta['minZ']) / meta['step']).astype(int)
    ok = (gi >= 0) & (gi < meta['nx']) & (gj >= 0) & (gj < meta['nz']) & (X > .08) & (X < W_ - .08) & (Z > .08) & (Z < D_ - .08)
    lvl = np.zeros(base.shape[:2], int)
    lvl[ok] = pl[gj[ok], gi[ok]]
    ov = base.copy() * .45
    for li in range(1, 10):
        sel = ok & (lvl == li)
        ov[sel] = ov[sel] * .4 + np.array(LV[li - 1]) * .6
    sel0 = ok & (lvl == 0)
    ov[sel0] = ov[sel0] * .35
    tiles.append((Image.fromarray((np.clip(ov, 0, 1) * 255).astype(np.uint8)), name,
                  'E1-key %.1f, key-E2 %.1f, strip %s' % (r['c5_route_in_room_e1_take_clear_width_m'], r['c5_route_in_room_take_e2_clear_width_m'],
                                                         ('%.1f m' % r['c5_staff_strip_through_clear_width_m']) if r['c5_staff_strip_through_clear_width_m'] is not None else 'none')))
ev = S.grid(tiles, cols=4, width=1920)

     'VL357', 'PARTIAL')
