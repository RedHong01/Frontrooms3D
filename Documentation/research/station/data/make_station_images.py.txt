def lin(c):
    return np.where(c <= 0.04045, c / 12.92, ((c + 0.055) / 1.055) ** 2.4)


def enc(c):
    c = np.clip(c, 0, 1)
    return np.where(c <= 0.0031308, c * 12.92, 1.055 * np.power(c, 1 / 2.4) - 0.055)


def rgb(p):
    return np.asarray(Image.open(p).convert('RGB')).astype(np.float32) / 255.0


def Emap(p):
    a = np.fromfile(p, dtype='<f4').reshape(540, 960, 4)[::-1, :, :3]
    return a @ LUM


def idmask(p, c):
    q = (rgb(p) > .5).astype(np.int32)
    return (q[..., 0] == c[0]) & (q[..., 1] == c[1]) & (q[..., 2] == c[2])


def ramp(t):
    t = np.clip(t, 0, 1)
    stops = np.array([[0.05, 0.05, 0.25], [0.10, 0.30, 0.80], [0.10, 0.75, 0.70], [0.95, 0.85, 0.15], [1.0, 0.95, 0.85]])
    x = t * (len(stops) - 1)
    i = np.minimum(np.floor(x).astype(int), len(stops) - 2)
    f = (x - i)[..., None]
    return stops[i] * (1 - f) + stops[i + 1] * f


def pct(v):
    return '%d %%' % round(100 * v) if v is not None else '-'


def save(img, name, title, sentence, vl, status, tail=None):
    out = S.frame(img, title, sentence, [vl, DATE, 'GAME VISUAL', REL + name, tail or TAIL], status)
    if out.width > 1920:
        out = out.resize((1920, round(out.height * 1920 / out.width)), Image.LANCZOS)
    path = os.path.join(OUT, name)
    S.save(out, path, quality=85)
    print('wrote', path, out.size)
    return path


def crop_box(mask, aspect=16 / 9., grow=1.7, minh=200):
    ys, xs = np.nonzero(mask)
    if len(ys) == 0:
        return (0, 0, mask.shape[1], mask.shape[0])
    cy, cx = (ys.min() + ys.max()) / 2, (xs.min() + xs.max()) / 2
    h = max((ys.max() - ys.min()) * grow, minh)
    w = h * aspect
    x0, y0 = int(max(0, cx - w / 2)), int(max(0, cy - h / 2))
    x1, y1 = int(min(mask.shape[1], x0 + w)), int(min(mask.shape[0], y0 + h))
    return (x0, y0, x1, y1)
