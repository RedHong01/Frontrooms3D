def srgb(v):
    v = min(max(v, 0.0), 1.0)
    return v * 12.92 if v <= 0.0031308 else 1.055 * v ** (1 / 2.4) - 0.055


def hexc(c):
    return "#" + "".join("%02x" % int(round(srgb(x) * 255)) for x in c)


em = sw["ExitSign_Diffuser"]["emission"]
glow_hex = hexc([x / max(em) for x in em])

def mm(v):
    v = round(v * 1000.0, 1)
    return int(v) if v == int(v) else v


def mat(slot, label, note=None, **kw):
    s = sw[slot]
    src = "Resources/Surfaces/%s.mat: " % slot
    if s["baseMap"]:
        src += "mean of %s x _BaseColor %s (tint sRGB->linear, product -> sRGB)" % (os.path.basename(s["baseMap"]), s["tint"])
    else:
        src += "no base map x _BaseColor %s" % s["tint"]
    src += "; metallic %g, smoothness = the _S mask's R (capped at 0.60) x _Smoothness %g" % (s["_Metallic"], s["_Smoothness"])
    if note:
        src += "; " + note
    d = dict(name=slot, slot=slot, label=label, hex=s["A_srgbMean_x_tintLinear"], source=src)
    d.update(kw)
    return d

def materials(mount):
    chrome_label = {"hanging": "Chrome (stems, collars, nuts, screws)", "flush": "Chrome (nuts, screws)"}[mount]
    m = [mat("ExitSign_Housing", "White enamel (pan, canopy)"), mat("ExitSign_Face", "Stencil plate, white enamel (the unused knockout stays scored)"),
         mat("ExitSign_Diffuser", "Red acrylic diffuser (seen through the letters and the punched chevron)"), GLOW]
    c = dict(name="Prop_Chrome", slot="Prop_Chrome", label=chrome_label, hex=sw["Prop_Chrome"]["A_srgbMean_x_tintLinear"], source=CHROME_NOTE)
    assert c["hex"] == "#d8d8d8", c
    m.append(c)
    return m

def side(kit):
    return json.load(open(os.path.join(MODELS, kit + ".json")))


def images(kit):
    return {v: "png/%s_%s.png" % (kit, v) for v in ("top", "front", "side", "persp")}


def slots(kit):
    return {v: "img:%s_%s" % (kit, v) for v in ("top", "front", "side", "persp")}

             "hold the stencil holes; chevrons/10_spec §4.1). LOD0 = the closed parent + 284 (the punched chevron, both faces); the plate is kept "

def tris(sc, parent):
    return dict(budget=sc["lodBudget"], lod0=sc["triangles"], lod1=sc["trianglesLod1"], lodDistances_m=sc["lodDistances"],
