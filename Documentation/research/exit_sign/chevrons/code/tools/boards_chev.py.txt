"""Boards for the exit-sign-chevrons build (VL333-VL336). numpy + Pillow, W/venv/bin/python.

  python boards_chev.py <work dir W/chev_work> <out images dir>

Images: graded frames (_g, the full game post) unless noted; crops are native pixels, enlarged with NEAREST where a
cell is larger than the crop (so the 9 / 12 m cells show the real pixel count).
"""

FONTB = os.path.expanduser("~/Library/Fonts/IBMPlexMono-Bold.otf")
S = json.load(open(os.path.join(W, "summary.json")))
R2 = {r["shot"]: r for r in json.load(open(os.path.join(W, "cap_r2_measure.json")))}

def font(size, bold=False):
    try:
        return ImageFont.truetype(FONTB if bold else FONT, size)
    except OSError:
        return ImageFont.load_default()


def img(cap, group, shot, kind="g"):
    p = os.path.join(W, cap, group, "%s_%s.png" % (shot, kind))
    if not os.path.exists(p):
        p = os.path.join(W, cap, group, "%s_m.png" % shot)
    return Image.open(p).convert("RGB")


def idmask(cap, idname):
    a = np.asarray(Image.open(os.path.join(W, cap, "id", idname + "_id.png")).convert("RGB"))
    r, g, b = a[..., 0] > 127, a[..., 1] > 127, a[..., 2] > 127
    return (r & ~g) | (~r & g & ~b) | (~r & ~g & b)


def ids_of(cap):
    out = {}
    for line in open(os.path.join(W, cap, "shots.txt")):
        p = [x.strip() for x in line.split("|")]
        if len(p) > 2 and p[2].startswith("id "):
            out[p[1]] = p[2][3:]
    return out


IDS = {"cap_r1": ids_of("cap_r1"), "cap_r2": ids_of("cap_r2")}


def sign_crop(cap, group, shot, margin=0.25, aspect=16 / 9.0, kind="g"):
    im = img(cap, group, shot, kind)
    m = idmask(cap, IDS[cap].get(shot, shot))
    ys, xs = np.nonzero(m)
    y0, y1, x0, x1 = ys.min(), ys.max(), xs.min(), xs.max()
    # the sign's face band only (drop the stems above): keep the lower 0.6 of a tall box
    h, w = y1 - y0, x1 - x0
    if h > 0.9 * w:
        y0 = int(y1 - 0.75 * w)
        h = y1 - y0
    cx, cy = (x0 + x1) / 2, (y0 + y1) / 2
    w2, h2 = w * (1 + margin) / 2, h * (1 + margin) / 2
    if w2 / max(h2, 1) < aspect:
        w2 = h2 * aspect
    else:
        h2 = w2 / aspect
    box = (int(cx - w2), int(cy - h2), int(cx + w2), int(cy + h2))
    return im.crop(box)


def label(d, xy, lines, size=19, fill=(255, 255, 255), bg=(12, 12, 12)):
    x, y = xy
    f = font(size)
    for t in lines:
        bb = d.textbbox((x, y), t, font=f)
        d.rectangle((bb[0] - 6, bb[1] - 4, bb[2] + 6, bb[3] + 4), fill=bg)
        d.text((x, y), t, font=f, fill=fill)
        y = bb[3] + 9


def fit(im, w, h):
    resample = Image.NEAREST if im.width < w * 0.8 else Image.LANCZOS
    return im.resize((w, h), resample)


def board(cells, cols, cw, ch, path, title, foot=None, cap=0):
    """cap > 0: captions in a band of that height under each image (dark text), else labels over the image."""
    rows = (len(cells) + cols - 1) // cols
    top = 60
    bottom = 40 if foot else 0
    rh = ch + cap
    out = Image.new("RGB", (cols * cw, top + rows * rh + bottom), (238, 236, 230))
    d = ImageDraw.Draw(out)
    d.text((16, 14), title, font=font(26, True), fill=(10, 10, 10))
    for i, (im, lines) in enumerate(cells):
        x, y = (i % cols) * cw, top + (i // cols) * rh
        if im is not None:
            out.paste(fit(im, cw - 4, ch - 4), (x + 2, y + 2))
        if cap:
            f = font(15)
            yy = y + ch + 2
            for t in lines:
                d.text((x + 8, yy), t, font=f, fill=(20, 20, 20))
                yy += 19
        else:
            label(d, (x + 10, y + 8), lines, size=19 if cw >= 600 else 15)
    if foot:
        d.text((16, top + rows * rh + 8), foot, font=font(17), fill=(40, 40, 40))
    out.save(path, quality=85)
    print("wrote", path, out.size)


def hs(r, n):
    h = r["holes"].get(n, {})
    return "hue %.1f sat %.2f" % tuple(h["body_hsv"][:2]) if h.get("body_hsv") else "-"
