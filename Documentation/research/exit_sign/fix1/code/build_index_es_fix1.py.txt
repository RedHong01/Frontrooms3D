# swatches by the K-sheet rule (swatch.py, read from the render clone = main's materials)
sw = {}
out = subprocess.run([sys.executable, os.path.join(TOOLS, "swatch.py"), CLONE, "ExitSign_Housing", "ExitSign_Face",
                      "ExitSign_Diffuser", "ExitSign_Diffuser_Dead", "Prop_Chrome"], capture_output=True, text=True, check=True).stdout
for line in out.splitlines():
    d = json.loads(line); sw[d["name"]] = d

def srgb(v):
    v = min(max(v, 0.0), 1.0)
    return v * 12.92 if v <= 0.0031308 else 1.055 * v ** (1 / 2.4) - 0.055
def hexc(c): return "#" + "".join("%02x" % int(round(srgb(x) * 255)) for x in c)
em = sw["ExitSign_Diffuser"]["emission"]
glow_hex = hexc([x / max(em) for x in em])

SCALEBAR = {125: (2.0, "2 m"), 250: (1.0, "1 m"), 500: (0.5, "50 cm"), 1000: (0.2, "20 cm"),
            2000: (0.1, "10 cm"), 4000: (0.05, "5 cm"), 8000: (0.025, "25 mm")}

def mm(v):
    v = round(v * 1000.0, 1)
    return int(v) if v == int(v) else v

def mat(slot, label, note=None, **kw):
    s = sw[slot]
    src = "Resources/Surfaces/%s.mat: " % slot
    if s["baseMap"]:
        src += "mean of %s x _BaseColor %s (tint sRGB->linear, product -> sRGB)" % (os.path.basename(s["baseMap"]), s["tint"])
    else:
        src += "no base map x _BaseColor %s" % s["tint"]
    src += "; metallic %g, smoothness = the _S mask's R (capped at 0.60) x _Smoothness %g" % (s["_Metallic"], s["_Smoothness"])
    if note: src += "; " + note
    d = dict(name=slot, slot=slot, label=label, hex=s["A_srgbMean_x_tintLinear"], source=src)
    d.update(kw)
    return d

GLOW = dict(name="ExitSign_Diffuser (glow)", slot="ExitSign_Diffuser", label="Lit letters (lamp glow)", hex=glow_hex,
            emissive=True, renderedHex="#e40f08",
            source="ExitSign_Diffuser.mat _EmissionColor (3.2, 0.013, 0.006) linear, shown at its peak (R = 1); the shader multiplies it by the lamp field "
                   "(ExitSign_Diffuser_F: letter means E/X/I/T 0.51 / 0.59 / 1 / 0.58, LIT). renderedHex = mean of the lit letters in Kit_ExitSign_front.png "
                   "(studio, no HDR, no post; median #f40d06). In game the letters pass bloom and ACES: exit_sign_fix1 measured the graded letter body at hue 6.9–8.0°, "
                   "saturation 0.80–0.83 (lit and dark rooms, 0.3–12 m; research/exit_sign/60_fix1.md §3)")
CHROME_NOTE = "existing K-sheet swatch (PROP KIT 2324:852), same rule"

def materials(dead, mount):
    chrome_label = {"wall": "Chrome (screws)", "end": "Chrome (screws)", "hanging": "Chrome (stems, collars, nuts, screws)",
                    "flush": "Chrome (nuts, screws)"}[mount]
    housing_label = "White enamel (pan)" if mount in ("wall", "end") else "White enamel (pan, canopy)"
    m = [mat("ExitSign_Housing", housing_label), mat("ExitSign_Face", "Stencil plate, white enamel")]
    if dead:
        m.append(mat("ExitSign_Diffuser_Dead", "Red acrylic, unlit", note="no emission: the letters read as dark red holes (rendered mean #3b0201)"))
    else:
        m.append(mat("ExitSign_Diffuser", "Red acrylic diffuser", note="ExitSign_Diffuser_Dead has the same albedo, tint, smoothness and metallic"))
        m.append(GLOW)
    c = dict(name="Prop_Chrome", slot="Prop_Chrome", label=chrome_label, hex=sw["Prop_Chrome"]["A_srgbMean_x_tintLinear"], source=CHROME_NOTE)
    assert c["hex"] == "#d8d8d8", c
    m.append(c)
    return m

def side(kit): return json.load(open(os.path.join(MODELS, kit + ".json")))
def images(kit): return {v: "png/%s_%s.png" % (kit, v) for v in ("top", "front", "side", "persp")}
def slots(kit): return {v: "img:%s_%s" % (kit, v) for v in ("top", "front", "side", "persp")}

TRIS_NOTE = "LOD1 from 5 m and never culled (lodDistances 0 = no further step); LOD2 not built (pipeline P-1)"
def tris(sc):
    return dict(budget=sc["lodBudget"], lod0=sc["triangles"], lod1=sc["trianglesLod1"], lodDistances_m=sc["lodDistances"], note=TRIS_NOTE)

sheets, kits = [], {}
for order, S in enumerate(SHEETS, 1):
    for dead in (False, True):
        k = S["kit"] + ("_Dead" if dead else "")
        a, sc = A[k], side(k)
        W, H, D = a["size_m"]
        ppm = int(a["sheetPpm"]); bar_m, bar_label = SCALEBAR[ppm]
        dims = dict(w=mm(W), d=mm(D), h=mm(H))
        rec = dict(
            kit=k, title=S["title"] + (" · dead" if dead else ""), lede=DEAD_LEDE if dead else S["lede"],
            role="variant" if dead else "primary", sheet=S["id"],
            dims_mm=dims, dims_label="W %s · D %s · H %s mm" % (dims["w"], dims["d"], dims["h"]),
            size_m=a["size_m"], boundsMin=a["boundsMin"], boundsMax=a["boundsMax"],
            aboveFloor_m=S["aboveFloor"], aboveFloorNote=S["aboveFloorNote"],
            placement=sc["placement"], era=ERA, family=S["family"], materials=materials(dead, S["mount"]),
            variants=[], viewYaw=a["viewYaw"], sheetPpm=ppm, pngPpm=int(a["pngPpm"]), px=a["px"],
            scaleBar=dict(label=bar_label, metres=bar_m, sheetPx=round(bar_m * ppm)),
            tags=sc["tags"], anchors=[x["name"] for x in sc["anchors"]], images=images(k), figmaSlots=slots(k), tris=tris(sc))
        if dead:
            rec.update(variantOf=S["kit"], swap=["ExitSign_Diffuser -> ExitSign_Diffuser_Dead"],
                       behaviour="A _Dead model is always Dead (FrontRoomsExitSign); the game places only the lit models today")
        else:
            rec["colour"] = COLOUR
        kits[k] = rec
    p = kits[S["kit"]]; v = S["kit"] + "_Dead"
    sheets.append(dict(
        id=S["id"], order=order, title=S["title"], primary=S["kit"], kit=S["kit"], lede=S["lede"], family=S["family"],
        dims_mm=p["dims_mm"], dims_label=p["dims_label"], era=ERA, materials=p["materials"],
        members=S["members"], parts=[], variants=[v], variantImages={v: "png/%s_persp.png" % v},
        variantNote=DEAD_LEDE, images=p["images"], figmaSlots=p["figmaSlots"],
        variantFigmaSlots={v: "img:%s_persp" % v},
        scaleBar=p["scaleBar"], sheetPpm=p["sheetPpm"], viewYaw=p["viewYaw"], tris=p["tris"], colour=COLOUR))
