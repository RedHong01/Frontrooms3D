func C(_ hex: UInt32, _ a: CGFloat = 1) -> NSColor {
  NSColor(srgbRed: CGFloat((hex >> 16) & 255) / 255, green: CGFloat((hex >> 8) & 255) / 255, blue: CGFloat(hex & 255) / 255, alpha: a)
}

func font(_ name: String, _ size: CGFloat) -> NSFont { NSFont(name: name, size: size) ?? NSFont.systemFont(ofSize: size) }

final class Sheet {
  let W = 1920, H = 1080
  let ctx: CGContext
  init() {
    ctx = CGContext(data: nil, width: W, height: H, bitsPerComponent: 8, bytesPerRow: 0, space: CGColorSpace(name: CGColorSpace.sRGB)!, bitmapInfo: CGImageAlphaInfo.noneSkipLast.rawValue)!
    ctx.translateBy(x: 0, y: CGFloat(H)); ctx.scaleBy(x: 1, y: -1)

    ctx.setFillColor(paperBG.cgColor); ctx.fill(CGRect(x: 0, y: 0, width: W, height: H))
  }
  func save(_ path: String) {
    let rep = NSBitmapImageRep(cgImage: ctx.makeImage()!)
    try! rep.representation(using: .jpeg, properties: [.compressionFactor: 0.85])!.write(to: URL(fileURLWithPath: path))
  }
}

// metres -> pixels; sx, sy separate (sy > 0); yUp for sections.
struct Panel { var ox: CGFloat; var oy: CGFloat; var sx: CGFloat; var sy: CGFloat; var yUp: Bool
  init(_ ox: CGFloat, _ oy: CGFloat, _ s: CGFloat, yUp: Bool, sy: CGFloat? = nil) { self.ox = ox; self.oy = oy; sx = s; self.sy = sy ?? s; self.yUp = yUp }
  func p(_ x: CGFloat, _ y: CGFloat) -> CGPoint { CGPoint(x: ox + x * sx, y: yUp ? oy - y * sy : oy + y * sy) }
  func r(_ x: CGFloat, _ y: CGFloat, _ w: CGFloat, _ h: CGFloat) -> CGRect {
    let a = p(x, y), b = p(x + w, y + h)
    return CGRect(x: min(a.x, b.x), y: min(a.y, b.y), width: abs(b.x - a.x), height: abs(b.y - a.y))
  }
}
func fill(_ r: CGRect, _ c: NSColor) { ctx.setFillColor(c.cgColor); ctx.fill(r) }
func stroke(_ r: CGRect, _ c: NSColor, _ w: CGFloat = 1, dash: [CGFloat] = []) { ctx.saveGState(); ctx.setLineDash(phase: 0, lengths: dash); ctx.setStrokeColor(c.cgColor); ctx.setLineWidth(w); ctx.stroke(r); ctx.restoreGState() }
func line(_ a: CGPoint, _ b: CGPoint, _ c: NSColor, _ w: CGFloat = 1, dash: [CGFloat] = []) {
  ctx.saveGState(); ctx.setStrokeColor(c.cgColor); ctx.setLineWidth(w); ctx.setLineDash(phase: 0, lengths: dash)
  ctx.move(to: a); ctx.addLine(to: b); ctx.strokePath(); ctx.restoreGState()
}
func poly(_ pts: [CGPoint], _ c: NSColor, stroke s: NSColor? = nil) {
  ctx.saveGState(); ctx.setFillColor(c.cgColor); ctx.move(to: pts[0]); for q in pts.dropFirst() { ctx.addLine(to: q) }; ctx.closePath(); ctx.fillPath()
  if let s = s { ctx.setStrokeColor(s.cgColor); ctx.setLineWidth(1); ctx.move(to: pts[0]); for q in pts.dropFirst() { ctx.addLine(to: q) }; ctx.closePath(); ctx.strokePath() }
  ctx.restoreGState()
}
func circle(_ c: CGPoint, _ r: CGFloat, _ col: NSColor) { ctx.setFillColor(col.cgColor); ctx.fillEllipse(in: CGRect(x: c.x - r, y: c.y - r, width: 2 * r, height: 2 * r)) }
enum Al { case l, c, r }
@discardableResult func text(_ s: String, _ x: CGFloat, _ y: CGFloat, _ f: NSFont, _ c: NSColor = ink, _ al: Al = .l, chip: NSColor? = nil) -> CGSize {
  let attrs: [NSAttributedString.Key: Any] = [.font: f, .foregroundColor: c]
  let sz = (s as NSString).size(withAttributes: attrs)
  let dx: CGFloat = al == .l ? 0 : (al == .c ? -sz.width / 2 : -sz.width)
  if let chip = chip { fill(CGRect(x: x + dx - 4, y: y - 1, width: sz.width + 8, height: sz.height + 2), chip) }
  (s as NSString).draw(at: CGPoint(x: x + dx, y: y), withAttributes: attrs)
  return sz
}
func lines(_ ls: [String], _ x: CGFloat, _ y: CGFloat, _ c: NSColor = mid, chip: NSColor? = nil) {
  for (i, l) in ls.enumerated() { text(l, x, y + CGFloat(i) * 16, fSmall, c, .l, chip: chip) }
}
func hatch(_ r: CGRect, _ c: NSColor, _ step: CGFloat = 7) {
  ctx.saveGState(); ctx.clip(to: r); ctx.setStrokeColor(c.cgColor); ctx.setLineWidth(0.8)
  var x = r.minX - r.height
  while x < r.maxX { ctx.move(to: CGPoint(x: x, y: r.maxY)); ctx.addLine(to: CGPoint(x: x + r.height, y: r.minY)); x += step }
  ctx.strokePath(); ctx.restoreGState()
}
func tick(_ q: CGPoint) { line(CGPoint(x: q.x - 4, y: q.y + 4), CGPoint(x: q.x + 4, y: q.y - 4), ink, 1.4) }
// horizontal dimension at pixel y between pixel xs
func hdim(_ x1: CGFloat, _ x2: CGFloat, _ y: CGFloat, _ label: String, above: Bool = true, ext: CGFloat? = nil) {
  if let e = ext { line(CGPoint(x: x1, y: e), CGPoint(x: x1, y: y), mid, 0.6); line(CGPoint(x: x2, y: e), CGPoint(x: x2, y: y), mid, 0.6) }
  line(CGPoint(x: x1, y: y), CGPoint(x: x2, y: y), mid, 1); tick(CGPoint(x: x1, y: y)); tick(CGPoint(x: x2, y: y))
  text(label, (x1 + x2) / 2, above ? y - 18 : y + 3, fSmall, ink, .c)
}
func vdim(_ x: CGFloat, _ y1: CGFloat, _ y2: CGFloat, _ label: String, right: Bool = true, ext: CGFloat? = nil) {
  if let e = ext { line(CGPoint(x: e, y: y1), CGPoint(x: x, y: y1), mid, 0.6); line(CGPoint(x: e, y: y2), CGPoint(x: x, y: y2), mid, 0.6) }
  line(CGPoint(x: x, y: y1), CGPoint(x: x, y: y2), mid, 1); tick(CGPoint(x: x, y: y1)); tick(CGPoint(x: x, y: y2))
  text(label, right ? x + 7 : x - 7, (y1 + y2) / 2 - 8, fSmall, ink, right ? .l : .r)
}
func leader(_ from: CGPoint, _ to: CGPoint, _ label: String, _ al: Al = .l, bold: Bool = false, chip: NSColor? = nil) {
  line(from, to, ink, 0.9); circle(from, 2.4, ink)
  text(label, to.x + (al == .l ? 4 : (al == .r ? -4 : 0)), to.y - 9, bold ? fLabB : fLab, ink, al, chip: chip)
}
func panel(_ x: CGFloat, _ y: CGFloat, _ w: CGFloat, _ h: CGFloat, _ tag: String, _ title: String, _ scale: String, _ body: () -> Void) {
  ctx.saveGState(); ctx.clip(to: CGRect(x: x + 1, y: y + 50, width: w - 2, height: h - 51)); body(); ctx.restoreGState()
  stroke(CGRect(x: x, y: y, width: w, height: h), faint, 1)
  line(CGPoint(x: x, y: y + 50), CGPoint(x: x + w, y: y + 50), faint, 1)
  text(tag, x + 16, y + 10, fPanel, accent)
  text(title, x + 46, y + 10, fPanel, ink)
  text(scale, x + w - 14, y + 20, fSmall, mid, .r)
}
func header(_ title: String, _ sub: String) { text(title, 72, 34, fTitle, ink); text(sub, 1848, 52, fLab, mid, .r) }
func legend(_ items: [(NSColor, String)]) {
  var x: CGFloat = 72
  for (c, s) in items {
    fill(CGRect(x: x, y: 95, width: 14, height: 14), c); stroke(CGRect(x: x, y: 95, width: 14, height: 14), mid, 0.6)
    x += 20 + text(s, x + 20, 93, fSmall, ink).width + 22
  }
}
// plan wall along x, centred on y=0; top face (y<0) = Level 0 side.
func wallPlan(_ P: Panel, _ x0: CGFloat, _ x1: CGFloat, top: NSColor, bot: NSColor, fin: CGFloat = 0.004) {
  let t: CGFloat = 0.16, g: CGFloat = 0.032
  fill(P.r(x0, -t / 2, x1 - x0, t), gypsum)
  let core = P.r(x0, -t / 2 + g, x1 - x0, t - 2 * g); fill(core, stud); hatch(core, faint, 8)
  line(P.p(x0, -t / 2 + g), P.p(x1, -t / 2 + g), mid, 0.6); line(P.p(x0, -t / 2 + g / 2), P.p(x1, -t / 2 + g / 2), faint, 0.5)
  line(P.p(x0, t / 2 - g), P.p(x1, t / 2 - g), mid, 0.6); line(P.p(x0, t / 2 - g / 2), P.p(x1, t / 2 - g / 2), faint, 0.5)
  fill(P.r(x0, -t / 2 - fin, x1 - x0, fin), top); fill(P.r(x0, t / 2, x1 - x0, fin), bot)
}
func chevrons(_ P: Panel, _ x0: CGFloat, _ x1: CGFloat, _ h: CGFloat, _ c: NSColor) {
  var i = 0
  while CGFloat(i) * 0.75 < x1 - x0 + 0.01 {
    for j in 0..<3 {
      let cx = x0 + 0.375 + CGFloat(i) * 0.75, cy = 0.56 + CGFloat(j) * 1.125
      if cx > x1 - 0.1 || cy > h - 0.1 { continue }
      poly([P.p(cx - 0.2, cy - 0.12), P.p(cx, cy + 0.12), P.p(cx + 0.2, cy - 0.12), P.p(cx + 0.13, cy - 0.12), P.p(cx, cy + 0.0), P.p(cx - 0.13, cy - 0.12)], c)
      fill(P.r(cx - 0.33, 0, 0.03, h), c.withAlphaComponent(c.alphaComponent * 0.6))
    }
    i += 1
  }
}
