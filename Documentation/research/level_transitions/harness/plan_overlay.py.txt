import json, math, sys
from PIL import Image, ImageDraw, ImageFont
D="/Users/redwang/Desktop/ArtCenter/Fall26T7/EGAM-401A-01 Individual Game Project/Frontrooms3D/Documentation/research/level_transitions"
FONT="/Users/redwang/Desktop/ArtCenter/Fall26T7/EGAM-401A-01 Individual Game Project/Frontrooms3D/Assets/Resources/Fonts/IBMPlexMono-Regular.ttf"
SP="/private/tmp/claude-501/-Users-redwang-Desktop-ArtCenter-Fall26T7-EGAM-401A-01-Individual-Game-Project/5656cffd-bc90-45f6-86a3-09b26549df8d/scratchpad"
plan=json.load(open(SP+"/trans/extras3/plan_516574485.json"))
shots=json.load(open(D+"/shots.json"))["shots"]
img=Image.open(SP+"/trans/extras3/plan_516574485.jpg").convert("RGB")
cx,cz=plan["centreMap"][0],plan["centreMap"][2]; w,d=plan["width"],plan["depth"]
def P(x,z): return ((x-(cx-w/2))/w*1920, ((cz+d/2)-z)/d*1080)
dr=ImageDraw.Draw(img,"RGBA")
cells={(c["x"],c["y"]):c for c in plan["plan"]["cells"]}
col={"Open":(230,40,40,255),"Arch":(255,150,0,255),"Door":(0,200,230,255),"Window":(220,60,220,255),"Wall":(255,255,255,200)}
for (x,y),c in cells.items():
    for dx,dy,key in ((1,0,"east"),(0,1,"north")):
        b=cells.get((x+dx,y+dy))
        if not b: continue
        if c["theme"]==b["theme"] and c["height"]==b["height"]: continue
        theme = c["theme"]!=b["theme"]
        k=c[key]
        if dx==1: p0=P((x+1)*3,y*3); p1=P((x+1)*3,(y+1)*3)
        else: p0=P(x*3,(y+1)*3); p1=P((x+1)*3,(y+1)*3)
        if theme: dr.line([p0,p1],fill=col[k],width=7 if k!="Wall" else 3)
        else: dr.line([p0,p1],fill=(80,80,80,230),width=3)
# low / tall tint
for (x,y),c in cells.items():
    if c["height"]=="Low":
        a=P(x*3,(y+1)*3); b=P((x+1)*3,y*3); dr.rectangle([a,b],fill=(0,0,0,55))
    if c["height"]=="Tall":
        a=P(x*3,(y+1)*3); b=P((x+1)*3,y*3); dr.rectangle([a,b],fill=(255,255,255,40))
f=ImageFont.truetype(FONT,30); fs=ImageFont.truetype(FONT,22)
for s in shots:
    ex,ez=s["eyeMap"][0],s["eyeMap"][2]; yaw=math.radians(s["yaw"])
    p=P(ex,ez); L=7.0; half=math.radians(52)
    pts=[p]
    for a in (yaw-half,yaw+half):
        pts.append(P(ex+math.sin(a)*L, ez+math.cos(a)*L))
    dr.polygon(pts,fill=(255,230,0,60),outline=(255,230,0,255))
    dr.ellipse([p[0]-9,p[1]-9,p[0]+9,p[1]+9],fill=(255,230,0,255),outline=(0,0,0,255))
    dr.text((p[0]+12,p[1]+(8 if p[1]<1000 else -48)),s["id"],font=f,fill=(255,230,0,255),stroke_width=3,stroke_fill=(0,0,0,255))
# legend
lx,ly=24,24
dr.rectangle([lx-12,ly-12,lx+560,ly+250],fill=(0,0,0,170))
items=[("Open edge, theme border",col["Open"]),("Arch, theme border",col["Arch"]),("Door, theme border",col["Door"]),("Window, theme border",col["Window"]),("Wall, theme border (two skins)",col["Wall"]),("Height border only (wall/door/window)",(80,80,80,230))]
for i,(t,c) in enumerate(items):
    y=ly+i*34
    dr.line([(lx,y+14),(lx+50,y+14)],fill=c,width=7)
    dr.text((lx+64,y),t,font=fs,fill=(255,255,255,255))
dr.text((lx,ly+6*34+4),"dark tint = Low 2.4 m; seed 516574485; 64 x 36 m; north up",font=fs,fill=(220,220,220,255))
img.save(D+"/images/cut_plan_1.jpg",quality=85)
print("ok")
