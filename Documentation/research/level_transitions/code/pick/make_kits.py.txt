def sub1(s, old, new, count=1):
    n = s.count(old)
    if n < 1: raise SystemExit("MISS: " + old[:90])
    if count == 1 and n != 1: raise SystemExit("AMBIG %d: %s" % (n, old[:90]))
    return s.replace(old, new) if count != 1 else s.replace(old, new, 1)
