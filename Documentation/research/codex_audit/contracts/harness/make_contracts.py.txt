def rep(text, old, new, label):
    n = text.count(old)
    if n != 1:
        sys.exit(f"{label}: anchor found {n} times")
    return text.replace(old, new)

def sha(b): return hashlib.sha1(b).hexdigest()

state = {}   # rel -> text (cumulative)
rows = []
shutil.rmtree(os.path.join(W, 'patched'), ignore_errors=True)
for name in ORDER:
    rel, edits = C[name]
    src = os.path.join(M, rel)
    bp = os.path.join(W, 'base', rel)
    os.makedirs(os.path.dirname(bp), exist_ok=True)
    cur = open(src, 'rb').read()
    if not os.path.exists(bp) or open(bp, 'rb').read() != cur:
        shutil.copyfile(src, bp)
    before = state.get(rel, cur.decode('utf-8'))
    after = before
    for old, new in edits:
        after = rep(after, old, new, name)
    state[rel] = after
    # files for the diff: this contract's own base (cumulative state before it) and its result
    d0 = os.path.join(W, 'tmp', name, 'a', rel); d1 = os.path.join(W, 'tmp', name, 'b', rel)
    for p, t in ((d0, before), (d1, after)):
        os.makedirs(os.path.dirname(p), exist_ok=True)
        open(p, 'w', encoding='utf-8', newline='').write(t)
    r = subprocess.run(['git', 'diff', '--no-index', '--no-prefix', 'a/' + rel, 'b/' + rel], cwd=os.path.join(W, 'tmp', name), capture_output=True, text=True)
    open(os.path.join(W, 'diffs', name + '.diff'), 'w').write(r.stdout)
    pp = os.path.join(W, 'patched', name, rel)
    os.makedirs(os.path.dirname(pp), exist_ok=True)
    open(pp, 'w', encoding='utf-8', newline='').write(after)
    rows.append((name, rel, sha(before.encode('utf-8')), sha(after.encode('utf-8')), sha(cur)))

print("contract\tfile\tbase_sha1(before this contract)\tafter_sha1\tmain_sha1")
for r in rows: print('\t'.join(r))
