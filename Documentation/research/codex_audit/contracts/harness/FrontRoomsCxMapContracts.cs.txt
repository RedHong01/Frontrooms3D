// CODEX AUDIT · REPORT STAGE · MAP CONTRACT CHECKS (clone W/proj_cxreport only; never copied into the real project).
// Checks the map contracts MAP-2 and MAP-3 on the code the clone holds (main, or main + the contract patches):
//  T1 shift independence (MAP-2): build a seed's 25 chunks, shift + rebuild every "even" chunk (x+y even, the
//     game's revisit shift), then compare every still-built "odd" chunk with a fresh rebuild of itself. The
//     generator's rule (FrontRoomsMap.cs "neighbours agree no matter which is built first") says they are equal.
//     Run with B0 off (control, the pre-Codex rule) and with B0 as compiled. Also counts the chunks the cache
//     generated beyond the built ones (neighbour data generated early).
//  T2 lamp predictor purity and agreement (MAP-3), with the V5 border rule forced on (Borders = true, as
//     -lightleadBorders does): LampModeOf on 64 far cells must generate no chunk; LampModeOf must equal the
//     built fixture's mode for every built lamp, before and after the cache drops every generated-but-unbuilt chunk.
//  T3 the project's FrontRoomsMapInteractionTests with Borders forced on (report copied per pass).
//  Signatures (renderers / lights / colliders hash per chunk) of every first build go to sig_<pass>.tsv so two
//  runs can be compared by a script.
// Run: Unity -batchmode -projectPath <clone> -buildTarget OSXUniversal -executeMethod FrontRoomsCxMapContracts.RunBatch
//      -cxLabel <name> -logFile <f>   (exits by itself)

    public static void RunBatch()
    {
        var code = 0;
        var label = Arg("-cxLabel") ?? "run";
        outDir = Path.Combine(Root, "Verification", "cx_contracts", label);
        Directory.CreateDirectory(outDir);
        Application.logMessageReceived += OnLog;
        try
        {
            EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
            var b0 = FrontRoomsTransitionKit.B0;
            var borders = FrontRoomsTransitionLightLead.Borders;
            Log("start · " + label + " · Unity " + Application.unityVersion + " · B0 " + b0 + " · LightLead.On " + FrontRoomsTransitionLightLead.On + " · Borders " + borders);

            FrontRoomsTransitionKit.B0 = false;
            ShiftTest("b0off");

            ShiftTest("b0on");

            PredictorTest("borders");

            FrontRoomsTransitionLightLead.Borders = borders;
            PredictorTest("default");
            if (Arg("-cxNoMI") == null) MapInteraction("default");
        }
        catch (Exception e)
        {
            Log("HARNESS EXCEPTION " + e);
            code = 2;
        }
        finally
        {
            Application.logMessageReceived -= OnLog;
            foreach (var o in owned) if (o != null) Object.DestroyImmediate(o);
            owned.Clear();
            Log("end · exceptions logged " + exceptions + " · exit " + code);
        }
        if (Application.isBatchMode) EditorApplication.Exit(code);
    }

    static void ShiftTest(string pass)
    {
        int totalOdd = 0, geomDiff = 0, lightDiff = 0, seedsWithDiff = 0;
        long generatedBeyond = 0, builtTotal = 0;
        using (var sig = new StreamWriter(Path.Combine(outDir, "sig_" + pass + ".tsv"), false, new UTF8Encoding(false)))
        {
            sig.WriteLine("seed\tchunk\tkind\tcount\thash");
            foreach (var seed in Seeds)
            {
                var w = BuildWorld(seed);
                var built = Chunks(w).Select(c => c.coord).ToList();
                var generated = w.Cache.Built.Count();
                builtTotal += built.Count;
                generatedBeyond += generated - built.Count;
                var first = Signatures(w, built);
                var posts = RegionPosts(w, built);
                var view1 = new Dictionary<GridCoord, PS[]>();
                foreach (var k in posts) view1[k] = Post(w, k);
                foreach (var kv in first.OrderBy(k => k.Key, StringComparer.Ordinal)) sig.WriteLine(seed + "\t" + kv.Key + "\t" + kv.Value.count + "\t" + kv.Value.hash);

                var even = built.Where(c => ((c.x + c.y) & 1) == 0).ToList();
                var odd = built.Where(c => ((c.x + c.y) & 1) != 0).ToList();
                foreach (var c in even) { w.Cache.Shift(c); w.RebuildChunk(c); }
                w.TickFixturesForTools(0f);
                Physics.SyncTransforms();
                var stale = Signatures(w, odd);
                var view2 = new Dictionary<GridCoord, PS[]>();
                foreach (var k in posts) view2[k] = Post(w, k);
                var evenSet = new HashSet<GridCoord>(even);
                if (FrontRoomsTransitionKit.B0) Coverage(w, pass, seed, posts, view1, view2, evenSet);
                foreach (var c in odd) w.RebuildChunk(c);
                w.TickFixturesForTools(0f);
                Physics.SyncTransforms();
                var fresh = Signatures(w, odd, detailTag: pass + "_" + seed);

                var sGeom = 0; var sLight = 0; var names = new List<string>();
                foreach (var c in odd)
                {
                    var k = c.x + "_" + c.y;
                    var g = stale[k + "\trenderers"].hash != fresh[k + "\trenderers"].hash || stale[k + "\tcolliders"].hash != fresh[k + "\tcolliders"].hash;
                    var l = stale[k + "\tlights"].hash != fresh[k + "\tlights"].hash;
                    if (g) { sGeom++; names.Add(k); DumpDiff(pass, seed, k); }
                    if (l) sLight++;
                }
                totalOdd += odd.Count; geomDiff += sGeom; lightDiff += sLight;
                if (sGeom + sLight > 0) seedsWithDiff++;
                Log("T1 " + pass + " · seed " + seed + " · built " + built.Count + " · cache generated " + generated + " (+" + (generated - built.Count) + " beyond built) · shifted " + even.Count
                    + " · still-built chunks that differ from their own rebuild: geometry " + sGeom + " / " + odd.Count + ", lights " + sLight + (names.Count > 0 ? " (" + string.Join(", ", names.Take(6)) + ")" : ""));
                Free(w);
                staleLines.Clear(); freshLines.Clear();
            }
        }
        CoverageSummary(pass);
        Log("T1 " + pass + " TOTAL · " + Seeds.Length + " seeds · still-built chunks differing from their own rebuild after neighbour shifts: geometry " + geomDiff + " / " + totalOdd
            + ", lights " + lightDiff + " / " + totalOdd + " · seeds affected " + seedsWithDiff + " · chunk data generated beyond the built chunks: " + generatedBeyond + " (built " + builtTotal + ")");
    }

    // ------------------------------------------------------------------ T2 (MAP-3)
    static void PredictorTest(string pass)
    {
        int seeds = 0, farGenerated = 0, lamps = 0, disagree = 0, disagreeAfterDrop = 0, regenerated = 0, borderChanged = 0;
        var dropMethod = typeof(FrontRoomsMapCache).GetMethod("Drop", IF);
        foreach (var seed in Seeds.Take(10))
        {
            seeds++;
            var w = BuildWorld(seed);
            var g0 = w.Cache.Built.Count();
            var mid = Chunks(w).Select(c => c.coord).OrderBy(c => c.x).ThenBy(c => c.y).ElementAt(12);
            var midCell = new GridCoord(mid.x * MapGrid.ChunkCells + 4, mid.y * MapGrid.ChunkCells + 4);
            for (var i = 0; i < 64; i++)
                w.LampModeOf(new GridCoord(midCell.x + 400 + i * 37, midCell.y - 300 + i * 23));
            var g1 = w.Cache.Built.Count();
            farGenerated += g1 - g0;

            var fixtures = typeof(FrontRoomsMapWorld).GetField("fixtureByCell", IF).GetValue(w) as IDictionary;
            var cells = new List<(GridCoord cell, int mode)>();
            foreach (DictionaryEntry e in fixtures)
            {
                var mode = (int)e.Value.GetType().GetField("mode", IF).GetValue(e.Value);
                cells.Add(((GridCoord)e.Key, mode));
            }
            lamps += cells.Count;
            using (var lw = new StreamWriter(Path.Combine(outDir, "lamps_" + pass + ".tsv"), seeds > 1, new UTF8Encoding(false)))
                foreach (var (cell, mode) in cells.OrderBy(c => c.cell.x).ThenBy(c => c.cell.y)) lw.WriteLine(seed + "\t" + cell.x + "\t" + cell.y + "\t" + mode);
            foreach (var (cell, mode) in cells)
                if ((int)w.LampModeOf(cell) != mode + 1) disagree++;

            // drop every generated-but-unbuilt chunk, then ask again
            var builtSet = new HashSet<GridCoord>(Chunks(w).Select(c => c.coord));
            foreach (var c in w.Cache.Built.Select(x => x.coord).ToList()) if (!builtSet.Contains(c)) dropMethod.Invoke(w.Cache, new object[] { c });
            var g2 = w.Cache.Built.Count();
            foreach (var (cell, mode) in cells)
                if ((int)w.LampModeOf(cell) != mode + 1) disagreeAfterDrop++;
            var g3 = w.Cache.Built.Count();
            regenerated += g3 - g2;

            // how many built lamps the border rule changed (re-roll without it)
            if (FrontRoomsTransitionLightLead.Borders)
            {
                FrontRoomsTransitionLightLead.Borders = false;
                foreach (var (cell, mode) in cells) if ((int)w.LampModeOf(cell) != mode + 1) borderChanged++;
                FrontRoomsTransitionLightLead.Borders = true;
            }
            Log("T2 " + pass + " · seed " + seed + " · far LampModeOf ×64 generated " + (g1 - g0) + " chunks · lamps " + cells.Count + " · predictor ≠ built " + disagree
                + " (cumulative) · after dropping " + (g1 - g2) + " unbuilt chunks: ≠ " + disagreeAfterDrop + " (cumulative), chunks regenerated " + (g3 - g2));
            Free(w);
        }
        Log("T2 " + pass + " TOTAL · " + seeds + " seeds · far-cell LampModeOf generated " + farGenerated + " chunks · built lamps " + lamps + " · predictor ≠ built " + disagree
            + " · after dropping unbuilt chunks ≠ " + disagreeAfterDrop + ", regenerated " + regenerated + " · lamps the border rule changed " + borderChanged);
    }

    static void MapInteraction(string pass)
    {
        EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
        var type = AppDomain.CurrentDomain.GetAssemblies().Select(a => a.GetType("FrontRoomsMapInteractionTests")).FirstOrDefault(t => t != null);
        if (type == null) { Log("T3 " + pass + " · FrontRoomsMapInteractionTests NOT FOUND"); return; }
        var report = Path.Combine(Root, "Verification", "map-interaction-tests.json");
        if (File.Exists(report)) File.Delete(report);
        string outcome;
        try { type.GetMethod("RunBatch", SF).Invoke(null, null); outcome = "no exception"; }
        catch (TargetInvocationException e) { outcome = "THREW " + e.InnerException?.GetType().Name + ": " + e.InnerException?.Message; }

        Log("T3 " + pass + " · FrontRoomsMapInteractionTests.RunBatch: " + outcome + " · report " + (File.Exists(report) ? "copied" : "missing"));
        EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
    }

    // ------------------------------------------------------------------ T1b post coverage (MAP-2)
    // For every post whose four cells lie in built chunks (start area left out), the state of its four pieces
    // (S, N, W, E: present, skin finishes, reaches as CornerReach returns them, height) is read with the map's own
    // PostPieceAt / CornerReach, after the first build (view 1) and after the shift + rebuild of the even chunks
    // (view 2). As built after the shift, a piece owned by a shifted chunk has view 2, every other piece view 1.
    // The post square is split into 4 quadrants (SW, SE, NW, NE); each skin covers the quadrants its reach gives.
    //  notch: two or more pieces meet and a quadrant is empty;
    //  overlap: a quadrant covered by skins of two different finishes (a z-fight at the post);
    //  high: a covered quadrant whose tallest cover is lower than the room it faces (a hole above a low ceiling).
    // Rules: "compiled" = what this clone's MapWorld does (reflection). Simulated from the same views:
    //  R0 = B0.1 everywhere (main as is); R1 = +1 on chunk lines and corners; RA = the verifier's run rule
    //  (+1 at corners); RB = the contract (B0.1 at corners, run rule, cross pieces to the cell line when the
    //  line is open, +1 with one line piece).
    struct PS { public bool p; public long fA, fB; public int rA, rB; public float h; }
    static readonly MethodInfo miCornerReach = typeof(FrontRoomsMapWorld).GetMethod("CornerReach", IF);
    static readonly MethodInfo miPostPieceAt = typeof(FrontRoomsMapWorld).GetMethod("PostPieceAt", IF);
    static readonly string[] RuleNames = { "compiled", "R0", "R1", "RA", "RB", "RD" };
    // [rule][kind 0 interior / 1 line / 2 corner][phase 0 build / 1 after shift][metric 0 posts,1 notch,2 overlap,3 high,4 mixedPosts,5 strip]
    static readonly Dictionary<string, long[,,,]> cov = new Dictionary<string, long[,,,]>();
    static int compiledMismatch;

    static List<GridCoord> RegionPosts(FrontRoomsMapWorld w, List<GridCoord> built)
    {
        var set = new HashSet<GridCoord>(built);
        var list = new List<GridCoord>();
        foreach (var c in built)
        {
            var o = MapGrid.ChunkOrigin(c);
            for (var j = 0; j < MapGrid.ChunkCells; j++)
            for (var i = 0; i < MapGrid.ChunkCells; i++)
            {
                var k = new GridCoord(o.x + i, o.y + j);
                GridCoord sw = new GridCoord(k.x - 1, k.y - 1), se = new GridCoord(k.x, k.y - 1), nw = new GridCoord(k.x - 1, k.y);
                if (!set.Contains(MapGrid.ChunkOf(sw)) || !set.Contains(MapGrid.ChunkOf(se)) || !set.Contains(MapGrid.ChunkOf(nw))) continue;
                if (w.InStartArea(sw) || w.InStartArea(se) || w.InStartArea(nw) || w.InStartArea(k)) continue;
                list.Add(k);
            }
        }
        return list;
    }

    static float CH(FrontRoomsMapWorld w, GridCoord c) => MapGrid.CeilingHeight(w.Cache.ZoneOf(c).height);

    static PS[] Post(FrontRoomsMapWorld w, GridCoord k)
    {
        GridCoord sw = new GridCoord(k.x - 1, k.y - 1), se = new GridCoord(k.x, k.y - 1), nw = new GridCoord(k.x - 1, k.y), ne = k;
        var pairs = new[] { (sw, se, false), (nw, ne, false), (sw, nw, true), (se, ne, true) };
        var r = new PS[4];
        for (var i = 0; i < 4; i++)
        {
            var (a, b, northward) = pairs[i];
            var pp = (FrontRoomsTransitionKit.PostPiece)miPostPieceAt.Invoke(w, new object[] { a, b, northward });
            var args = new object[] { k, i, 0, 0 };
            miCornerReach.Invoke(w, args);
            r[i] = new PS { p = pp.present, fA = pp.finishA, fB = pp.finishB, rA = (int)args[2], rB = (int)args[3], h = Mathf.Max(CH(w, a), CH(w, b)) };
        }
        return r;
    }

    static GridCoord Owner(GridCoord k, int role)
    {
        GridCoord sw = new GridCoord(k.x - 1, k.y - 1), se = new GridCoord(k.x, k.y - 1), nw = new GridCoord(k.x - 1, k.y);
        return MapGrid.ChunkOf(role == 0 || role == 2 ? sw : role == 1 ? nw : se);
    }

    // reach of `role` under a simulated rule, as the owning chunk saw the post (view v)
    static void Rule(string rule, FrontRoomsMapWorld w, GridCoord k, PS[] v, int role, out int ra, out int rb)
    {
        ra = rb = 1;
        var o = MapGrid.ChunkOrigin(MapGrid.ChunkOf(k));
        bool onX = k.x == o.x, onY = k.y == o.y;
        var pieces = new FrontRoomsTransitionKit.PostPiece[4];
        for (var i = 0; i < 4; i++) pieces[i] = new FrontRoomsTransitionKit.PostPiece { present = v[i].p, finishA = v[i].fA, finishB = v[i].fB, officeA = v[i].fA >> 4 == OfficeWallId(w), officeB = v[i].fB >> 4 == OfficeWallId(w) };
        if (rule == "R0" || (!onX && !onY)) { FrontRoomsTransitionKit.CornerReach(pieces, role, out ra, out rb); return; }
        if (rule == "R1") return;
        var corner = onX && onY;
        if (corner) { if (rule == "RB" || rule == "RD") FrontRoomsTransitionKit.CornerReach(pieces, role, out ra, out rb); return; }
        if (OneFinish(w, k)) return;
        var line = onX ? role == 0 || role == 1 : role == 2 || role == 3;
        bool lo = onX ? v[0].p : v[2].p, hi = onX ? v[1].p : v[3].p;
        if (lo && hi) { ra = rb = line ? 0 : -1; return; }
        if ((rule == "RB" || rule == "RD") && !lo && !hi) { ra = rb = line ? 1 : 0; return; }
        if (rule != "RD" || (lo && hi) || (!lo && !hi)) return;
        // RD, one line piece L: safe when L and the near cross piece are both at least as tall as the room the
        // far cross piece's other quadrant faces; then the far cross piece stops at the face and B0.1 runs on L + near.
        int loR = onX ? 0 : 2, hiR = onX ? 1 : 3, nearR = onX ? 2 : 0, farR = onX ? 3 : 1;
        var L = lo ? loR : hiR;
        GridCoord sw = new GridCoord(k.x - 1, k.y - 1), se = new GridCoord(k.x, k.y - 1), nw = new GridCoord(k.x - 1, k.y), ne = k;
        var risk = onX ? (L == loR ? ne : se) : (L == loR ? ne : nw);
        var hL = onX ? (L == loR ? Mathf.Max(CH(w, sw), CH(w, se)) : Mathf.Max(CH(w, nw), CH(w, ne))) : (L == loR ? Mathf.Max(CH(w, sw), CH(w, nw)) : Mathf.Max(CH(w, se), CH(w, ne)));
        var hNear = onX ? Mathf.Max(CH(w, sw), CH(w, nw)) : Mathf.Max(CH(w, sw), CH(w, se));
        if (CH(w, risk) > Mathf.Min(hL, hNear) + .001f) return;
        if (role == farR) { ra = rb = -1; return; }
        var sub = new FrontRoomsTransitionKit.PostPiece[4];
        sub[L] = pieces[L];
        sub[nearR] = pieces[nearR];
        FrontRoomsTransitionKit.CornerReach(sub, role, out ra, out rb);
    }

    static long officeWallId = long.MinValue;
    static long OfficeWallId(FrontRoomsMapWorld w)
    {
        if (officeWallId == long.MinValue)
        {
            var office = typeof(FrontRoomsMapWorld).GetField("office", IF).GetValue(w);
            var wall = office.GetType().GetField("wall", IF).GetValue(office) as Material;
            officeWallId = wall != null ? wall.GetInstanceID() : 0;
        }
        return officeWallId;
    }

    static bool OneFinish(FrontRoomsMapWorld w, GridCoord k)
    {
        GridCoord sw = new GridCoord(k.x - 1, k.y - 1), se = new GridCoord(k.x, k.y - 1), nw = new GridCoord(k.x - 1, k.y), ne = k;
        long F(GridCoord c)
        {
            var z = w.Cache.ZoneOf(c);
            var theme = typeof(FrontRoomsMapWorld).GetMethod("Theme", IF).Invoke(w, new object[] { z.theme });
            var wall = theme.GetType().GetField("wall", IF).GetValue(theme) as Material;
            var h = MapGrid.CeilingHeight(z.height);
            var hc = h < 2.65f ? 0 : h < 4f ? 1 : 2;
            return ((long)wall.GetInstanceID() << 4) | (long)hc;
        }
        var f = F(sw);
        return F(se) == f && F(nw) == f && F(ne) == f;
    }

    static readonly Dictionary<GridCoord, long> cellFCache = new Dictionary<GridCoord, long>();
    static long CellF(FrontRoomsMapWorld w, GridCoord c)
    {
        if (cellFCache.TryGetValue(c, out var f)) return f;
        var z = w.Cache.ZoneOf(c);
        var theme = typeof(FrontRoomsMapWorld).GetMethod("Theme", IF).Invoke(w, new object[] { z.theme });
        var wall = theme.GetType().GetField("wall", IF).GetValue(theme) as Material;
        var h = MapGrid.CeilingHeight(z.height);
        var hc = h < 2.65f ? 0 : h < 4f ? 1 : 2;
        f = ((long)wall.GetInstanceID() << 4) | (long)hc;
        cellFCache[c] = f;
        return f;
    }

    // quadrant masks covered by a skin: index [role][skin 0=A,1=B][reach+1]
    static int Quads(int role, int skin, int r)
    {
        // quadrants: 0 SW, 1 SE, 2 NW, 3 NE
        int near0, far1;
        switch (role)
        {
            case 0: near0 = skin == 0 ? 0 : 1; far1 = skin == 0 ? 2 : 3; break;   // S: A west (SW, +NW), B east (SE, +NE)
            case 1: near0 = skin == 0 ? 2 : 3; far1 = skin == 0 ? 0 : 1; break;   // N: A (NW, +SW), B (NE, +SE)
            case 2: near0 = skin == 0 ? 0 : 2; far1 = skin == 0 ? 1 : 3; break;   // W: A south (SW, +SE), B north (NW, +NE)
            default: near0 = skin == 0 ? 1 : 3; far1 = skin == 0 ? 0 : 2; break;  // E: A (SE, +SW), B (NE, +NW)
        }
        var m = 0;
        if (r >= 0) m |= 1 << near0;
        if (r >= 1) m |= 1 << far1;
        return m;
    }

    static void Evaluate(FrontRoomsMapWorld w, GridCoord k, bool[] present, long[,] fin, int[,] reach, float[] height, long[] counts)
    {
        // counts: 0 posts, 1 notch, 2 overlap, 3 high, 4 mixed posts
        counts[0]++;
        var n = 0;
        for (var i = 0; i < 4; i++) if (present[i]) n++;
        var cellH = new[] { CH(w, new GridCoord(k.x - 1, k.y - 1)), CH(w, new GridCoord(k.x, k.y - 1)), CH(w, new GridCoord(k.x - 1, k.y)), CH(w, k) };
        // pieces adjacent to each quadrant's cell: SW: S, W · SE: S, E · NW: N, W · NE: N, E
        var adj = new[] { new[] { 0, 2 }, new[] { 0, 3 }, new[] { 1, 2 }, new[] { 1, 3 } };
        var mixedPost = false;
        long firstF = long.MinValue;
        bool notch = false, overlap = false, high = false;
        for (var q = 0; q < 4; q++)
        {
            long f0 = long.MinValue; var diff = false; var covered = false; var topH = 0f;
            for (var i = 0; i < 4; i++)
            {
                if (!present[i]) continue;
                for (var s = 0; s < 2; s++)
                {
                    var f = fin[i, s];
                    if (firstF == long.MinValue) firstF = f; else if (f != firstF) mixedPost = true;
                    if ((Quads(i, s, reach[i, s]) & (1 << q)) == 0) continue;
                    covered = true;
                    topH = Mathf.Max(topH, height[i]);
                    if (f0 == long.MinValue) f0 = f; else if (f != f0) diff = true;
                }
            }
            if (!covered && n >= 2) notch = true;
            if (diff) overlap = true;
            var faces = present[adj[q][0]] || present[adj[q][1]];
            if (covered && faces && topH < cellH[q] - .01f) high = true;
        }
        // strip: a covered quadrant whose face on an open side (no piece there) shows a finish other than its room's
        var strip = false;
        var sides = new[] { new[] { 0, 2 }, new[] { 0, 3 }, new[] { 1, 2 }, new[] { 1, 3 } };   // SW: S, W · SE: S, E · NW: N, W · NE: N, E
        var cellF = new[] { CellF(w, new GridCoord(k.x - 1, k.y - 1)), CellF(w, new GridCoord(k.x, k.y - 1)), CellF(w, new GridCoord(k.x - 1, k.y)), CellF(w, k) };
        for (var q = 0; q < 4 && !strip; q++)
        {
            if (present[sides[q][0]] && present[sides[q][1]]) continue;
            for (var i = 0; i < 4 && !strip; i++)
            {
                if (!present[i]) continue;
                for (var s2 = 0; s2 < 2; s2++)
                    if ((Quads(i, s2, reach[i, s2]) & (1 << q)) != 0 && fin[i, s2] != cellF[q]) { strip = true; break; }
            }
        }
        if (strip) counts[5]++;
        if (notch) counts[1]++;
        if (overlap) counts[2]++;
        if (high) counts[3]++;
        if (mixedPost) counts[4]++;
    }

    static void Coverage(FrontRoomsMapWorld w, string pass, int seed, List<GridCoord> posts, Dictionary<GridCoord, PS[]> v1, Dictionary<GridCoord, PS[]> v2, HashSet<GridCoord> shifted)
    {
        if (!cov.ContainsKey(pass)) cov[pass] = new long[RuleNames.Length, 3, 2, 6];
        cellFCache.Clear();
        var t = cov[pass];
        var counts = new long[6];
        foreach (var k in posts)
        {
            var o = MapGrid.ChunkOrigin(MapGrid.ChunkOf(k));
            bool onX = k.x == o.x, onY = k.y == o.y;
            var kind = onX && onY ? 2 : onX || onY ? 1 : 0;
            for (var phase = 0; phase < 2; phase++)
            for (var ri = 0; ri < RuleNames.Length; ri++)
            {
                var present = new bool[4]; var fin = new long[4, 2]; var reach = new int[4, 2]; var height = new float[4];
                for (var role = 0; role < 4; role++)
                {
                    var v = phase == 1 && shifted.Contains(Owner(k, role)) ? v2[k] : v1[k];
                    present[role] = v[role].p; fin[role, 0] = v[role].fA; fin[role, 1] = v[role].fB; height[role] = v[role].h;
                    int ra, rb;
                    if (ri == 0) { ra = v[role].rA; rb = v[role].rB; }
                    else Rule(RuleNames[ri], w, k, v, role, out ra, out rb);
                    reach[role, 0] = ra; reach[role, 1] = rb;
                }
                Array.Clear(counts, 0, counts.Length);
                Evaluate(w, k, present, fin, reach, height, counts);
                for (var m = 0; m < 6; m++) t[ri, kind, phase, m] += counts[m];
            }
        }
    }

    static void CoverageSummary(string pass)
    {
        if (!cov.TryGetValue(pass, out var t)) return;
        var kinds = new[] { "inside a chunk", "on a chunk line", "at a chunk corner" };
        var sb = new StringBuilder();
        sb.AppendLine("rule\tposts\tphase\tposts\tmixed posts\tnotch\toverlap\thigh\tstrip");
        for (var ri = 0; ri < RuleNames.Length; ri++)
        for (var kind = 0; kind < 3; kind++)
        for (var phase = 0; phase < 2; phase++)
            sb.AppendLine(RuleNames[ri] + "\t" + kinds[kind] + "\t" + (phase == 0 ? "first build" : "after shifts") + "\t" + t[ri, kind, phase, 0] + "\t" + t[ri, kind, phase, 4] + "\t" + t[ri, kind, phase, 1] + "\t" + t[ri, kind, phase, 2] + "\t" + t[ri, kind, phase, 3] + "\t" + t[ri, kind, phase, 5]);
        File.WriteAllText(Path.Combine(outDir, "post_coverage_" + pass + ".tsv"), sb.ToString());
        for (var ri = 0; ri < RuleNames.Length; ri++)

                    + ", after shifts " + t[ri, kind, 1, 1] + "/" + t[ri, kind, 1, 2] + "/" + t[ri, kind, 1, 3] + "/" + t[ri, kind, 1, 5] + " ·");
            Log(line.ToString());
        }
    }

    // ------------------------------------------------------------------ ShotBatch: frames of a MAP-2 notch
    // Edit mode, in the game scene (its environment, camera and post), map root (4992, 0, 4992), like the level-
    // transition audit harness. Builds a seed, shifts + rebuilds the even chunks (as T1), finds a chunk-line post
    // where the as-built pieces leave a quadrant empty (a notch), and renders it from inside the room that quadrant
    // faces: (1) right after the shift, (2) after the stale neighbour chunk is rebuilt. -cxSeed s -cxPost x,y
    // -cxQuad q shoot a given post (for the same frame on another build). Writes <out>/<seed>_<x>_<y>_<step>.jpg.
    const string ScenePath = "Assets/Scenes/FrontRooms3D.unity";
    const int ShotW = 1920, ShotH = 1080;
    static Camera shotCam;
    static RenderTexture shotRT;
    static Texture2D shotTex;
    struct Env { public AmbientMode mode; public Color sky, eq, ground, light; public float ambInt, reflInt, fogDensity; public bool fog; public FogMode fogMode; public Color fogColor; public Material skybox; }
    static Env sceneEnv;

    public static void ShotBatch()
    {
        var code = 0;
        var label = Arg("-cxLabel") ?? "shot";
        outDir = Path.Combine(Root, "Verification", "cx_contracts", label);
        Directory.CreateDirectory(outDir);
        try
        {
            ShaderUtil.allowAsyncCompilation = false;
            EditorSceneManager.OpenScene(ScenePath, OpenSceneMode.Single);
            sceneEnv = new Env
            {
                mode = RenderSettings.ambientMode, sky = RenderSettings.ambientSkyColor, eq = RenderSettings.ambientEquatorColor, ground = RenderSettings.ambientGroundColor,
                light = RenderSettings.ambientLight, ambInt = RenderSettings.ambientIntensity, reflInt = RenderSettings.reflectionIntensity, fog = RenderSettings.fog,
                fogMode = RenderSettings.fogMode, fogColor = RenderSettings.fogColor, fogDensity = RenderSettings.fogDensity, skybox = RenderSettings.skybox,
            };
            var game = Object.FindFirstObjectByType<FrontRooms3DGame>();
            var hunter = game != null ? game.transform.Find("Hunter") : null;
            if (hunter != null) hunter.gameObject.SetActive(false);
            var seeds = int.TryParse(Arg("-cxSeed"), out var one) ? new[] { one } : Seeds;
            var max = int.TryParse(Arg("-cxMax"), out var m) ? m : 2;
            GridCoord? want = null;
            var pa = Arg("-cxPost");
            if (pa != null) { var parts = pa.Split(','); want = new GridCoord(int.Parse(parts[0]), int.Parse(parts[1])); }
            var wantQuad = int.TryParse(Arg("-cxQuad"), out var wq) ? wq : -1;
            Log("shot · " + label + " · B0 " + FrontRoomsTransitionKit.B0 + " · seeds " + string.Join(",", seeds) + (want != null ? " · post " + want : ""));
            var shots = 0;
            foreach (var seed in seeds)
            {
                if (shots >= max) break;
                var w = BuildWorld(seed);
                RestoreEnv();
                var built = Chunks(w).Select(c => c.coord).ToList();
                var posts = RegionPosts(w, built);
                var v1 = new Dictionary<GridCoord, PS[]>();
                foreach (var k in posts) v1[k] = Post(w, k);
                var even = built.Where(c => ((c.x + c.y) & 1) == 0).ToList();
                foreach (var c in even) { w.Cache.Shift(c); w.RebuildChunk(c); }
                Physics.SyncTransforms();
                var v2 = new Dictionary<GridCoord, PS[]>();
                foreach (var k in posts) v2[k] = Post(w, k);
                var evenSet = new HashSet<GridCoord>(even);
                var candidates = new List<(GridCoord k, int q)>();
                foreach (var k in want != null ? new List<GridCoord> { want.Value } : posts)
                {
                    if (!v1.ContainsKey(k)) { Log("  post " + k + " not in the region"); continue; }
                    var empty = EmptyQuads(k, v1, v2, evenSet, out var n);
                    if (want != null) { Log("  post " + k + " · pieces " + n + " · empty quadrants " + string.Join(",", empty)); candidates.Add((k, wantQuad >= 0 ? wantQuad : empty.FirstOrDefault())); }
                    else if (n >= 2 && empty.Count > 0) candidates.Add((k, empty[0]));
                }
                Log("seed " + seed + " · posts " + posts.Count + " · notch candidates " + candidates.Count);
                foreach (var (k, q) in candidates)
                {
                    if (shots >= max) break;
                    if (!Shoot(w, seed, k, q, "1_after_shift")) continue;
                    // rebuild the not-shifted (stale) chunks that own a piece at this post
                    foreach (var c in Enumerable.Range(0, 4).Select(r => Owner(k, r)).Distinct().Where(c => !evenSet.Contains(c)).ToList()) w.RebuildChunk(c);
                    Physics.SyncTransforms();
                    Shoot(w, seed, k, q, "2_stale_chunk_rebuilt");
                    shots++;
                    break;   // one post per seed: the rebuild changed the state
                }
                Free(w);
            }
            Log("shots " + shots);
        }
        catch (Exception e) { Log("HARNESS EXCEPTION " + e); code = 2; }
        finally
        {
            foreach (var o in owned) if (o != null) Object.DestroyImmediate(o);
            owned.Clear();
        }
        if (Application.isBatchMode) EditorApplication.Exit(code);
    }

    static void RestoreEnv()
    {
        RenderSettings.ambientMode = sceneEnv.mode; RenderSettings.ambientSkyColor = sceneEnv.sky; RenderSettings.ambientEquatorColor = sceneEnv.eq;
        RenderSettings.ambientGroundColor = sceneEnv.ground; RenderSettings.ambientLight = sceneEnv.light; RenderSettings.ambientIntensity = sceneEnv.ambInt;
        RenderSettings.reflectionIntensity = sceneEnv.reflInt; RenderSettings.fog = sceneEnv.fog; RenderSettings.fogMode = sceneEnv.fogMode;
        RenderSettings.fogColor = sceneEnv.fogColor; RenderSettings.fogDensity = sceneEnv.fogDensity; RenderSettings.skybox = sceneEnv.skybox;
        DynamicGI.UpdateEnvironment();
    }

    // empty quadrants of a post as built after the shift (compiled reaches; a piece of a shifted chunk has view 2)
    static List<int> EmptyQuads(GridCoord k, Dictionary<GridCoord, PS[]> v1, Dictionary<GridCoord, PS[]> v2, HashSet<GridCoord> shifted, out int n)
    {
        var mask = 0; n = 0;
        for (var role = 0; role < 4; role++)
        {
            var v = shifted.Contains(Owner(k, role)) ? v2[k] : v1[k];
            if (!v[role].p) continue;
            n++;
            mask |= Quads(role, 0, v[role].rA) | Quads(role, 1, v[role].rB);
        }
        var list = new List<int>();
        for (var q = 0; q < 4; q++) if ((mask & (1 << q)) == 0) list.Add(q);
        return list;
    }

    static bool Shoot(FrontRoomsMapWorld w, int seed, GridCoord k, int q, string step)
    {
        var root = w.transform.position;
        var corner = new Vector3(k.x * MapGrid.CellSize, 0f, k.y * MapGrid.CellSize);
        var dir = new Vector3((q & 1) == 0 ? -1f : 1f, 0f, (q & 2) == 0 ? -1f : 1f).normalized;   // into the quadrant's cell
        var target = corner + dir * .04f + Vector3.up * 1.3f;
        foreach (var dist in new[] { .9f, .7f, 1.2f, .55f })
        foreach (var spin in new[] { 0f, 20f, -20f, 35f, -35f })
        {
            var d = Quaternion.Euler(0f, spin, 0f) * dir;
            var eye = corner + d * dist + Vector3.up * 1.45f;
            var we = root + eye;
            if (Physics.CheckSphere(we, .12f, ~0, QueryTriggerInteraction.Ignore)) continue;
            var to = (root + target) - we;
            if (Physics.Raycast(we, to.normalized, out var hit, to.magnitude, ~0, QueryTriggerInteraction.Ignore) && hit.distance < to.magnitude - .14f) continue;
            var eyeT = w.transform.Find("CAPTURE / eye");
            if (eyeT != null) eyeT.localPosition = new Vector3(eye.x, 0f, eye.z);
            w.TickFixturesForTools(0f);
            var path = Path.Combine(outDir, seed + "_" + k.x + "_" + k.y + "_" + step + ".jpg");
            Render(we, root + target, 50f, path);
            var wide = Path.Combine(outDir, seed + "_" + k.x + "_" + k.y + "_" + step + "_wide.jpg");
            var back = corner + d * (dist + 1.1f) + Vector3.up * 1.6f;
            Render(root + back, root + target, 76f, wide);
            Log("  shot " + step + " · seed " + seed + " · post " + k + " (map " + corner.x + ", " + corner.z + ") · empty quadrant " + "SW SE NW NE".Split(' ')[q] + " · eye " + eye.ToString("F2") + " · " + Path.GetFileName(path));
            return true;
        }
        Log("  no free eye for post " + k + " quadrant " + q);
        return false;
    }

    static void Render(Vector3 eyeWorld, Vector3 targetWorld, float fov, string path)
    {
        if (shotCam == null)
        {
            foreach (var c in Object.FindObjectsByType<Camera>(FindObjectsInactive.Include, FindObjectsSortMode.None))
                if (c.name == "First-person camera") { shotCam = c; break; }
            if (shotCam == null) { shotCam = FrontRooms3DGame.CreateCamera(null); shotCam.gameObject.hideFlags = HideFlags.DontSave; }
            shotCam.gameObject.SetActive(true);
            shotCam.enabled = true;
            FrontRoomsPostStack.ConfigureCamera(shotCam);
            var volume = FrontRoomsPostStack.Ensure(null);
            if (volume != null) volume.gameObject.hideFlags = HideFlags.DontSave;
            shotRT = new RenderTexture(ShotW, ShotH, 24, RenderTextureFormat.ARGB32, RenderTextureReadWrite.sRGB) { antiAliasing = 4 };
            shotRT.Create();
            shotTex = new Texture2D(ShotW, ShotH, TextureFormat.RGB24, false);
        }
        var t = shotCam.transform;
        t.SetPositionAndRotation(eyeWorld, Quaternion.LookRotation(targetWorld - eyeWorld, Vector3.up));
        shotCam.fieldOfView = fov;
        shotCam.nearClipPlane = .06f;
        shotCam.farClipPlane = 60f;
        var keep = shotCam.targetTexture;
        shotCam.targetTexture = shotRT;
        shotCam.Render();
        shotCam.Render();
        shotCam.targetTexture = keep;
        var active = RenderTexture.active;
        RenderTexture.active = shotRT;
        shotTex.ReadPixels(new Rect(0, 0, ShotW, ShotH), 0, 0);
        shotTex.Apply(false);
        RenderTexture.active = active;
        File.WriteAllBytes(path, shotTex.EncodeToJPG(90));
    }

    // ------------------------------------------------------------------ helpers
    static FrontRoomsMapWorld BuildWorld(int seed)
    {
        var profile = Object.Instantiate(FrontRoomsLevelProfiles.Resolve());
        profile.hideFlags = HideFlags.DontSave;
        profile.generation.seed = seed;
        owned.Add(profile);
        var root = new GameObject("CX CONTRACT / seed " + seed) { hideFlags = HideFlags.DontSave };
        root.transform.position = RootPosition;
        var world = root.AddComponent<FrontRoomsMapWorld>();
        world.Profile = profile;
        world.BuildForCapture();
        world.TickFixturesForTools(0f);
        Physics.SyncTransforms();
        return world;
    }

    static void Free(FrontRoomsMapWorld w)
    {
        if (w == null) return;
        var go = w.gameObject;
        w.Release();
        Object.DestroyImmediate(go);
        Physics.SyncTransforms();
    }

    static List<(GridCoord coord, Transform root)> Chunks(FrontRoomsMapWorld w)
    {
        var list = new List<(GridCoord, Transform)>();
        var built = typeof(FrontRoomsMapWorld).GetField("built", IF).GetValue(w) as IDictionary;
        foreach (DictionaryEntry e in built)
        {
            var rv = e.Value.GetType().GetField("root", IF)?.GetValue(e.Value);
            var root = rv is GameObject g ? g.transform : rv as Transform;
            list.Add(((GridCoord)e.Key, root));
        }
        return list;
    }

    static string R(float v, float q) => Mathf.RoundToInt(v / q).ToString(CultureInfo.InvariantCulture);
    static string RV(Vector3 v, float q) => R(v.x, q) + "," + R(v.y, q) + "," + R(v.z, q);

    static string MeshHash(Mesh m)
    {
        if (m == null) return "null";
        var sb = new StringBuilder();
        sb.Append(m.vertexCount).Append('/').Append(m.subMeshCount).Append('/');
        if (m.isReadable)
        {
            var v = m.vertices;
            long acc = 17;
            for (var i = 0; i < v.Length; i++) acc = acc * 31 + Mathf.RoundToInt(v[i].x * 1e4f) * 7 + Mathf.RoundToInt(v[i].y * 1e4f) * 13 + Mathf.RoundToInt(v[i].z * 1e4f) * 19;
            sb.Append(acc);
            for (var s = 0; s < m.subMeshCount; s++) sb.Append('/').Append(m.GetIndexCount(s));
        }
        else sb.Append(RV(m.bounds.center, .001f)).Append(RV(m.bounds.size, .001f));
        return sb.ToString();
    }

    static string HPath(Transform t, Transform root)
    {
        var parts = new List<string>();
        for (; t != null && t != root; t = t.parent) parts.Add(t.name);
        parts.Reverse();
        return string.Join("/", parts);
    }

    static readonly Dictionary<string, List<string>> staleLines = new Dictionary<string, List<string>>();
    static readonly Dictionary<string, List<string>> freshLines = new Dictionary<string, List<string>>();

    static Dictionary<string, (int count, string hash)> Signatures(FrontRoomsMapWorld w, List<GridCoord> only, string detailTag = null)
    {
        var want = new HashSet<GridCoord>(only);
        var result = new Dictionary<string, (int, string)>();
        foreach (var (coord, root) in Chunks(w))
        {
            if (root == null || !want.Contains(coord)) continue;
            var key = coord.x + "_" + coord.y;
            var rl = new List<string>();
            foreach (var r in root.GetComponentsInChildren<Renderer>(true))
            {
                var mf = r.GetComponent<MeshFilter>();
                var mats = string.Join("+", r.sharedMaterials.Select(m => m == null ? "null" : m.name));
                rl.Add("R|" + HPath(r.transform, root) + "|" + (mf != null ? MeshHash(mf.sharedMesh) : r.GetType().Name) + "|" + mats + "|" + RV(r.bounds.center - w.transform.position, .001f) + "|" + RV(r.bounds.size, .001f) + "|" + r.enabled);
            }
            var ll = new List<string>();
            foreach (var l in root.GetComponentsInChildren<Light>(true))
                ll.Add("L|" + HPath(l.transform, root) + "|" + RV(l.transform.position - w.transform.position, .001f) + "|" + R(l.color.r, .001f) + "," + R(l.color.g, .001f) + "," + R(l.color.b, .001f) + "|" + R(l.intensity, .001f) + "|" + R(l.range, .001f) + "|" + l.enabled);
            var cl = new List<string>();
            foreach (var c in root.GetComponentsInChildren<Collider>(true))
                cl.Add("C|" + HPath(c.transform, root) + "|" + c.GetType().Name + "|" + c.isTrigger + "|" + c.enabled + "|" + RV(c.bounds.center - w.transform.position, .001f) + "|" + RV(c.bounds.size, .001f) + (c is MeshCollider mc ? "|" + MeshHash(mc.sharedMesh) : ""));
            rl.Sort(StringComparer.Ordinal); ll.Sort(StringComparer.Ordinal); cl.Sort(StringComparer.Ordinal);
            result[key + "\trenderers"] = (rl.Count, Hash(rl));
            result[key + "\tlights"] = (ll.Count, Hash(ll));
            result[key + "\tcolliders"] = (cl.Count, Hash(cl));
            var all = rl.Concat(cl).ToList();
            if (detailTag == null) staleLines[key] = all; else freshLines[key] = all;
        }
        return result;
    }

    static int dumps;
    static void DumpDiff(string pass, int seed, string key)
    {
        if (dumps >= 40 || !staleLines.TryGetValue(key, out var a) || !freshLines.TryGetValue(key, out var b)) return;
        dumps++;
        var sa = new HashSet<string>(a); var sb = new HashSet<string>(b);
        var text = new StringBuilder();
        text.AppendLine("pass " + pass + " seed " + seed + " chunk " + key + " · lines only in the still-built chunk (-) / only in its fresh rebuild (+)");
        foreach (var x in a.Where(x => !sb.Contains(x))) text.Append("- ").AppendLine(x);
        foreach (var x in b.Where(x => !sa.Contains(x))) text.Append("+ ").AppendLine(x);
        var dir = Path.Combine(outDir, "diff");
        Directory.CreateDirectory(dir);
        File.WriteAllText(Path.Combine(dir, pass + "_" + seed + "_" + key + ".txt"), text.ToString());
    }

    static string Hash(List<string> lines)
    {
        using (var sha = SHA1.Create())
        {
            var bytes = sha.ComputeHash(Encoding.UTF8.GetBytes(string.Join("\n", lines)));
            return BitConverter.ToString(bytes, 0, 8).Replace("-", "").ToLowerInvariant();
        }
    }
