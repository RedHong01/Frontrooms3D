#!/usr/bin/env python3
"""Recompile a clone's Assembly-CSharp / -Editor with Unity's own Roslyn and response files, for several
platform define sets (approximation: same reference assemblies, only the defines change).
usage: csc_check.py <clone> <outdir>"""
import sys, os, subprocess, re, collections
clone, outdir = sys.argv[1], sys.argv[2]
os.makedirs(outdir, exist_ok=True)
U = '/Applications/Unity/Hub/Editor/6000.3.10f1/Unity.app/Contents'
DOTNET = U + '/Resources/Scripting/NetCoreRuntime/dotnet'
CSC = U + '/Resources/Scripting/DotNetSdkRoslyn/csc.dll'
dag = os.path.join(clone, 'Library/Bee/artifacts')
dag = os.path.join(dag, [d for d in os.listdir(dag) if d.endswith('.dag')][0])
EDITOR = {'UNITY_EDITOR', 'UNITY_EDITOR_64', 'UNITY_EDITOR_OSX', 'UNITY_EDITOR_WIN', 'UNITY_EDITOR_LINUX'}
def variant(lines, name):
    out = []
    for l in lines:
        m = re.match(r'-define:(\S+)', l)
        if m:
            d = m.group(1)
            if name == 'editor_win':
                d = {'UNITY_EDITOR_OSX': 'UNITY_EDITOR_WIN', 'UNITY_STANDALONE_OSX': 'UNITY_STANDALONE_WIN', 'PLATFORM_STANDALONE_OSX': 'PLATFORM_STANDALONE_WIN'}.get(d, d)
            elif name in ('player_webgl', 'player_mac'):
                if d in EDITOR or d.startswith('UNITY_EDITOR'): continue
                if name == 'player_webgl':
                    if d in ('UNITY_STANDALONE_OSX', 'UNITY_STANDALONE', 'PLATFORM_STANDALONE_OSX', 'PLATFORM_STANDALONE'): continue
            out.append('-define:' + d); continue
        if l.startswith('-out:') or l.startswith('-refout:'): continue
        out.append(l)
    if name == 'player_webgl': out += ['-define:UNITY_WEBGL', '-define:PLATFORM_WEBGL']
    return out
summary = []
for asm in ('Assembly-CSharp', 'Assembly-CSharp-Editor'):
    rsp = os.path.join(dag, asm + '.rsp')
    lines = open(rsp).read().splitlines()
    for name in ('editor_mac', 'editor_win', 'player_mac', 'player_webgl'):
        if asm.endswith('Editor') and name.startswith('player'): continue
        v = variant(lines, name)
        vd = os.path.join(outdir, name); os.makedirs(vd, exist_ok=True)
        if asm.endswith('Editor'):
            # reference the runtime assembly compiled with the same defines (same assembly name, so InternalsVisibleTo holds)
            v = [('-r:"' + os.path.join(vd, 'Assembly-CSharp.dll') + '"') if (l.startswith('-r:') and l.rstrip('"').endswith('/Assembly-CSharp.ref.dll')) else l for l in v]
        tmp = os.path.join(vd, f'{asm}.rsp')
        v += [f'-out:{os.path.join(vd, asm + ".dll")}']
        open(tmp, 'w').write('\n'.join(v) + '\n')
        r = subprocess.run([DOTNET, 'exec', CSC, '/nologo', '@' + tmp], cwd=clone, capture_output=True, text=True)
        out = r.stdout + r.stderr
        open(os.path.join(vd, f'{asm}.txt'), 'w').write(out)
        errs = [l for l in out.splitlines() if ': error ' in l]
        warns = [l for l in out.splitlines() if ': warning ' in l]
        summary.append(f'{asm} {name}: exit {r.returncode} · {len(errs)} errors · {len(warns)} warnings')
        for l in (errs + warns)[:30]: summary.append('    ' + l.replace(clone + '/', ''))
open(os.path.join(outdir, 'summary.txt'), 'w').write('\n'.join(summary) + '\n')
print('\n'.join(summary))
