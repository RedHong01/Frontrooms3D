import sys,subprocess,struct,os,tempfile
def scan(path):
    tmp=tempfile.mktemp(suffix='.bmp')
    subprocess.run(['sips','-s','format','bmp',path,'--out',tmp],capture_output=True)
    b=open(tmp,'rb').read(); os.remove(tmp)
    off=struct.unpack_from('<I',b,10)[0]; w,h=struct.unpack_from('<ii',b,18); bpp=struct.unpack_from('<H',b,28)[0]
    flip=h>0; h=abs(h); step=bpp//8; row=(w*step+3)&~3
    n=0; xs=[];ys=[]
    for y in range(0,h,2):
        base=off+y*row
        for x in range(0,w,2):
            i=base+x*step; bl,g,r=b[i],b[i+1],b[i+2]
            if r>170 and bl>170 and g<100: n+=1; xs.append(x); ys.append(h-1-y if flip else y)
    return n,(min(xs),min(ys),max(xs),max(ys)) if n else None
for p in sys.argv[1:]:
    n,bb=scan(p)
    if n: print(f"{p}\tmagenta px(sampled) {n}\tbbox {bb}")
