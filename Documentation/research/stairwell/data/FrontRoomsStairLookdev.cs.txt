/// <summary>
/// CLONE-ONLY look-dev for the C22 stairwell behind the floor's EXIT door (workflow c22-stairwell-look,
/// stage 2 Directions). Spec: Documentation/research/stairwell/10_directions.md; research 01_research.md.
/// Builds, in an empty scene at world (4992, 0, 4992), a code-true 1990 switchback stair in 1 x 2 map
/// cells (clear 2.84 x 5.84 m, floor to floor 3.50 m, 2 x 10 risers of 175 mm, treads 280 mm, flights
/// 1.22 m, well 0.40 m, landings 1.66 m, doors centred in the landing end wall), periodic over floors

///   A  painted block + steel pan stair     B  bare concrete service stair

/// wall (z = 0, stair face 0.08) to the far end wall (face 5.92). The maze is z < 0.
/// ASCII only, no C# tuples (clone editor-script rule).
/// </summary>

    // ------------------------------------------------------------- batch
    public static void RunBatch()
    {
        var args = Environment.GetCommandLineArgs();
        var dirs = "A,B,C";
        string onlyShots = null;
        for (var i = 0; i < args.Length; i++)
        {
            if (args[i] == "-stairDirs" && i + 1 < args.Length) dirs = args[i + 1];
            if (args[i] == "-stairOut" && i + 1 < args.Length) outRoot = args[i + 1];
            if (args[i] == "-stairShots" && i + 1 < args.Length) onlyShots = args[i + 1];
            if (args[i] == "-stairNoE") ePass = false;
            if (args[i] == "-stairNoKits") saveKits = false;
            if (args[i] == "-stairLamp" && i + 1 < args.Length) lampMode = args[i + 1];
            if (args[i] == "-stairLampGain" && i + 1 < args.Length) lampGain = float.Parse(args[i + 1], CultureInfo.InvariantCulture);
            if (args[i] == "-stairNoEndDado") noEndDado = true;
            if (args[i] == "-stairRemedy") remedy = true;
        }
        if (string.IsNullOrEmpty(outRoot)) outRoot = Path.Combine(Directory.GetParent(Application.dataPath).FullName, "Verification", "stair_lookdev");
        EnsureAssets();
        foreach (var d in dirs.Split(','))
        {
            var key = d.Trim().ToUpperInvariant();
            if (key.Length == 0) continue;
            try { RunOne(key, onlyShots); }
            catch (Exception e) { Debug.LogError("[StairLookdev] direction " + key + " failed: " + e); File.AppendAllText(Path.Combine(outRoot, "errors.txt"), key + ": " + e + "\n"); }
        }
        Debug.Log("[StairLookdev] ALL DONE -> " + outRoot);
    }

    static void Log(string s) { log.Add(s); Debug.Log("[StairLookdev] " + dir + " " + s); }

    static void RunOne(string key, string onlyShots)
    {
        dir = key;
        outDir = Path.Combine(outRoot, dir);
        Directory.CreateDirectory(outDir);
        ids.Clear(); mats.Clear(); log.Clear(); stairLights.Clear(); kitGroups.Clear();
        var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
        EditorSceneManager.SetActiveScene(scene);
        root = new GameObject("STAIR LOOKDEV " + dir).transform;
        root.position = RootWorld;
        FrontRoomsLook.ApplyAmbient();
        FrontRoomsPostStack.Ensure(root);
        SetupReflections();
        SceneFill();
        shaft = Group(root, "Stair shaft " + dir);
        mazeRoot = Group(root, "Level 0 maze side");
        kitsRoot = Group(shaft, "floor 0 kits");

        Materials();
        BuildMaze();
        BuildShaft();
        BuildExitDoor();
        BuildRelays();
        Physics.SyncTransforms();

        var shots = Shots();
        RenderAll(shots, onlyShots);
        Stats();
        if (saveKits) { try { SaveKits(); } catch (Exception e) { Log("kit save failed: " + e.Message); } }
        File.WriteAllLines(Path.Combine(outDir, "log.txt"), log);
        Debug.Log("[StairLookdev] " + dir + " done -> " + outDir);
    }

    static void EnsureAssets()
    {
        foreach (var a in new[] { "Kit_EmergencyTwinHead", "Kit_EmergencyTwinHead_Yoke", "Kit_EmergencyTwinHead_Lamp", "Kit_WrapFixture_4ft" })

        }
        foreach (var t in new[] { "Stair_Concrete", "Stair_ConcreteFloor" })
            foreach (var s in new[] { "_A", "_N", "_S" })
                AssetDatabase.ImportAsset(TexDir + t + s + ".png", ImportAssetOptions.ForceUpdate);
        FrontRoomsKitLibrary.ClearCache();
    }

    static Texture2D T(string stem) { return AssetDatabase.LoadAssetAtPath<Texture2D>(TexDir + stem + ".png"); }

    static void ConfigureSurface(Material m, string tex, Vector2 tile, Color tint, bool meshUV, float smooth, float bump, float macroTone, float macroDirt)
    {
        m.SetTexture("_BaseMap", tex != null ? T(tex + "_A") : null);
        m.SetTexture("_BumpMap", tex != null ? T(tex + "_N") : null);
        m.SetTexture("_MaskMap", tex != null ? T(tex + "_S") : null);
        if (tex != null && T(tex + "_A") == null) Log("WARNING texture missing " + tex);
        m.SetColor("_BaseColor", tint);
        m.SetTextureScale("_BaseMap", Vector2.one); m.SetTextureOffset("_BaseMap", Vector2.zero);
        m.SetVector("_TileSize", new Vector4(tile.x, tile.y, 0, 0));
        m.SetFloat("_BumpScale", bump);
        m.SetFloat("_Smoothness", smooth);
        m.SetFloat("_Metallic", 0f);
        m.SetFloat("_OcclusionStrength", .8f);
        m.SetTexture("_MacroMap", T("MacroWear_M"));
        m.SetFloat("_MacroTone", macroTone);
        m.SetFloat("_MacroDirt", macroDirt);
        m.SetColor("_StainColor", new Color(.42f, .38f, .30f));
        m.SetFloat("_StainStrength", 0f);
        m.SetFloat("_WetStrength", 0f);
        // The shader's floor / ceiling grime bands key on WORLD y (FrontRoomsSurface.shader:455); a stair
        // spans many floors, so every stair surface keeps them at 0.
        m.SetFloat("_FloorGrime", 0f);
        m.SetFloat("_CeilingGrime", 0f);
        m.SetFloat("_CeilingHeight", MazeH);
        m.SetFloat("_MeshUV", meshUV ? 1f : 0f);
        if (meshUV) m.EnableKeyword("_FR_MESH_UV"); else m.DisableKeyword("_FR_MESH_UV");
        m.SetFloat("_UseEmission", 0f); m.DisableKeyword("_EMISSION");
    }

    /// <summary>World-planar FrontRooms/Surface (walls, floors, concrete).</summary>
    static Material Surf(string name, string tex, float tile, Color tint, float smooth = 1f, float bump = 1f, float macroTone = .2f, float macroDirt = .2f)
    {
        Material c;
        if (mats.TryGetValue(name, out c)) return c;
        var m = new Material(Shader.Find("FrontRooms/Surface")) { name = name };
        ConfigureSurface(m, tex, new Vector2(tile, tile), tint, false, smooth, bump, macroTone, macroDirt);
        mats[name] = m;
        return m;
    }

    /// <summary>Flat-tint FrontRooms/Surface with mesh UVs (props shade like the shell).</summary>
    static Material Flat(string name, Color c, float smooth = .3f, float metal = 0f)
    {
        Material cached;
        if (mats.TryGetValue(name, out cached)) return cached;
        var m = new Material(Shader.Find("FrontRooms/Surface")) { name = name };
        ConfigureSurface(m, null, Vector2.one, c, true, smooth, 0f, 0f, 0f);
        m.SetFloat("_Metallic", metal);
        mats[name] = m;
        return m;
    }

    static Material Glow(string name, Color c, Color emission)
    {
        Material cached;
        if (mats.TryGetValue(name, out cached)) return cached;
        var m = new Material(Shader.Find("Universal Render Pipeline/Lit")) { name = name };
        m.SetColor("_BaseColor", c);
        m.SetFloat("_Smoothness", .2f);
        m.EnableKeyword("_EMISSION");
        m.SetColor("_EmissionColor", emission);
        m.globalIlluminationFlags = MaterialGlobalIlluminationFlags.RealtimeEmissive;
        mats[name] = m;
        return m;
    }

    /// <summary>A copy of a project surface with the world-y grime bands off (for stair walls below y 0).</summary>
    static Material NoGrime(Material src, string name)
    {
        Material cached;
        if (mats.TryGetValue(name, out cached)) return cached;
        var m = new Material(src) { name = name };
        m.SetFloat("_FloorGrime", 0f);
        m.SetFloat("_CeilingGrime", 0f);
        mats[name] = m;
        return m;
    }

    static Cubemap flatCube;
    static void SetupReflections()
    {
        flatCube = new Cubemap(16, TextureFormat.RGBAHalf, false) { name = "Stair_FlatAmbientCube" };
        var sky = FrontRoomsLook.AmbientSky.linear; var eq = FrontRoomsLook.AmbientEquator.linear; var gr = FrontRoomsLook.AmbientGround.linear;
        foreach (CubemapFace f in Enum.GetValues(typeof(CubemapFace)))
        {
            if (f == CubemapFace.Unknown) continue;
            var col = f == CubemapFace.PositiveY ? sky : f == CubemapFace.NegativeY ? gr : eq;
            var px = new Color[16 * 16];
            for (var i = 0; i < px.Length; i++)
            {
                var row = i / 16;
                var c = col;
                if (f != CubemapFace.PositiveY && f != CubemapFace.NegativeY) c = row < 8 ? Color.Lerp(sky, eq, row / 8f) : Color.Lerp(eq, gr, (row - 8) / 8f);
                px[i] = c;
            }
            flatCube.SetPixels(px, f);
        }
        flatCube.Apply();
        RenderSettings.defaultReflectionMode = DefaultReflectionMode.Custom;
        RenderSettings.customReflectionTexture = flatCube;
    }

    /// <summary>The shipped scene's "Soft ambient direction" (FrontRooms3D.unity): directional 0.16,
    /// rgb (.769, .816, .800), soft shadows at strength 0.18, so 82 % of it reaches every interior.</summary>
    static void SceneFill()
    {
        var go = new GameObject("Soft ambient direction (scene copy)");
        go.transform.SetParent(root, false);
        go.transform.rotation = new Quaternion(0.5540323f, -0.21201216f, 0.1484525f, 0.7912401f);
        var l = go.AddComponent<Light>();
        l.type = LightType.Directional;
        l.color = new Color(0.76862746f, 0.8156863f, 0.8f);
        l.intensity = .16f;
        l.shadows = LightShadows.Soft;
        l.shadowStrength = .18f;
        l.shadowBias = .045f; l.shadowNormalBias = .28f; l.shadowNearPlane = .1f;
    }

    static Renderer Box(Transform parent, string name, Vector3 centre, Vector3 size, Material m, Id id)
    {
        var go = GameObject.CreatePrimitive(PrimitiveType.Cube);
        go.name = name;
        go.transform.SetParent(parent, false);
        go.transform.position = root.TransformPoint(centre);
        go.transform.rotation = root.rotation;
        go.transform.localScale = size;
        UnityEngine.Object.DestroyImmediate(go.GetComponent<Collider>());
        var r = go.GetComponent<MeshRenderer>();
        r.sharedMaterial = m;
        ids[r] = id;
        return r;
    }

    /// <summary>Box between two points (centre line a..b), w across (horizontal), h perpendicular.</summary>
    static Renderer Seg(Transform parent, string name, Vector3 a, Vector3 b, float w, float h, Material m, Id id)
    {
        var go = GameObject.CreatePrimitive(PrimitiveType.Cube);
        go.name = name;
        go.transform.SetParent(parent, false);
        var d = b - a;
        go.transform.position = root.TransformPoint((a + b) * .5f);
        var up = Mathf.Abs(Vector3.Dot(d.normalized, Vector3.up)) > .98f ? Vector3.forward : Vector3.up;
        go.transform.rotation = Quaternion.LookRotation(d, up);
        go.transform.localScale = new Vector3(w, h, d.magnitude);
        UnityEngine.Object.DestroyImmediate(go.GetComponent<Collider>());
        var r = go.GetComponent<MeshRenderer>();
        r.sharedMaterial = m;
        ids[r] = id;
        return r;
    }

    static Renderer Cyl(Transform parent, string name, Vector3 a, Vector3 b, float radius, Material m, Id id)
    {
        var go = GameObject.CreatePrimitive(PrimitiveType.Cylinder);
        go.name = name;
        go.transform.SetParent(parent, false);
        go.transform.position = root.TransformPoint((a + b) * .5f);
        go.transform.rotation = Quaternion.FromToRotation(Vector3.up, (b - a).normalized);
        go.transform.localScale = new Vector3(radius * 2f, (b - a).magnitude * .5f, radius * 2f);
        UnityEngine.Object.DestroyImmediate(go.GetComponent<Collider>());
        var r = go.GetComponent<MeshRenderer>();
        r.sharedMaterial = m;
        ids[r] = id;
        return r;
    }

    static Light Spot(Transform parent, string name, Vector3 pos, Vector3 aim, float outer, float inner, Color c, float intensity, float range, bool shadows)
    {
        var go = new GameObject(name);
        go.transform.SetParent(parent, false);
        go.transform.position = root.TransformPoint(pos);
        go.transform.rotation = Quaternion.LookRotation(aim.normalized, Mathf.Abs(aim.normalized.y) > .98f ? Vector3.forward : Vector3.up);
        var l = go.AddComponent<Light>();
        l.type = LightType.Spot;
        l.spotAngle = outer; l.innerSpotAngle = inner;
        l.color = c; l.intensity = intensity; l.range = range;
        l.shadows = shadows ? LightShadows.Soft : LightShadows.None;
        l.shadowStrength = .92f; l.shadowNearPlane = .05f;
        return l;
    }

    static GameObject KitSpawn(string asset, Transform parent, Vector3 pos, Quaternion rot, Id id)
    {
        var go = FrontRoomsKitLibrary.Spawn(asset, parent, Vector3.zero, Quaternion.identity, null, false);
        if (go == null) { Log("MISSING kit asset " + asset); return null; }
        go.transform.position = root.TransformPoint(pos);
        go.transform.rotation = root.rotation * rot;
        foreach (var r in go.GetComponentsInChildren<Renderer>(true)) ids[r] = id;
        return go;
    }

    static int SlotIndex(Renderer r, string slot)
    {
        var m = r.sharedMaterials;
        for (var i = 0; i < m.Length; i++) if (m[i] != null && m[i].name.Replace(" (Instance)", "").StartsWith(slot)) return i;
        return -1;
    }

    static void Recolour(GameObject go, Material m)
    {
        if (go == null || m == null) return;
        foreach (var r in go.GetComponentsInChildren<Renderer>(true))
        {
            var arr = r.sharedMaterials;
            if (arr.Length > 0) arr[0] = m;
            r.sharedMaterials = arr;
        }
    }

    // ---------------------------------------------------------------- mesh builder
    class MB
    {
        public readonly List<Vector3> v = new List<Vector3>();
        public readonly List<Vector3> n = new List<Vector3>();
        public readonly List<Vector2> uv = new List<Vector2>();
        public readonly List<int> tri = new List<int>();

        public void Quad(Vector3 a, Vector3 b, Vector3 c, Vector3 d, Vector3 normal)
        {
            var i = v.Count;
            var t = Mathf.Abs(normal.y) > .5f ? Vector3.right : Vector3.Cross(Vector3.up, normal).normalized;
            var bt = Vector3.Cross(normal, t);
            foreach (var p in new[] { a, b, c, d }) { v.Add(p); n.Add(normal); uv.Add(new Vector2(Vector3.Dot(p, t), Vector3.Dot(p, bt))); }
            Tri(i, i + 1, i + 2, normal);
            Tri(i, i + 2, i + 3, normal);
        }

        void Tri(int a, int b, int c, Vector3 normal)
        {
            if (Vector3.Dot(Vector3.Cross(v[b] - v[a], v[c] - v[a]), normal) < 0f) { var s = b; b = c; c = s; }
            tri.Add(a); tri.Add(b); tri.Add(c);
        }

        public Mesh Build(string name)
        {
            var m = new Mesh { name = name };
            if (v.Count > 65000) m.indexFormat = IndexFormat.UInt32;
            m.SetVertices(v); m.SetNormals(n); m.SetUVs(0, uv); m.SetTriangles(tri, 0);
            m.RecalculateTangents(); m.RecalculateBounds();
            return m;
        }
    }

    static Renderer MeshObj(Transform parent, string name, Mesh mesh, Material m, Id id)
    {
        var go = new GameObject(name);
        go.transform.SetParent(parent, false);
        go.transform.position = root.position;
        go.transform.rotation = root.rotation;
        go.AddComponent<MeshFilter>().sharedMesh = mesh;
        var r = go.AddComponent<MeshRenderer>();
        r.sharedMaterial = m;
        ids[r] = id;
        return r;
    }

    // ---------------------------------------------------------------- the shaft
    static void BuildShaft()
    {
        var yTop = YD(FloorMin) + 3.3f;
        var yBot = YD(FloorMax + 1) - .4f;
        var h = yTop - yBot;
        var yc = (yTop + yBot) * .5f;
        // long walls (faces x = -+1.42) and far wall (face z 5.92); 0.16 thick on the cell lines
        // side walls start at z 0 (the maze face of the door wall is at z -0.08; a wall end there would show in the maze)
        Box(shaft, "shaft wall W", new Vector3(-1.5f, yc, 3.04f), new Vector3(.16f, h, 6.08f), wallM, Id.Background);

        Box(shaft, "shaft wall far", new Vector3(0f, yc, 6f), new Vector3(3.16f, h, .16f), wallM, Id.Background);
        Box(shaft, "shaft roof slab", new Vector3(0f, yTop + .1f, 3f), new Vector3(3.16f, .2f, 6.16f), landBodyM, Id.Background);
        // door wall, stair layer z 0..0.08, one segment per floor [YD(n), YD(n) + FF] with a 1.0 x 2.1 opening
        for (var n = FloorMin; n <= FloorMax + 1; n++)
        {
            var yd = YD(n);
            var s0 = n == FloorMax + 1 ? yBot : yd;
            var s1 = n == FloorMin ? yTop : yd + FF;
            Box(shaft, "door wall L f" + n, new Vector3(-1.0f, (s0 + s1) * .5f, .04f), new Vector3(1.0f, s1 - s0, .08f), wallM, Id.Background);
            Box(shaft, "door wall R f" + n, new Vector3(1.0f, (s0 + s1) * .5f, .04f), new Vector3(1.0f, s1 - s0, .08f), wallM, Id.Background);
            Box(shaft, "door wall header f" + n, new Vector3(0f, (yd + 2.1f + s1) * .5f, .04f), new Vector3(1.0f, s1 - yd - 2.1f, .08f), wallM, Id.Background);
            if (n == FloorMax + 1) Box(shaft, "door wall sill pit", new Vector3(0f, (s0 + yd) * .5f, .04f), new Vector3(1.0f, yd - s0, .08f), wallM, Id.Background);
        }
        // the far side of each lower door: the new floor (not modelled) -> a closed leaf is enough
        for (var n = FloorMin; n <= FloorMax; n++) BuildFloor(n);

    }

    static void BuildFloorBottom(int n)
    {
        var y = YD(n);
        Box(shaft, "pit floor slab", new Vector3(0f, y - .2f, 3f), new Vector3(2.84f, .4f, 5.84f), landBodyM, Id.OtherFloor);
        Box(shaft, "pit floor top", new Vector3(0f, y - .006f, 3f), new Vector3(2.84f, .012f, 5.84f), landTopM, Id.OtherFloor);
        StairDoor(n);
        FloorNumber(n);
        LandingLight(n, true);
    }

    static void BuildFloor(int n)
    {
        var yd = YD(n); var yt = YT(n);
        var pathD = n == 0 || n == 1;
        // landings: D (door) z 0.08..1.74 and T (turn) z 4.26..5.92, full width 2.84
        Landing(n, "D", yd, ZNear, ZD, pathD ? Id.Walk : Id.OtherFloor);
        Landing(n, "T", yt, ZT, ZFarFace, n == 0 ? Id.Walk : Id.OtherFloor);
        // flight 1: D(n) -> T(n), x -1.42..-0.20, descends along +z ; flight 2: T(n) -> D(n+1), x 0.20..1.42, along -z
        Flight(n, 1, -XFace, -WellHalf, ZD, yd, 1f);
        Flight(n, 2, WellHalf, XFace, ZT, yt, -1f);
        Guards(n);
        WallRails(n);
        if (dir == "A" || (dir == "B" && remedy)) Dado(n);
        if (dir == "C") Skirts(n);
        StairDoor(n);
        FloorNumber(n);
        LandingLight(n, true);
        LandingLight(n, false);
        if (dir != "C") Services(n);
        if (n == 0) Gate(n, true);
        if (n == 1) Gate(n, false);
    }

    static void Flight(int n, int f, float x0, float x1, float zTop, float yTop, float dz)
    {
        var p = Kit("Flight", n);
        var walk = OnPath(n, f) ? Id.Walk : Id.OtherFloor;
        var w = x1 - x0; var xc = (x0 + x1) * .5f;
        if (dir == "A")
        {
            // steel pan stair: channel stringers, pans with 40 mm concrete fill, closed plate risers, open soffit
            for (var k = 1; k <= Treads; k++)
            {
                var y = yTop - k * Rise;
                var za = Zk(zTop, dz, k - 1); var zb = Zk(zTop, dz, k);
                var zc = (za + zb) * .5f + dz * .0125f;   // fill reaches 25 mm onto the nosing
                Box(p, "tread fill f" + n + "." + f + "." + k, new Vector3(xc, y - .02f, zc), new Vector3(w - .06f, .04f, Going + .025f), treadM, walk);
                Box(p, "tread pan f" + n + "." + f + "." + k, new Vector3(xc, y - .045f, zc), new Vector3(w - .06f, .01f, Going + .025f), steelM, Id.Background);
                Box(p, "pan nosing lip f" + n + "." + f + "." + k, new Vector3(xc, y - .02f, zb + dz * .025f), new Vector3(w - .06f, .045f, .006f), nosingM, Id.Background);
            }
            for (var k = 0; k <= Treads; k++)
            {
                var zr = Zk(zTop, dz, k) + dz * .003f;
                var yr = yTop - k * Rise - Rise * .5f;
                Box(p, "riser plate f" + n + "." + f + "." + k, new Vector3(xc, yr - .02f, zr), new Vector3(w - .06f, Rise, .005f), steelM, Id.Background);
            }
            // stringers (C10 channel ~ 250 mm), along the pitch line, tops 60 mm above the nosings
            var pitch = Rise / Going;
            var a = new Vector3(0f, yTop + .1f * pitch - .065f, zTop - dz * .1f);
            var b = new Vector3(0f, yTop - Treads * Rise - .1f * pitch - .065f, Zk(zTop, dz, Treads) + dz * .1f);
            foreach (var sx in new[] { x0 + .03f, x1 - .03f })
            {
                a.x = sx; b.x = sx;
                Seg(p, "stringer f" + n + "." + f, a, b, .05f, .25f, steelM, Id.Background);
            }
        }
        else
        {
            // monolithic flight (cast-in-place concrete in B; the same solid, carpeted and plastered, in C)
            var treadsMB = new MB(); var risersMB = new MB(); var bodyMB = new MB();
            var cosA = Going / Mathf.Sqrt(Going * Going + Rise * Rise);
            var wv = .12f / cosA;
            Func<float, float> ys = z => yTop - Rise * Mathf.Abs(z - zTop) / Going - Rise - wv;
            for (var k = 1; k <= Treads; k++)
            {
                var y = yTop - k * Rise;
                var za = Zk(zTop, dz, k - 1); var zb = Zk(zTop, dz, k);
                treadsMB.Quad(new Vector3(x0, y, za), new Vector3(x1, y, za), new Vector3(x1, y, zb), new Vector3(x0, y, zb), Vector3.up);
                foreach (var sx in new[] { x0, x1 })
                    bodyMB.Quad(new Vector3(sx, y, za), new Vector3(sx, y, zb), new Vector3(sx, ys(zb), zb), new Vector3(sx, ys(za), za), sx == x0 ? Vector3.left : Vector3.right);
                // riser k at zb, from y down to y - Rise, facing +dz
                risersMB.Quad(new Vector3(x0, y, zb), new Vector3(x1, y, zb), new Vector3(x1, y - Rise, zb), new Vector3(x0, y - Rise, zb), new Vector3(0f, 0f, dz));
            }
            // riser 0 (the landing edge) carries the finish too, 1 mm proud of the slab edge
            var z0r = zTop + dz * .001f;
            risersMB.Quad(new Vector3(x0, yTop, z0r), new Vector3(x1, yTop, z0r), new Vector3(x1, yTop - Rise, z0r), new Vector3(x0, yTop - Rise, z0r), new Vector3(0f, 0f, dz));
            var z9 = Zk(zTop, dz, Treads);
            var sl = new Vector3(0f, -Rise * Treads, z9 - zTop).normalized;
            var sn = Vector3.Cross(Vector3.right, sl); if (sn.y > 0f) sn = -sn;
            bodyMB.Quad(new Vector3(x0, ys(zTop), zTop), new Vector3(x1, ys(zTop), zTop), new Vector3(x1, ys(z9), z9), new Vector3(x0, ys(z9), z9), sn);
            bodyMB.Quad(new Vector3(x0, yTop - Rise, zTop), new Vector3(x1, yTop - Rise, zTop), new Vector3(x1, ys(zTop), zTop), new Vector3(x0, ys(zTop), zTop), new Vector3(0f, 0f, -dz));
            var tag = "f" + n + "." + f;
            MeshObj(p, "flight treads " + tag, treadsMB.Build("treads " + dir + tag), treadM, walk);
            MeshObj(p, "flight risers " + tag, risersMB.Build("risers " + dir + tag), riserM, Id.Background);
            MeshObj(p, "flight body " + tag, bodyMB.Build("body " + dir + tag), bodyM, Id.Background);
            // nosings: B safety-yellow 50 mm stripes, C brass nosing bars over the carpet
            for (var k = 1; k <= Treads; k++)
            {
                var y = yTop - k * Rise;
                var zb = Zk(zTop, dz, k);
                if (dir == "B") Box(p, "nosing stripe " + tag + "." + k, new Vector3(xc, y + .0006f, zb - dz * .025f), new Vector3(w - .02f, .002f, .05f), nosingM, Id.Background);
                else Box(p, "brass nosing " + tag + "." + k, new Vector3(xc, y + .004f, zb - dz * .012f), new Vector3(w - .04f, .012f, .028f), nosingM, Id.Background);
            }
        }
    }

    /// <summary>Guard on the well side of each flight and across the well at each landing; newels.</summary>
    static void Guards(int n)
    {
        var p = Kit("Guard", n);
        // flight 1 well edge x = -0.20 (rail centre -0.23), flight 2 well edge x = +0.20
        GuardRun(p, n, 1, -WellHalf - .03f, ZD, YD(n), 1f);
        GuardRun(p, n, 2, WellHalf + .03f, ZT, YT(n), -1f);
        // across the well at the landing edges (42 in = 1.067 m)
        LandingGuard(p, n, "D", ZD - .03f, YD(n));
        LandingGuard(p, n, "T", ZT + .03f, YT(n));
    }

    static void GuardRun(Transform p, int n, int f, float x, float zTop, float yTop, float dz)
    {
        var tag = "f" + n + "." + f;
        var zBot = Zk(zTop, dz, Treads);                       // last nosing line
        var yBotNose = yTop - Treads * Rise;
        var railH = .91f;                                     // 34-38 in above the nosing line
        var a = new Vector3(x, yTop + railH, zTop); var b = new Vector3(x, yBotNose + railH, zBot);
        if (dir == "A")
        {
            Cyl(p, "guard rail " + tag, a, b, .021f, railM, Id.Background);
            Seg(p, "guard rail wear " + tag, a + Vector3.up * .017f, b + Vector3.up * .017f, .012f, .006f, railTopM, Id.Background);
            var ba = new Vector3(x, yTop + .10f, zTop); var bb = new Vector3(x, yBotNose + .10f, zBot);
            Seg(p, "guard bottom rail " + tag, ba, bb, .025f, .025f, steelM, Id.Background);
            var count = Mathf.FloorToInt(Mathf.Abs(zBot - zTop) / .11f);
            for (var i = 1; i < count; i++)
            {
                var t = i / (float)count;
                var top = Vector3.Lerp(a, b, t) - Vector3.up * .02f; var bot = Vector3.Lerp(ba, bb, t);
                Box(p, "picket " + tag + "." + i, (top + bot) * .5f, new Vector3(.0127f, (top - bot).magnitude, .0127f), steelM, Id.Background);
            }
        }
        else if (dir == "B")
        {
            Cyl(p, "guard top rail " + tag, a, b, .024f, railM, Id.Background);
            Seg(p, "guard rail polish " + tag, a + Vector3.up * .02f, b + Vector3.up * .02f, .012f, .006f, railTopM, Id.Background);
            var ma = new Vector3(x, yTop + .48f, zTop); var mb = new Vector3(x, yBotNose + .48f, zBot);
            Cyl(p, "guard mid rail " + tag, ma, mb, .024f, steelM, Id.Background);
            for (var i = 0; i <= 2; i++)
            {
                var t = i / 2f;
                var zp = Mathf.Lerp(zTop, zBot, t) - dz * (i == 0 ? -.06f : i == 2 ? .06f : 0f);
                var yNose = yTop - Rise * Mathf.Abs(zp - zTop) / Going;
                var yTread = yTop - Rise * Mathf.Ceil(Mathf.Abs(zp - zTop) / Going - .001f);
                Cyl(p, "guard post " + tag + "." + i, new Vector3(x, yTread, zp), new Vector3(x, yNose + railH + .02f, zp), .024f, steelM, Id.Background);
                Cyl(p, "post flange " + tag + "." + i, new Vector3(x, yTread, zp), new Vector3(x, yTread + .012f, zp), .055f, steelM, Id.Background);
            }
        }
        else
        {
            // oak: moulded handrail on square balusters (2 per tread), turned-ish newels at the ends
            Seg(p, "oak handrail " + tag, a + Vector3.up * .02f, b + Vector3.up * .02f, .062f, .055f, railM, Id.Background);
            Seg(p, "oak shoe rail " + tag, new Vector3(x, yTop + .02f, zTop), new Vector3(x, yBotNose + .02f, zBot), .06f, .03f, steelM, Id.Background);
            for (var k = 1; k <= Treads; k++)
                for (var j = 0; j < 2; j++)
                {
                    var zc = Zk(zTop, dz, k - 1) + dz * Going * (.25f + .5f * j);
                    var yTread = yTop - k * Rise;
                    var yRail = yTop + railH - Rise * Mathf.Abs(zc - zTop) / Going;
                    Box(p, "baluster " + tag + "." + k + "." + j, new Vector3(x, (yTread + yRail) * .5f, zc), new Vector3(.032f, yRail - yTread, .032f), steelM, Id.Background);
                }
            foreach (var e in new[] { 0, 1 })
            {
                var zc = e == 0 ? zTop - dz * .05f : zBot + dz * .05f;

    /// <summary>A: the darker dado band, 2 mm proud, following landings and flights (1.1 m above them).</summary>
    static void Dado(int n)
    {
        var p = Kit("Paint", n);
        var mb = new MB();
        var top = dadoTop; var under = .35f;
        var yd = YD(n); var yt = YT(n);
        foreach (var side in new[] { -1, 1 })
        {
            var x = side * (XFace - .002f);
            var nrm = side < 0 ? Vector3.right : Vector3.left;
            // D landing and T landing, flat
            mb.Quad(new Vector3(x, yd - .06f, ZNear), new Vector3(x, yd - .06f, ZD), new Vector3(x, yd + top, ZD), new Vector3(x, yd + top, ZNear), nrm);
            mb.Quad(new Vector3(x, yt - .06f, ZT), new Vector3(x, yt - .06f, ZFarFace), new Vector3(x, yt + top, ZFarFace), new Vector3(x, yt + top, ZT), nrm);
            // west wall carries flight 1 (D n -> T n); east wall carries flight 2 (T n -> D n+1)
            if (side < 0) mb.Quad(new Vector3(x, yd - under, ZD), new Vector3(x, yt - under, ZT), new Vector3(x, yt + top, ZT), new Vector3(x, yd + top, ZD), nrm);
            else mb.Quad(new Vector3(x, yt - under, ZT), new Vector3(x, YD(n + 1) - under, ZD), new Vector3(x, YD(n + 1) + top, ZD), new Vector3(x, yt + top, ZT), nrm);
        }
        // end walls: door wall at the D landing (beside the door), far wall at the T landing
        var zn = ZNear + .002f; var zf = ZFarFace - .002f;
        if (!noEndDado)
        {
            mb.Quad(new Vector3(-XFace, yd - .06f, zn), new Vector3(-.5f, yd - .06f, zn), new Vector3(-.5f, yd + top, zn), new Vector3(-XFace, yd + top, zn), Vector3.forward);

        }
        MeshObj(p, "dado band f" + n, mb.Build("dado " + n), dadoM, Id.Background);
        // a thin dark line at the band top (the cut-in edge)
    }

    static GameObject StairDoor(int n)
    {
        var p = Kit("Door", n);

        var origin = new Vector3(.5f, y, 0f);   // hinge-jamb edge on the wall centre line
        var frameKit = dir == "C" ? "Kit_DoorFrame_Wood" : "Kit_DoorFrame_Steel";
        var leafKit = dir == "C" ? "Kit_DoorLeaf_Veneer" : "Kit_DoorLeaf_Steel";
        var frame = KitSpawn(frameKit, p, origin, DoorRot, Id.Door);
        var leaf = KitSpawn(leafKit, p, origin, DoorRot, Id.Door);
        if (leaf == null) return null;
        if (n == 0) exitLeaf = leaf.transform;
        if (doorLeafM != null) Recolour(leaf, doorLeafM);
        if (dir != "C" && frame != null) Recolour(frame, Flat(dir + " frame enamel", dir == "A" ? Hex("5F5A50") : Hex("3E3833"), .45f));
        // closer on the pull face (stair side): body at closer_mount_s, regular arm to the frame shoe
        Closer(leaf.transform);
        // stair-side signs (placeholders; strings and type are not ours): floor sign plate at 5 ft on the leaf,
        // the re-entry plate under it

    }

    static void Closer(Transform leaf)
    {
        var body = FrontRoomsKitLibrary.Spawn("Kit_DoorCloser_Body", leaf, new Vector3(.022f, 2.0275f, .545f), Quaternion.Euler(0f, 90f, 0f), null, false);
        if (body == null) { Log("MISSING closer body"); return; }
        foreach (var r in body.GetComponentsInChildren<Renderer>(true)) ids[r] = Id.Door;
        // regular arm: spindle door (0.062, 2.068, 0.520) -> elbow -> shoe on the frame head (0.125, 2.13, 0.10)
        var s = new Vector2(.062f, .520f); var shoe = new Vector2(.125f, .100f);
        var d = (shoe - s).magnitude;
        var aLen = .24f; var fLen = .26f;
        var along = (aLen * aLen - fLen * fLen + d * d) / (2f * d);
        var hgt = Mathf.Sqrt(Mathf.Max(0f, aLen * aLen - along * along));
        var u = (shoe - s) / d; var perp = new Vector2(u.y, -u.x); if (perp.x < 0f) perp = -perp;
        var e = s + u * along + perp * hgt;
        var arm = FrontRoomsKitLibrary.Spawn("Kit_DoorCloser_Arm", leaf, new Vector3(s.x, 2.068f, s.y), Quaternion.Euler(0f, Mathf.Atan2(-(e.y - s.y), e.x - s.x) * Mathf.Rad2Deg, 0f), null, false);
        var fore = FrontRoomsKitLibrary.Spawn("Kit_DoorCloser_Forearm", leaf, new Vector3(e.x, 2.08f, e.y), Quaternion.Euler(0f, Mathf.Atan2(-(shoe.y - e.y), shoe.x - e.x) * Mathf.Rad2Deg, 0f), null, false);
        var sh = FrontRoomsKitLibrary.Spawn("Kit_DoorCloser_Shoe", leaf, new Vector3(shoe.x, 2.13f, shoe.y), Quaternion.identity, null, false);
        foreach (var g in new[] { arm, fore, sh }) if (g != null) foreach (var r in g.GetComponentsInChildren<Renderer>(true)) ids[r] = Id.Door;
    }

    /// <summary>Seven-segment placeholder digit on the door wall, read from the stair side (+z): its right is -x.</summary>
    static void Digit(Transform p, int d, Vector3 bl, float h, float s, Material m, bool stencil, string tag)
    {
        var w = h * .58f;
        // segments a b c d e f g
        var on = new[] {
            new[] { true, true, true, true, true, true, false }, new[] { false, true, true, false, false, false, false },
            new[] { true, true, false, true, true, false, true }, new[] { true, true, true, true, false, false, true },
            new[] { false, true, true, false, false, true, true }, new[] { true, false, true, true, false, true, true },
            new[] { true, false, true, true, true, true, true }, new[] { true, true, true, false, false, false, false },
            new[] { true, true, true, true, true, true, true }, new[] { true, true, true, true, false, true, true } }[Mathf.Clamp(d, 0, 9)];
        var gap = stencil ? s * .55f : 0f;
        var hh = h * .5f;
        // centre, size (x, y) of each segment in the digit's plane (x right, y up from bottom-left)
        var cs = new[] {
            new Vector4(w * .5f, h - s * .5f, w - 2f * gap, s), new Vector4(w - s * .5f, hh + hh * .5f, s, hh - gap * 2f),
            new Vector4(w - s * .5f, hh * .5f, s, hh - gap * 2f), new Vector4(w * .5f, s * .5f, w - 2f * gap, s),
            new Vector4(s * .5f, hh * .5f, s, hh - gap * 2f), new Vector4(s * .5f, hh + hh * .5f, s, hh - gap * 2f),
            new Vector4(w * .5f, hh, w - 2f * gap - (stencil ? 0f : 0f), s) };
        for (var i = 0; i < 7; i++)
        {
            if (!on[i]) continue;
            Box(p, "numeral seg " + tag + "." + i, bl + new Vector3(-cs[i].x, cs[i].y, .001f), new Vector3(cs[i].z, cs[i].w, .003f), m, Id.Sign);
        }
    }

    // ---------------------------------------------------------------- lights
    static Material wrapLensLit, globeLit, pilotRed;

    /// <summary>One normal fixture per landing (D and T). A: 4 ft 2-lamp wraparound on the soffit; B: caged
    /// vapour-tight incandescent globe on the wall; C: the maze's own troffer in a 2.9 m drop ceiling.</summary>
    static void LandingLight(int n, bool dLanding)
    {
        var p = Kit("Light", n);
        var y = dLanding ? YD(n) : YT(n);
        var shadows = n >= -1 && n <= 1;
        var zc = dLanding ? (ZNear + ZD) * .5f : (ZT + ZFarFace) * .5f;
        var edge = lampMode == "edge";
        var zEdge = dLanding ? ZNear + .30f : ZFarFace - .30f;
        var name = (dLanding ? "D" : "T") + n;
        Light l = null;
        if (dir == "A")
        {
            var soffit = y + FF - .15f;    // underside of the landing above (steel pan landing ~0.15 deep)
            var zl = edge ? zEdge : zc;
            var go = KitSpawn("Kit_WrapFixture_4ft", p, new Vector3(0f, soffit, zl), Quaternion.identity, Id.Emitter);
            if (go != null)
            {
                var r = go.GetComponentInChildren<Renderer>();
                r.shadowCastingMode = ShadowCastingMode.Off;
                var li = SlotIndex(r, "Run_WrapLens");
                if (li >= 0)
                {
                    if (wrapLensLit == null)
                    {
                        wrapLensLit = new Material(r.sharedMaterials[li]) { name = "A wrap lens lit" };
                        wrapLensLit.SetColor("_EmissionColor", new Color(1f, 1f, 1f) * 1.6f);
                        wrapLensLit.SetFloat("_UseEmission", 1f); wrapLensLit.EnableKeyword("_EMISSION");
                    }
                    var arr = r.sharedMaterials; arr[li] = wrapLensLit; r.sharedMaterials = arr;
                }
            }
            l = Spot(p, "wrap light " + name, new Vector3(0f, soffit - .09f, zl), Vector3.down, 170f, 120f, new Color(.95f, .98f, 1f), 3.2f, 6.5f, shadows);
        }
        else if (dir == "B")
        {
            // wall globe: D on the east wall over the landing, T on the far wall
            Vector3 mount, outN;
            if (dLanding) { mount = edge ? new Vector3(.95f, y + 2.30f, ZNear) : new Vector3(XFace, y + 2.35f, 1.05f); outN = edge ? Vector3.forward : Vector3.left; }
            else { mount = new Vector3(0f, y + 2.35f, ZFarFace); outN = Vector3.back; }
            var box = mount + outN * .03f;
            Cyl(p, "globe box " + name, mount, mount + outN * .05f, .055f, steelM, Id.Background);
            if (globeLit == null) globeLit = Glow("B globe lit (100 W A19 behind glass)", new Color(1f, .93f, .8f), new Color(1f, .82f, .6f) * 4.0f);
            var gc = box + outN * .12f;
            var gr = Ball(p, "globe glass " + name, gc, .13f, globeLit, Id.Emitter);
            gr.shadowCastingMode = ShadowCastingMode.Off;
            // cage: 4 wires + a ring
            for (var i = 0; i < 4; i++)
            {
                var ang = i * Mathf.PI * .5f;
                var side = Vector3.Cross(outN, Vector3.up).normalized;
                var off = (side * Mathf.Cos(ang) + Vector3.up * Mathf.Sin(ang)) * .075f;
                var wr = Cyl(p, "globe cage wire " + name + "." + i, box + outN * .03f + off * .8f, gc + outN * .07f + off * .25f, .003f, steelM, Id.Background);
                wr.shadowCastingMode = ShadowCastingMode.Off;
            }
            if (edge)
            {
                // a bare globe lights its own wall too: a point light (omni), its hot halo on the end wall
                var pgo = new GameObject("globe light " + name);
                pgo.transform.SetParent(p, false);
                pgo.transform.position = root.TransformPoint(gc);
                l = pgo.AddComponent<Light>();
                l.type = LightType.Point; l.color = new Color(1f, .84f, .66f); l.intensity = 2.6f; l.range = 6f;
                l.shadows = shadows ? LightShadows.Soft : LightShadows.None; l.shadowStrength = .92f; l.shadowNearPlane = .05f;
            }
            else l = Spot(p, "globe light " + name, gc, outN + Vector3.down * .45f, 168f, 110f, new Color(1f, .84f, .66f), 3.6f, 6f, shadows);
        }
        else
        {
            if (edge) l = MapLampAcross(p, new Vector3(0f, y + MazeH, dLanding ? ZNear + .45f : ZFarFace - .45f), shadows);
            else l = MapLamp(p, new Vector3(0f, y + MazeH, zc - .3f), shadows, Id.Emitter);
            l.range = 6.5f;
        }
        if (l != null) { l.intensity *= lampGain; stairLights.Add(l); }
        // twin-head battery unit over each door (A, B): normal power, so only the red AC pilot glows
        if (dLanding && dir != "C") TwinHead(p, n);
    }

    static void TwinHead(Transform p, int n)
    {
        var y = YD(n);
        var cab = KitSpawn("Kit_EmergencyTwinHead", p, new Vector3(0f, y + 2.36f, ZNear + .05f), Quaternion.identity, Id.Background);
        if (cab == null) return;
        var cr = cab.GetComponentInChildren<Renderer>();
        var pi = SlotIndex(cr, "Run_NeonPilot");
        if (pi >= 0)
        {
            if (pilotRed == null) pilotRed = Glow("twin-head AC pilot (lit)", new Color(.5f, .06f, .04f), new Color(1f, .08f, .04f) * 1.4f);
            var arr = cr.sharedMaterials; arr[pi] = pilotRed; cr.sharedMaterials = arr;
        }
        var aims = new[] { new Vector3(-.81f, y - 1.0f, 3.2f), new Vector3(.6f, y, 1.2f) };
        var mounts = new[] { new Vector3(.105f, .191f, .008f), new Vector3(-.105f, .191f, .008f) };
        for (var i = 0; i < 2; i++)
        {
            var yoke = FrontRoomsKitLibrary.Spawn("Kit_EmergencyTwinHead_Yoke", cab.transform, mounts[i], Quaternion.identity, null, false);
            if (yoke == null) continue;
            var head = FrontRoomsKitLibrary.Spawn("Kit_EmergencyTwinHead_Lamp", yoke.transform, new Vector3(0f, .085f, 0f), Quaternion.identity, null, false);
            if (head == null) continue;
            var dirW = root.TransformDirection(aims[i] - root.InverseTransformPoint(head.transform.position));
            var dCab = cab.transform.InverseTransformDirection(dirW);
            yoke.transform.localRotation = Quaternion.Euler(0f, Mathf.Atan2(dCab.x, dCab.z) * Mathf.Rad2Deg, 0f);
            var dYoke = yoke.transform.InverseTransformDirection(dirW);
            head.transform.localRotation = Quaternion.Euler(Mathf.Atan2(-dYoke.y, dYoke.z) * Mathf.Rad2Deg, 0f, 0f);
            foreach (var r in yoke.GetComponentsInChildren<Renderer>(true)) ids[r] = Id.Background;
        }
        // EMT feed to the soffit
        var feed = root.InverseTransformPoint(cab.transform.TransformPoint(new Vector3(0f, .182f, -.012f)));
        Cyl(p, "unit feed EMT f" + n, feed, new Vector3(feed.x, y + FF - .2f, feed.z), .0117f, Flat("EMT", Hex("B4B6B2"), .5f, .85f), Id.Background);

    // ---------------------------------------------------------------- the Relay
    static void BuildRelays()
    {
        relayStair = MakeRelay("Relay (stair, turn landing)", new Vector3(-.81f, YT(0), 5.15f), 180f);
        relayMaze = MakeRelay("Relay (maze, before the exit)", new Vector3(.05f, 0f, -1.7f), 180f);
    }

    static Transform MakeRelay(string name, Vector3 pos, float yaw)
    {
        var t = FrontRooms3DGame.CreateHunter(root);
        t.name = name;
        var rig = t.GetComponent<FrontRoomsRelayRig>();
        if (rig != null) rig.TickAnimation(.35f, FrontRoomsRelayRig.MotionState.Walk, true, 1f);
        t.position = root.TransformPoint(pos);
        t.rotation = root.rotation * Quaternion.Euler(0f, yaw, 0f);
        t.localScale = Vector3.one;
        var b = Bounds(t);
        var nativeTop = b.max.y - t.position.y;
        relayScale = nativeTop > .1f ? RelayHeight / nativeTop : 1f;
        t.localScale = Vector3.one * relayScale;
        b = Bounds(t);
        foreach (var r in t.GetComponentsInChildren<Renderer>(true)) { ids[r] = Id.Relay; r.shadowCastingMode = ShadowCastingMode.On; }
        Log(name + ": the game's Relay rig (FrontRooms3DGame.CreateHunter, walk pose) scaled x" + relayScale.ToString("0.000") + " from a native top of " + nativeTop.ToString("0.00") + " m to " + (b.max.y - t.position.y).ToString("0.00") + " m (ModuleUnits.RelayHeight 2.05), width " + b.size.x.ToString("0.00"));
        ShowRelay(t, false);
        return t;
    }

    static void ShowRelay(Transform t, bool on)
    {
        foreach (var r in t.GetComponentsInChildren<Renderer>(true)) r.enabled = on;
    }

    static Bounds Bounds(Transform t)
    {
        var rs = t.GetComponentsInChildren<Renderer>(true);
        var b = new Bounds(t.position, Vector3.zero);
        foreach (var r in rs) b.Encapsulate(r.bounds);
        return b;
    }

    static Camera MakeCamera()
    {
        var go = new GameObject("STAIR LOOKDEV camera");
        var cam = go.AddComponent<Camera>();
        cam.nearClipPlane = .02f;
        cam.farClipPlane = 80f;
        cam.clearFlags = CameraClearFlags.SolidColor;
        cam.backgroundColor = new Color(0x22 / 255f, 0x23 / 255f, 0x1C / 255f);
        FrontRoomsPostStack.ConfigureCamera(cam);
        var rtCam = cam.GetComponent("FrontRoomsGlassRTCamera");

    static void Pose(Camera cam, Shot s)
    {
        var eye = root.TransformPoint(s.eye);
        cam.transform.SetPositionAndRotation(eye, Quaternion.LookRotation(root.TransformPoint(s.look) - eye, Vector3.up));
        cam.fieldOfView = s.fov;
    }

    static void RenderAll(List<Shot> shots, string onlyShots)
    {
        var filter = string.IsNullOrEmpty(onlyShots) ? null : new HashSet<string>(onlyShots.Split(','));
        var cam = MakeCamera();
        var rt = new RenderTexture(1920, 1080, 24, RenderTextureFormat.ARGB32) { antiAliasing = 4 };
        cam.targetTexture = rt;
        var poses = new StringBuilder("shot\teye\tlook\tfov\trelay\n");
        foreach (var s in shots)
        {
            if (filter != null && !filter.Contains(s.name)) continue;
            ShowRelay(relayStair, s.relay == 1);
            ShowRelay(relayMaze, s.relay == 2);
            Pose(cam, s);
            cam.targetTexture = rt;
            cam.Render(); cam.Render();
            Save(rt, Path.Combine(outDir, s.name + ".png"));
            RenderId(cam, s);
            if (s.relay > 0) RenderRelayParts(cam, s, s.relay == 1 ? relayStair : relayMaze);
            if (ePass) RenderE(cam, s);
            poses.Append(s.name + "\t" + V(s.eye) + "\t" + V(s.look) + "\t" + s.fov.ToString("0", CultureInfo.InvariantCulture) + "\t" + s.relay + "\n");
            Log("shot " + s.name + " rendered");
        }
        File.WriteAllText(Path.Combine(outDir, "poses.tsv"), poses.ToString());
        cam.targetTexture = null;
        rt.Release();
        UnityEngine.Object.DestroyImmediate(cam.gameObject);
        ShowRelay(relayStair, false); ShowRelay(relayMaze, false);
    }

    static string V(Vector3 v) { return v.x.ToString("0.###", CultureInfo.InvariantCulture) + "," + v.y.ToString("0.###", CultureInfo.InvariantCulture) + "," + v.z.ToString("0.###", CultureInfo.InvariantCulture); }

    static void SwapMaterials(Func<Renderer, Material> pick, Dictionary<Renderer, Material[]> saved, List<Renderer> hidden)
    {
        foreach (var r in root.GetComponentsInChildren<Renderer>(false))
        {
            if (!r.enabled) continue;
            saved[r] = r.sharedMaterials;
            var m = pick(r);
            if (m == null) { hidden.Add(r); r.enabled = false; continue; }
            var arr = new Material[r.sharedMaterials.Length];
            for (var i = 0; i < arr.Length; i++) arr[i] = m;
            r.sharedMaterials = arr;
        }
    }

    static void Restore(Dictionary<Renderer, Material[]> saved, List<Renderer> hidden)
    {
        foreach (var kv in saved) if (kv.Key != null) kv.Key.sharedMaterials = kv.Value;
        foreach (var r in hidden) if (r != null) r.enabled = true;
    }

    /// <summary>Flat unlit ID colours (pure 0/1 channels), no post, no MSAA, no fog.</summary>
    static void RenderId(Camera cam, Shot s)
    {
        var unlit = Shader.Find("Universal Render Pipeline/Unlit");
        var idMats = new Dictionary<Id, Material>();
        foreach (Id id in Enum.GetValues(typeof(Id))) { var m = new Material(unlit); m.SetColor("_BaseColor", IdColour(id)); idMats[id] = m; }
        var saved = new Dictionary<Renderer, Material[]>(); var hidden = new List<Renderer>();
        SwapMaterials(r => { Id v; return idMats[ids.TryGetValue(r, out v) ? v : Id.Background]; }, saved, hidden);
        var fog = RenderSettings.fog; RenderSettings.fog = false;
        var data = cam.GetUniversalAdditionalCameraData();
        data.renderPostProcessing = false;
        var prev = cam.targetTexture;
        var rt = new RenderTexture(1920, 1080, 24, RenderTextureFormat.ARGB32) { antiAliasing = 1 };
        cam.targetTexture = rt; cam.allowMSAA = false;
        cam.Render();
        Save(rt, Path.Combine(outDir, s.name + "_id.png"));
        cam.targetTexture = prev; cam.allowMSAA = true; rt.Release();
        data.renderPostProcessing = true;
        RenderSettings.fog = fog;
        Restore(saved, hidden);
    }

    /// <summary>Relay parts mask: pale head red, dark body (the rig's body material) green, hands/feet/face
    /// (detail material) blue, everything else black. Unlit, no post, no MSAA, no fog.</summary>
    static void RenderRelayParts(Camera cam, Shot s, Transform relay)
    {
        var unlit = Shader.Find("Universal Render Pipeline/Unlit");
        Func<Color, Material> mk = c => { var m = new Material(unlit); m.SetColor("_BaseColor", c); return m; };
        var black = mk(Color.black); var red = mk(Color.red); var green = mk(Color.green); var blue = mk(Color.blue);
        var relayRs = new HashSet<Renderer>(relay.GetComponentsInChildren<Renderer>(true));
        var saved = new Dictionary<Renderer, Material[]>(); var hidden = new List<Renderer>();
        SwapMaterials(r =>
        {
            if (!relayRs.Contains(r)) return black;
            if (r.gameObject.name.Contains("blank head")) return red;
            var m0 = saved.ContainsKey(r) && saved[r].Length > 0 ? saved[r][0] : null;
            return m0 != null && m0.name.Contains("body") ? green : blue;
        }, saved, hidden);
        var fog = RenderSettings.fog; RenderSettings.fog = false;
        var data = cam.GetUniversalAdditionalCameraData();
        data.renderPostProcessing = false;
        var prev = cam.targetTexture;
        var rt = new RenderTexture(1920, 1080, 24, RenderTextureFormat.ARGB32) { antiAliasing = 1 };
        cam.targetTexture = rt; cam.allowMSAA = false;
        cam.Render();
        Save(rt, Path.Combine(outDir, s.name + "_relayparts.png"));
        cam.targetTexture = prev; cam.allowMSAA = true; rt.Release();
        data.renderPostProcessing = true;
        RenderSettings.fog = fog;
        Restore(saved, hidden);
    }

    /// <summary>Illuminance pass: every surface white Lambert (albedo 1, no specular, no reflections), real
    /// lights + the game's ambient, no post, no fog; linear float RGBA, 960 x 540, raw little-endian floats.</summary>
    static void RenderE(Camera cam, Shot s)
    {
        var white = new Material(Shader.Find("Universal Render Pipeline/Lit")) { name = "E pass white" };
        white.SetColor("_BaseColor", Color.white);
        white.SetFloat("_Smoothness", 0f); white.SetFloat("_Metallic", 0f);
        white.SetFloat("_SpecularHighlights", 0f); white.EnableKeyword("_SPECULARHIGHLIGHTS_OFF");
        white.SetFloat("_EnvironmentReflections", 0f); white.EnableKeyword("_ENVIRONMENTREFLECTIONS_OFF");
        var saved = new Dictionary<Renderer, Material[]>(); var hidden = new List<Renderer>();
        SwapMaterials(r => { Id v; var id = ids.TryGetValue(r, out v) ? v : Id.Background; return id == Id.Emitter ? null : white; }, saved, hidden);
        var fog = RenderSettings.fog; RenderSettings.fog = false;
        var data = cam.GetUniversalAdditionalCameraData();
        data.renderPostProcessing = false;
        var prev = cam.targetTexture;
        var rt = new RenderTexture(960, 540, 24, RenderTextureFormat.ARGBFloat) { antiAliasing = 1 };
        cam.targetTexture = rt; cam.allowMSAA = false;
        cam.Render();
        var prevActive = RenderTexture.active;
        RenderTexture.active = rt;
        var tex = new Texture2D(960, 540, TextureFormat.RGBAFloat, false, true);
        tex.ReadPixels(new Rect(0, 0, 960, 540), 0, 0);
        tex.Apply();
        RenderTexture.active = prevActive;
        var raw = tex.GetRawTextureData();
        File.WriteAllBytes(Path.Combine(outDir, s.name + "_E960x540.f32"), raw);
        UnityEngine.Object.DestroyImmediate(tex);
        cam.targetTexture = prev; cam.allowMSAA = true; rt.Release();
        data.renderPostProcessing = true;
        RenderSettings.fog = fog;
        Restore(saved, hidden);
    }

    static void Save(RenderTexture rt, string path)
    {
        var prev = RenderTexture.active;
        RenderTexture.active = rt;
        var image = new Texture2D(rt.width, rt.height, TextureFormat.RGB24, false);
        image.ReadPixels(new Rect(0, 0, rt.width, rt.height), 0, 0);
        image.Apply();
        RenderTexture.active = prev;
        File.WriteAllBytes(path, image.EncodeToPNG());
        UnityEngine.Object.DestroyImmediate(image);
    }

    // ---------------------------------------------------------------- stats and kits
    static int Tris(Transform t)
    {
        var sum = 0;
        foreach (var mf in t.GetComponentsInChildren<MeshFilter>(true))
        {
            if (mf.sharedMesh == null) continue;
            for (var i = 0; i < mf.sharedMesh.subMeshCount; i++) sum += (int)(mf.sharedMesh.GetIndexCount(i) / 3);
        }
        foreach (var smr in t.GetComponentsInChildren<SkinnedMeshRenderer>(true)) if (smr.sharedMesh != null) sum += smr.sharedMesh.triangles.Length / 3;
        return sum;
    }

    /// <summary>Saves the floor-0 kit groups as blockout prefabs (meshes and materials as assets) in the clone.</summary>
    static void SaveKits()
    {
        var baseDir = "Assets/StairBlockout/" + dir;
        if (!AssetDatabase.IsValidFolder("Assets/StairBlockout")) AssetDatabase.CreateFolder("Assets", "StairBlockout");
        if (!AssetDatabase.IsValidFolder(baseDir)) AssetDatabase.CreateFolder("Assets/StairBlockout", dir);
        if (!AssetDatabase.IsValidFolder(baseDir + "/Meshes")) AssetDatabase.CreateFolder(baseDir, "Meshes");
        if (!AssetDatabase.IsValidFolder(baseDir + "/Materials")) AssetDatabase.CreateFolder(baseDir, "Materials");
        var mi = 0; var me = 0;
        foreach (var kv in kitGroups)
        {
            foreach (var r in kv.Value.GetComponentsInChildren<Renderer>(true))
            {
                var arr = r.sharedMaterials;
                for (var i = 0; i < arr.Length; i++)
                {
                    if (arr[i] == null || EditorUtility.IsPersistent(arr[i])) continue;
                    var safe = new string(arr[i].name.Select(ch => char.IsLetterOrDigit(ch) ? ch : '_').ToArray());
                    try { AssetDatabase.CreateAsset(arr[i], baseDir + "/Materials/" + safe + "_" + (mi++) + ".mat"); }
                    catch (Exception e) { Log("material not saved: " + arr[i].name + " " + e.Message); }
                }
                var mf = r.GetComponent<MeshFilter>();
                if (mf != null && mf.sharedMesh != null && !EditorUtility.IsPersistent(mf.sharedMesh))
                {
                    try { AssetDatabase.CreateAsset(mf.sharedMesh, baseDir + "/Meshes/mesh_" + (me++) + ".asset"); }
                    catch (Exception e) { Log("mesh not saved: " + mf.sharedMesh.name + " " + e.Message); }
                }
            }
            PrefabUtility.SaveAsPrefabAsset(kv.Value.gameObject, baseDir + "/BK_" + dir + "_" + kv.Key + ".prefab");
        }
        AssetDatabase.SaveAssets();
        Log("lamp mode " + lampMode + "; saved " + kitGroups.Count + " blockout kit prefabs to " + baseDir + " (" + mi + " materials, " + me + " meshes)");
    }
