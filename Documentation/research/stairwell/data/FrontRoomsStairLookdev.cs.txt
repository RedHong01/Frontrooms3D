    // ------------------------------------------------------------- batch
    public static void RunBatch()
    {
        var args = Environment.GetCommandLineArgs();
        var dirs = "A,B,C";
        string onlyShots = null;
        for (var i = 0; i < args.Length; i++)
        {
            if (args[i] == "-stairDirs" && i + 1 < args.Length) dirs = args[i + 1];
            if (args[i] == "-stairOut" && i + 1 < args.Length) outRoot = args[i + 1];
            if (args[i] == "-stairShots" && i + 1 < args.Length) onlyShots = args[i + 1];
            if (args[i] == "-stairNoE") ePass = false;
            if (args[i] == "-stairNoKits") saveKits = false;
            if (args[i] == "-stairLamp" && i + 1 < args.Length) lampMode = args[i + 1];
            if (args[i] == "-stairLampGain" && i + 1 < args.Length) lampGain = float.Parse(args[i + 1], CultureInfo.InvariantCulture);
            if (args[i] == "-stairNoEndDado") noEndDado = true;
            if (args[i] == "-stairRemedy") remedy = true;
            if (args[i] == "-stairFixtureGain" && i + 1 < args.Length) fixtureGain = float.Parse(args[i + 1], CultureInfo.InvariantCulture);
            if (args[i] == "-stairLadder") ladder = true;
            if (args[i] == "-stairLadderLamps" && i + 1 < args.Length) ladderLamps = args[i + 1];
            if (args[i] == "-stairSignGain" && i + 1 < args.Length) signGain = float.Parse(args[i + 1], CultureInfo.InvariantCulture);
            if (args[i] == "-stairLadderSeed" && i + 1 < args.Length) ladderSeed = int.Parse(args[i + 1], CultureInfo.InvariantCulture);
        }
        if (string.IsNullOrEmpty(outRoot)) outRoot = Path.Combine(Directory.GetParent(Application.dataPath).FullName, "Verification", "stair_lookdev");
        EnsureAssets();
        foreach (var d in dirs.Split(','))
        {
            var key = d.Trim().ToUpperInvariant();
            if (key.Length == 0) continue;
            try { RunOne(key, onlyShots); }
            catch (Exception e) { Debug.LogError("[StairLookdev] direction " + key + " failed: " + e); File.AppendAllText(Path.Combine(outRoot, "errors.txt"), key + ": " + e + "\n"); }
        }
        Debug.Log("[StairLookdev] ALL DONE -> " + outRoot);
    }

    static void Log(string s) { log.Add(s); Debug.Log("[StairLookdev] " + dir + " " + s); }

    static void RunOne(string key, string onlyShots)
    {
        dir = key;
        outDir = Path.Combine(outRoot, dir);
        Directory.CreateDirectory(outDir);
        ids.Clear(); mats.Clear(); log.Clear(); stairLights.Clear(); kitGroups.Clear();
        lockRs.Clear(); trimRs.Clear(); signRs.Clear(); leafRs.Clear(); barRs.Clear(); lampLog.Clear();
        flatDrv = null; endDrv = null;
        exitLeaf = null; farLeaf = null; farLever = null; lockedSet = null; unlockedSet = null; flatSign = null; endSign = null;
        wrapLensLit = null; globeLit = null; pilotRed = null;
        var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
        EditorSceneManager.SetActiveScene(scene);
        root = new GameObject("STAIR LOOKDEV " + dir).transform;
        root.position = RootWorld;
        FrontRoomsLook.ApplyAmbient();
        FrontRoomsPostStack.Ensure(root);
        SetupReflections();
        SceneFill();
        shaft = Group(root, "Stair shaft " + dir);
        mazeRoot = Group(root, "Level 0 maze side");
        kitsRoot = Group(shaft, "floor 0 kits");

        Materials();
        if (ladder)
        {
            // the exit approach ladder: a 63 x 33 m Level 0 hall in front of the exit door; the stair is not built
            BuildHall();
            StairDoor(0);
        }
        else
        {
            BuildMaze();
            BuildShaft();
        }
        BuildExitDoor();
        BuildRelays();
        Physics.SyncTransforms();

        var shots = Shots();
        RenderAll(shots, onlyShots);
        Stats();
        if (saveKits && !ladder) { try { SaveKits(); } catch (Exception e) { Log("kit save failed: " + e.Message); } }
        File.WriteAllLines(Path.Combine(outDir, "log.txt"), log);
        Debug.Log("[StairLookdev] " + dir + " done -> " + outDir);
    }

    // ---------------------------------------------------------------- assets
    static void EnsureAssets()
    {
        foreach (var a in new[] { "Kit_EmergencyTwinHead", "Kit_EmergencyTwinHead_Yoke", "Kit_EmergencyTwinHead_Lamp", "Kit_WrapFixture_4ft" })
        {
            AssetDatabase.ImportAsset("Assets/Resources/Props/Models/" + a + ".json", ImportAssetOptions.ForceUpdate);
            AssetDatabase.ImportAsset("Assets/Resources/Props/Models/" + a + ".fbx", ImportAssetOptions.ForceUpdate);
        }
        foreach (var t in new[] { "Stair_Concrete", "Stair_ConcreteFloor" })
            foreach (var s in new[] { "_A", "_N", "_S" })
                AssetDatabase.ImportAsset(TexDir + t + s + ".png", ImportAssetOptions.ForceUpdate);
        // placeholder numerals (gen_stair_numerals.py; the face and size belong to graphic design)
        if (AssetDatabase.IsValidFolder("Assets/StairLookdevTex"))
            foreach (var g in AssetDatabase.FindAssets("t:Texture2D", new[] { "Assets/StairLookdevTex" }))
            {
                var path = AssetDatabase.GUIDToAssetPath(g);
                var imp = AssetImporter.GetAtPath(path) as TextureImporter;
                if (imp == null) continue;
                imp.alphaIsTransparency = true; imp.mipmapEnabled = true; imp.sRGBTexture = true;
                imp.wrapMode = path.Contains("mesh") ? TextureWrapMode.Repeat : TextureWrapMode.Clamp;
                imp.SaveAndReimport();
            }
        FrontRoomsKitLibrary.ClearCache();
    }

    /// <summary>A placeholder numeral or plate face: an alpha-clipped URP Lit quad material.</summary>
    static Material DecalMat(string name, string tex, Color tint, float smooth)
    {
        Material c;
        if (mats.TryGetValue(name, out c)) return c;
        var m = new Material(Shader.Find("Universal Render Pipeline/Lit")) { name = name };
        var t = AssetDatabase.LoadAssetAtPath<Texture2D>(NumTexDir + tex + ".png");
        if (t == null) Log("WARNING numeral texture missing " + tex);
        m.SetTexture("_BaseMap", t);
        m.SetColor("_BaseColor", tint);
        m.SetFloat("_Smoothness", smooth);
        m.SetFloat("_AlphaClip", 1f); m.SetFloat("_Cutoff", .5f); m.EnableKeyword("_ALPHATEST_ON");
        m.SetFloat("_EnvironmentReflections", 0f); m.EnableKeyword("_ENVIRONMENTREFLECTIONS_OFF");
        mats[name] = m;
        return m;
    }

    /// <summary>A flat quad on a wall (normal nrm), centred at c, size w x h, with a decal material.</summary>
    static Renderer Decal(Transform parent, string name, Vector3 c, Vector3 nrm, float w, float h, Material m, Id id)
    {
        var go = GameObject.CreatePrimitive(PrimitiveType.Quad);
        go.name = name;
        go.transform.SetParent(parent, false);
        go.transform.position = root.TransformPoint(c);
        // a Unity quad faces -z: its visible side looks along +forward when forward = -nrm
        go.transform.rotation = root.rotation * Quaternion.LookRotation(-nrm, Vector3.up);
        go.transform.localScale = new Vector3(w, h, 1f);
        UnityEngine.Object.DestroyImmediate(go.GetComponent<Collider>());
        var r = go.GetComponent<MeshRenderer>();
        r.sharedMaterial = m;
        r.shadowCastingMode = ShadowCastingMode.Off;
        ids[r] = id;
        return r;
    }

    static Texture2D T(string stem) { return AssetDatabase.LoadAssetAtPath<Texture2D>(TexDir + stem + ".png"); }

    static void ConfigureSurface(Material m, string tex, Vector2 tile, Color tint, bool meshUV, float smooth, float bump, float macroTone, float macroDirt)
    {
        m.SetTexture("_BaseMap", tex != null ? T(tex + "_A") : null);
        m.SetTexture("_BumpMap", tex != null ? T(tex + "_N") : null);
        m.SetTexture("_MaskMap", tex != null ? T(tex + "_S") : null);
        if (tex != null && T(tex + "_A") == null) Log("WARNING texture missing " + tex);
        m.SetColor("_BaseColor", tint);
        m.SetTextureScale("_BaseMap", Vector2.one); m.SetTextureOffset("_BaseMap", Vector2.zero);
        m.SetVector("_TileSize", new Vector4(tile.x, tile.y, 0, 0));
        m.SetFloat("_BumpScale", bump);
        m.SetFloat("_Smoothness", smooth);
        m.SetFloat("_Metallic", 0f);
        m.SetFloat("_OcclusionStrength", .8f);
        m.SetTexture("_MacroMap", T("MacroWear_M"));
        m.SetFloat("_MacroTone", macroTone);
        m.SetFloat("_MacroDirt", macroDirt);
        m.SetColor("_StainColor", new Color(.42f, .38f, .30f));
        m.SetFloat("_StainStrength", 0f);
        m.SetFloat("_WetStrength", 0f);
        // The shader's floor / ceiling grime bands key on WORLD y (FrontRoomsSurface.shader:455); a stair
        // spans many floors, so every stair surface keeps them at 0.
        m.SetFloat("_FloorGrime", 0f);
        m.SetFloat("_CeilingGrime", 0f);
        m.SetFloat("_CeilingHeight", MazeH);
        m.SetFloat("_MeshUV", meshUV ? 1f : 0f);
        if (meshUV) m.EnableKeyword("_FR_MESH_UV"); else m.DisableKeyword("_FR_MESH_UV");
        m.SetFloat("_UseEmission", 0f); m.DisableKeyword("_EMISSION");
    }

    /// <summary>World-planar FrontRooms/Surface (walls, floors, concrete).</summary>
    static Material Surf(string name, string tex, float tile, Color tint, float smooth = 1f, float bump = 1f, float macroTone = .2f, float macroDirt = .2f)
    {
        Material c;
        if (mats.TryGetValue(name, out c)) return c;
        var m = new Material(Shader.Find("FrontRooms/Surface")) { name = name };
        ConfigureSurface(m, tex, new Vector2(tile, tile), tint, false, smooth, bump, macroTone, macroDirt);
        mats[name] = m;
        return m;
    }

    /// <summary>Flat-tint FrontRooms/Surface with mesh UVs (props shade like the shell).</summary>
    static Material Flat(string name, Color c, float smooth = .3f, float metal = 0f)
    {
        Material cached;
        if (mats.TryGetValue(name, out cached)) return cached;
        var m = new Material(Shader.Find("FrontRooms/Surface")) { name = name };
        ConfigureSurface(m, null, Vector2.one, c, true, smooth, 0f, 0f, 0f);
        m.SetFloat("_Metallic", metal);
        mats[name] = m;
        return m;
    }

    static Material Glow(string name, Color c, Color emission)
    {
        Material cached;
        if (mats.TryGetValue(name, out cached)) return cached;
        var m = new Material(Shader.Find("Universal Render Pipeline/Lit")) { name = name };
        m.SetColor("_BaseColor", c);
        m.SetFloat("_Smoothness", .2f);
        m.EnableKeyword("_EMISSION");
        m.SetColor("_EmissionColor", emission);
        m.globalIlluminationFlags = MaterialGlobalIlluminationFlags.RealtimeEmissive;
        mats[name] = m;
        return m;
    }

    /// <summary>A copy of a project surface with the world-y grime bands off (for stair walls below y 0).</summary>
    static Material NoGrime(Material src, string name)
    {
        Material cached;
        if (mats.TryGetValue(name, out cached)) return cached;
        var m = new Material(src) { name = name };
        m.SetFloat("_FloorGrime", 0f);
        m.SetFloat("_CeilingGrime", 0f);
        mats[name] = m;
        return m;
    }

    static Cubemap flatCube;
    static void SetupReflections()
    {
        flatCube = new Cubemap(16, TextureFormat.RGBAHalf, false) { name = "Stair_FlatAmbientCube" };
        var sky = FrontRoomsLook.AmbientSky.linear; var eq = FrontRoomsLook.AmbientEquator.linear; var gr = FrontRoomsLook.AmbientGround.linear;
        foreach (CubemapFace f in Enum.GetValues(typeof(CubemapFace)))
        {
            if (f == CubemapFace.Unknown) continue;
            var col = f == CubemapFace.PositiveY ? sky : f == CubemapFace.NegativeY ? gr : eq;
            var px = new Color[16 * 16];
            for (var i = 0; i < px.Length; i++)
            {
                var row = i / 16;
                var c = col;
                if (f != CubemapFace.PositiveY && f != CubemapFace.NegativeY) c = row < 8 ? Color.Lerp(sky, eq, row / 8f) : Color.Lerp(eq, gr, (row - 8) / 8f);
                px[i] = c;
            }
            flatCube.SetPixels(px, f);
        }
        flatCube.Apply();
        RenderSettings.defaultReflectionMode = DefaultReflectionMode.Custom;
        RenderSettings.customReflectionTexture = flatCube;
    }

    // ---------------------------------------------------------------- mesh builder
    class MB
    {
        public readonly List<Vector3> v = new List<Vector3>();
        public readonly List<Vector3> n = new List<Vector3>();
        public readonly List<Vector2> uv = new List<Vector2>();
        public readonly List<int> tri = new List<int>();

        public void Quad(Vector3 a, Vector3 b, Vector3 c, Vector3 d, Vector3 normal)
        {
            var i = v.Count;
            var t = Mathf.Abs(normal.y) > .5f ? Vector3.right : Vector3.Cross(Vector3.up, normal).normalized;
            var bt = Vector3.Cross(normal, t);
            foreach (var p in new[] { a, b, c, d }) { v.Add(p); n.Add(normal); uv.Add(new Vector2(Vector3.Dot(p, t), Vector3.Dot(p, bt))); }
            Tri(i, i + 1, i + 2, normal);
            Tri(i, i + 2, i + 3, normal);
        }

        void Tri(int a, int b, int c, Vector3 normal)
        {
            if (Vector3.Dot(Vector3.Cross(v[b] - v[a], v[c] - v[a]), normal) < 0f) { var s = b; b = c; c = s; }
            tri.Add(a); tri.Add(b); tri.Add(c);
        }

        public Mesh Build(string name)
        {
            var m = new Mesh { name = name };
            if (v.Count > 65000) m.indexFormat = IndexFormat.UInt32;
            m.SetVertices(v); m.SetNormals(n); m.SetUVs(0, uv); m.SetTriangles(tri, 0);
            m.RecalculateTangents(); m.RecalculateBounds();
            return m;
        }
    }

    // ---------------------------------------------------------------- the exit approach ladder (Level 0 hall)
    const float HallHalfX = 30f, HallDepth = 33f;

    /// <summary>FIX (critic gameplay 3, 5, 9): a 63 x 33 m Level 0 hall in front of the exit door, tiled per 3 m cell,
    /// one map troffer per cell with a tier-5 lamp draw (FrontRoomsLevelProfile Defaults tier 5: steady .40, stutter
    /// .25, failing .18, dead .12, dim .05). Still levels: steady 1.0, stutter .97 (between dips), failing .47 (mean of
    /// (.25 + .45 wave)), dead 0, dim .42 (FrontRoomsMapWorld.Level). ladderLamps: "hold" = the cells within 2 of the
    /// exit door held Steady (the proposed rule), "dead" = the same cells Dead (the worst case), "steady" = all steady.
    /// Lamps within 9 m of the door cast shadows (as the map's castsShadow near the player).</summary>
    static void BuildHall()
    {
        var carpet = FrontRoomsSurfaces.Room(RoomRule.Lobby, FrontRoomsSurfaces.Slot.Floor);
        var ceil = FrontRoomsSurfaces.Room(RoomRule.Lobby, FrontRoomsSurfaces.Slot.Ceiling);
        var paper = FrontRoomsSurfaces.Room(RoomRule.Lobby, FrontRoomsSurfaces.Slot.Wall);
        var nx = Mathf.RoundToInt(HallHalfX / 3f);   // cells cx -11..11 (x = 3 cx)
        var nz = Mathf.RoundToInt(HallDepth / 3f);   // cells cz 0..10 (z = -1.5 - 3 cz)
        for (var cx = -nx; cx <= nx; cx++)
            for (var cz = 0; cz < nz; cz++)
            {
                var c = new Vector3(cx * 3f, 0f, -1.5f - 3f * cz);
                Box(mazeRoot, "hall carpet " + cx + "," + cz, new Vector3(c.x, -.1f, c.z), new Vector3(3f, .2f, 3f), carpet, Id.MazeFloor);
                Box(mazeRoot, "hall ceiling " + cx + "," + cz, new Vector3(c.x, MazeH + .08f, c.z), new Vector3(3f, .16f, 3f), ceil, Id.Background);
            }
        var w = nx * 6f + 3f; var zMin = -3f * nz;
        Box(mazeRoot, "hall wall S", new Vector3(0f, MazeH * .5f, zMin - .08f), new Vector3(w, MazeH, .16f), paper, Id.Background);
        Box(mazeRoot, "hall wall W", new Vector3(-w * .5f - .08f, MazeH * .5f, zMin * .5f), new Vector3(.16f, MazeH, -zMin), paper, Id.Background);
        Box(mazeRoot, "hall wall E", new Vector3(w * .5f + .08f, MazeH * .5f, zMin * .5f), new Vector3(.16f, MazeH, -zMin), paper, Id.Background);
        var half = w * .5f;
        Box(mazeRoot, "hall wall N left", new Vector3(-(half + .5f) * .5f, MazeH * .5f, -.04f), new Vector3(half - .5f, MazeH, .08f), paper, Id.Background);
        Box(mazeRoot, "hall wall N right", new Vector3((half + .5f) * .5f, MazeH * .5f, -.04f), new Vector3(half - .5f, MazeH, .08f), paper, Id.Background);
        Box(mazeRoot, "hall wall N header", new Vector3(0f, (2.1f + MazeH) * .5f, -.04f), new Vector3(1.0f, MazeH - 2.1f, .08f), paper, Id.Background);
        Box(mazeRoot, "hall wall N back", new Vector3(0f, MazeH * .5f, .2f), new Vector3(w, MazeH, .2f), Flat("hall back (unseen)", Hex("202020"), .1f), Id.Background);
        Box(mazeRoot, "hall cove N left", new Vector3(-(half + .5f) * .5f, .05f, -.09f), new Vector3(half - .5f, .1f, .02f), coveM, Id.Background);
        Box(mazeRoot, "hall cove N right", new Vector3((half + .5f) * .5f, .05f, -.09f), new Vector3(half - .5f, .1f, .02f), coveM, Id.Background);
        // tier-5 draw, deterministic per seed
        var rng = (uint)(ladderSeed * 2654435761u + 12345u);
        var t5 = new[] { .40f, .25f, .18f, .12f, .05f };
        var names = new[] { "steady", "stutter", "failing", "dead", "dim" };
        var levels = new[] { 1f, .97f, .47f, 0f, .42f };
        var counts = new int[5];
        for (var cx = -nx; cx <= nx; cx++)
            for (var cz = 0; cz < nz; cz++)
            {
                rng ^= rng << 13; rng ^= rng >> 17; rng ^= rng << 5;
                var roll = (rng & 0xffffff) / 16777216f;
                var mode = 0; var acc = 0f;
                for (var k = 0; k < 5; k++) { acc += t5[k]; if (roll < acc) { mode = k; break; } mode = 4; }
                var nearExit = Mathf.Abs(cx) <= 2 && cz <= 1;
                if (nearExit && ladderLamps == "hold") mode = 0;
                if (nearExit && ladderLamps == "dead") mode = 3;
                if (ladderLamps == "steady") mode = 0;
                counts[mode]++;
                var c = new Vector3(cx * 3f, MazeH, -1.5f - 3f * cz);
                var shadow = Vector3.Distance(new Vector3(c.x, 0f, c.z), Vector3.zero) < 9f;
                var l = MapLamp(mazeRoot, c, shadow, Id.Emitter);
                var lv = levels[mode];
                l.intensity *= lv;
                if (lv <= 0f) l.enabled = false;
                // the lens follows the level (dead = a dark lens)
                var lensR = l.transform.parent.GetComponentsInChildren<Renderer>().LastOrDefault(r => r.name == "map lamp lens");
                if (lensR != null && lv < .999f)
                {
                    var lm = new Material(lensR.sharedMaterial) { name = "L0 map lamp lens x" + lv.ToString("0.00") };
                    lm.SetColor("_EmissionColor", new Color(1f, .96f, .84f) * 2.6f * Mathf.Max(lv, .02f));
                    lensR.sharedMaterial = lm;
                }
                if (nearExit) lampLog.Add("exit cell " + cx + "," + cz + " " + names[mode] + (shadow ? " shadowed" : ""));
            }
        Log("hall lamps (" + ladderLamps + ", seed " + ladderSeed + "): steady " + counts[0] + ", stutter " + counts[1] + ", failing " + counts[2] + ", dead " + counts[3] + ", dim " + counts[4] + "; " + string.Join("; ", lampLog.ToArray()));
    }

    {

        var yc = (yTop + yBot) * .5f;

        // side walls start at z 0 (the maze face of the door wall is at z -0.08; a wall end there would show in the maze)

        // door wall, stair layer z 0..0.08, one segment per floor [YD(n), YD(n) + FF] with a 1.0 x 2.1 opening
        for (var n = FloorMin; n <= FloorMax + 1; n++)
        {
            var yd = YD(n);
            var s0 = n == FloorMax + 1 ? yBot : yd;
            var s1 = n == FloorMin ? yTop : yd + FF;
            Box(shaft, "door wall L f" + n, new Vector3(-1.0f, (s0 + s1) * .5f, .04f), new Vector3(1.0f, s1 - s0, .08f), wallM, Id.Background);

        }
        // the far side of each lower door: the new floor (not modelled) -> a closed leaf is enough
        for (var n = FloorMin; n <= FloorMax; n++) BuildFloor(n);
        BuildFloorBottom(FloorMax + 1);
    }

        Box(shaft, "far floor top (under the turn)", new Vector3(0f, y - .006f, (ZD + ZFarFace) * .5f), new Vector3(2.84f, .012f, ZFarFace - ZD), landTopM, Id.OtherFloor);
        if (dir == "A") Box(shaft, "far floor cove", new Vector3(0f, y + .05f, ZFarFace - .01f), new Vector3(2.84f, .1f, .02f), Flat("A cove base (black vinyl)", Hex("1F1E1C"), .35f), Id.Background);
        if (dir == "C")
        {
            var ceil = FrontRoomsSurfaces.Room(RoomRule.Lobby, FrontRoomsSurfaces.Slot.Ceiling);
            Box(shaft, "far landing drop ceiling", new Vector3(0f, y + MazeH + .02f, (ZNear + ZD) * .5f), new Vector3(2.84f, .04f, ZD - ZNear), ceil, Id.Background);
            Box(shaft, "far landing bulkhead", new Vector3(0f, y + MazeH + .2f, ZD), new Vector3(2.84f, .4f, .04f), wallM, Id.Background);
            Box(shaft, "far landing cove", new Vector3(0f, y + .05f, ZNear + .01f), new Vector3(2.84f, .1f, .02f), coveM, Id.Background);
        }

        StairDoor(n);
        FloorNumber(n);
        LandingLight(n, true);
    }

    static void BuildFloor(int n)
    {
        var yd = YD(n); var yt = YT(n);
        var pathD = n == 0 || n == 1;
        // landings: D (door) z 0.08..1.74 and T (turn) z 4.26..5.92, full width 2.84
        Landing(n, "D", yd, ZNear, ZD, pathD ? Id.Walk : Id.OtherFloor);
        Landing(n, "T", yt, ZT, ZFarFace, n == 0 ? Id.Walk : Id.OtherFloor);
        // flight 1: D(n) -> T(n), x -1.42..-0.20, descends along +z ; flight 2: T(n) -> D(n+1), x 0.20..1.42, along -z
        Flight(n, 1, -XFace, -WellHalf, ZD, yd, 1f);
        Flight(n, 2, WellHalf, XFace, ZT, yt, -1f);
        Guards(n);
        WallRails(n);
        if (dir == "A" || (dir == "B" && remedy)) Dado(n);
        if (dir == "C") Skirts(n);
        StairDoor(n);
        FloorNumber(n);
        LandingLight(n, true);
        LandingLight(n, false);
        if (dir != "C") Services(n);
        // FIX (critic era 3, gameplay 2): no gates. There is no up flight (the slab closes the shaft over the exit
        // landing) and no down flight (the far landing is the shaft floor). Gate() stays for the option in 10 1.5.
        if (n == 0 && dir != "C") TopOfShaft();
    }

    static void Flight(int n, int f, float x0, float x1, float zTop, float yTop, float dz)
    {
        var p = Kit("Flight", n);
        var walk = OnPath(n, f) ? Id.Walk : Id.OtherFloor;
        var w = x1 - x0; var xc = (x0 + x1) * .5f;
        if (dir == "A")
        {
            // steel pan stair: channel stringers, pans with 40 mm concrete fill, closed plate risers, open soffit
            for (var k = 1; k <= Treads; k++)
            {
                var y = yTop - k * Rise;
                var za = Zk(zTop, dz, k - 1); var zb = Zk(zTop, dz, k);
                var zc = (za + zb) * .5f + dz * .0125f;   // fill reaches 25 mm onto the nosing
                Box(p, "tread fill f" + n + "." + f + "." + k, new Vector3(xc, y - .02f, zc), new Vector3(w - .06f, .04f, Going + .025f), treadM, walk);
                Box(p, "tread pan f" + n + "." + f + "." + k, new Vector3(xc, y - .045f, zc), new Vector3(w - .06f, .01f, Going + .025f), steelM, Id.Background);
                Box(p, "pan nosing lip f" + n + "." + f + "." + k, new Vector3(xc, y - .02f, zb + dz * .025f), new Vector3(w - .06f, .045f, .006f), nosingM, Id.Background);
                if (chipM != null && (n == 0 || n == 1)) NosingChips(p, n, f, k, xc, w, y, zb + dz * .028f, dz);
            }
            for (var k = 0; k <= Treads; k++)
            {
                var zr = Zk(zTop, dz, k) + dz * .003f;
                var yr = yTop - k * Rise - Rise * .5f;
                Box(p, "riser plate f" + n + "." + f + "." + k, new Vector3(xc, yr - .02f, zr), new Vector3(w - .06f, Rise, .005f), steelM, Id.Background);
            }
            // stringers (C10 channel ~ 250 mm), along the pitch line, tops 60 mm above the nosings
            var pitch = Rise / Going;
            var a = new Vector3(0f, yTop + .1f * pitch - .065f, zTop - dz * .1f);
            var b = new Vector3(0f, yTop - Treads * Rise - .1f * pitch - .065f, Zk(zTop, dz, Treads) + dz * .1f);
            foreach (var sx in new[] { x0 + .03f, x1 - .03f })
            {
                a.x = sx; b.x = sx;
                Seg(p, "stringer web f" + n + "." + f, a, b, .006f, .254f, steelM, Id.Background);
                // FIX (critic era 6): C10 channel flanges (66 x 11 mm), toes turned away from the treads
                var toe = sx < xc ? -1f : 1f;
                var nrm = Vector3.Cross(Vector3.right, (b - a).normalized); if (nrm.y < 0f) nrm = -nrm;
                foreach (var s in new[] { 1f, -1f })
                {
                    var off = nrm * (s * (.127f - .0055f)) + Vector3.right * (toe * .030f);
                    Seg(p, "stringer flange f" + n + "." + f + (s > 0 ? "t" : "b"), a + off, b + off, .066f, .011f, steelM, Id.Background);
                }
            }
        }
        else
        {
            // monolithic flight (cast-in-place concrete in B; the same solid, carpeted and plastered, in C)
            var treadsMB = new MB(); var risersMB = new MB(); var bodyMB = new MB();
            var cosA = Going / Mathf.Sqrt(Going * Going + Rise * Rise);
            var wv = .12f / cosA;
            Func<float, float> ys = z => yTop - Rise * Mathf.Abs(z - zTop) / Going - Rise - wv;
            for (var k = 1; k <= Treads; k++)
            {
                var y = yTop - k * Rise;
                var za = Zk(zTop, dz, k - 1); var zb = Zk(zTop, dz, k);
                treadsMB.Quad(new Vector3(x0, y, za), new Vector3(x1, y, za), new Vector3(x1, y, zb), new Vector3(x0, y, zb), Vector3.up);
                foreach (var sx in new[] { x0, x1 })
                    bodyMB.Quad(new Vector3(sx, y, za), new Vector3(sx, y, zb), new Vector3(sx, ys(zb), zb), new Vector3(sx, ys(za), za), sx == x0 ? Vector3.left : Vector3.right);
                // riser k at zb, from y down to y - Rise, facing +dz
                risersMB.Quad(new Vector3(x0, y, zb), new Vector3(x1, y, zb), new Vector3(x1, y - Rise, zb), new Vector3(x0, y - Rise, zb), new Vector3(0f, 0f, dz));
            }
            // riser 0 (the landing edge) carries the finish too, 1 mm proud of the slab edge
            var z0r = zTop + dz * .001f;
            risersMB.Quad(new Vector3(x0, yTop, z0r), new Vector3(x1, yTop, z0r), new Vector3(x1, yTop - Rise, z0r), new Vector3(x0, yTop - Rise, z0r), new Vector3(0f, 0f, dz));
            var z9 = Zk(zTop, dz, Treads);
            var sl = new Vector3(0f, -Rise * Treads, z9 - zTop).normalized;
            var sn = Vector3.Cross(Vector3.right, sl); if (sn.y > 0f) sn = -sn;
            bodyMB.Quad(new Vector3(x0, ys(zTop), zTop), new Vector3(x1, ys(zTop), zTop), new Vector3(x1, ys(z9), z9), new Vector3(x0, ys(z9), z9), sn);
            bodyMB.Quad(new Vector3(x0, yTop - Rise, zTop), new Vector3(x1, yTop - Rise, zTop), new Vector3(x1, ys(zTop), zTop), new Vector3(x0, ys(zTop), zTop), new Vector3(0f, 0f, -dz));
            var tag = "f" + n + "." + f;
            MeshObj(p, "flight treads " + tag, treadsMB.Build("treads " + dir + tag), treadM, walk);
            MeshObj(p, "flight risers " + tag, risersMB.Build("risers " + dir + tag), riserM, Id.Background);
            MeshObj(p, "flight body " + tag, bodyMB.Build("body " + dir + tag), bodyM, Id.Background);
            // nosings: B safety-yellow 50 mm stripes, C brass nosing bars over the carpet
            for (var k = 1; k <= Treads; k++)
            {
                var y = yTop - k * Rise;
                var zb = Zk(zTop, dz, k);
                if (dir == "B")
                {
                    if (chipM != null && (n == 0 || n == 1)) WornStripe(p, tag + "." + k, xc, w, y, zb - dz * .025f, k);
                    else Box(p, "nosing stripe " + tag + "." + k, new Vector3(xc, y + .0006f, zb - dz * .025f), new Vector3(w - .02f, .002f, .05f), nosingM, Id.Background);
                }
                else Box(p, "brass nosing " + tag + "." + k, new Vector3(xc, y + .004f, zb - dz * .012f), new Vector3(w - .04f, .012f, .028f), nosingM, Id.Background);
            }
        }
    }

    static float Hash01(int a, int b, int c)
    {
        var h = (uint)(a * 73856093) ^ (uint)(b * 19349663) ^ (uint)(c * 83492791);
        h ^= h << 13; h ^= h >> 17; h ^= h << 5;
        return (h & 0xffffff) / 16777216f;
    }

    /// <summary>A: red-oxide primer showing through the stair enamel in a few chips on the nosing lip, concentrated on
    /// the walk line (the inner third of the flight, by the guard rail and by the wall rail).</summary>
    static void NosingChips(Transform p, int n, int f, int k, float xc, float w, float y, float zFace, float dz)
    {
        for (var i = 0; i < 5; i++)
        {
            var u = Hash01(n * 31 + f, k, i);
            var v = Hash01(n * 17 + f, k, i + 11);
            // two walk lines, 0.30 m in from each side; most chips there, a few anywhere
            var side = u < .5f ? -1f : 1f;
            var x = i < 4 ? xc + side * (w * .5f - .30f) + (v - .5f) * .28f : xc + (v - .5f) * (w - .1f);
            var len = .015f + Hash01(k, i, n + 5) * .045f;
            var hgt = .006f + Hash01(i, k, f + 9) * .016f;
            Box(p, "nosing chip f" + n + "." + f + "." + k + "." + i, new Vector3(x, y - .004f - Hash01(f, i, k + 3) * .02f, zFace), new Vector3(len, hgt, .0015f), chipM, Id.Background);
        }
    }

    /// <summary>B: a 50 mm safety-yellow stripe worn through on the walk lines: crisp only at the ends, a ghost of
    /// paint on concrete where feet land.</summary>
    static void WornStripe(Transform p, string tag, float xc, float w, float y, float z, int k)
    {
        var x0 = xc - (w - .02f) * .5f; var x1 = xc + (w - .02f) * .5f;
        // worn zones: 0.45 m wide, centred 0.30 m in from each side (the two walk lines)
        var wa0 = x0 + .30f - .225f; var wa1 = x0 + .30f + .225f;
        var wb0 = x1 - .30f - .225f; var wb1 = x1 - .30f + .225f;
        var segs = new List<Vector2> { new Vector2(x0, wa0), new Vector2(wa1, wb0), new Vector2(wb1, x1) };
        foreach (var s in segs)
            if (s.y - s.x > .01f) Box(p, "nosing stripe " + tag, new Vector3((s.x + s.y) * .5f, y + .0006f, z), new Vector3(s.y - s.x, .002f, .05f), nosingM, Id.Background);
        foreach (var s in new[] { new Vector2(wa0, wa1), new Vector2(wb0, wb1) })
            for (var i = 0; i < 2; i++)
            {
                var a = Mathf.Lerp(s.x + .05f, s.y - .05f, Hash01(k, i, 7)); var len = .012f + Hash01(i, k, 9) * .03f;
                Box(p, "nosing stripe ghost " + tag + "." + i, new Vector3(a, y + .0005f, z + (Hash01(k, i, 2) - .5f) * .03f), new Vector3(len, .002f, .008f + Hash01(i, k, 4) * .012f), chipM, Id.Background);
            }
    }

    /// <summary>Guard on the well side of each flight, across the well at the turn landing, and along the whole top
    /// edge of the exit landing (with no flight above, the flight-2 side of that edge is a 3.5 m drop).</summary>
    static void Guards(int n)
    {
        var p = Kit("Guard", n);
        GuardRun(p, n, 1, -WellHalf - .03f, ZD, YD(n), 1f);
        GuardRun(p, n, 2, WellHalf + .03f, ZT, YT(n), -1f);
        // 42 in = 1.067 m guards at the landing edges
        var xw0 = -WellHalf - .03f; var xw1 = WellHalf + .03f;
        LandingGuard(p, n, "D", ZD - .03f, YD(n), xw0, n == FloorMin ? XFace - .02f : xw1);
        LandingGuard(p, n, "T", ZT + .03f, YT(n), xw0, xw1);
    }

    static void GuardRun(Transform p, int n, int f, float x, float zTop, float yTop, float dz)
    {
        var tag = "f" + n + "." + f;
        var zBot = Zk(zTop, dz, Treads);                       // last nosing line
        var yBotNose = yTop - Treads * Rise;
        var railH = .91f;                                     // 34-38 in above the nosing line
        var a = new Vector3(x, yTop + railH, zTop); var b = new Vector3(x, yBotNose + railH, zBot);
        if (dir == "A")
        {
            Cyl(p, "guard rail " + tag, a, b, .021f, railM, Id.Background);
            Seg(p, "guard rail wear " + tag, a + Vector3.up * .017f, b + Vector3.up * .017f, .012f, .006f, railTopM, Id.Background);
            var ba = new Vector3(x, yTop + .10f, zTop); var bb = new Vector3(x, yBotNose + .10f, zBot);
            Seg(p, "guard bottom rail " + tag, ba, bb, .025f, .025f, steelM, Id.Background);
            // FIX (critic era 4): pickets at <= 0.10 m centres, under 4 in clear
            var count = Mathf.CeilToInt(Mathf.Abs(zBot - zTop) / .10f);
            for (var i = 1; i < count; i++)
            {
                var t = i / (float)count;
                var top = Vector3.Lerp(a, b, t) - Vector3.up * .02f; var bot = Vector3.Lerp(ba, bb, t);
                Box(p, "picket " + tag + "." + i, (top + bot) * .5f, new Vector3(.0127f, (top - bot).magnitude, .0127f), steelM, Id.Background);
            }
        }
        else if (dir == "B")
        {
            Cyl(p, "guard top rail " + tag, a, b, .024f, railM, Id.Background);
            Seg(p, "guard rail polish " + tag, a + Vector3.up * .02f, b + Vector3.up * .02f, .012f, .006f, railTopM, Id.Background);
            var ba = new Vector3(x, yTop + .10f, zTop); var bb = new Vector3(x, yBotNose + .10f, zBot);
            Cyl(p, "guard bottom rail " + tag, ba, bb, .021f, steelM, Id.Background);
            for (var i = 0; i <= 2; i++)
            {
                var t = i / 2f;
                var zp = Mathf.Lerp(zTop, zBot, t) - dz * (i == 0 ? -.06f : i == 2 ? .06f : 0f);
                var yNose = yTop - Rise * Mathf.Abs(zp - zTop) / Going;
                var yTread = yTop - Rise * Mathf.Ceil(Mathf.Abs(zp - zTop) / Going - .001f);
                Cyl(p, "guard post " + tag + "." + i, new Vector3(x, yTread, zp), new Vector3(x, yNose + railH + .02f, zp), .024f, steelM, Id.Background);
                Cyl(p, "post flange " + tag + "." + i, new Vector3(x, yTread, zp), new Vector3(x, yTread + .012f, zp), .055f, steelM, Id.Background);
            }
            // FIX (critic era 4): woven-wire mesh infill between the bottom rail and the top rail
            var mb = new MB();
            var nrm = x < 0f ? Vector3.left : Vector3.right;
            mb.Quad(ba + Vector3.up * .02f, bb + Vector3.up * .02f, b - Vector3.up * .03f, a - Vector3.up * .03f, nrm);
            MeshObj(p, "guard mesh " + tag, mb.Build("mesh " + dir + tag), meshM, Id.Background);
        }
        else
        {
            // FIX (critic era 4): a round oak rail, 48 mm (1.9 in, inside the 1.5-2 in grip), on 32 mm square
            // balusters, 3 per tread (93 mm centres, 61 mm clear), turned-ish newels at the ends
            Cyl(p, "oak handrail " + tag, a + Vector3.up * .02f, b + Vector3.up * .02f, .024f, railM, Id.Background);
            Seg(p, "oak shoe rail " + tag, new Vector3(x, yTop + .02f, zTop), new Vector3(x, yBotNose + .02f, zBot), .06f, .03f, steelM, Id.Background);
            for (var k = 1; k <= Treads; k++)
                for (var j = 0; j < 3; j++)
                {
                    var zc = Zk(zTop, dz, k - 1) + dz * Going * ((1f + 2f * j) / 6f);
                    var yTread = yTop - k * Rise;
                    var yRail = yTop + railH - Rise * Mathf.Abs(zc - zTop) / Going;
                    Box(p, "baluster " + tag + "." + k + "." + j, new Vector3(x, (yTread + yRail) * .5f, zc), new Vector3(.032f, yRail - yTread, .032f), steelM, Id.Background);
                }
            foreach (var e in new[] { 0, 1 })
            {
                var zc = e == 0 ? zTop - dz * .05f : zBot + dz * .05f;
                var yb = e == 0 ? yTop : yTop - 10 * Rise;
                Box(p, "newel " + tag + "." + e, new Vector3(x, yb + .58f, zc), new Vector3(.09f, 1.16f, .09f), steelM, Id.Background);
                Box(p, "newel cap " + tag + "." + e, new Vector3(x, yb + 1.18f, zc), new Vector3(.12f, .04f, .12f), railM, Id.Background);
            }
        }
    }

    static void LandingGuard(Transform p, int n, string which, float z, float y, float x0, float x1)
    {
        var tag = which + n;
        var h = 1.067f;
        var w = x1 - x0;
        if (dir == "C")
        {
            Cyl(p, "landing oak rail " + tag, new Vector3(x0, y + h, z), new Vector3(x1, y + h, z), .024f, railM, Id.Background);
            Box(p, "landing oak shoe " + tag, new Vector3((x0 + x1) * .5f, y + .015f, z), new Vector3(w, .03f, .06f), steelM, Id.Background);
            var cnt = Mathf.CeilToInt(w / .09f);
            for (var i = 1; i < cnt; i++) Box(p, "landing baluster " + tag + "." + i, new Vector3(Mathf.Lerp(x0, x1, i / (float)cnt), y + h * .5f, z), new Vector3(.032f, h, .032f), steelM, Id.Background);
            Box(p, "landing newel " + tag, new Vector3(x1, y + .6f, z), new Vector3(.09f, 1.2f, .09f), steelM, Id.Background);
            return;
        }
        Cyl(p, "landing guard rail " + tag, new Vector3(x0 - .02f, y + h, z), new Vector3(x1 + .02f, y + h, z), dir == "A" ? .021f : .024f, railM, Id.Background);

    /// <summary>A: the band, 2 mm proud, following the flights and wrapping round the landings (BSS 120 Fig. 2.7.5),
    /// FIX (critic era 8): no dead end at the landing corners; a 12 mm dark cut-in line along its top is the slope
    /// cue (critic era 7: the band is now lighter than the steel by more than 15 L*).</summary>
    static void Dado(int n)
    {
        var p = Kit("Paint", n);
        var mb = new MB(); var ln = new MB();
        var top = dadoTop; var under = .35f;
        var yd = YD(n); var yt = YT(n);
        foreach (var side in new[] { -1, 1 })
        {
            var x = side * (XFace - .002f);
            var nrm = side < 0 ? Vector3.right : Vector3.left;
            SideBand(mb, ln, x, ZNear, ZD, yd - .06f, yd - .06f, yd + top, yd + top, nrm);
            SideBand(mb, ln, x, ZT, ZFarFace, yt - .06f, yt - .06f, yt + top, yt + top, nrm);
            // west wall carries flight 1 (D n -> T n); east wall carries flight 2 (T n -> D n+1)
            if (side < 0) SideBand(mb, ln, x, ZD, ZT, yd - under, yt - under, yd + top, yt + top, nrm);
            else SideBand(mb, ln, x, ZT, ZD, yt - under, YD(n + 1) - under, yt + top, YD(n + 1) + top, nrm);
        }
        // end walls: door wall at the D landing (either side of the door), far wall at the T landing
        if (!noEndDado)
        {
            var zn = ZNear + .002f; var zf = ZFarFace - .002f;
            EndBand(mb, ln, zn, -XFace, -.56f, yd - .06f, yd + top, Vector3.forward);
            EndBand(mb, ln, zn, .56f, XFace, yd - .06f, yd + top, Vector3.forward);
            EndBand(mb, ln, zf, -XFace, XFace, yt - .06f, yt + top, Vector3.back);
        }
        MeshObj(p, "dado band f" + n, mb.Build("dado " + n), dadoM, Id.Background);
        if (cutInM != null) MeshObj(p, "band cut-in line f" + n, ln.Build("cut-in " + n), cutInM, Id.Background);
    }

    /// <summary>A, far floor: the band round the far landing (side walls and the door wall).</summary>
    static void DadoBottom(int n)
    {
        var mb = new MB(); var ln = new MB();
        var top = dadoTop; var yd = YD(n);
        foreach (var side in new[] { -1, 1 })
        {
            var x = side * (XFace - .002f);
            var nrm = side < 0 ? Vector3.right : Vector3.left;
            if (side < 0) SideBand(mb, ln, x, ZNear, ZD, yd - .06f, yd - .06f, yd + top, yd + top, nrm);
            else SideBand(mb, ln, x, ZNear, ZD - .002f, yd - .06f, yd - .06f, yd + top, yd + top, nrm);
        }
        if (!noEndDado)
        {
            var zn = ZNear + .002f;
            EndBand(mb, ln, zn, -XFace, -.56f, yd - .06f, yd + top, Vector3.forward);
            EndBand(mb, ln, zn, .56f, XFace, yd - .06f, yd + top, Vector3.forward);
        }
        MeshObj(shaft, "dado band far floor", mb.Build("dado far"), dadoM, Id.Background);
        if (cutInM != null) MeshObj(shaft, "band cut-in line far floor", ln.Build("cut-in far"), cutInM, Id.Background);
    }

    /// <summary>A band quad on a side wall (x fixed) from zA to zB, bottoms loA / loB, tops topA / topB, and its
    /// 12 mm cut-in line (1 mm further out) along the top.</summary>
    static void SideBand(MB mb, MB ln, float x, float zA, float zB, float loA, float loB, float topA, float topB, Vector3 nrm)
    {
        mb.Quad(new Vector3(x, loA, zA), new Vector3(x, loB, zB), new Vector3(x, topB, zB), new Vector3(x, topA, zA), nrm);
        var xl = x + nrm.x * .001f;
        ln.Quad(new Vector3(xl, topA - .008f, zA), new Vector3(xl, topB - .008f, zB), new Vector3(xl, topB + .004f, zB), new Vector3(xl, topA + .004f, zA), nrm);
    }

    static void EndBand(MB mb, MB ln, float z, float xA, float xB, float lo, float top, Vector3 nrm)
    {
        mb.Quad(new Vector3(xA, lo, z), new Vector3(xB, lo, z), new Vector3(xB, top, z), new Vector3(xA, top, z), nrm);
        var zl = z + nrm.z * .001f;
        ln.Quad(new Vector3(xA, top - .008f, zl), new Vector3(xB, top - .008f, zl), new Vector3(xB, top + .004f, zl), new Vector3(xA, top + .004f, zl), nrm);
    }

    static GameObject StairDoor(int n)
    {
        var p = Kit("Door", n);
        var y = YD(n);
        var origin = new Vector3(.5f, y, 0f);   // hinge-jamb edge on the wall centre line
        var frameKit = dir == "C" ? "Kit_DoorFrame_Wood" : "Kit_DoorFrame_Steel";
        var leafKit = dir == "C" ? "Kit_DoorLeaf_Veneer" : "Kit_DoorLeaf_Steel";
        var frame = KitSpawn(frameKit, p, origin, DoorRot, Id.Door);
        var leaf = KitSpawn(leafKit, p, origin, DoorRot, Id.Door);
        if (leaf == null) return null;
        if (n == 0)
        {
            exitLeaf = leaf.transform;
            foreach (var r in leaf.GetComponentsInChildren<Renderer>(true)) leafRs.Add(r);
        }
        if (doorLeafM != null) Recolour(leaf, doorLeafM);
        if (dir != "C" && frame != null) Recolour(frame, Flat(dir + " frame enamel", dir == "A" ? Hex("5F5A50") : Hex("3E3833"), .45f));
        // closer on the pull face (stair side): body at closer_mount_s, regular arm to the frame shoe
        Closer(leaf.transform);
        var live = n == FloorMax + 1;   // the far door: the one door the player opens from the stair side
        if (live)
        {
            // FIX (critic gameplay 2): pull trim on the stair face (a lever at the escutcheon) and a dark
            // floor-access plate at 5 ft. The exit door stays blank with its NO RE-ENTRY plate.
            farLeaf = leaf.transform; farLeafRest = farLeaf.rotation;
            var lever = FrontRoomsKitLibrary.Spawn("Kit_Lock_Lever", leaf.transform, new Vector3(.022f, .968f, .92f), Quaternion.Euler(0f, 90f, 0f), null, false);
            if (lever != null)
            {
                farLever = lever.transform; farLeverRest = farLever.localRotation;
                foreach (var r in lever.GetComponentsInChildren<Renderer>(true)) { ids[r] = Id.Door; trimRs.Add(r); }
                var rose = Cyl(p, "far lever rose", LeafPoint(leaf.transform, new Vector3(.022f, .968f, .92f)), LeafPoint(leaf.transform, new Vector3(.032f, .968f, .92f)), .03f, Flat("lever rose (satin chrome)", Hex("B4B6B2"), .6f, .9f), Id.Door);
                trimRs.Add(rose);
            }
            else Log("MISSING Kit_Lock_Lever");
            var plate = Box(p, "far door floor-access plate (placeholder) f" + n, LeafPoint(leaf.transform, new Vector3(.026f, 1.524f, .5f)), new Vector3(.30f, .20f, .004f), farPlateM, Id.Sign);
            trimRs.Add(plate);
        }
        else
        {
            // the exit door's stair face: blank, closer, a NO RE-ENTRY plate at 5 ft (NYC LL5 1973 C26-608.5;
            // strings narrative's)
            Box(p, "door re-entry plate (placeholder) f" + n, LeafPoint(leaf.transform, new Vector3(.026f, 1.524f, .5f)), new Vector3(.30f, .20f, .004f), dir == "B" ? Flat("B plate red", Hex("A3271E"), .4f) : signPlateM, Id.Sign);
        }
        // FIX (critic era 13): the stair ID sign (stair, level, terminus, roof access) on the WALL beside the latch
        // jamb at 5 ft, readable with the door open or shut (UBC 1991 3306(p)). C keeps only its suite plate.
        if (dir != "C") Box(p, "stair ID sign (placeholder) f" + n, new Vector3(-.80f, y + 1.524f, ZNear + .003f), new Vector3(.30f, .36f, .006f), idPlateM, Id.Sign);
        return leaf;
    }

    static Vector3 LeafPoint(Transform leaf, Vector3 local)
    {
        return root.InverseTransformPoint(leaf.TransformPoint(local));
    }

    static void Closer(Transform leaf)
    {
        var body = FrontRoomsKitLibrary.Spawn("Kit_DoorCloser_Body", leaf, new Vector3(.022f, 2.0275f, .545f), Quaternion.Euler(0f, 90f, 0f), null, false);
        if (body == null) { Log("MISSING closer body"); return; }
        foreach (var r in body.GetComponentsInChildren<Renderer>(true)) ids[r] = Id.Door;
        // regular arm: spindle door (0.062, 2.068, 0.520) -> elbow -> shoe on the frame head (0.125, 2.13, 0.10)
        var s = new Vector2(.062f, .520f); var shoe = new Vector2(.125f, .100f);
        var d = (shoe - s).magnitude;
        var aLen = .24f; var fLen = .26f;
        var along = (aLen * aLen - fLen * fLen + d * d) / (2f * d);
        var hgt = Mathf.Sqrt(Mathf.Max(0f, aLen * aLen - along * along));
        var u = (shoe - s) / d; var perp = new Vector2(u.y, -u.x); if (perp.x < 0f) perp = -perp;
        var e = s + u * along + perp * hgt;
        var arm = FrontRoomsKitLibrary.Spawn("Kit_DoorCloser_Arm", leaf, new Vector3(s.x, 2.068f, s.y), Quaternion.Euler(0f, Mathf.Atan2(-(e.y - s.y), e.x - s.x) * Mathf.Rad2Deg, 0f), null, false);
        var fore = FrontRoomsKitLibrary.Spawn("Kit_DoorCloser_Forearm", leaf, new Vector3(e.x, 2.08f, e.y), Quaternion.Euler(0f, Mathf.Atan2(-(shoe.y - e.y), shoe.x - e.x) * Mathf.Rad2Deg, 0f), null, false);
        var sh = FrontRoomsKitLibrary.Spawn("Kit_DoorCloser_Shoe", leaf, new Vector3(shoe.x, 2.13f, shoe.y), Quaternion.identity, null, false);
        foreach (var g in new[] { arm, fore, sh }) if (g != null) foreach (var r in g.GetComponentsInChildren<Renderer>(true)) ids[r] = Id.Door;
    }

    /// <summary>The exit door on the exit floor (n = 0): crossbar on the push face (maze side), lit EXIT
    /// sign over the frame (C23: Kit_ExitSign locked to LIT), and the two A13 lock states.</summary>
    static void BuildExitDoor()
    {
        var p = Kit("ExitDoor", 0);
        var leaf = exitLeaf;
        if (leaf == null) { Log("no exit leaf found"); return; }
        var bar = FrontRoomsKitLibrary.Spawn("Kit_ExitDevice_Crossbar", leaf, new Vector3(-.022f, 1.000f, .900f), Quaternion.Euler(0f, -90f, 0f), new Vector3(-1f, 1f, 1f), false);
        if (bar != null) { bar.transform.localScale = new Vector3(-1f, 1f, 1f); foreach (var r in bar.GetComponentsInChildren<Renderer>(true)) { ids[r] = Id.Door; barRs.Add(r); } }
        else Log("MISSING crossbar");
        var gain = signGain > 0f ? signGain : SignGainDefault;
        // EXIT sign: frame anchor exit_sign_p (-0.080, 2.180, 0.500) in door root -> world (0, 2.18, -0.08), face -z
        var sign = KitSpawn("Kit_ExitSign", p, new Vector3(0f, 2.18f, -.08f), Quaternion.Euler(0f, 180f, 0f), Id.Sign);
        if (sign != null)
        {
            flatSign = sign;
            flatDrv = SignDriver(sign, gain);
            foreach (var r in sign.GetComponentsInChildren<Renderer>(true)) signRs.Add(r);
            Log("EXIT sign LIT, level " + flatDrv.Level.ToString("0.00") + ", EmissionGain " + gain.ToString("0.00"));
        }
        // FIX (critic gameplay 5): the end-mounted double-face sign for an exit on a corridor side wall (shots with
        // FEndSign swap it in for the flat sign)
        var end = KitSpawn("Kit_ExitSign_End", p, new Vector3(0f, 2.18f, -.08f), Quaternion.Euler(0f, 180f, 0f), Id.Sign);
        if (end != null)
        {
            endSign = end;
            endDrv = SignDriver(end, gain);
            foreach (var r in end.GetComponentsInChildren<Renderer>(true)) signRs.Add(r);
            end.SetActive(false);
        }
        else Log("MISSING Kit_ExitSign_End");
        BuildLockStates(p);
        // LOD diagnostics for the door-head strip (critic era 9)
        foreach (var lg in leaf.GetComponentsInChildren<LODGroup>(true))
        {
            var lods = lg.GetLODs();
            for (var i = 0; i < lods.Length; i++)
                foreach (var r in lods[i].renderers)
                    if (r != null) Log("exit leaf LOD" + i + " " + r.name + " bounds y " + r.bounds.min.y.ToString("0.000") + ".." + r.bounds.max.y.ToString("0.000") + " screenRel " + lods[i].screenRelativeTransitionHeight.ToString("0.000"));
        }
    }

    static FrontRoomsExitSign SignDriver(GameObject sign, float gain)
    {
        var drv = sign.GetComponent<FrontRoomsExitSign>();
        if (drv == null) drv = sign.AddComponent<FrontRoomsExitSign>();
        drv.SelfTick = false;
        drv.EmissionGain = gain;
        drv.Configure(0u, FrontRoomsExitSign.Temperament.Lit, null, 0f);
        return drv;
    }

    static void SetSignGain(float gain)
    {
        foreach (var d in new[] { flatDrv, endDrv })
        {
            if (d == null) continue;
            d.EmissionGain = gain;
            d.Configure(0u, FrontRoomsExitSign.Temperament.Lit, null, 0f);
        }
    }

    // ---------------------------------------------------------------- A13 lock states (maze side of the exit door)
    const float ChainWire = .0127f, ChainLen = .064f, ChainWid = .044f, ChainPitch = .041f;   // 1/2 in proof coil

    /// <summary>FIX (critic gameplay 1, A13): LOCKED = a chain wrapped twice round the crossbar near the latch end,
    /// run through an eye bolt in the wall beside the latch jamb, its two ends joined by a padlock that hangs below
    /// the bar. UNLOCKED = the same chain and the open padlock on the floor against the wall, latch side (one state
    /// change, no blink). The EXIT sign stays LIT in both.</summary>
    static void BuildLockStates(Transform p)
    {
        lockedSet = new GameObject("A13 LOCKED (chain + padlock)");
        lockedSet.transform.SetParent(p, false);
        unlockedSet = new GameObject("A13 UNLOCKED (chain on the floor)");
        unlockedSet.transform.SetParent(p, false);
        var L = lockedSet.transform; var U = unlockedSet.transform;
        // eye bolt (both states): wall beside the latch jamb, maze face z -0.08
        var eye = new Vector3(-.64f, 1.0f, -.115f);
        Cyl(p, "eye bolt plate", new Vector3(-.64f, 1.0f, -.08f), new Vector3(-.64f, 1.0f, -.086f), .03f, padlockM, Id.Door);
        Cyl(p, "eye bolt shank", new Vector3(-.64f, 1.0f, -.086f), new Vector3(-.64f, 1.0f, -.095f), .008f, padlockM, Id.Door);
        for (var i = 0; i < 10; i++)
        {
            var a0 = i * Mathf.PI * 2f / 10f; var a1 = (i + 1) * Mathf.PI * 2f / 10f;
            var e0 = eye + new Vector3(0f, Mathf.Sin(a0), -Mathf.Cos(a0) + 1f) * .022f - new Vector3(0f, 0f, .022f);
            var e1 = eye + new Vector3(0f, Mathf.Sin(a1), -Mathf.Cos(a1) + 1f) * .022f - new Vector3(0f, 0f, .022f);
            Seg(p, "eye bolt ring " + i, e0, e1, .009f, .009f, padlockM, Id.Door);
        }
        // LOCKED chain path: the wrap round the bar (axis along x at y 1.0, z -0.092), then to the eye, the sag
        // to the padlock, and from the wrap start down to the padlock
        var path = new List<Vector3>();
        var bc = new Vector3(0f, 1.0f, -.092f); var rr = .034f;
        var pre = new List<Vector3> { new Vector3(-.465f, .80f, -.13f), new Vector3(-.40f, .815f, -.13f), new Vector3(-.31f, .87f, -.128f), new Vector3(-.235f, .945f, -.126f) };
        path.AddRange(pre);
        for (var k = 0; k <= 24; k++)
        {
            var t = k / 24f; var ang = t * Mathf.PI * 4f;
            path.Add(new Vector3(-.22f - .08f * t, bc.y - rr * Mathf.Cos(ang), bc.z - rr * Mathf.Sin(ang)));
        }
        path.Add(new Vector3(-.45f, 1.0f, -.13f));
        path.Add(eye + new Vector3(.02f, 0f, -.02f));
        path.Add(eye + new Vector3(-.01f, -.01f, -.03f));
        path.Add(new Vector3(-.62f, .90f, -.13f));
        path.Add(new Vector3(-.56f, .825f, -.13f));
        path.Add(new Vector3(-.495f, .80f, -.13f));
        var nLocked = ChainAlong(L, "chain", path);
        Padlock(L, new Vector3(-.48f, .765f, -.135f), false);
        // UNLOCKED: the chain on the floor, in a loose coil against the wall on the latch side, the padlock open
        var heap = new List<Vector3>();
        var hc = new Vector3(-.78f, 0f, -.24f);
        for (var k = 0; k <= 40; k++)
        {
            var t = k / 40f; var ang = t * Mathf.PI * 4.4f;
            var rad = .14f - .05f * t;
            heap.Add(hc + new Vector3(Mathf.Cos(ang) * rad, .008f + .016f * t, Mathf.Sin(ang) * rad * .8f));
        }
        heap.Add(hc + new Vector3(.18f, .008f, .06f));
        var nHeap = ChainAlong(U, "chain (dropped)", heap);
        Padlock(U, hc + new Vector3(.24f, .012f, .08f), true);
        foreach (var r in lockedSet.GetComponentsInChildren<Renderer>(true)) { lockRs.Add(r); ids[r] = Id.Door; }
        foreach (var r in unlockedSet.GetComponentsInChildren<Renderer>(true)) { lockRs.Add(r); ids[r] = Id.Door; }

    /// <summary>Links of a 1/2 in chain along a polyline (alternating planes), each link four thin boxes.</summary>
    static int ChainAlong(Transform p, string name, List<Vector3> pts)
    {
        var cum = new List<float> { 0f };
        for (var i = 1; i < pts.Count; i++) cum.Add(cum[i - 1] + (pts[i] - pts[i - 1]).magnitude);
        var total = cum[cum.Count - 1];
        var n = Mathf.Max(1, Mathf.FloorToInt(total / ChainPitch));
        var seg = 0;
        for (var k = 0; k < n; k++)
        {
            var d = (k + .5f) * total / n;
            while (seg < pts.Count - 2 && cum[seg + 1] < d) seg++;
            var tt = Mathf.InverseLerp(cum[seg], cum[seg + 1], d);
            var c = Vector3.Lerp(pts[seg], pts[seg + 1], tt);
            var t = (pts[seg + 1] - pts[seg]).normalized;
            var nref = Mathf.Abs(Vector3.Dot(t, Vector3.up)) > .9f ? Vector3.forward : Vector3.up;
            var u0 = Vector3.Cross(t, nref).normalized;
            var u = (k % 2 == 0) ? u0 : Vector3.Cross(t, u0).normalized;
            var hl = ChainLen * .5f - ChainWire * .5f; var hw = ChainWid * .5f - ChainWire * .5f;
            Seg(p, name + " link " + k + " a", c - t * hl + u * hw, c + t * hl + u * hw, ChainWire, ChainWire, chainM, Id.Door);
            Seg(p, name + " link " + k + " b", c - t * hl - u * hw, c + t * hl - u * hw, ChainWire, ChainWire, chainM, Id.Door);
            Seg(p, name + " link " + k + " c", c - t * hl - u * hw, c - t * hl + u * hw, ChainWire, ChainWire, chainM, Id.Door);
            Seg(p, name + " link " + k + " d", c + t * hl - u * hw, c + t * hl + u * hw, ChainWire, ChainWire, chainM, Id.Door);
        }
        return n;
    }

    /// <summary>Seven-segment placeholder digit on the door wall, read from the stair side (+z): its right is -x.</summary>
    static void Digit(Transform p, int d, Vector3 bl, float h, float s, Material m, bool stencil, string tag)
    {
        var w = h * .58f;
        // segments a b c d e f g
        var on = new[] {
            new[] { true, true, true, true, true, true, false }, new[] { false, true, true, false, false, false, false },
            new[] { true, true, false, true, true, false, true }, new[] { true, true, true, true, false, false, true },
            new[] { false, true, true, false, false, true, true }, new[] { true, false, true, true, false, true, true },
            new[] { true, false, true, true, true, true, true }, new[] { true, true, true, false, false, false, false },
            new[] { true, true, true, true, true, true, true }, new[] { true, true, true, true, false, true, true } }[Mathf.Clamp(d, 0, 9)];
        var gap = stencil ? s * .55f : 0f;
        var hh = h * .5f;
        // centre, size (x, y) of each segment in the digit's plane (x right, y up from bottom-left)
        var cs = new[] {
            new Vector4(w * .5f, h - s * .5f, w - 2f * gap, s), new Vector4(w - s * .5f, hh + hh * .5f, s, hh - gap * 2f),
            new Vector4(w - s * .5f, hh * .5f, s, hh - gap * 2f), new Vector4(w * .5f, s * .5f, w - 2f * gap, s),
            new Vector4(s * .5f, hh * .5f, s, hh - gap * 2f), new Vector4(s * .5f, hh + hh * .5f, s, hh - gap * 2f),
            new Vector4(w * .5f, hh, w - 2f * gap - (stencil ? 0f : 0f), s) };
        for (var i = 0; i < 7; i++)
        {
            if (!on[i]) continue;
            Box(p, "numeral seg " + tag + "." + i, bl + new Vector3(-cs[i].x, cs[i].y, .001f), new Vector3(cs[i].z, cs[i].w, .003f), m, Id.Sign);
        }
    }

    /// <summary>The steady-fixture gain per direction (critic gameplay 10: walk-path post Y median about 0.05-0.065,
    /// about 1 stop under the maze's 0.13), unless -stairFixtureGain overrides it.</summary>
    static float FixtureGain()
    {
        if (fixtureGain >= 0f) return fixtureGain;
        return dir == "A" ? 2.4f : dir == "B" ? 2.9f : 1.5f;
    }

    /// <summary>One normal fixture per landing (D and T). A: 4 ft 2-lamp wraparound on the soffit (on the far wall
    /// at the top turn landing, which has no landing above it); B: caged vapour-tight incandescent globe on the wall;
    /// C: the maze's own troffer in a 2.9 m drop ceiling. FIX (critic era 5): A's and C's linear sources are two
    /// spots 0.6 m apart along the lamp (a long soft wash, not a round hot pool); B's cage and box cast no shadow
    /// and the globe's shadow bias is raised (the dashed arcs were shadow acne).</summary>
    static void LandingLight(int n, bool dLanding)
    {
        var p = Kit("Light", n);
        var y = dLanding ? YD(n) : YT(n);
        var shadows = n >= -1 && n <= 1;
        var zc = dLanding ? (ZNear + ZD) * .5f : (ZT + ZFarFace) * .5f;
        var edge = lampMode == "edge";
        // FIX (fix stage): the door-landing fixture moves 0.80 m off the door wall so it lights the door face, not
        // only the wall above it (at 0.30 m the far door measured darker than 3/4 of the frame); the turn-landing
        // fixture stays 0.30 m off the far wall and washes the wall you walk toward
        var zEdge = dLanding ? ZNear + .80f : ZFarFace - .30f;
        var name = (dLanding ? "D" : "T") + n;
        var gain = lampGain * FixtureGain();
        var lights = new List<Light>();
        if (dir == "A")
        {
            var top = n == FloorMin && dLanding;                 // under the roof slab
            var wallMount = n == FloorMin && !dLanding;          // the top turn landing: nothing above it
            var soffit = top ? YD(FloorMin) + 3.3f : y + FF - .15f;
            var zl = edge ? zEdge : zc;
            Vector3 pos; Quaternion rot; Vector3 aim; Vector3 lightAt;
            if (wallMount)
            {
                pos = new Vector3(0f, y + 2.55f, ZFarFace); rot = Quaternion.Euler(90f, 0f, 0f);
                aim = new Vector3(0f, -.55f, -1f); lightAt = new Vector3(0f, y + 2.55f, ZFarFace - .10f);
            }
            else
            {
                pos = new Vector3(0f, soffit, zl); rot = Quaternion.identity;
                aim = Vector3.down; lightAt = new Vector3(0f, soffit - .09f, zl);
            }
            var go = KitSpawn("Kit_WrapFixture_4ft", p, pos, rot, Id.Emitter);
            if (go != null)
            {
                var r = go.GetComponentInChildren<Renderer>();
                r.shadowCastingMode = ShadowCastingMode.Off;
                var li = SlotIndex(r, "Run_WrapLens");
                if (li >= 0)
                {
                    if (wrapLensLit == null)
                    {
                        wrapLensLit = new Material(r.sharedMaterials[li]) { name = "A wrap lens lit" };
                        wrapLensLit.SetColor("_EmissionColor", new Color(1f, 1f, 1f) * 1.6f * Mathf.Sqrt(FixtureGain()));
                        wrapLensLit.SetFloat("_UseEmission", 1f); wrapLensLit.EnableKeyword("_EMISSION");
                    }
                    var arr = r.sharedMaterials; arr[li] = wrapLensLit; r.sharedMaterials = arr;
                }
            }
            foreach (var sx in new[] { -.30f, .30f })
                lights.Add(Spot(p, "wrap light " + name + (sx < 0f ? " W" : " E"), lightAt + new Vector3(sx, 0f, 0f), aim, 170f, 120f, new Color(.95f, .98f, 1f), 1.6f, 6.5f, shadows));
        }
        else if (dir == "B")
        {
            // wall globe: D on the door wall beside the door, T on the far wall
            Vector3 mount, outN;
            if (dLanding) { mount = edge ? new Vector3(.95f, y + 2.30f, ZNear) : new Vector3(XFace, y + 2.35f, 1.05f); outN = edge ? Vector3.forward : Vector3.left; }
            else { mount = new Vector3(0f, y + 2.35f, ZFarFace); outN = Vector3.back; }
            var box = mount + outN * .03f;
            var bx = Cyl(p, "globe box " + name, mount, mount + outN * .05f, .055f, steelM, Id.Background);
            bx.shadowCastingMode = ShadowCastingMode.Off;
            if (globeLit == null) globeLit = Glow("B globe lit (100 W A19 behind glass)", new Color(1f, .93f, .8f), new Color(1f, .82f, .6f) * 4.0f * Mathf.Sqrt(FixtureGain()));
            var gc = box + outN * .15f;
            var gr = Ball(p, "globe glass " + name, gc, .13f, globeLit, Id.Emitter);
            gr.shadowCastingMode = ShadowCastingMode.Off;
            for (var i = 0; i < 4; i++)
            {
                var ang = i * Mathf.PI * .5f;
                var side = Vector3.Cross(outN, Vector3.up).normalized;
                var off = (side * Mathf.Cos(ang) + Vector3.up * Mathf.Sin(ang)) * .075f;

            if (edge)
            {

                pgo.transform.SetParent(p, false);
                pgo.transform.position = root.TransformPoint(gc);
                l = pgo.AddComponent<Light>();
                l.type = LightType.Point; l.color = new Color(1f, .84f, .66f); l.intensity = 2.6f; l.range = 6f;
                l.shadows = shadows ? LightShadows.Soft : LightShadows.None; l.shadowStrength = .92f; l.shadowNearPlane = .1f;
                l.shadowBias = .2f; l.shadowNormalBias = 1.2f;
            }
            else l = Spot(p, "globe light " + name, gc, outN + Vector3.down * .45f, 168f, 110f, new Color(1f, .84f, .66f), 3.6f, 6f, shadows);
            lights.Add(l);
        }
        else
        {

            var cc = new Vector3(0f, y + MazeH, dLanding ? ZNear + .80f : ZFarFace - .45f);

                lights.Add(l);
            }
        }
        foreach (var l in lights) { l.intensity *= gain; stairLights.Add(l); }
        // twin-head battery unit over each door (A, B): normal power, so only the red AC pilot glows

    }

    static Transform MakeRelay(string name, Vector3 pos, float yaw)
    {
        var t = FrontRooms3DGame.CreateHunter(root);
        t.name = name;
        var rig = t.GetComponent<FrontRoomsRelayRig>();
        if (rig != null) rig.TickAnimation(.35f, FrontRoomsRelayRig.MotionState.Walk, true, 1f);
        t.position = root.TransformPoint(pos);
        t.rotation = root.rotation * Quaternion.Euler(0f, yaw, 0f);
        t.localScale = Vector3.one;
        var b = Bounds(t);
        var nativeTop = b.max.y - t.position.y;
        relayScale = nativeTop > .1f ? RelayHeight / nativeTop : 1f;
        t.localScale = Vector3.one * relayScale;

    }

    static void Pose(Camera cam, Shot s)
    {
        var eye = root.TransformPoint(s.eye);
        cam.transform.SetPositionAndRotation(eye, Quaternion.LookRotation(root.TransformPoint(s.look) - eye, Vector3.up));
        cam.fieldOfView = s.fov;
    }

    /// <summary>Sets the scene state a shot asks for: A13 lock state, far-door hold, forced LOD0, end sign, sign gain.</summary>
    static void ApplyState(Shot s)
    {
        var unlocked = (s.flags & FUnlocked) != 0;
        if (lockedSet != null) lockedSet.SetActive(!unlocked);
        if (unlockedSet != null) unlockedSet.SetActive(unlocked);
        var held = (s.flags & FHeld) != 0;
        if (farLeaf != null)
        {
            // the hold: the lever pressed down 40 degrees, the leaf pulled 2 mm at the latch edge (stack pressure)
            farLeaf.rotation = held ? Quaternion.AngleAxis(Mathf.Asin(.002f / .98f) * Mathf.Rad2Deg, root.up) * farLeafRest : farLeafRest;
            if (farLever != null) farLever.localRotation = held ? farLeverRest * Quaternion.Euler(0f, 0f, -40f) : farLeverRest;
        }
        var end = (s.flags & FEndSign) != 0;
        if (flatSign != null) flatSign.SetActive(!end);
        if (endSign != null) endSign.SetActive(end);
        SetSignGain(s.signGain > 0f ? s.signGain : (signGain > 0f ? signGain : SignGainDefault));
        // FIX (critic era 9): the strip at the exit-door head was the frame head's and the sign's shadow on the leaf
        // recessed 58 mm behind them (map-lamp default bias: a lit sliver, then the displaced shadow), at LOD0 and
        // LOD1 alike. C23 spec: the exit frame and the EXIT sign cast no shadow. FAux restores the old state.
        var aux = (s.flags & FAux) != 0;
        foreach (var r in signRs) if (r != null) r.shadowCastingMode = aux ? ShadowCastingMode.On : ShadowCastingMode.Off;
        if (exitLeaf != null && exitLeaf.parent != null)
            foreach (var r in exitLeaf.parent.GetComponentsInChildren<Renderer>(true))
                if (r.name.StartsWith("Kit_DoorFrame")) r.shadowCastingMode = aux ? ShadowCastingMode.On : ShadowCastingMode.Off;
        var lod0 = (s.flags & FLod0) != 0;
        foreach (var lg in root.GetComponentsInChildren<LODGroup>(true)) lg.ForceLOD(lod0 ? 0 : -1);
    }

    static void RenderAll(List<Shot> shots, string onlyShots)
    {
        var filter = string.IsNullOrEmpty(onlyShots) ? null : new HashSet<string>(onlyShots.Split(','));
        var cam = MakeCamera();
        var rt = new RenderTexture(1920, 1080, 24, RenderTextureFormat.ARGB32) { antiAliasing = 4 };
        cam.targetTexture = rt;
        var poses = new StringBuilder("shot\teye\tlook\tfov\trelay\tflags\tsignGain\n");
        foreach (var s in shots)
        {
            if (filter != null && !filter.Contains(s.name)) continue;
            ShowRelay(relayStair, s.relay == 1);
            ShowRelay(relayMaze, s.relay == 2);
            ApplyState(s);
            Physics.SyncTransforms();
            Pose(cam, s);
            cam.targetTexture = rt;
            cam.Render(); cam.Render();
            Save(rt, Path.Combine(outDir, s.name + ".png"));
            RenderId(cam, s);
            RenderDoorParts(cam, s);
            if (s.relay > 0) RenderRelayParts(cam, s, s.relay == 1 ? relayStair : relayMaze);
            if (ePass && (s.flags & FNoE) == 0) RenderE(cam, s);
            if ((s.flags & FRaw) != 0) RenderRaw(cam, s);
            poses.Append(s.name + "\t" + V(s.eye) + "\t" + V(s.look) + "\t" + s.fov.ToString("0", CultureInfo.InvariantCulture) + "\t" + s.relay + "\t" + s.flags + "\t" + s.signGain.ToString("0.0", CultureInfo.InvariantCulture) + "\n");
            Log("shot " + s.name + " rendered");
        }
        File.WriteAllText(Path.Combine(outDir, "poses.tsv"), poses.ToString());
        cam.targetTexture = null;
        rt.Release();
        UnityEngine.Object.DestroyImmediate(cam.gameObject);
        ShowRelay(relayStair, false); ShowRelay(relayMaze, false);
        ApplyState(new Shot("rest", Vector3.zero, Vector3.forward, 76f, 0));
    }

    /// <summary>Door parts mask (unlit, no post, no MSAA, no fog), per material slot: red = the A13 chain and padlock,
    /// green = the EXIT letters (diffuser), blue = the sign face, yellow = sign housing and chrome, magenta = the exit
    /// leaf's enamel, cyan = the crossbar, white = the far door's lever and floor-access plate, black = the rest.</summary>
    static void RenderDoorParts(Camera cam, Shot s)
    {
        var unlit = Shader.Find("Universal Render Pipeline/Unlit");
        Func<Color, Material> mk = c => { var m = new Material(unlit); m.SetColor("_BaseColor", c); return m; };
        var black = mk(Color.black); var red = mk(Color.red); var green = mk(Color.green); var blue = mk(Color.blue);
        var yellow = mk(new Color(1f, 1f, 0f)); var magenta = mk(new Color(1f, 0f, 1f)); var cyan = mk(new Color(0f, 1f, 1f)); var white = mk(Color.white);
        var saved = new Dictionary<Renderer, Material[]>();
        foreach (var r in root.GetComponentsInChildren<Renderer>(false))
        {
            if (!r.enabled) continue;
            saved[r] = r.sharedMaterials;
            var src = r.sharedMaterials;
            var arr = new Material[src.Length];
            for (var i = 0; i < arr.Length; i++)
            {
                var nm = src[i] != null ? src[i].name.Replace(" (Instance)", "") : "";
                Material m = black;
                if (lockRs.Contains(r)) m = red;
                else if (signRs.Contains(r)) m = nm.StartsWith("ExitSign_Diffuser") ? green : nm.StartsWith("ExitSign_Face") ? blue : yellow;
                else if (leafRs.Contains(r)) m = (i == 0) ? magenta : black;
                else if (barRs.Contains(r)) m = cyan;
                else if (trimRs.Contains(r)) m = white;
                arr[i] = m;
            }
            r.sharedMaterials = arr;
        }
        var fog = RenderSettings.fog; RenderSettings.fog = false;
        var data = cam.GetUniversalAdditionalCameraData();
        data.renderPostProcessing = false;
        var prev = cam.targetTexture;
        var rt = new RenderTexture(1920, 1080, 24, RenderTextureFormat.ARGB32) { antiAliasing = 1 };
        cam.targetTexture = rt; cam.allowMSAA = false;
        cam.Render();
        Save(rt, Path.Combine(outDir, s.name + "_parts.png"));
        cam.targetTexture = prev; cam.allowMSAA = true; rt.Release();
        data.renderPostProcessing = true;
        RenderSettings.fog = fog;
        foreach (var kv in saved) if (kv.Key != null) kv.Key.sharedMaterials = kv.Value;
    }

    static string V(Vector3 v) { return v.x.ToString("0.###", CultureInfo.InvariantCulture) + "," + v.y.ToString("0.###", CultureInfo.InvariantCulture) + "," + v.z.ToString("0.###", CultureInfo.InvariantCulture); }

    static void SwapMaterials(Func<Renderer, Material> pick, Dictionary<Renderer, Material[]> saved, List<Renderer> hidden)
    {
        foreach (var r in root.GetComponentsInChildren<Renderer>(false))
        {
            if (!r.enabled) continue;
            saved[r] = r.sharedMaterials;
            var m = pick(r);
            if (m == null) { hidden.Add(r); r.enabled = false; continue; }
            var arr = new Material[r.sharedMaterials.Length];
            for (var i = 0; i < arr.Length; i++) arr[i] = m;
            r.sharedMaterials = arr;
        }
    }

    static void Restore(Dictionary<Renderer, Material[]> saved, List<Renderer> hidden)
    {
        foreach (var kv in saved) if (kv.Key != null) kv.Key.sharedMaterials = kv.Value;
        foreach (var r in hidden) if (r != null) r.enabled = true;
    }

    /// <summary>Illuminance pass: every surface white Lambert (albedo 1, no specular, no reflections), real
    /// lights + the game's ambient, no post, no fog; linear float RGBA, 960 x 540, raw little-endian floats.</summary>
    static void RenderE(Camera cam, Shot s)
    {
        var white = new Material(Shader.Find("Universal Render Pipeline/Lit")) { name = "E pass white" };
        white.SetColor("_BaseColor", Color.white);
        white.SetFloat("_Smoothness", 0f); white.SetFloat("_Metallic", 0f);
        white.SetFloat("_SpecularHighlights", 0f); white.EnableKeyword("_SPECULARHIGHLIGHTS_OFF");
        white.SetFloat("_EnvironmentReflections", 0f); white.EnableKeyword("_ENVIRONMENTREFLECTIONS_OFF");
        var saved = new Dictionary<Renderer, Material[]>(); var hidden = new List<Renderer>();
        SwapMaterials(r => { Id v; var id = ids.TryGetValue(r, out v) ? v : Id.Background; return id == Id.Emitter ? null : white; }, saved, hidden);
        var fog = RenderSettings.fog; RenderSettings.fog = false;
        var data = cam.GetUniversalAdditionalCameraData();
        data.renderPostProcessing = false;
        var prev = cam.targetTexture;
        var rt = new RenderTexture(960, 540, 24, RenderTextureFormat.ARGBFloat) { antiAliasing = 1 };
        cam.targetTexture = rt; cam.allowMSAA = false;
        cam.Render();
        var prevActive = RenderTexture.active;
        RenderTexture.active = rt;
        var tex = new Texture2D(960, 540, TextureFormat.RGBAFloat, false, true);
        tex.ReadPixels(new Rect(0, 0, 960, 540), 0, 0);
        tex.Apply();
        RenderTexture.active = prevActive;
        var raw = tex.GetRawTextureData();
        File.WriteAllBytes(Path.Combine(outDir, s.name + "_E960x540.f32"), raw);
        UnityEngine.Object.DestroyImmediate(tex);
        cam.targetTexture = prev; cam.allowMSAA = true; rt.Release();
        data.renderPostProcessing = true;
        RenderSettings.fog = fog;
        Restore(saved, hidden);
    }

    // ---------------------------------------------------------------- stats and kits
    static int Tris(Transform t)
    {
        var sum = 0;
        foreach (var mf in t.GetComponentsInChildren<MeshFilter>(true))
        {
            if (mf.sharedMesh == null) continue;
            for (var i = 0; i < mf.sharedMesh.subMeshCount; i++) sum += (int)(mf.sharedMesh.GetIndexCount(i) / 3);
        }
        foreach (var smr in t.GetComponentsInChildren<SkinnedMeshRenderer>(true)) if (smr.sharedMesh != null) sum += smr.sharedMesh.triangles.Length / 3;
        return sum;
    }

    static void SaveKits()
    {
        var baseDir = "Assets/StairBlockout/" + dir;
        if (!AssetDatabase.IsValidFolder("Assets/StairBlockout")) AssetDatabase.CreateFolder("Assets", "StairBlockout");
        if (!AssetDatabase.IsValidFolder(baseDir)) AssetDatabase.CreateFolder("Assets/StairBlockout", dir);
        if (!AssetDatabase.IsValidFolder(baseDir + "/Meshes")) AssetDatabase.CreateFolder(baseDir, "Meshes");
        if (!AssetDatabase.IsValidFolder(baseDir + "/Materials")) AssetDatabase.CreateFolder(baseDir, "Materials");
        var mi = 0; var me = 0;
        foreach (var kv in kitGroups)
        {
            foreach (var r in kv.Value.GetComponentsInChildren<Renderer>(true))
            {
                var arr = r.sharedMaterials;
                for (var i = 0; i < arr.Length; i++)
                {
                    if (arr[i] == null || EditorUtility.IsPersistent(arr[i])) continue;
                    var safe = new string(arr[i].name.Select(ch => char.IsLetterOrDigit(ch) ? ch : '_').ToArray());
                    try { AssetDatabase.CreateAsset(arr[i], baseDir + "/Materials/" + safe + "_" + (mi++) + ".mat"); }
                    catch (Exception e) { Log("material not saved: " + arr[i].name + " " + e.Message); }
                }
                var mf = r.GetComponent<MeshFilter>();
                if (mf != null && mf.sharedMesh != null && !EditorUtility.IsPersistent(mf.sharedMesh))
                {
                    try { AssetDatabase.CreateAsset(mf.sharedMesh, baseDir + "/Meshes/mesh_" + (me++) + ".asset"); }
                    catch (Exception e) { Log("mesh not saved: " + mf.sharedMesh.name + " " + e.Message); }
                }
            }
            PrefabUtility.SaveAsPrefabAsset(kv.Value.gameObject, baseDir + "/BK_" + dir + "_" + kv.Key + ".prefab");
        }
        AssetDatabase.SaveAssets();
        Log("lamp mode " + lampMode + "; saved " + kitGroups.Count + " blockout kit prefabs to " + baseDir + " (" + mi + " materials, " + me + " meshes)");
    }
