def P(run, d, shot, suffix='.png'):
    return os.path.join(run, d, shot + suffix)


def img(run, d, shot):
    return Image.open(P(run, d, shot)).convert('RGB')


def m(d, shot, src=None):
    return (src or met)['dirs'].get(d, {}).get(shot, {})


def parts(run, d, shot):
    q = np.asarray(Image.open(P(run, d, shot, '_parts.png')).convert('RGB')) > 127
    return q


def bbox_of(mask):
    ys, xs = np.nonzero(mask)
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def door_crop(run, d, shot, w=640, h=360, pad=1.6):
    """crop round the exit door + sign (leaf, sign, lock from the parts mask), 16:9, at least w x h px"""
    q = parts(run, d, shot)
    mk = (q[..., 0] & ~q[..., 1] & q[..., 2]) | (q[..., 1] & ~q[..., 0]) | (q[..., 0] & ~q[..., 1] & ~q[..., 2]) | (~q[..., 0] & ~q[..., 1] & q[..., 2])
    im = img(run, d, shot)
    if mk.sum() < 5:
        return im
    x0, y0, x1, y1 = bbox_of(mk)
    cx, cy = (x0 + x1) / 2, (y0 + y1) / 2
    hh = max((y1 - y0) * pad / 2, h / 2 / 4)
    hw = hh * 16 / 9
    box = [int(cx - hw), int(cy - hh), int(cx + hw), int(cy + hh)]
    dx = -min(0, box[0]) - max(0, box[2] - 1920); dy = -min(0, box[1]) - max(0, box[3] - 1080)
    box = [box[0] + dx, box[1] + dy, box[2] + dx, box[3] + dy]
    return im.crop(box)


def f2(x):
    return '-' if x is None else f'{x:.2f}'

def lum(o):
    s = f"med Y {o.get('medianY', 0):.3f}"
    if 'walkY_median' in o:
        s += f" · walk {o['walkY_median']:.3f}"
    if 'E_Walk' in o:
        s += f" · {o['E_Walk']['fc_eq_p2']}/{o['E_Walk']['fc_eq_median']} fc-eq"
    return s

def ramp(t):
    t = np.clip(t, 0, 1)
    stops = np.array([[0.20, 0.0, 0.35], [0.0, 0.35, 0.85], [0.0, 0.75, 0.55], [0.95, 0.85, 0.1], [1.0, 0.3, 0.1]])
    x = t * (len(stops) - 1)
    i = np.minimum(np.floor(x).astype(int), len(stops) - 2)
    f = (x - i)[..., None]
    return stops[i] * (1 - f) + stops[i + 1] * f

def relay_wide(run, d, s, metsrc):
    q = np.asarray(Image.open(P(run, d, s, '_id.png')).convert('RGB')) > 127
    rel = q[..., 0] & q[..., 1] & q[..., 2]
    door = q[..., 0] & ~q[..., 1]
    x0, y0, x1, y1 = bbox_of(rel | door)
    cx, cy = (x0 + x1) / 2, (y0 + y1) / 2
    hh = (y1 - y0) * .7; hw = max(hh * 16 / 9, (x1 - x0) * .7); hh = hw * 9 / 16
    box = [int(cx - hw), int(cy - hh), int(cx + hw), int(cy + hh)]
    dx = -min(0, box[0]) - max(0, box[2] - 1920); dy = -min(0, box[1]) - max(0, box[3] - 1080)
    return img(run, d, s).crop([box[0] + dx, box[1] + dy, box[2] + dx, box[3] + dy])
