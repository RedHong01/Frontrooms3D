  a window flashes when >= 25 % of its cells make the same transition within 4 frames; a flash = a pair of opposing

def load(p):
    return np.asarray(Image.open(p).convert("RGB")).astype(np.float32) / 255.0


def lin(c):
    return np.where(c <= 0.04045, c / 12.92, ((c + 0.055) / 1.055) ** 2.4)


def enc(c):
    c = np.clip(c, 0, 1)
    return np.where(c <= 0.0031308, c * 12.92, 1.055 * np.power(c, 1 / 2.4) - 0.055)


def Y(img):
    return lin(img) @ np.array([.2126, .7152, .0722], np.float32)


PROT = np.array([[0.152286, 1.052583, -0.204868], [0.114503, 0.786281, 0.099216], [-0.003882, -0.048116, 1.051998]], np.float32)


def protan(img):
    return enc(np.clip(lin(img) @ PROT.T, 0, 1))


def stats(img):
    y = Y(img)
    p = np.percentile(y, [5, 50, 95, 99.5])
    im = img * 255
    s = im.sum(-1) + 1e-6
    rf = im[..., 0] / s
    return dict(Y_p5=round(float(p[0]), 4), Y_p50=round(float(p[1]), 4), Y_p95=round(float(p[2]), 4), Y_p995=round(float(p[3]), 4),
                contrast_p95_p5=round(float((p[2] + .05) / (p[0] + .05)), 2),
                sat_red_pct=round(float(((rf >= 0.8) & (s > 30)).mean() * 100), 2),
                red_dominant_pct=round(float(((rf > 0.6) & (s > 30)).mean() * 100), 2))


PAL = {"Background": (0, 0, 0), "Relay": (1, 1, 1), "Goal": (1, 0, 0), "Dolly": (0, 1, 0), "Cart": (0, 0, 1),
       "Truck": (1, 1, 0), "Units": (1, 0, 1), "Main": (0, 1, 1), "Emitter": (.5, .5, .5), "Door": (1, .5, 0), "Sign": (.5, 0, 1)}


def idmap(p):
    im = load(p)
    keys = list(PAL.keys())
    cols = np.array([enc(np.array(PAL[k], np.float32)) for k in keys], np.float32)
    idx = ((im[:, :, None, :] - cols[None, None, :, :]) ** 2).sum(-1).argmin(-1)
    return {k: idx == i for i, k in enumerate(keys)}


def ring(mask, px=12):
    m = Image.fromarray((mask * 255).astype(np.uint8))
    dil = np.asarray(m.filter(ImageFilter.MaxFilter(2 * px + 1))) > 127
    return dil & ~mask


def ratio(a, b):
    hi, lo = max(a, b), min(a, b)
    return round(hi / max(lo, 1e-6), 2)


def within(a, b, tol):
    if b == 0:
        return abs(a) < 1e-3
    return abs(a - b) / abs(b) <= tol


def relay_read(img, ids, name):
    rel = ids["Relay"]
    if rel.sum() < 20:
        return dict(px=int(rel.sum()), note="relay not visible")
    y, yp = Y(img), Y(protan(img))
    rg = ring(rel)
    ys, xs = np.nonzero(rel)
    cy = (ys.min() + ys.max()) / 2
    rows = np.arange(img.shape[0])[:, None]
    up, lo = rg & (rows < cy), rg & (rows >= cy)
    a, b = float(y[rel].mean()), float(y[rg].mean())
    return dict(px=int(rel.sum()), bbox_h=int(ys.max() - ys.min() + 1), bbox=[int(xs.min()), int(ys.min()), int(xs.max() - xs.min() + 1), int(ys.max() - ys.min() + 1)],
                relayY=round(a, 4), ringY=round(b, 4), ratio=ratio(a, b), ratio_protan=ratio(float(yp[rel].mean()), float(yp[rg].mean())),
                ratio_upper=ratio(a, float(y[up].mean())) if up.sum() else None, ratio_lower=ratio(a, float(y[lo].mean())) if lo.sum() else None,
                glare_pct=round(float((y > .5).mean() * 100), 3), outline=outline(img, ids, "Relay"), PASS=bool(ratio(a, b) >= 4 and ratio(float(yp[rel].mean()), float(yp[rg].mean())) >= 4))


def obstacle(img, ids, k):
    m = ids[k]
    if m.sum() < 20:
        return "not visible"
    y = Y(img)
    r = ring(m) & ~(ids["Dolly"] | ids["Cart"] | ids["Truck"] | ids["Relay"])
    a, b = float(y[m].mean()), float(y[r].mean())
    return dict(px=int(m.sum()), meanY=round(a, 4), ringY=round(b, 4), ratio=ratio(a, b), PASS=bool(ratio(a, b) >= 2))


def boxsum(a, r=3):
    c = np.pad(a, ((r + 1, r), (r + 1, r))).cumsum(0).cumsum(1)
    return c[2 * r + 1:, 2 * r + 1:] - c[:-2 * r - 1, 2 * r + 1:] - c[2 * r + 1:, :-2 * r - 1] + c[:-2 * r - 1, :-2 * r - 1]


def outline(img, ids, k):
    y = Y(img)
    m = ids[k]
    obst = ids["Dolly"] | ids["Cart"] | ids["Truck"] | ids["Relay"]
    bg = ~obst
    mi, bi = m.astype(np.float64), bg.astype(np.float64)
    sin, nin = boxsum(y * mi), boxsum(mi)
    sout, nout = boxsum(y * bi), boxsum(bi)
    edge = m & (nout > 3)
    if edge.sum() < 10:
        return None
    a = sin[edge] / np.maximum(nin[edge], 1)
    b = sout[edge] / np.maximum(nout[edge], 1)
    r = np.maximum(a, b) / np.maximum(np.minimum(a, b), 1e-5)
    return dict(edge_px=int(edge.sum()), median=round(float(np.median(r)), 2), share_ge2=round(float((r >= 2).mean()), 3))

def flashes(grid_y, grid_r, fps=60):
    """grid_y, grid_r: (frames, GH, GW). Returns (general flashes in worst 1 s, red flashes in worst 1 s, detail)."""
    F, GH, GW = grid_y.shape

    def cell_transitions(series_fn):
        # per cell: list of (frame, direction)
        out = np.zeros((F, GH, GW), np.int8)
        for j in range(GH):
            for i in range(GW):
                ev = series_fn(j, i)
                for f, d in ev:
                    out[f, j, i] = d
        return out

    def y_events2(j, i):
        # zig-zag on the cell's linear Y: a transition = >= 0.1 from the last extreme, the darker side < 0.8
        s = grid_y[:, j, i]
        ev = []
        hi = lo = float(s[0])
        dirn = 0
        for f in range(1, F):
            v = float(s[f])
            if dirn == 0:
                hi, lo = max(hi, v), min(lo, v)
                if v - lo >= .1 and lo < .8:
                    ev.append((f, 1)); dirn = 1; hi = v
                elif hi - v >= .1 and v < .8:
                    ev.append((f, -1)); dirn = -1; lo = v
            elif dirn == 1:
                if v > hi:
                    hi = v
                elif hi - v >= .1 and v < .8:
                    ev.append((f, -1)); dirn = -1; lo = v
            else:
                if v < lo:
                    lo = v
                elif v - lo >= .1 and lo < .8:
                    ev.append((f, 1)); dirn = 1; hi = v
        return ev

    def r_events(j, i):
        r = grid_r[:, j, i] >= .5
        s = grid_y[:, j, i]
        ev = []
        for f in range(1, F):
            if r[f] != r[f - 1] and abs(s[f] - s[max(0, f - 4)]) >= .1:
                ev.append((f, 1 if r[f] else -1))
        return ev

    res_ = {}
    for kind, fn in (("general", y_events2), ("red", r_events)):
        tr = cell_transitions(fn)
        # window events: >= 25 % of the window's cells make the same transition within 4 frames
        wh, ww = GH // 3, GW // 3
        sy, sx = max(1, GH // 12), max(1, GW // 12)
        worst, worst_at, nwin_ev = 0, None, 0
        for y0 in range(0, GH - wh + 1, sy):
            for x0 in range(0, GW - ww + 1, sx):
                sub = tr[:, y0:y0 + wh, x0:x0 + ww]
                n = wh * ww
                ev = []
                last = {1: -99, -1: -99}
                up = (sub == 1).reshape(F, -1).sum(1)
                dn = (sub == -1).reshape(F, -1).sum(1)
                cu = np.convolve(up, np.ones(4, int), "full")[:F]
                cd = np.convolve(dn, np.ones(4, int), "full")[:F]
                for f in range(F):
                    for d, c in ((1, cu[f]), (-1, cd[f])):
                        if c >= .25 * n and f - last[d] > 4:
                            ev.append((f, d)); last[d] = f
                ev.sort()
                # collapse consecutive same-direction events
                seq = []
                for f, d in ev:
                    if seq and seq[-1][1] == d:
                        continue
                    seq.append((f, d))
                nwin_ev += len(seq)
                # flashes = pairs of opposing transitions; most in any 1 s span
                for a in range(len(seq)):
                    cnt = 0
                    for b in range(a, len(seq)):
                        if seq[b][0] - seq[a][0] < fps:
                            cnt += 1
                    fl = cnt // 2
                    if fl > worst:
                        worst, worst_at = fl, (int(seq[a][0]), int(y0), int(x0))
        res_[kind] = dict(worst_flashes_per_s=int(worst), at_frame_row_col=worst_at, window_transitions=int(nwin_ev))
    return res_

        ms = list(means)
        mx = max(ms)
        trans, start, dirn = [], 0, 0
        for i in range(1, len(ms)):
            dd = ms[i] - ms[i - 1]
            s = 0 if abs(dd) < 1e-6 else (1 if dd > 0 else -1)
            if s != 0 and s != dirn:
                if dirn != 0 and abs(ms[i - 1] - ms[start]) > .1 * mx:
                    trans.append([start, i - 1, round(ms[start], 4), round(ms[i - 1], 4)])
                start, dirn = i - 1, s
        if dirn != 0 and abs(ms[-1] - ms[start]) > .1 * mx:
            trans.append([start, len(ms) - 1, round(ms[start], 4), round(ms[-1], 4)])
        # black gap (Reduce Flashing rule): frames whose mean is below 2 % of the pre-event mean
        pre = float(np.mean(means[:20])) if Fr > 20 else float(means[0])
        black = int((means < .02 * max(pre, 1e-6)).sum())
        mono = None
        csv = [l.split(",") for l in open(os.path.join(d, "frames.csv")).read().strip().splitlines()[1:]]
        tl[name] = dict(frames=Fr, meta={k: v for k, v in meta.items() if k not in ("layout",)}, frame_mean_min=round(float(means.min()), 5), frame_mean_max=round(float(mx), 5),
                        frame_mean_transitions_gt10pct=trans, black_frames_lt2pct_of_pre=black,
                        general=fl["general"], red=fl["red"], max_red_sat_frac=round(max(float(r[4]) for r in csv), 5),
                        PASS=bool(fl["general"]["worst_flashes_per_s"] <= 3 and fl["red"]["worst_flashes_per_s"] <= 3))
        if name.endswith("_rf1"):
            tl[name]["reduce_flashing_no_flash"] = bool(fl["general"]["worst_flashes_per_s"] == 0 and fl["red"]["worst_flashes_per_s"] == 0)
            tl[name]["reduce_flashing_no_black_gap"] = bool(black == 0) if ("trip" in name) else None
        tl[name]["mean_curve"] = [round(float(v), 5) for v in means]
    # sprint reads (T10): obstacle contrast at the read frames
    sp = os.path.join(TD, "corridor_sprint_rf0")
    if os.path.isdir(sp):
        rd = {}
        for n, k in (("read_dolly", "Dolly"), ("read_cart", "Cart"), ("read_truck", "Truck")):
            p, pi = os.path.join(sp, n + ".png"), os.path.join(sp, n + "_id.png")
            if os.path.exists(p) and os.path.exists(pi):
                img, idm = load(p), idmap(pi)
                rd[n] = obstacle(img, idm, k)
                o = outline(img, idm, k)
                if o:
                    rd[n + "_outline"] = o
        tl["sprint_reads"] = rd
        # pool rhythm: frame-mean oscillation frequency after the heads are up (t > 0.6 s)
        raw = np.fromfile(os.path.join(sp, "grid_f32.bin"), np.float32)
        GW, GH = 48, 27
        Fr = raw.size // (2 * GW * GH)
        g = raw[:Fr * 2 * GW * GH].reshape(Fr, 2, GH, GW)[:, 0]
        floor = g[:, GH * 2 // 3:, GW // 3: 2 * GW // 3].reshape(Fr, -1).mean(1)   # the floor ahead (lower third, middle)
        start = 15 + 36
        x = floor[start:] - floor[start:].mean()
        if len(x) > 32:
            spec = np.abs(np.fft.rfft(x * np.hanning(len(x))))
            freqs = np.fft.rfftfreq(len(x), 1 / 60)
            k = int(np.argmax(spec[1:]) + 1)
            tl["sprint_pool_rhythm"] = dict(peak_hz=round(float(freqs[k]), 3), bar_hz=0.92, PASS=bool(freqs[k] <= .92 + 1e-6),
                                            note="the floor ahead (lower third, middle third) after t 0.6 s; corridor bar 0.92 Hz")
    res["timeline"] = tl

# ====================================================================== kits (T4: LOD pop)
KD = os.path.join(CAP, "kits")
if os.path.isdir(KD):
    rows = [l.split("\t") for l in open(os.path.join(KD, "kits.tsv")).read().strip().splitlines()[1:]]
    pairs = {}
    for kit, shot, dist, lod, cx, cy, n, note in rows:
        if shot.startswith("lodpair"):
            pairs.setdefault((kit, shot), []).append((int(lod), float(cx), float(cy), float(dist), note))
    kres = {}
    for (kit, shot), v in sorted(pairs.items()):
        if len(v) != 2:
            continue
        v.sort()
        a = load(os.path.join(KD, f"{kit}_{shot}_lod{v[0][0]}.png"))
        b = load(os.path.join(KD, f"{kit}_{shot}_lod{v[1][0]}.png"))
        cx, cy = int(v[0][1]), int(v[0][2])
        x0, y0 = max(0, cx - 32), max(0, cy - 32)
        wa, wb = Y(a)[y0:y0 + 64, x0:x0 + 64], Y(b)[y0:y0 + 64, x0:x0 + 64]
        diff = np.abs(a * 255 - b * 255).max(-1)
        pop = abs(float(wa.mean()) - float(wb.mean())) / max(float(wa.mean()), 1e-6) * 100
        kres[kit + " " + shot] = dict(distance_m=v[0][3], lods=[v[0][0], v[1][0]], win_meanY=[round(float(wa.mean()), 4), round(float(wb.mean()), 4)],
                                      pop_pct=round(pop, 2), px_changed_gt8=int((diff > 8).sum()), PASS=bool(pop <= 2.0))
    res["kits_lod"] = kres
