/// <summary>
/// CLONE-ONLY tool (W/tools/feature/run_b; never merged). Run! B production setup in the clone
/// (run_b_production/10_plan.md stages 2-3): the WebGL tab of every new B texture capped at 1024
/// (the desktop tabs belong to FrontRoomsSurfaceTextureImporter), the B surface and kit-slot
/// materials in Resources/Surfaces, the FrontRoomsPost_Run_B profile, a forced re-import of the 18 B kits
/// (so FrontRoomsKitImporter remaps their slots to the new materials), and a report.
///   Unity -batchmode -projectPath W/proj_runb -buildTarget OSXUniversal -executeMethod FrontRoomsRunBSetup.RunBatch -quit -logFile ...
///   ... -executeMethod FrontRoomsRunBSetup.ReimportScripts -quit   (TOOLS.md known issue, before the first Play run)
/// Output: Verification/run_b/setup_report.txt. ASCII only, no C# tuples (clone editor-script rule).
/// </summary>
public static class FrontRoomsRunBSetup
{

    static readonly StringBuilder Log = new StringBuilder();
    static void L(string s) { Log.AppendLine(s); Debug.Log("[RunBSetup] " + s); }

    public static void ReimportScripts()
    {
        AssetDatabase.ImportAsset("Assets/Scripts", ImportAssetOptions.ImportRecursive | ImportAssetOptions.ForceUpdate);
        Debug.Log("[RunBSetup] Assets/Scripts re-imported");
    }

    public static void RunBatch()
    {
        Log.Length = 0;
        L("Run B setup " + DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss") + " Unity " + Application.unityVersion);
        AssetDatabase.Refresh();
        Textures_();
        Materials();
        Post();
        AssetDatabase.SaveAssets();
        ReimportKits();
        Report();
        var dir = Path.GetFullPath("Verification/run_b");
        Directory.CreateDirectory(dir);
        File.WriteAllText(Path.Combine(dir, "setup_report.txt"), Log.ToString());
        Debug.Log("[RunBSetup] DONE " + Path.Combine(dir, "setup_report.txt"));
        // TOOLS.md known issue: force a re-import of Assets/Scripts before the clone's first Play run.
        ReimportScripts();
    }

    // ------------------------------------------------------------------ textures
    static void Textures_()
    {
        foreach (var t in Textures)
        {
            var path = TexDir + t + ".png";
            var importer = AssetImporter.GetAtPath(path) as TextureImporter;
            if (importer == null) { L("MISSING texture " + path); continue; }
            var web = importer.GetPlatformTextureSettings("WebGL");
            var want = !web.overridden || web.maxTextureSize != 1024;
            if (want)
            {
                web.overridden = true;
                web.maxTextureSize = 1024;
                web.format = TextureImporterFormat.Automatic;
                web.textureCompression = TextureImporterCompression.Compressed;
                importer.SetPlatformTextureSettings(web);
                importer.SaveAndReimport();
            }
        }
        L("textures: " + Textures.Length + " B textures, WebGL tab max 1024 (desktop tabs: FrontRoomsSurfaceTextureImporter)");
    }

    static Texture2D T(string stem)
    {
        return string.IsNullOrEmpty(stem) ? null : AssetDatabase.LoadAssetAtPath<Texture2D>(TexDir + stem + ".png");
    }

    static Color Hex(string h)
    {
        Color c;
        ColorUtility.TryParseHtmlString("#" + h, out c);
        return c;
    }

    /// <summary>The tint that turns the baked field colour into another colour (per channel, in linear light).</summary>
    static Color Ratio(string target, string baked)
    {
        var a = Hex(target).linear; var b = Hex(baked).linear;
        return new Color(Mathf.LinearToGammaSpace(a.r / b.r), Mathf.LinearToGammaSpace(a.g / b.g), Mathf.LinearToGammaSpace(a.b / b.b), 1f);
    }

    static Material Surface(string name)
    {
        var path = SurfDir + name + ".mat";
        var shader = Shader.Find("FrontRooms/Surface");
        var m = AssetDatabase.LoadAssetAtPath<Material>(path);
        if (m == null) { m = new Material(shader) { name = name }; AssetDatabase.CreateAsset(m, path); }
        m.shader = shader;
        return m;
    }

    static void Configure(Material m, string tex, Vector2 tile, Color tint, bool meshUV, float smooth, float bump, float macroTone, float macroDirt, float floorGrime, float ceilingGrime, string emission, Color emissionColor, float metallic)
    {
        m.SetTexture("_BaseMap", T(tex == null ? null : tex + "_A"));
        m.SetTexture("_BumpMap", T(tex == null ? null : tex + "_N"));
        m.SetTexture("_MaskMap", T(tex == null ? null : tex + "_S"));
        m.SetColor("_BaseColor", tint);
        m.SetTextureScale("_BaseMap", Vector2.one);
        m.SetTextureOffset("_BaseMap", Vector2.zero);
        m.SetVector("_TileSize", new Vector4(tile.x, tile.y, 0f, 0f));
        m.SetFloat("_BumpScale", bump);
        m.SetFloat("_Smoothness", smooth);
        m.SetFloat("_Metallic", metallic);
        m.SetFloat("_OcclusionStrength", .8f);
        m.SetTexture("_MacroMap", T("MacroWear_M"));
        m.SetFloat("_MacroTone", macroTone);
        m.SetFloat("_MacroDirt", macroDirt);
        m.SetColor("_StainColor", new Color(.42f, .38f, .30f));
        m.SetFloat("_StainStrength", 0f);
        m.SetFloat("_WetStrength", 0f);
        m.SetFloat("_FloorGrime", floorGrime);
        m.SetFloat("_FloorGrimeHeight", .35f);
        m.SetFloat("_CeilingGrime", ceilingGrime);
        m.SetFloat("_CeilingHeight", 2.9f);
        m.SetFloat("_MeshUV", meshUV ? 1f : 0f);
        if (meshUV) m.EnableKeyword("_FR_MESH_UV"); else m.DisableKeyword("_FR_MESH_UV");
        if (emission != null)
        {
            m.SetTexture("_EmissionMap", emission.Length == 0 ? null : T(emission + "_E"));
            m.SetColor("_EmissionColor", emissionColor);
            m.SetFloat("_UseEmission", 1f);
            m.EnableKeyword("_EMISSION");
            m.globalIlluminationFlags = MaterialGlobalIlluminationFlags.RealtimeEmissive;
        }
        else
        {
            m.SetFloat("_UseEmission", 0f);
            m.DisableKeyword("_EMISSION");
        }
        EditorUtility.SetDirty(m);
    }

    static void Shell(string name, string tex, Vector2 tile, Color tint, float smooth, float macroTone, float macroDirt, float floorGrime, float ceilingGrime)
    {
        Configure(Surface(name), tex, tile, tint, false, smooth, 1f, macroTone, macroDirt, floorGrime, ceilingGrime, null, Color.black, 0f);
    }

    static void Slot(string name, Color tint, float smooth, float metallic = 0f, string tex = null, string emission = null, float bump = 1f)
    {
        Configure(Surface(name), tex, Vector2.one, tint, true, smooth, bump, 0f, 0f, 0f, 0f, emission, Color.black, metallic);
    }

    // ------------------------------------------------------------------ post
    static T Comp<T>(VolumeProfile p) where T : VolumeComponent
    {
        T c;
        if (!p.TryGet<T>(out c)) { c = p.Add<T>(true); c.name = typeof(T).Name; AssetDatabase.AddObjectToAsset(c, p); }
        c.active = true;
        return c;
    }

    static void Post()
    {
        var profile = AssetDatabase.LoadAssetAtPath<VolumeProfile>(PostPath);
        if (profile == null) { profile = ScriptableObject.CreateInstance<VolumeProfile>(); AssetDatabase.CreateAsset(profile, PostPath); }
        // r8 values (D1: lift w 0, not the brief's -0.01).
        Comp<LiftGammaGain>(profile).lift.Override(new Vector4(1f, 1f, 1f, 0f));
        var ca = Comp<ColorAdjustments>(profile);
        ca.contrast.Override(12f);
        ca.saturation.Override(-10f);
        ca.postExposure.Override(-.1f);
        Comp<WhiteBalance>(profile).temperature.Override(-6f);
        var bl = Comp<Bloom>(profile);
        bl.threshold.Override(1.0f);
        bl.intensity.Override(.6f);
        Comp<Vignette>(profile).intensity.Override(.32f);
        EditorUtility.SetDirty(profile);
        L("post: " + PostPath);
    }

    // ------------------------------------------------------------------ kits
    static void ReimportKits()
    {
        foreach (var k in Kits)
        {
            AssetDatabase.ImportAsset(ModelDir + k + ".json", ImportAssetOptions.ForceUpdate);
            AssetDatabase.ImportAsset(ModelDir + k + ".fbx", ImportAssetOptions.ForceUpdate);
        }
        FrontRoomsKitLibrary.ClearCache();
        L("kits: " + Kits.Length + " re-imported");
    }

    // ------------------------------------------------------------------ report
    static void Report()
    {
        long total = 0;
        foreach (var t in Textures)
        {
            var tex = AssetDatabase.LoadAssetAtPath<Texture2D>(TexDir + t + ".png");
            if (tex == null) { L("tex MISSING " + t); continue; }
            var size = Profiler.GetRuntimeMemorySizeLong(tex);
            total += size;
            L("tex " + t + " " + tex.width + "x" + tex.height + " " + tex.format + " mips " + tex.mipmapCount + " " + (size / 1048576f).ToString("0.00") + " MB");
        }
        L("tex TOTAL " + (total / 1048576f).ToString("0.00") + " MB (budget 40 MB)");
        foreach (var k in Kits)
        {
            var go = AssetDatabase.LoadAssetAtPath<GameObject>(ModelDir + k + ".fbx");
            if (go == null) { L("kit MISSING " + k); continue; }
            var sb = new StringBuilder();
            sb.Append("kit ").Append(k);
            var group = go.GetComponent<LODGroup>();
            if (group != null)
            {
                var lods = group.GetLODs();
                for (var i = 0; i < lods.Length; i++)
                {
                    var tris = 0;
                    foreach (var r in lods[i].renderers)
                    {
                        var mf = r != null ? r.GetComponent<MeshFilter>() : null;
                        if (mf != null && mf.sharedMesh != null) tris += mf.sharedMesh.triangles.Length / 3;
                    }
                    sb.Append(" LOD").Append(i).Append(' ').Append(tris).Append(" tris @").Append(lods[i].screenRelativeTransitionHeight.ToString("0.0000"));
                }
            }
            else
            {
                var mf = go.GetComponentInChildren<MeshFilter>();
                sb.Append(" single ").Append(mf != null && mf.sharedMesh != null ? mf.sharedMesh.triangles.Length / 3 : 0).Append(" tris");
            }
            var mats = new List<string>();
            foreach (var r in go.GetComponentsInChildren<Renderer>(true))
                foreach (var m in r.sharedMaterials)
                {
                    if (m == null) continue;
                    var key = m.name + (AssetDatabase.GetAssetPath(m).StartsWith(SurfDir) ? "" : "(fbx)");
                    if (!mats.Contains(key)) mats.Add(key);
                }
            sb.Append(" slots ").Append(string.Join(",", mats.ToArray()));
            L(sb.ToString());
        }
    }
