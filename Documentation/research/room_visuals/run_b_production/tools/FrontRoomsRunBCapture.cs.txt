/// <summary>
/// CLONE-ONLY capture harness for the Run! B production build (W/tools/feature/run_b; never merged). Stage "capture"
/// of the run_b production workflow (run_b_production/10_plan.md s.9: T4, T5, T6, T7, T9, T10, T11).
/// Every entry renders through the PRODUCTION code: FrontRoomsRunRig + FrontRoomsRunDress (corridor preset, own lamps)
/// and the real FrontRoomsRoomStream Run room (stream preset, host lamps). The r8 look-dev rig is not used.
///   CorridorBatch  (-quit): the r8 proposal shots at the r8 cameras (s1a ... r_truck, c4_k with the 2.05 m stand-in),
///                  ID masks, projected sample points, stats -> Verification/run_b/capture/corridor
///                  [-runbNoCull: the rig's 16 m light culling off]
///   StreamBatch    (-quit): the edit-mode preview's Run room, Ward and B armed / tripped, the Relay stand-in in the
///                  stream room (gap, rear door), ID masks -> .../capture/stream
///   TimelineBatch  (-quit): 60 fps sequences (trip, reset, stutter, stage-1 warning, sprint pass; Reduce Flashing off

    static Camera MakeCamera(string name, int w, int h, int msaa, float fov)
    {
        var go = new GameObject(name) { hideFlags = HideFlags.DontSave };
        var cam = go.AddComponent<Camera>();
        cam.nearClipPlane = .02f;
        cam.farClipPlane = 80f;
        cam.fieldOfView = fov;
        cam.clearFlags = CameraClearFlags.SolidColor;
        cam.backgroundColor = new Color(0x22 / 255f, 0x23 / 255f, 0x1C / 255f);
        FrontRoomsPostStack.ConfigureCamera(cam);
        // As the r8 harness: main's ConfigureCamera opts every camera into the G14 Metal RT glass; no RT glass here.
        var rtCam = cam.GetComponent("FrontRoomsGlassRTCamera");
        if (rtCam != null) UnityEngine.Object.DestroyImmediate(rtCam);
        var rt = new RenderTexture(w, h, 24, RenderTextureFormat.ARGB32) { antiAliasing = msaa, name = name + " rt" };
        rt.Create();
        cam.targetTexture = rt;
        cam.enabled = false;
        return cam;
    }

    static void Release(Camera cam)
    {
        if (cam == null) return;
        var rt = cam.targetTexture;
        cam.targetTexture = null;
        if (rt != null) rt.Release();
        UnityEngine.Object.DestroyImmediate(cam.gameObject);
    }

    static Texture2D scratch;

    static Texture2D Read(RenderTexture rt)
    {
        if (scratch == null || scratch.width != rt.width || scratch.height != rt.height)
        {
            if (scratch != null) UnityEngine.Object.DestroyImmediate(scratch);
            scratch = new Texture2D(rt.width, rt.height, TextureFormat.RGB24, false);
        }
        var prev = RenderTexture.active;
        RenderTexture.active = rt;
        scratch.ReadPixels(new Rect(0, 0, rt.width, rt.height), 0, 0);
        scratch.Apply();
        RenderTexture.active = prev;
        return scratch;
    }

    static void SavePng(RenderTexture rt, string path) { File.WriteAllBytes(path, Read(rt).EncodeToPNG()); }
    static void SaveJpg(RenderTexture rt, string path, int q) { File.WriteAllBytes(path, Read(rt).EncodeToJPG(q)); }

    static void Pose(Camera cam, Transform space, Vector3 eye, Vector3 look, float fov)
    {
        var e = space.TransformPoint(eye);
        cam.transform.SetPositionAndRotation(e, Quaternion.LookRotation(space.TransformPoint(look) - e, Vector3.up));
        cam.fieldOfView = fov;
    }

    // A batch run uses the Forward+ probe atlas only after a RenderSettings change (20_build.md s.6.1): nudge the
    // reflection intensity by 0.001 / 0.002 (alternating) whenever the probe set may have changed.
    static float reflBase = -1f;
    static int nudge;
    static void RefreshProbes()
    {
        if (reflBase < 0f) reflBase = RenderSettings.reflectionIntensity;
        nudge = nudge == 1 ? 2 : 1;
        RenderSettings.reflectionIntensity = reflBase + .001f * nudge;
    }

    static int probeSig = -1;
    /// <summary>Refresh the atlas only when the rig's enabled probe set changed (sequences).</summary>
    static void RefreshIfProbesChanged(FrontRoomsRunRig rig)
    {
        if (rig == null) return;
        var sig = 0;
        if (rig.ArmedProbes != null) for (var i = 0; i < rig.ArmedProbes.Length; i++) if (rig.ArmedProbes[i] != null && rig.ArmedProbes[i].enabled) sig |= 1 << i;
        if (rig.TrippedProbes != null) for (var i = 0; i < rig.TrippedProbes.Length; i++) if (rig.TrippedProbes[i] != null && rig.TrippedProbes[i].enabled) sig |= 1 << (8 + i);
        if (sig != probeSig) { probeSig = sig; RefreshProbes(); }
    }

    static Cubemap flatCube;

    /// <summary>The r8 harness's flat ambient cube as the default reflection (an empty scene would reflect the blue sky).</summary>
    static void SetupReflections()
    {
        flatCube = new Cubemap(16, TextureFormat.RGBAHalf, false) { name = "RunB_FlatAmbientCube" };
        var sky = FrontRoomsLook.AmbientSky.linear; var eq = FrontRoomsLook.AmbientEquator.linear; var gr = FrontRoomsLook.AmbientGround.linear;
        foreach (CubemapFace f in Enum.GetValues(typeof(CubemapFace)))
        {
            if (f == CubemapFace.Unknown) continue;
            var col = f == CubemapFace.PositiveY ? sky : f == CubemapFace.NegativeY ? gr : eq;
            var px = new Color[16 * 16];
            for (var i = 0; i < px.Length; i++)
            {
                var row = i / 16;
                var c = col;
                if (f != CubemapFace.PositiveY && f != CubemapFace.NegativeY) c = row < 8 ? Color.Lerp(sky, eq, row / 8f) : Color.Lerp(eq, gr, (row - 8) / 8f);
                px[i] = c;
            }
            flatCube.SetPixels(px, f);
        }
        flatCube.Apply();
        RenderSettings.defaultReflectionMode = DefaultReflectionMode.Custom;
        RenderSettings.customReflectionTexture = flatCube;
    }

    static Material Lit(string name, Color c, float smooth, float metal = 0f, Color? emission = null)
    {
        var m = new Material(Shader.Find("Universal Render Pipeline/Lit")) { name = name };
        m.SetColor("_BaseColor", c);
        m.SetFloat("_Smoothness", smooth);
        m.SetFloat("_Metallic", metal);
        if (emission.HasValue) { m.EnableKeyword("_EMISSION"); m.SetColor("_EmissionColor", emission.Value); m.globalIlluminationFlags = MaterialGlobalIlluminationFlags.None; }
        return m;
    }

    static Material Surf(string name, Color fallback) { var m = FrontRoomsSurfaces.TryGet(name); return m != null ? m : Lit(name + " (fallback)", fallback, .3f); }
    static Color Hex(string h) { Color c; ColorUtility.TryParseHtmlString("#" + h, out c); return c; }

    static Transform Group(Transform parent, string name)
    {
        var t = new GameObject(name).transform;
        t.SetParent(parent, false);
        return t;
    }

    static Renderer Box(Transform parent, string name, Vector3 centre, Vector3 size, Material m)
    {
        var go = GameObject.CreatePrimitive(PrimitiveType.Cube);
        UnityEngine.Object.DestroyImmediate(go.GetComponent<Collider>());
        go.name = name;
        go.transform.SetParent(parent, false);
        go.transform.localPosition = centre;
        go.transform.localScale = size;
        var r = go.GetComponent<Renderer>();
        r.sharedMaterial = m;
        return r;
    }

    static Renderer Cyl(Transform parent, string name, Vector3 a, Vector3 b, float radius, Material m)
    {
        var go = GameObject.CreatePrimitive(PrimitiveType.Cylinder);
        UnityEngine.Object.DestroyImmediate(go.GetComponent<Collider>());
        go.name = name;
        go.transform.SetParent(parent, false);
        go.transform.localPosition = (a + b) * .5f;
        go.transform.localRotation = Quaternion.FromToRotation(Vector3.up, (b - a).normalized);
        go.transform.localScale = new Vector3(radius * 2f, (b - a).magnitude * .5f, radius * 2f);
        var r = go.GetComponent<Renderer>();
        r.sharedMaterial = m;
        return r;
    }

    static Renderer Ball(Transform parent, string name, Vector3 c, Vector3 size, Material m)
    {
        var go = GameObject.CreatePrimitive(PrimitiveType.Sphere);
        UnityEngine.Object.DestroyImmediate(go.GetComponent<Collider>());
        go.name = name;
        go.transform.SetParent(parent, false);
        go.transform.localPosition = c;
        go.transform.localScale = size;
        var r = go.GetComponent<Renderer>();
        r.sharedMaterial = m;
        return r;
    }

    static Light Spot(Transform parent, string name, Vector3 pos, Quaternion rot, float outer, float inner, Color c, float intensity, float range, LightShadows shadows)
    {
        var go = new GameObject(name);
        go.transform.SetParent(parent, false);
        go.transform.localPosition = pos;
        go.transform.localRotation = rot;
        var l = go.AddComponent<Light>();
        l.type = LightType.Spot;
        l.spotAngle = outer;
        l.innerSpotAngle = inner;
        l.color = c;
        l.intensity = intensity;
        l.range = range;
        l.shadows = shadows;
        l.shadowStrength = .9f;
        return l;
    }

    // ================================================================== ID masks (the r8 palette)
    enum Id { Background, Relay, Goal, Dolly, Cart, Truck, Units, Main, Emitter, Door, Sign }
    static readonly Color[] IdColour =
    {
        Color.black, Color.white, Color.red, Color.green, Color.blue, Color.yellow, Color.magenta, Color.cyan,
        new Color(.5f, .5f, .5f), new Color(1f, .5f, 0f), new Color(.5f, 0f, 1f),
    };

    static bool Under(Transform t, string part)
    {
        for (; t != null; t = t.parent) if (t.name.IndexOf(part, StringComparison.OrdinalIgnoreCase) >= 0) return true;
        return false;
    }

    static Id Classify(Renderer r, Transform relay, Transform goal)
    {
        var t = r.transform;
        if (relay != null && t.IsChildOf(relay)) return Id.Relay;
        if (goal != null && t.IsChildOf(goal)) return Id.Goal;
        if (Under(t, "obstacle Kit_ChairDolly8")) return Id.Dolly;
        if (Under(t, "obstacle Kit_JanitorCart")) return Id.Cart;
        if (Under(t, "obstacle Kit_HandTruck")) return Id.Truck;
        if (Under(t, "twin-head unit")) return Id.Units;
        if (Under(t, "run b sprinkler main")) return Id.Main;
        if (Under(t, "run b side door")) return Id.Door;
        if (r.GetComponentInParent<FrontRoomsExitSign>() != null || Under(t, "exit sign")) return Id.Sign;
        if (Under(t, "Kit_WrapFixture") || Under(t, "wrap")) return Id.Emitter;
        return Id.Background;
    }

    static bool IsFx(Renderer r)
    {
        var m = r.sharedMaterial;
        return m != null && m.shader != null && (m.shader.name.Contains("Beam") || m.shader.name.Contains("Halo"));
    }

    /// <summary>Flat unlit ID colours, no post, no MSAA, no fog. Only renderers under <paramref name="scope"/> are
    /// coloured; every other enabled renderer in the scene renders black (the stream scene holds the whole game).</summary>
    static void RenderId(Camera cam, Transform scope, Transform relay, Transform goal, string path)
    {
        var unlit = Shader.Find("Universal Render Pipeline/Unlit");
        var idMats = new Material[IdColour.Length];
        for (var i = 0; i < idMats.Length; i++) { idMats[i] = new Material(unlit); idMats[i].SetColor("_BaseColor", IdColour[i]); }
        var saved = new Dictionary<Renderer, Material[]>();
        var hidden = new List<Renderer>();
        foreach (var r in UnityEngine.Object.FindObjectsByType<Renderer>(FindObjectsInactive.Exclude, FindObjectsSortMode.None))
        {
            if (!r.enabled) continue;
            if (IsFx(r) || r is ParticleSystemRenderer) { hidden.Add(r); r.enabled = false; continue; }
            saved[r] = r.sharedMaterials;
            var id = Id.Background;
            if (relay != null && r.transform.IsChildOf(relay)) id = Id.Relay;
            else if (scope != null && r.transform.IsChildOf(scope)) id = Classify(r, relay, goal);
            var arr = new Material[r.sharedMaterials.Length];
            for (var i = 0; i < arr.Length; i++) arr[i] = idMats[(int)id];
            r.sharedMaterials = arr;
        }
        var fog = RenderSettings.fog;
        RenderSettings.fog = false;
        var data = cam.GetUniversalAdditionalCameraData();
        var post = data.renderPostProcessing;
        data.renderPostProcessing = false;
        var prevRt = cam.targetTexture;
        var rt = new RenderTexture(prevRt.width, prevRt.height, 24, RenderTextureFormat.ARGB32) { antiAliasing = 1 };
        cam.targetTexture = rt;
        cam.allowMSAA = false;
        cam.Render();
        SavePng(rt, path);
        cam.targetTexture = prevRt;
        cam.allowMSAA = true;
        rt.Release();
        data.renderPostProcessing = post;
        RenderSettings.fog = fog;
        foreach (var kv in saved) kv.Key.sharedMaterials = kv.Value;
        foreach (var r in hidden) r.enabled = true;
        foreach (var m in idMats) UnityEngine.Object.DestroyImmediate(m);
    }

    // ================================================================== the corridor (Corridor1x9 + stand-ins)
    sealed class Corridor
    {
        public Transform root;
        public RunRigLayout layout;
        public FrontRoomsRunRig rig;
        public Transform relay;
        public Transform goal;
        public Transform ante;
        public Transform stub;
        public SphericalHarmonicsL2 armedSH, trippedSH;
    }

    static Corridor BuildCorridor(bool newScene)
    {
        if (newScene)
        {
            var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
            EditorSceneManager.SetActiveScene(scene);
        }
        var c = new Corridor();
        c.root = new GameObject("RUNB CAPTURE / corridor").transform;
        c.root.position = RootWorld;
        FrontRoomsLook.ApplyAmbient();
        FrontRoomsPostStack.Ensure(c.root);
        SetupReflections();
        c.layout = RunRigLayout.Corridor1x9();
        var dress = FrontRoomsRunDress.Build(c.root, c.layout);
        c.rig = FrontRoomsRunRig.Create(c.root, c.layout, RunRigHost.OwnLamps, null, dress);
        c.ante = Group(c.root, "stand-in / Level 0 anteroom (r8)");
        c.stub = Group(c.root, "stand-in / Level 0 stub (r8)");
        c.goal = Group(c.root, "stand-in / goal RN-X (r8, call 2 open)");
        BuildAnteroom(c.ante);
        BuildStub(c.stub);
        BuildGoalDoor(c.goal);
        c.relay = BuildRelay(c.root, new Vector3(0f, 0f, 11.5f), 0f);
        c.armedSH = FrontRoomsRunRig.BounceSH(c.layout.armedUp, c.layout.armedSide, c.layout.armedDown, c.layout.armedTint);
        c.trippedSH = FrontRoomsRunRig.BounceSH(c.layout.trippedUp, c.layout.trippedSide, c.layout.trippedDown, c.layout.trippedTint);
        foreach (var r in c.goal.GetComponentsInChildren<Renderer>(true)) r.lightProbeUsage = LightProbeUsage.CustomProvided;
        return c;
    }

    /// <summary>Drive the rig to trip time t through its own state machine (Trip + Tick at 120 Hz).</summary>
    static void Drive(Corridor c, float t)
    {
        c.rig.ResetImmediate();
        if (t >= 0f)
        {
            c.rig.Trip();
            var n = Mathf.RoundToInt(t * 120f);
            for (var i = 0; i < n; i++) c.rig.Tick(1f / 120f);
        }
        c.rig.Tick(0f);
        StandInSH(c, t);
        RefreshProbes();
    }

    /// <summary>The r8 harness kept the goal door (and the Relay) in the corridor's SH; the stand-ins get the rig's
    /// state SH here (armed x wrap level + tripped x mean head level, as FrontRoomsRunRig.ApplyAmbient).</summary>
    static void StandInSH(Corridor c, float t)
    {
        var w = c.rig.WrapLevel(t);
        var hl = c.rig.MeanHeadLevel(t);
        var sh = new SphericalHarmonicsL2[] { c.armedSH * w + c.trippedSH * hl };
        var block = new MaterialPropertyBlock();
        var list = new List<Renderer>(c.goal.GetComponentsInChildren<Renderer>(true));
        foreach (var r in c.relay.GetComponentsInChildren<Renderer>(true)) if (r.lightProbeUsage == LightProbeUsage.CustomProvided) list.Add(r);
        foreach (var r in list)
            for (var k = 0; k < r.sharedMaterials.Length; k++)
            {
                block.Clear();
                block.CopySHCoefficientArraysFrom(sh);
                r.SetPropertyBlock(block, k);
            }
    }

    static void RelayAmbient(Corridor c, bool rigSH)
    {
        foreach (var r in c.relay.GetComponentsInChildren<Renderer>(true))
        {
            r.lightProbeUsage = rigSH ? LightProbeUsage.CustomProvided : LightProbeUsage.BlendProbes;
            if (!rigSH) r.SetPropertyBlock(null);
        }
    }

    static Light MapLamp(Transform parent, Vector3 cellCentre, bool shadows)
    {
        var lens = Lit("L0 map lamp lens", new Color(1f, .98f, .92f), .1f, 0f, new Color(1f, .96f, .84f) * 2.6f);
        var p = cellCentre + new Vector3(0f, 0f, .3f);
        var r = Box(parent, "map lamp lens", new Vector3(p.x, H - .0125f, p.z), new Vector3(.6f, .025f, 1.2f), lens);
        r.shadowCastingMode = ShadowCastingMode.Off;
        return Spot(parent, "map lamp", new Vector3(p.x, H - .025f - .06f, p.z), Quaternion.Euler(90f, 0f, 0f), 162f, 96f, new Color(1f, .96f, .88f), 5f, 10f, shadows ? LightShadows.Soft : LightShadows.None);
    }

    static void L0Walls(Transform parent, float x0, float x1, float z0, float z1, bool south, bool north, bool west, bool east)
    {
        var paper = FrontRoomsSurfaces.Room(RoomRule.Lobby, FrontRoomsSurfaces.Slot.Wall);
        var cove = FrontRoomsSurfaces.CoveBase;
        const float t = .16f;
        if (south) { Box(parent, "L0 wall", new Vector3((x0 + x1) * .5f, H * .5f, z0), new Vector3(x1 - x0 + t, H, t), paper); Box(parent, "L0 cove", new Vector3((x0 + x1) * .5f, .05f, z0), new Vector3(x1 - x0 + t, .1f, t + .02f), cove); }
        if (north) { Box(parent, "L0 wall", new Vector3((x0 + x1) * .5f, H * .5f, z1), new Vector3(x1 - x0 + t, H, t), paper); Box(parent, "L0 cove", new Vector3((x0 + x1) * .5f, .05f, z1), new Vector3(x1 - x0 + t, .1f, t + .02f), cove); }
        if (west) { Box(parent, "L0 wall", new Vector3(x0, H * .5f, (z0 + z1) * .5f), new Vector3(t, H, z1 - z0 + t), paper); Box(parent, "L0 cove", new Vector3(x0, .05f, (z0 + z1) * .5f), new Vector3(t + .02f, .1f, z1 - z0 + t), cove); }
        if (east) { Box(parent, "L0 wall", new Vector3(x1, H * .5f, (z0 + z1) * .5f), new Vector3(t, H, z1 - z0 + t), paper); Box(parent, "L0 cove", new Vector3(x1, .05f, (z0 + z1) * .5f), new Vector3(t + .02f, .1f, z1 - z0 + t), cove); }
    }

    /// <summary>The r8 Level 0 anteroom (3 x 2 cells) behind the entry face. Its north wall sits behind the production
    /// shell's entry slab (z -0.10 .. 0), with the same 1.4 x 2.2 arch.</summary>
    static void BuildAnteroom(Transform ante)
    {
        var carpet = FrontRoomsSurfaces.Room(RoomRule.Lobby, FrontRoomsSurfaces.Slot.Floor);
        var ceil = FrontRoomsSurfaces.Room(RoomRule.Lobby, FrontRoomsSurfaces.Slot.Ceiling);
        var paper = FrontRoomsSurfaces.Room(RoomRule.Lobby, FrontRoomsSurfaces.Slot.Wall);
        Box(ante, "L0 carpet", new Vector3(0f, -.1f, -3.09f), new Vector3(9.2f, .2f, 6.02f), carpet);
        Box(ante, "L0 ceiling", new Vector3(0f, H + .1f, -3.09f), new Vector3(9.2f, .2f, 6.02f), ceil);
        L0Walls(ante, -4.5f, 4.5f, -6.1f, -.1f, true, false, true, true);
        Box(ante, "L0 north wall L", new Vector3(-2.6f, H * .5f, -.14f), new Vector3(3.8f, H, .08f), paper);
        Box(ante, "L0 north wall R", new Vector3(2.6f, H * .5f, -.14f), new Vector3(3.8f, H, .08f), paper);
        Box(ante, "L0 north header", new Vector3(0f, (2.2f + H) * .5f, -.14f), new Vector3(1.4f, H - 2.2f, .08f), paper);
        for (var cx = -1; cx <= 1; cx++)
            for (var cz = 0; cz < 2; cz++)
                MapLamp(ante, new Vector3(cx * 3f, 0f, -4.6f + cz * 3f), cz == 1 && (cx == 0 || cx == 1));
    }

    /// <summary>The r8 Level 0 stub (3 x 3 m) behind the goal face (production shell slab z 27 .. 27.1, gap x +-0.5, 2.2 m).</summary>
    static void BuildStub(Transform stub)
    {
        var carpet = FrontRoomsSurfaces.Room(RoomRule.Lobby, FrontRoomsSurfaces.Slot.Floor);

    public static void CorridorBatch()
    {
        Log.Length = 0;
        var dir = Dir("corridor" + (Has("-runbNoCull") ? "_nocull" : ""));
        var only = Arg("-runbShots");
        var filter = string.IsNullOrEmpty(only) ? null : new HashSet<string>(only.Split(','));
        SetReduce(false);
        var c = BuildCorridor(true);
        var cam = MakeCamera("RUNB CAPTURE camera", 1920, 1080, 4, 72f);
        if (!Has("-runbNoCull")) c.rig.SetViewer(cam.transform);
        L("corridor built: rig own lamps " + c.rig.OwnLampCount + ", heads " + c.rig.HeadCount + ", probes " + c.rig.ProbeCount + ", cull " + (Has("-runbNoCull") ? "off" : "16 m (viewer = camera)"));
        Pose(cam, c.root, new Vector3(0f, 1.62f, 13.5f), new Vector3(0f, 1.3f, 27f), 72f);   // mid-corridor: every rig light within 16 m
        Stats(c, dir);
        var samples = new StringBuilder("{\n");
        var first = true;
        foreach (var s in CorridorShots())
        {
            if (filter != null && !filter.Contains(s.name)) continue;
            var t0 = DateTime.Now;
            c.relay.gameObject.SetActive(s.relay);
            c.relay.localScale = Vector3.one * (s.relayScale > 0f ? s.relayScale : 1f);
            RelayAmbient(c, s.relaySH);
            Pose(cam, c.root, s.eye, s.look, s.fov);
            if (s.name == "h_lamp_tripped_03")
            {
                // the unit-1 (z 9) head nearest the camera side: 0.3 m from its lens, 45 degrees off its beam axis
                Light best = null;
                for (var i = 0; i < c.rig.HeadCount; i++)
                {
                    var hl = c.rig.HeadLight(i);
                    if (hl == null) continue;
                    var lp = c.root.InverseTransformPoint(hl.transform.position);
                    if (Mathf.Abs(lp.z - 9f) < .4f && (best == null || lp.z < c.root.InverseTransformPoint(best.transform.position).z)) best = hl;
                }
                if (best != null)
                {
                    var lensP = best.transform.position;
                    var fwd = best.transform.forward;
                    var sideV = Vector3.Cross(Vector3.up, fwd).normalized;
                    var eyeW = lensP + (fwd * .707f + sideV * .707f).normalized * .3f;
                    cam.transform.SetPositionAndRotation(eyeW, Quaternion.LookRotation(lensP - eyeW, Vector3.up));
                }
            }
            Drive(c, s.t);
            cam.Render(); cam.Render();
            SavePng(cam.targetTexture, Path.Combine(dir, s.name + ".png"));
            int total, on, sh;
            c.rig.CountLights(out total, out on, out sh);
            L("shot " + s.name + " t=" + F(s.t) + " state " + c.rig.State + " rig lights on " + on + "/" + total + " (shadowed " + sh + ") " + (DateTime.Now - t0).TotalSeconds.ToString("0.0", IC) + " s");
            if (!first) samples.Append(",\n");
            first = false;
            samples.Append(" \"" + s.name + "\": {\"t\": " + F(s.t) + ", \"lights_on\": " + on + ", \"shadowed\": " + sh + ", \"axis\": [");
            var fa = true;
            for (var z = .5f; z <= 26.5f; z += .5f)
            {
                var p = cam.WorldToScreenPoint(c.root.TransformPoint(new Vector3(0f, .003f, z)));
                if (p.z <= 0f || p.x < 0 || p.x >= 1920 || p.y < 0 || p.y >= 1080) continue;
                samples.Append((fa ? "" : ", ") + "[" + F(z) + ", " + F(p.x) + ", " + F(1080f - p.y) + "]");
                fa = false;
            }
            samples.Append("], \"points\": {");
            var names = new List<string> { "pool6", "pool12", "pool18", "pool25_5", "goal_leaf", "relay_head" };
            var pts = new List<Vector3> { new Vector3(0f, .003f, 6f), new Vector3(0f, .003f, 12f), new Vector3(0f, .003f, 18f), new Vector3(0f, .003f, 25.5f), new Vector3(0f, 1.0f, 27f), new Vector3(0f, 2.3f, 11.5f) };
            for (var i = 0; i < c.rig.HeadCount; i++)
            {
                var hl = c.rig.HeadLight(i);
                if (hl == null) continue;
                names.Add("head" + i + (c.rig.HeadIsGoal(i) ? "_goal" : ""));
                pts.Add(c.root.InverseTransformPoint(hl.transform.position));
            }
            for (var i = 0; i < names.Count; i++)
            {
                var p = cam.WorldToScreenPoint(c.root.TransformPoint(pts[i]));
                samples.Append((i == 0 ? "" : ", ") + "\"" + names[i] + "\": [" + F(p.x) + ", " + F(1080f - p.y) + ", " + F(p.z) + "]");
            }
            samples.Append("}}");
            if (s.id) RenderId(cam, c.root, c.relay, c.goal, Path.Combine(dir, s.name + "_id.png"));
        }
        samples.Append("\n}\n");
        File.WriteAllText(Path.Combine(dir, "samples.json"), samples.ToString());
        Release(cam);
        L("CORRIDOR DONE " + dir);
        File.WriteAllText(Path.Combine(dir, "log.txt"), Log.ToString());
    }

    static void Stats(Corridor c, string dir)
    {
        var sb = new StringBuilder();
        long all = 0, lod0 = 0; var renderers = 0; var sub = 0;
        var mats = new HashSet<Material>();
        foreach (var r in c.root.GetComponentsInChildren<Renderer>(true))
        {
            if (r.transform.IsChildOf(c.ante) || r.transform.IsChildOf(c.stub) || r.transform.IsChildOf(c.goal) || r.transform.IsChildOf(c.relay)) continue;
            renderers++;
            var mf = r.GetComponent<MeshFilter>();
            if (mf == null || mf.sharedMesh == null) continue;
            long t = 0;
            for (var i = 0; i < mf.sharedMesh.subMeshCount; i++) t += mf.sharedMesh.GetIndexCount(i) / 3;
            all += t; sub += mf.sharedMesh.subMeshCount;
            if (!r.name.EndsWith("_LOD1", StringComparison.Ordinal) && !r.name.EndsWith("_LOD2", StringComparison.Ordinal)) lod0 += t;
            foreach (var m in r.sharedMaterials) if (m != null) mats.Add(m);
        }
        sb.AppendLine("B corridor (dress + rig, stand-ins excluded): renderers " + renderers + ", submeshes " + sub + ", tris all LODs " + all + ", LOD0-only " + lod0 + ", materials " + mats.Count);
        var lights = c.root.GetComponentsInChildren<Light>(true);
        var rigLights = 0;
        foreach (var l in lights) if (l.transform.IsChildOf(c.rig.transform) || l.GetComponentInParent<FrontRoomsRunRig>() != null) rigLights++;
        sb.AppendLine("lights in scene " + lights.Length + " (rig-owned " + rigLights + ")");
        foreach (var t in new[] { -1f, .2f, 3f })
        {
            Drive(c, t);
            int total, on, sh;
            c.rig.CountLights(out total, out on, out sh);
            sb.AppendLine("t=" + F(t) + " " + c.rig.State + ": rig lights on " + on + "/" + total + ", shadowed " + sh);
        }
        File.WriteAllText(Path.Combine(dir, "stats.txt"), sb.ToString());
        L(sb.ToString());
    }

    static Transform FindDeep(Transform t, string name)
    {
        if (t.name == name) return t;
        foreach (Transform c in t) { var f = FindDeep(c, name); if (f != null) return f; }
        return null;
    }

    static Transform RunRoom(FrontRoomsRoomStream s) { return FindDeep(s.transform, "Profile " + (int)RoomRule.Run + " / " + RoomRule.Run); }

    /// <summary>The preview does not tick the stream: switch the host lamps off by hand, as TickRunB's mains hook does in
    /// play (output x MainsAt = 0, lens x LensAt ~ 0 at t +3, halos off).</summary>
    static int HostLampsOff(Transform run)
    {
        var n = 0;
        foreach (var l in run.GetComponentsInChildren<Light>(true))
            if (l.name.StartsWith("fluorescent", StringComparison.Ordinal)) { l.enabled = false; n++; }
        var block = new MaterialPropertyBlock();
        foreach (var r in run.GetComponentsInChildren<Renderer>(true))
        {
            if (r.name.StartsWith("run b soffit halo", StringComparison.Ordinal)) { r.enabled = false; continue; }
            var mats = r.sharedMaterials;
            for (var m = 0; m < mats.Length; m++)
                if (mats[m] != null && mats[m].name.StartsWith("Run_WrapLens", StringComparison.Ordinal)) { block.Clear(); block.SetColor(EmissionId, Color.black); r.SetPropertyBlock(block, m); }
        }
        return n;
    }

    static void StreamPose(Camera cam, Transform space, float z0, string pose)
    {
        switch (pose)
        {
            case "fwd": Pose(cam, space, new Vector3(.6f, 1.62f, z0 + 1.6f), new Vector3(0f, 1.3f, z0 + 12f), 72f); break;
            case "back": Pose(cam, space, new Vector3(3.6f, 1.62f, z0 + 10.4f), new Vector3(-.4f, 1.0f, z0 + 1.5f), 72f); break;
            case "unit": Pose(cam, space, new Vector3(4.55f, 1.95f, z0 + 1.9f), new Vector3(5.5f, 2.42f, z0 + 3.0f), 72f); break;
            case "wrap": Pose(cam, space, new Vector3(.95f, 1.75f, z0 + .7f), new Vector3(0f, 2.84f, z0 + 1.9f), 72f); break;
            case "lane": Pose(cam, space, new Vector3(0f, 1.62f, z0 + 1.6f), new Vector3(0f, 1.0f, z0 + 12f), 76f); break;

        }
    }

    // ================================================================== TimelineBatch (T9 photosafety, T10 sprint)
    const int GW = 48, GH = 27, TW = 640, TH = 360;
    static readonly float[] LinLut = BuildLut();
    static float[] BuildLut()
    {
        var t = new float[256];
        for (var i = 0; i < 256; i++) { var c = i / 255f; t[i] = c <= .04045f ? c / 12.92f : Mathf.Pow((c + .055f) / 1.055f, 2.4f); }
        return t;
    }

    sealed class Seq
    {
        public string name; public BinaryWriter grid; public StringBuilder csv; public string dir; public int frame; public int jpgEvery = 4;
    }

    static Seq Begin(string root, string name)
    {
        var d = Path.Combine(root, name);
        Directory.CreateDirectory(d);
        var s = new Seq { name = name, dir = d, csv = new StringBuilder("frame,t,meanY,p95Y,red_sat_frac,note\n") };
        s.grid = new BinaryWriter(File.Open(Path.Combine(d, "grid_f32.bin"), FileMode.Create));
        return s;
    }

    static void Frame(Seq s, Camera cam, float t, string note)
    {
        cam.Render();
        var tex = Read(cam.targetTexture);
        var px = tex.GetPixels32();
        var cellY = new double[GW * GH];
        var cellR = new double[GW * GH];
        var ys = new float[px.Length];
        double sum = 0; var red = 0;
        // GetPixels32 starts at the bottom row; grid rows are written top-down.
        for (var i = 0; i < px.Length; i++)
        {
            var c = px[i];
            var y = .2126f * LinLut[c.r] + .7152f * LinLut[c.g] + .0722f * LinLut[c.b];
            ys[i] = y; sum += y;
            var x = i % TW; var row = TH - 1 - i / TW;
            var cell = (row * GH / TH) * GW + (x * GW / TW);
            cellY[cell] += y;
            var ss = c.r + c.g + c.b;
            if (ss > 30 && c.r >= .8f * ss) { red++; cellR[cell] += 1; }
        }
        var per = (double)(TW / GW) * (TH / GH);
        for (var i = 0; i < cellY.Length; i++) s.grid.Write((float)(cellY[i] / per));
        for (var i = 0; i < cellR.Length; i++) s.grid.Write((float)(cellR[i] / per));
        Array.Sort(ys);
        s.csv.Append(s.frame + "," + F(t) + "," + (sum / px.Length).ToString("0.000000", IC) + "," + ys[(int)(ys.Length * .95f)].ToString("0.00000", IC) + "," + ((float)red / px.Length).ToString("0.00000", IC) + "," + note + "\n");
        if (s.frame % s.jpgEvery == 0) File.WriteAllBytes(Path.Combine(s.dir, "f" + s.frame.ToString("000") + ".jpg"), tex.EncodeToJPG(88));
        s.frame++;
    }

    static void End(Seq s, string meta)
    {
        s.grid.Close();
        File.WriteAllText(Path.Combine(s.dir, "frames.csv"), s.csv.ToString());
        File.WriteAllText(Path.Combine(s.dir, "meta.json"), "{\"name\": \"" + s.name + "\", \"frames\": " + s.frame + ", \"fps\": 60, \"grid\": [" + GW + ", " + GH + "], \"size\": [" + TW + ", " + TH + "], \"layout\": \"float32: per frame Y[GH*GW] then redfrac[GH*GW], rows top-down\", " + meta + "}\n");
        L("sequence " + s.name + ": " + s.frame + " frames");
    }

    /// <summary>The real stage-1 burst (FrontRoomsWarnLamps): 3 dips 0.55 s apart, attack 0.10 to depth 0.55, hold 0.15,
    /// release 0.22. Reduce Flashing: one dip to 0.7 with 0.6 s attack and release.</summary>
    static float WarnAt(float t, bool reduced)
    {
        if (reduced)
        {
            const float a = FrontRoomsWarnLamps.ReducedSeconds, depth = FrontRoomsWarnLamps.ReducedDepth;
            if (t < 0f) return 1f;
            if (t < a) return Mathf.Lerp(1f, depth, t / a);
            if (t < 2f * a) return Mathf.Lerp(depth, 1f, (t - a) / a);
            return 1f;
        }
        var m = 1f;
        for (var d = 0; d < 3; d++)
        {
            var s = t - d * FrontRoomsWarnLamps.DipSpacing;
            if (s < 0f) continue;
            const float at = FrontRoomsWarnLamps.DipAttack, hold = .15f, rel = FrontRoomsWarnLamps.DipRelease, depth2 = FrontRoomsWarnLamps.DipDepthMin;
            float v;
            if (s < at) v = Mathf.Lerp(1f, depth2, s / at);
            else if (s < at + hold) v = depth2;
            else if (s < at + hold + rel) v = Mathf.Lerp(depth2, 1f, (s - at - hold) / rel);
            else v = 1f;
            m = Mathf.Min(m, v);
        }
        return m;
    }

            var front = go.transform.forward;    // sidecars: frontAxis +Z
            var ctr = b.center;
            var ext = Vector3.Dot(b.extents, new Vector3(Mathf.Abs(front.x), Mathf.Abs(front.y), Mathf.Abs(front.z)));
            var side = Vector3.Cross(Vector3.up, front).normalized;
            // lights relative to the kit
            keyL.transform.position = ctr + front * 1.6f + side * -1.2f + Vector3.up * 1.4f; keyL.transform.LookAt(ctr);
            fill.transform.position = ctr + front * 1.8f + side * 1.6f + Vector3.up * .3f; fill.transform.LookAt(ctr);
            rim.transform.position = ctr - front * 1.2f + side * .8f + Vector3.up * 1.6f; rim.transform.LookAt(ctr);
            var lg = go.GetComponentInChildren<LODGroup>(true);
            var lodCount = lg != null ? lg.lodCount : 1;
            // (1) whole kit, 3/4 view, fitted
            var fitD = Mathf.Max(.35f, b.extents.magnitude / Mathf.Tan(38f * Mathf.Deg2Rad) * 1.15f);
            var dir34 = (front + side * .55f + Vector3.up * .35f).normalized;
            cam.transform.SetPositionAndRotation(ctr + dir34 * fitD, Quaternion.LookRotation(-dir34, Vector3.up));
            if (lg != null) lg.ForceLOD(0);
            RefreshProbes();
            cam.Render(); cam.Render();
            SavePng(cam.targetTexture, Path.Combine(dir, kit + "_fit.png"));
            report.Append(kit + "\tfit\t" + F(fitD) + "\t0\t960\t540\t" + lodCount + "\t3/4 view fitted\n");
            // (2) 0.3 m close-up on the front face (camera 0.3 m from the nearest front surface, vFOV 76)
            var faceCentre = ctr + front * ext;
            var detail = faceCentre;
            Vector3 hero;
            if (HeroPoint(kit, go.transform, info, out hero)) detail = hero;
            // ceiling kits are seen from below (as the player sees them), the rest face-on; 0.3 m from the detail point
            var ceilingKit = kit == "Kit_WrapFixture_4ft" || kit.StartsWith("Kit_Sprinkler", StringComparison.Ordinal) || kit == "Kit_PipeHanger_Band" || kit == "Kit_CableTray_3m";
            var camDir = ceilingKit ? (front * .5f - Vector3.up * .866f).normalized : (front + Vector3.up * .067f + side * .1f).normalized;
            var camPos = detail + camDir * .3f;
            cam.transform.SetPositionAndRotation(camPos, Quaternion.LookRotation(detail - camPos, Vector3.up));
            cam.Render(); cam.Render();
            SavePng(cam.targetTexture, Path.Combine(dir, kit + "_03m.png"));
            report.Append(kit + "\tclose_0.3m\t0.3\t0\t960\t540\t" + lodCount + "\tfront face" + (detail != faceCentre ? " hero point" : "") + "\n");
            // (3) LOD pairs at the authored switch distances (same camera, ForceLOD)
            if (lg != null && info != null && info.lodDistances != null)
            {
                for (var k = 0; k < 2 && k + 1 < lodCount; k++)
                {
                    var d = k < info.lodDistances.Length ? info.lodDistances[k] : 0f;
                    if (d <= 0f) continue;
                    var dirL = (front + Vector3.up * .25f + side * .2f).normalized;
                    // the camera at the switch distance from the LODGroup's reference point (its world bounds centre)
                    var refP = lg.transform.TransformPoint(lg.localReferencePoint);
                    var cp = refP + dirL * d * .995f;
                    cam.transform.SetPositionAndRotation(cp, Quaternion.LookRotation(refP - cp, Vector3.up));
                    var sp = cam.WorldToScreenPoint(refP);
                    for (var lod = k; lod <= k + 1; lod++)
                    {
                        lg.ForceLOD(lod);
                        cam.Render(); cam.Render();
                        SavePng(cam.targetTexture, Path.Combine(dir, kit + "_lodpair" + k + "_lod" + lod + ".png"));
                        report.Append(kit + "\tlodpair" + k + "\t" + F(d * .995f) + "\t" + lod + "\t" + F(sp.x) + "\t" + F(1080f - sp.y) + "\t" + lodCount + "\tauthored d" + k + (k + 1) + " " + F(d) + "\n");
                    }
                }
                lg.ForceLOD(-1);
            }
            UnityEngine.Object.DestroyImmediate(go);
            L("kit " + kit + " done (LODs " + lodCount + ")");
        }
        File.WriteAllText(Path.Combine(dir, "kits.tsv"), report.ToString());
        Release(cam);
        L("KITS DONE " + dir);
        File.WriteAllText(Path.Combine(dir, "log.txt"), Log.ToString());
    }

    /// <summary>Hero detail points (kit-local) for the 0.3 m close-ups, from the sidecar anchors where they exist.</summary>
    static bool HeroPoint(string kit, Transform t, FrontRoomsKitLibrary.Info info, out Vector3 p)
    {
        p = Vector3.zero;
        Vector3 a;
        switch (kit)
        {
            case "Kit_EmergencyTwinHead_Lamp":
                if (info != null && info.TryAnchor("lens", out a)) { p = t.TransformPoint(a); return true; }
                return false;
            case "Kit_WrapFixture_4ft":
                // the end cap and the prismatic lens: 0.45 m from the centre along the fixture, at the lens bottom
                p = t.TransformPoint(new Vector3(.45f, info != null ? info.boundsMin[1] : -.04f, 0f)); return true;
            default:
                return false;
        }
    }

    // ================================================================== CostBatch (T11, Play mode)
    static readonly List<IEnumerator<int>> costStack = new List<IEnumerator<int>>();
    static int costWaitUntil = -1;
    static bool costFinished;
    static readonly StringBuilder costTsv = new StringBuilder();

    public static void CostBatch()
    {
        var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
        EditorSceneManager.SetActiveScene(scene);
        SessionState.SetBool(CostKey, true);
        SessionState.SetFloat(CostDeadlineKey, (float)EditorApplication.timeSinceStartup + 2400f);
        EditorApplication.update -= CostTick;
        EditorApplication.update += CostTick;
        EditorApplication.EnterPlaymode();
    }

    static void CostTick()
    {
        var timedOut = EditorApplication.timeSinceStartup > SessionState.GetFloat(CostDeadlineKey, float.MaxValue);
        if (costFinished || timedOut || SessionState.GetBool(CostDoneKey, false))
        {
            if (timedOut && !costFinished && !SessionState.GetBool(CostDoneKey, false)) { L("COST TIMEOUT"); CostFinish(); }
            if (EditorApplication.isPlaying) { EditorApplication.ExitPlaymode(); return; }
            if (EditorApplication.isPlayingOrWillChangePlaymode) return;
            EditorApplication.update -= CostTick;
            SessionState.EraseBool(CostKey);
            SessionState.EraseBool(CostDoneKey);
            if (Application.isBatchMode) EditorApplication.Exit(0);
            return;
        }
        if (!EditorApplication.isPlaying) return;
        if (costStack.Count == 0) { costStack.Add(CostMain()); L("cost routine created"); }
        if (Time.frameCount < costWaitUntil) return;
        try
        {
            var top = costStack[0];
            if (!top.MoveNext()) { CostFinish(); return; }
            costWaitUntil = Time.frameCount + Mathf.Max(1, top.Current);
        }
        catch (Exception e) { L("COST EXCEPTION " + e); CostFinish(); }
    }

    static void CostFinish()
    {
        costFinished = true;
        SessionState.SetBool(CostDoneKey, true);
        try
        {
            var dir = Dir("cost");
            File.WriteAllText(Path.Combine(dir, "cost.tsv"), costTsv.ToString());
            File.WriteAllText(Path.Combine(dir, "log.txt"), Log.ToString());
        }
        catch (Exception e) { Debug.LogError(e); }
    }

    sealed class Rec
    {
        public string name; public ProfilerRecorder r;
    }

    static List<Rec> StartRecorders()
    {
        var want = new[] { "Draw Calls Count", "SetPass Calls Count", "Batches Count", "Triangles Count", "Vertices Count", "Shadow Casters Count",
            "Main Thread", "Render Thread", "CPU Main Thread Frame Time", "CPU Render Thread Frame Time", "CPU Total Frame Time", "GPU Frame Time" };
        var avail = new List<ProfilerRecorderHandle>();
        ProfilerRecorderHandle.GetAvailable(avail);
        var list = new List<Rec>();
        var names = new StringBuilder();
        foreach (var h in avail)
        {
            var d = ProfilerRecorderHandle.GetDescription(h);
            foreach (var w in want)
                if (d.Name == w)
                {
                    var exists = false;
                    foreach (var x in list) if (x.name == w) exists = true;
                    if (exists) continue;
                    list.Add(new Rec { name = w, r = ProfilerRecorder.StartNew(d.Category, d.Name, 1) });
                    names.Append(w + " [" + d.Category.Name + "]; ");
                }
        }
        L("recorders: " + names);
        return list;
    }

    static IEnumerator<int> CostMain()
    {
        SetReduce(false);
        FrontRoomsLook.ApplyAmbient();
        FrontRoomsPostStack.Ensure(null);
        SetupReflections();
        var cam = MakeCamera("RUNB COST camera", 1920, 1080, 4, 72f);
        var recs = StartRecorders();
        var logs = new List<string>();
        Application.LogCallback cb = (msg, st, type) => { if (msg.IndexOf("shadow", StringComparison.OrdinalIgnoreCase) >= 0 || type == LogType.Error || type == LogType.Exception) logs.Add(type + ": " + msg); };
        Application.logMessageReceived += cb;
        costTsv.Append("condition\tframes\tdraws_med\tsetpass_med\tbatches_med\ttris_med\tverts_med\tshadowcasters_med\trender_call_ms_med\trender_call_ms_p95\tframe_ms_med\tframe_ms_p95\tframe_ms_max\tmain_thread_ms_med\tgpu_ms_med\tlights_visible\tvis_renderers\tvis_submeshes\tvis_tris\tpath\n");
        var z0 = (int)RoomRule.Run * FrontRoomsRoomStream.RoomLength;
        var holder = new GameObject("RUNB COST holder").transform;
        // ---- stream room: Ward, then B armed / tripped, same poses
        for (var pass = 0; pass < 2; pass++)
        {
            var look = pass == 0 ? RunLook.Ward : RunLook.EmergencyPower;
            var s = BuildStreamPreview(holder, RootWorld, look);
            yield return 10;
            var run = RunRoom(s);
            var rig = run != null ? run.GetComponentInChildren<FrontRoomsRunRig>(true) : null;
            if (rig != null) rig.SetViewer(cam.transform);
            foreach (var p in new[] { "fwd", "back", "lane" })
            {
                StreamPose(cam, s.transform, z0, p);
                if (rig != null) rig.Tick(0f);
                RefreshProbes();
                var e = Measure(cam, recs, (pass == 0 ? "stream_ward_" : "stream_b_armed_") + p, 300, null);
                while (e.MoveNext()) yield return e.Current;
            }
            if (rig != null)
            {
                // first second after the trip: per-frame times while the rig ticks (host lamps off at the mains drop)
                StreamPose(cam, s.transform, z0, "fwd");
                rig.Trip();
                var after = 0;
                var e = Measure(cam, recs, "stream_b_trip_first_second_fwd", 60, () => { rig.Tick(1f / 60f); after++; if (after == 4) HostLampsOff(run); });
                while (e.MoveNext()) yield return e.Current;
                for (var i = 0; i < 180; i++) rig.Tick(1f / 60f);
                HostLampsOff(run);
                RefreshProbes();
                foreach (var p in new[] { "fwd", "back", "lane" })
                {
                    StreamPose(cam, s.transform, z0, p);
                    rig.Tick(0f);
                    var e2 = Measure(cam, recs, "stream_b_tripped_" + p, 300, null);
                    while (e2.MoveNext()) yield return e2.Current;
                }
            }
            UnityEngine.Object.Destroy(s.gameObject);
            yield return 5;
        }
        // ---- corridor preset
        var c = BuildCorridor(false);
        c.rig.SetViewer(cam.transform);
        yield return 10;
        var shots = CorridorShots();
        foreach (var name in new[] { "s1a", "s1b", "c4_k_global" })
        {
            Shot sh = shots[0];
            foreach (var x in shots) if (x.name == name) sh = x;
            c.relay.gameObject.SetActive(sh.relay);
            c.relay.localScale = Vector3.one * (sh.relayScale > 0f ? sh.relayScale : 1f);
            Pose(cam, c.root, sh.eye, sh.look, sh.fov);
            Drive(c, sh.t);
            var e = Measure(cam, recs, "corridor_" + name, 300, null);
            while (e.MoveNext()) yield return e.Current;
        }
        {
            c.relay.gameObject.SetActive(false);
            Pose(cam, c.root, new Vector3(0f, 1.62f, 1.5f), new Vector3(0f, 1.3f, 27f), 72f);
            Drive(c, -1f);
            c.rig.Trip();
            var e = Measure(cam, recs, "corridor_trip_first_second_s1", 60, () => c.rig.Tick(1f / 60f));
            while (e.MoveNext()) yield return e.Current;
        }
        Application.logMessageReceived -= cb;
        foreach (var x in recs) x.r.Dispose();
        L("log lines with 'shadow' / errors during the cost run: " + logs.Count);
        foreach (var x in logs) L("  " + x);
        L("COST DONE");
        yield return 1;
    }

    static IEnumerator<int> Measure(Camera cam, List<Rec> recs, string condition, int frames, Action perFrame)
    {
        var vals = new Dictionary<string, List<double>>();
        foreach (var r in recs) vals[r.name] = new List<double>();
        var renderMs = new List<double>();
        var frameMs = new List<double>();
        var gpu = new List<double>();
        var timings = new FrameTiming[1];
        var sw = new System.Diagnostics.Stopwatch();
        // -runbCostLoop: the camera renders in the player loop (enabled) so the render counters see it; otherwise a
        // manual Render() per frame (timed; the counters stay 0 for manual renders, measured 14:32).
        var loop = Has("-runbCostLoop");
        cam.enabled = loop;
        for (var i = 0; i < 20; i++) { if (!loop) cam.Render(); yield return 1; }
        var last = Time.realtimeSinceStartupAsDouble;
        for (var i = 0; i < frames; i++)
        {
            if (perFrame != null) perFrame();
            sw.Reset(); sw.Start();
            if (!loop) cam.Render();
            sw.Stop();
            renderMs.Add(sw.Elapsed.TotalMilliseconds);
            yield return 1;
            var now = Time.realtimeSinceStartupAsDouble;
            frameMs.Add((now - last) * 1000.0);
            last = now;
            foreach (var r in recs) vals[r.name].Add(r.r.LastValue);
            FrameTimingManager.CaptureFrameTimings();
            if (FrameTimingManager.GetLatestTimings(1, timings) > 0) gpu.Add(timings[0].gpuFrameTime);
        }
        cam.enabled = false;
        // visible-set estimate (LOD-aware, camera + shadow visibility): renderers, submeshes (= draws before
        // batching, one pass), triangles
        cam.Render();
        int visR = 0, visSub = 0; long visTris = 0;
        foreach (var r in UnityEngine.Object.FindObjectsByType<Renderer>(FindObjectsSortMode.None))
        {
            if (!r.enabled || !r.gameObject.activeInHierarchy || !r.isVisible) continue;
            visR++;
            var mf = r.GetComponent<MeshFilter>();
            if (mf == null || mf.sharedMesh == null) continue;
            visSub += mf.sharedMesh.subMeshCount;
            for (var k = 0; k < mf.sharedMesh.subMeshCount; k++) visTris += mf.sharedMesh.GetIndexCount(k) / 3;
        }
        var lightsVisible = 0;
        foreach (var l in UnityEngine.Object.FindObjectsByType<Light>(FindObjectsSortMode.None))
            if (l.enabled && l.intensity > 0f && l.isActiveAndEnabled && (l.type == LightType.Directional || (l.transform.position - cam.transform.position).magnitude < l.range + 12f)) lightsVisible++;
        Func<string, double> med = n => { List<double> v; return vals.TryGetValue(n, out v) ? Pct(v, .5) : -1; };
        costTsv.Append(condition + "\t" + frames + "\t" + med("Draw Calls Count") + "\t" + med("SetPass Calls Count") + "\t" + med("Batches Count") + "\t" + med("Triangles Count") + "\t" + med("Vertices Count") + "\t" + med("Shadow Casters Count")
            + "\t" + Pct(renderMs, .5).ToString("0.00", IC) + "\t" + Pct(renderMs, .95).ToString("0.00", IC) + "\t" + Pct(frameMs, .5).ToString("0.00", IC) + "\t" + Pct(frameMs, .95).ToString("0.00", IC) + "\t" + Pct(frameMs, 1.0).ToString("0.00", IC)
            + "\t" + (med("Main Thread") > 0 ? (med("Main Thread") / 1e6).ToString("0.00", IC) : (med("CPU Main Thread Frame Time") > 0 ? (med("CPU Main Thread Frame Time") / 1e6).ToString("0.00", IC) : "-"))
            + "\t" + (gpu.Count > 0 ? Pct(gpu, .5).ToString("0.00", IC) : (med("GPU Frame Time") > 0 ? (med("GPU Frame Time") / 1e6).ToString("0.00", IC) : "-")) + "\t" + lightsVisible + "\t" + visR + "\t" + visSub + "\t" + visTris + "\t" + (loop ? "player loop" : "manual Render") + "\n");
        CostFinishPartial();
        L("cost " + condition + " done");
    }

    static void CostFinishPartial()
    {
        try { File.WriteAllText(Path.Combine(Dir("cost"), "cost.tsv"), costTsv.ToString()); } catch { }
    }

    static double Pct(List<double> v, double q)
    {
        if (v == null || v.Count == 0) return -1;
        var a = v.ToArray();
        Array.Sort(a);
        var i = (int)Math.Min(a.Length - 1, Math.Max(0, Math.Round(q * (a.Length - 1))));
        return a[i];
    }
