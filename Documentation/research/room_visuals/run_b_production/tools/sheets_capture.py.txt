def want(n):
    return not ONLY or n in ONLY


def img(p):
    return Image.open(p).convert("RGB")


def save(im, name, title, sentence, vl, status):
    out = S.frame(im, title=title, sentence=sentence, meta=[vl, DATE, "GAME VISUAL", "RUN B CAPTURE"], status=status)
    if out.width > 1920:
        out = out.resize((1920, round(out.height * 1920 / out.width)), Image.LANCZOS)
    p = S.save(out, os.path.join(OUT, name + ".jpg"), quality=88)
    print(p, out.size)


def fv(x, d=3):
    return ("%." + str(d) + "f") % x

# ---------------------------------------------------------------- T9: timelines
if want("t9") and "timeline" in M:
    T = M["timeline"]
    TD = os.path.join(CAP, "timeline")
    tiles = []
    for name in sorted(T):
        if not isinstance(T[name], dict) or "frames" not in T[name]:
            continue
        t = T[name]
        curve = t["mean_curve"]
        W_, H_ = 960, 210
        ch = Image.new("RGB", (W_, H_), (244, 241, 232))
        d = ImageDraw.Draw(ch)
        mx = max(max(curve), 1e-4)
        n = len(curve)
        for sec in range(0, int(n / 60) + 1):                     # 1 s ticks
            x = 20 + sec * 60 * (W_ - 40) / max(1, n - 1)
            d.line([(x, H_ - 20), (x, H_ - 12)], fill=(120, 118, 110), width=2)
        pts = [(20 + i * (W_ - 40) / max(1, n - 1), H_ - 20 - (H_ - 40) * v / mx) for i, v in enumerate(curve)]
        d.line([(20, H_ - 20), (W_ - 20, H_ - 20)], fill=(189, 186, 176), width=2)
        d.line(pts, fill=(10, 10, 10), width=2)
        # filmstrip: 6 frames
        frames = sorted(f for f in os.listdir(os.path.join(TD, name)) if f.endswith(".jpg") and f.startswith("f"))
        pick = [frames[int(i * (len(frames) - 1) / 5)] for i in range(6)] if frames else []
        strip = Image.new("RGB", (W_, 90), (10, 10, 10))
        for k, f in enumerate(pick):
            fi = img(os.path.join(TD, name, f)).resize((160, 90))
            strip.paste(fi, (k * 160, 0))
        tile = Image.new("RGB", (W_, H_ + 90), (10, 10, 10))
        tile.paste(ch, (0, 0)); tile.paste(strip, (0, H_))
        g_ = t["general"]["worst_flashes_per_s"]; r_ = t["red"]["worst_flashes_per_s"]
        extra = ""
        if name.endswith("_rf1") and "trip" in name:
            extra = "  black frames %d" % t["black_frames_lt2pct_of_pre"]
        tiles.append((tile, name.replace("_", " ").upper(), "%s  general %d/s  red %d/s  max Y %.3f%s" % ("PASS" if t["PASS"] else "FAIL", g_, r_, max(curve), extra)))
