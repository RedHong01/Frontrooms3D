run() { tag=$1; shots=$2; shift 2; mkdir -p $R/$tag; S=$(date +%s); echo "$tag start $(date)" >> $R/status_C.txt

  echo "$tag exit $ec $(( $(date +%s) - S )) s errors=$(grep -c 'error CS' $R/$tag.log) png=$(ls $R/$tag/*.png 2>/dev/null | wc -l)" >> $R/status_C.txt; }
run C_armed s1a,s2
run C_tripped s1b,x2
run C_relay c4,c4b,c4_k,c4b_k
