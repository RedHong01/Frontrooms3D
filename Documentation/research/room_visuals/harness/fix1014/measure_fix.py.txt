"""Room-visuals fix stage (2026-10-14): measure the 10-08 fix renders.

H1  floor reflections: probe-on vs probe-off of the same tripped shot (A and C), floor ROI.
M6  Relay readability: the brief's proxy (1.84 m top) vs the contract-size stand-in (x 1.114, 2.05 m top)
    in the same run, each direction's own mask source and a 12 px ring (linear Rec. 709 Y, hi/lo).
usage: python measure_fix.py <runs dir> <out json>
"""
import json
import os
import sys

import numpy as np
from PIL import Image, ImageFilter

R, OUT = sys.argv[1], sys.argv[2]


def load(p):
    return np.asarray(Image.open(p).convert("RGB")).astype(np.float32) / 255.0


def lin(c):
    return np.where(c <= 0.04045, c / 12.92, ((c + 0.055) / 1.055) ** 2.4)


def enc(c):
    c = np.clip(c, 0, 1)
    return np.where(c <= 0.0031308, c * 12.92, 1.055 * np.power(c, 1 / 2.4) - 0.055)


def Y(img):
    return lin(img) @ np.array([.2126, .7152, .0722], np.float32)


PROT = np.array([[0.152286, 1.052583, -0.204868], [0.114503, 0.786281, 0.099216], [-0.003882, -0.048116, 1.051998]], np.float32)


def protan(img):
    return enc(np.clip(lin(img) @ PROT.T, 0, 1))


def ring(mask, px=12):
    m = Image.fromarray((mask * 255).astype(np.uint8))
    dil = np.asarray(m.filter(ImageFilter.MaxFilter(2 * px + 1))) > 127
    return dil & ~mask


def ratio(a, b):
    hi, lo = max(a, b), min(a, b)
    return round(hi / max(lo, 1e-6), 2)


def block(a, k=8):
    h, w = a.shape[:2]
    a = a[: h - h % k, : w - w % k]
    return a.reshape(h // k, k, w // k, k).mean((1, 3))


res = {"H1_probe_effect": {}, "M6_relay": {}}

# ---------------- H1: probes on vs off (tripped S1b), A and C
for key, on, off in (("A_red_ward", "A_tripped/s1b.png", "A_noprobe/s1b.png"), ("C_house_lights", "C_tripped/s1b.png", "C_noprobe/s1b.png")):
    a, b = load(os.path.join(R, on)), load(os.path.join(R, off))
    ya, yb = Y(a), Y(b)
    h, w = ya.shape
    roi = (slice(int(h * .70), h), slice(int(w * .30), int(w * .70)))   # the floor in front of the camera
    lev = np.abs(a - b).max(-1) * 255
    ba, bb = block(ya), block(yb)                                         # 8 x 8 means: grain-proof
    rel = np.abs(ba - bb) / np.maximum(bb, 1e-3)
    fa, fb = ya[roi], yb[roi]
    res["H1_probe_effect"][key] = dict(
        frame_px_changed_ge3_levels_pct=round(float((lev >= 3).mean() * 100), 2),
        frame_blocks_changed_gt5pct=round(float((rel > .05).mean() * 100), 2),
        max_level_change=int(lev.max()),
        floor_roi_meanY_probes=round(float(fa.mean()), 4), floor_roi_meanY_noprobes=round(float(fb.mean()), 4),
        floor_roi_gain_pct=round(float((fa.mean() / max(fb.mean(), 1e-6) - 1) * 100), 1),
        floor_roi_blocks_changed_gt5pct=round(float((rel[int(rel.shape[0] * .70):, int(rel.shape[1] * .30):int(rel.shape[1] * .70)] > .05).mean() * 100), 1),
        floor_roi_p99_probes=round(float(np.percentile(fa, 99)), 4), floor_roi_p99_noprobes=round(float(np.percentile(fb, 99)), 4))

# ---------------- M6: A (ID render), Relay body+head vs 12 px ring
A_ID = {"relay_body": (1, .5, 0), "relay_head": (.5, 0, 1)}


def a_masks(p):
    im = load(p)
    q = np.where(im < .25, 0.0, np.where(im < .85, .5, 1.0))
    return {k: np.all(np.abs(q - np.array(c)) < .01, axis=-1) for k, c in A_ID.items()}


for shot in ("c4", "c4_k"):
    img = load(os.path.join(R, "A_tripped", shot + ".png"))
    m = a_masks(os.path.join(R, "A_tripped", shot + "_id.png"))
    whole = m["relay_body"] | m["relay_head"]
    if whole.sum() == 0:
        res["M6_relay"]["A_" + shot] = dict(px=0)
        continue
    y, yp = Y(img), Y(protan(img))
    rg = ring(whole)
    ys, xs = np.nonzero(whole)
    res["M6_relay"]["A_" + shot] = dict(px=int(whole.sum()), bbox_h=int(ys.max() - ys.min() + 1),
                                        bodyY=round(float(y[m["relay_body"]].mean()), 4), ringY=round(float(y[rg].mean()), 4),
                                        ratio=ratio(float(y[m["relay_body"]].mean()), float(y[rg].mean())),
                                        ratio_protan=ratio(float(yp[m["relay_body"]].mean()), float(yp[rg].mean())))

# ---------------- M6: B (ID palette, Relay = white)
PAL = {"Background": (0, 0, 0), "Relay": (1, 1, 1), "Goal": (1, 0, 0), "Dolly": (0, 1, 0), "Cart": (0, 0, 1),
       "Truck": (1, 1, 0), "Units": (1, 0, 1), "Main": (0, 1, 1), "Emitter": (.5, .5, .5), "Door": (1, .5, 0), "Sign": (.5, 0, 1)}


def b_relay(p):
    im = load(p)
    keys = list(PAL.keys())
    cols = np.array([enc(np.array(PAL[k], np.float32)) for k in keys], np.float32)
    idx = ((im[:, :, None, :] - cols[None, None, :, :]) ** 2).sum(-1).argmin(-1)
    return idx == keys.index("Relay")


for shot in ("c4", "c4_k"):
    img = load(os.path.join(R, "B_tripped", shot + ".png"))
    rel = b_relay(os.path.join(R, "B_tripped", shot + "_id.png"))
    y, yp = Y(img), Y(protan(img))
    rg = ring(rel)
    ys, xs = np.nonzero(rel)
    res["M6_relay"]["B_" + shot] = dict(px=int(rel.sum()), bbox_h=int(ys.max() - ys.min() + 1),
                                        relayY=round(float(y[rel].mean()), 4), ringY=round(float(y[rg].mean()), 4),
                                        ratio=ratio(float(y[rel].mean()), float(y[rg].mean())),
                                        ratio_protan=ratio(float(yp[rel].mean()), float(yp[rg].mean())))


# ---------------- M6: C (mask PNGs); on axis (c4) and 0.65 m off axis (c4b); ring kept inside the lites
def c_mask(p):
    if not os.path.exists(p):
        return None
    return np.asarray(Image.open(p).convert("L")) > 60


for shot in ("c4", "c4_k", "c4b", "c4b_k"):
    base = os.path.join(R, "C_relay", shot)
    img = load(base + ".png")
    rel, body, lites = c_mask(base + "_mask_relay.png"), c_mask(base + "_mask_relayBody.png"), c_mask(base + "_mask_lites.png")
    y, yp = Y(img), Y(protan(img))
    d = dict(relay_px=int(rel.sum()) if rel is not None else 0)
    if rel is not None and rel.sum() > 0:
        rg = ring(rel)
        d.update(relayY=round(float(y[rel].mean()), 4), ringY=round(float(y[rg].mean()), 4), ratio_plain=ratio(float(y[rel].mean()), float(y[rg].mean())))
    if body is not None and body.sum() > 0 and lites is not None:
        rgl = ring(body) & lites & ~body
        vis = body & lites
        d.update(body_in_lites_px=int(vis.sum()), ring_in_lites_px=int(rgl.sum()))
        if rgl.sum() > 0:
            d.update(bodyY=round(float(y[body].mean()), 4), ring_in_liteY=round(float(y[rgl].mean()), 4),
                     ratio_in_lites=ratio(float(y[body].mean()), float(y[rgl].mean())),
                     ratio_in_lites_protan=ratio(float(yp[body].mean()), float(yp[rgl].mean())))
    res["M6_relay"]["C_" + shot] = d

json.dump(res, open(OUT, "w"), indent=1)
print(json.dumps(res, indent=1))
