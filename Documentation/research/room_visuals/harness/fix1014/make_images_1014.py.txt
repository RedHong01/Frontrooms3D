def font(px):

def jpg(src, name, crop=None):
    im = Image.open(src).convert("RGB")
    if crop:
        im = im.crop(crop)
    if im.width > 1920:
        im = im.resize((1920, round(im.height * 1920 / im.width)), Image.LANCZOS)
    im.save(os.path.join(DOC, name + ".jpg"), "JPEG", quality=85, optimize=True)
    return name

# 3. verification composites (labels in IBM Plex Mono, the deck's mono face)
def label(dr, xy, text, size=26, dark=True):
    f = font(size)
    x, y = xy
    w = dr.textlength(text, font=f)
    dr.rectangle([x, y, x + w + 20, y + size + 14], fill=(10, 10, 10) if dark else (244, 223, 59))
    dr.text((x + 10, y + 5), text, font=f, fill=(255, 255, 255) if dark else (10, 10, 10))


def lin(c):
    return np.where(c <= 0.04045, c / 12.92, ((c + 0.055) / 1.055) ** 2.4)


def diff_img(a, b, gain=8):
    A = lin(np.asarray(Image.open(a).convert("RGB")).astype(np.float32) / 255)
    B = lin(np.asarray(Image.open(b).convert("RGB")).astype(np.float32) / 255)
    d = np.abs(A - B).max(-1) * gain
    d = np.clip(d, 0, 1) ** (1 / 2.2) * 255
    return Image.fromarray(d.astype(np.uint8)).convert("RGB")
