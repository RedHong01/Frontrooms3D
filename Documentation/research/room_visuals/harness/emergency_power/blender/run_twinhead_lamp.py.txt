"""Lamp head of the twin-head emergency unit (see run_twinhead.py): a 4.5 in
(PAR-36 class) 6 V sealed-beam lamp in a spun almond housing with a chrome
retaining bezel. CLONE-ONLY look-dev asset, Run! direction B.

Origin = the trunnion axis (the head's pivot in the yoke), at the housing
centre. The lens faces -Y in Blender (+Z in Unity): the rig aims the head by
rotating this asset, and puts the spot light 15 mm in front of the lens.
The lens is a domed disc with decal UVs (the fluted-lens normal map and the
lit filament/reflector emission map, Run_SealedBeamLens_*).
Slots: Prop_SteelAlmond, Prop_Chrome, Run_SealedBeamLens.
"""

import kitlib

NAME = "Kit_EmergencyTwinHead_Lamp"
SMOOTH_ANGLE = 50.0

STEEL = "Prop_SteelAlmond"
CHROME = "Prop_Chrome"
LENS = "Run_SealedBeamLens"
kitlib.register_slot(LENS, (0.62, 0.62, 0.60), 0.1)

R = 0.0575


def build(kit):
    # Housing: a spun cup, rounded back, a rolled lip under the bezel.
    housing = [(0.0, -0.052), (0.020, -0.0505), (0.034, -0.0460), (0.045, -0.0380), (0.052, -0.0270),
               (0.0560, -0.0130), (R, -0.0020), (R, 0.0200), (0.0570, 0.0215), (R, 0.0230),
               (R, 0.0340), (0.0585, 0.0360), (0.0585, 0.0390)]
    kit.lathe(housing, (0, 0, 0), STEEL, verts=48, rot=(90, 0, 0), name="housing")
    # Chrome retaining bezel.
    bezel = [(0.0500, 0.0385), (0.0588, 0.0383), (0.0601, 0.0410), (0.0590, 0.0458), (0.0545, 0.0476), (0.0503, 0.0468)]
    kit.lathe(bezel, (0, 0, 0), CHROME, verts=48, rot=(90, 0, 0), close_top=False, close_bottom=False, name="bezel")
    # Domed fluted lens (decal UVs across the disc).
    lens_profile = [(0.0512, 0.0418), (0.0500, 0.0442), (0.0450, 0.0458), (0.0350, 0.0473), (0.0220, 0.0483), (0.0, 0.0487)]
    lens = kit.lathe(lens_profile, (0, 0, 0), LENS, verts=48, rot=(90, 0, 0), uv="decal", close_top=False, name="lens")
    lens["fr_decal_axes"] = "xy"
    # Trunnion pins into the yoke arms.
    for sx in (-1, 1):
        kit.cylinder(0.0055, 0.010, (sx * (R + 0.003), 0, 0), STEEL, verts=16, rot=(0, 90, 0), bevel=0.0008, segments=1, name="trunnion pin")
    # Cord gland at the back, low.
    kit.cylinder(0.0060, 0.012, (0, 0.050, -0.018), STEEL, verts=16, rot=(90, 0, 0), bevel=0.001, segments=1, name="cord gland")
    kit.no_collider()
    kit.anchor("lens", (0, -0.049, 0))
    kit.tag("run")
