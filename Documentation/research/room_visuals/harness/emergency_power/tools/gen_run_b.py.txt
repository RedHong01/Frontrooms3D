def resize(arr, w, h):
    im = Image.fromarray(arr.astype(np.float32), "F").resize((w, h), Image.BILINEAR)
    return np.asarray(im).astype(np.float32)

def mask(smooth, cavity, wet=None):
    z = np.zeros_like(smooth)
    return np.stack([np.clip(smooth, 0, 1), np.clip(cavity, 0, 1), z if wet is None else wet, np.ones_like(smooth)], -1)

def block():
    N = 1024                      # 0.4 m -> 0.39 mm / px
    yy, xx = np.mgrid[0:N, 0:N].astype(np.float32)
    v = 1 - (yy + .5) / N         # texture v (0 at the bottom row, = floor line)
    u = (xx + .5) / N
    course = np.floor(v * 2)      # two 200 mm courses per tile
    uo = (u + .5 * course) % 1.0  # running bond: odd course shifted half a block
    jw = 10 / 400                 # 10 mm joint in tile units (u and v are both 0.4 m)
    # distance to the nearest joint centre line (bed joints at v = 0, .5; head joints at uo = 0)
    dv = np.minimum(np.abs((v * 2) % 1.0), 1 - np.abs((v * 2) % 1.0)) * .5
    du = np.minimum(uo, 1 - uo)
    d = np.minimum(dv, du)
    hw = jw * .5
    joint = np.clip(1 - d / hw, 0, 1)
    concave = np.where(d < hw, -(1 - (d / hw) ** 2), 0.0)       # tooled: a smooth hollow
    edge = np.clip(1 - (d - hw) / (1.2 / 400), 0, 1) * (d >= hw)  # arris roll-over 1.2 mm
    bid = (np.floor(uo * 1) + 7 * course).astype(int)
    # block face under paint: open-texture CMU, pits filled by two coats but still there
    pores = band(N, N, 160, 520, 201)
    pits = np.clip((pores - .70) / .12, 0, 1)
    pits = blur(pits, .8)
    grain = band(N, N, 40, 140, 202)
    r = np.random.default_rng(203)
    tone = r.normal(0, .012, 64)[bid % 64]
    height = .55 * concave - .35 * edge * 0 + (-.30 * pits + .10 * grain) * (1 - joint)
    alb = (0.955 + tone - .045 * pits - .02 * (grain - .5)) * (1 - .06 * joint)
    # paint drips/runs: rare, faint, vertical
    drip = np.zeros((N, N), np.float32)
    for _ in range(6):
        x0 = int(r.integers(0, N)); y0 = int(r.integers(0, N)); L = int(r.integers(30, 120))
        for k in range(L):
            drip[(y0 + k) % N, (x0 + int(2 * np.sin(k * .05))) % N] = 1 - k / L
    drip = blur(drip, 1.1)
    height += .25 * drip
    col = np.repeat(np.clip(alb, 0, 1)[..., None], 3, -1)
    col *= np.array([1.0, .995, .985])
    smooth = .66 - .20 * pits - .10 * joint - .05 * (grain - .5)
    cav = 1 - .45 * joint - .25 * pits
    write("Run_Block", col, normal_from_height(blur(height, .7), 2.2), mask(smooth, cav))

def service_vct():
    W, H = 2048, 1024             # 3.0 x 1.5 m -> 1.46 mm / px
    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
    x = (xx + .5) / W * 3.0 - 1.5                 # corridor-local metres, axis at x = 0
    z = (1 - (yy + .5) / H) * 1.5
    t = .3
    gx = (x + .15) / t; gz = z / t
    col_id = np.floor(gx).astype(int); row_id = np.floor(gz).astype(int)
    tid = (col_id + 20) * 7 + row_id
    r = np.random.default_rng(301)
    tone = r.normal(0, .022, 4096)[tid % 4096]
    accent = (np.abs(np.abs(x) - .6) < .15)
    base = np.where(accent[..., None], hexc("#7E7A70")[None, None, :], hexc("#B9B4A6")[None, None, :])
    # VCT through-chip: directional flecks (laid in quarter turns tile by tile)
    chipsA = resize(band(H * 4, W, 120, 500, 302), W, H)
    chipsB = resize(band(H, W * 4, 120, 500, 303), W, H)
    turn = ((col_id + row_id) % 2 == 0)
    chips = np.where(turn, chipsA, chipsB)
    darkf = np.clip((chips - .60) / .2, 0, 1)
    lightf = np.clip((.30 - chips) / .2, 0, 1)
    col = base * (1 + tone[..., None])
    col = col * (1 - .20 * darkf[..., None]) + .05 * lightf[..., None]
    fx = np.minimum(gx % 1, 1 - gx % 1) * t; fz = np.minimum(gz % 1, 1 - gz % 1) * t
    seam = np.exp(-(np.minimum(fx, fz) / .0011) ** 2)
    col *= (1 - .22 * seam[..., None])
    # Wax: a burnished traffic lane on the axis, yellowed build-up toward the walls.
    lane = np.exp(-(x / .55) ** 4)
    build = np.clip((np.abs(x) - .85) / .45, 0, 1)
    col = col * (1 - .10 * build[..., None]) * (1 - build[..., None] * np.array([0, .03, .09]))
    wear = spectral(H, W, 1.9, 304)
    # Heel scuffs: mostly in the lane; cart-wheel tracks at x = +-0.27.
    img = Image.new("L", (W, H), 0); dr = ImageDraw.Draw(img)
    for _ in range(260):
        xs = r.normal(0, .45); zs = r.random() * 1.5
        px, py = (xs + 1.5) / 3.0 * W, (1 - zs / 1.5) * H
        a = r.normal(0, .5) + (np.pi / 2 if r.random() < .5 else 0); l = r.random() * 35 + 8
        for oy in (-H, 0, H):
            dr.line([(px, py + oy), (px + np.cos(a) * l, py + oy + np.sin(a) * l)], fill=int(150 + r.integers(0, 105)), width=int(r.integers(1, 4)))
    sc = blur(np.asarray(img).astype(np.float32) / 255, 1.0)
    track = (np.exp(-((np.abs(x) - .27) / .012) ** 2) * (0.6 + .4 * band(H, W, 2, 12, 305)))
    col *= (1 - .32 * sc[..., None]) * (1 - .10 * track[..., None])
    smooth = .46 + .26 * lane - .14 * wear - .20 * sc - .25 * seam + .08 * build
    height = -.9 * seam + .08 * band(H, W, 300, 900, 306)
    write("Run_ServiceVCT", np.clip(col, 0, 1), normal_from_height(blur(height, .6), 1.4), mask(smooth, 1 - .35 * seam))
    # 2026-10-07 (r3): Red's decision 2026-10-03 -- the Run! floor is GLOSSY waxed 12 in VCT.
    # Waxed field ~.80, burnished lane ~.85, hazier yellowed build-up at the walls ~.75; the
    # seams hold dirt and stay matte (~.38), so the tile grid still reads in the reflections;
    # heel scuffs and wear cut through the wax.
    gloss = .80 + .05 * lane - .07 * wear - .22 * sc - .42 * seam - .05 * build
    save_rgba(f"{OUT}/Run_ServiceVCTGloss_S.png", mask(np.clip(gloss, .30, .90), 1 - .35 * seam))
    print("wrote Run_ServiceVCTGloss_S")

def soffit():
    N = 1024                      # 2.4 m -> 2.3 mm / px
    yy, xx = np.mgrid[0:N, 0:N].astype(np.float32)
    u = (xx + .5) / N * 2.4; v = (1 - (yy + .5) / N) * 2.4
    mott = spectral(N, N, 2.2, 401)
    fine = band(N, N, 200, 600, 402)
    # plywood form seams every 1.2 x 2.4 m and form-tie cones on a 0.6 m grid
    su = np.minimum(u % 1.2, 1.2 - u % 1.2); sv = np.minimum(v % 2.4, 2.4 - v % 2.4)
    seam = np.exp(-(np.minimum(su, sv) / .004) ** 2)
    tu = np.minimum(u % .6, .6 - u % .6) - 0; tv = np.minimum((v + .3) % .6, .6 - (v + .3) % .6)
    tie = np.exp(-((tu ** 2 + tv ** 2) / .012 ** 2))
    alb = .93 + .07 * (mott - .5) - .05 * seam - .25 * tie
    col = np.repeat(np.clip(alb, 0, 1)[..., None], 3, -1)
    height = -.6 * tie + .3 * seam + .15 * fine + .2 * mott
    write("Run_SoffitPaint", col, normal_from_height(blur(height, .8), 1.2), mask(.22 + .05 * fine - .1 * tie, 1 - .5 * tie - .2 * seam))

def cap_font(path, cap_px):
    """Font size whose capital H is cap_px tall."""
    size = cap_px
    for _ in range(6):
        f = ImageFont.truetype(path, int(size))
        bb = f.getbbox("H")
        h = bb[3] - bb[1]
        size *= cap_px / max(1, h)
    return ImageFont.truetype(path, int(round(size)))


def text_centered(dr, cx, cy, text, font, fill):
    bb = dr.textbbox((0, 0), text, font=font)
    dr.text((cx - (bb[0] + bb[2]) / 2, cy - (bb[1] + bb[3]) / 2), text, font=font, fill=fill)

    a = np.asarray(img).astype(np.float32) / 255
    letters = (a[..., 0] > .5) & (a[..., 1] < .35)
    # stamped stencil face: the letters are the red diffuser, slightly translucent
    emit = np.zeros_like(a)
    emit[letters] = (1.0, .13, .07)
    halo = blur(emit, 3.0) * .35

    # a 3 mm darker rim where the face meets the housing lip

    m = int(3 * pxmm)
    rim[:m, :] = rim[-m:, :] = 1; rim[:, :m] = rim[:, -m:] = 1

    write("Run_ExitSignB", a, None, None, np.clip(emit, 0, 1))

def sealed_beam():
    N = 512
    yy, xx = np.mgrid[0:N, 0:N].astype(np.float32)
    x = (xx + .5) / N * 2 - 1; y = 1 - (yy + .5) / N * 2
    rr = np.sqrt(x * x + y * y)
    # flute blocks: 14 columns x 10 rows of small cylindrical lenslets (spread horizontally)
    bx = (x + 1) * 7; by = (y + 1) * 5
    fu = bx % 1; fv = by % 1
    lens_h = np.sqrt(np.clip(1 - (2 * fu - 1) ** 2, 0, 1)) * (1 - .4 * (2 * fv - 1) ** 8)
    edge = np.exp(-(np.minimum(np.minimum(fu, 1 - fu), np.minimum(fv, 1 - fv)) / .05) ** 2)
    inside = rr < .985
    height = (lens_h - .3 * edge) * inside
    # off: the silvered parabola through the glass
    refl = .52 + .22 * np.cos(rr * 9.0) * np.exp(-rr * 1.2) + .10 * (1 - rr)
    alb = (refl * (0.92 + .08 * lens_h) - .08 * edge) * inside + .25 * (~inside)
    col = np.stack([alb * 1.0, alb * .98, alb * .93], -1)
    # on: filament image (hot bar) + broad glow, fluted
    core = np.exp(-((x / .16) ** 2 + (y / .10) ** 2))
    glow = np.exp(-(rr / .62) ** 2)
    ring = np.exp(-((rr - .78) / .05) ** 2) * .25
    e = (1.0 * core + .55 * glow + ring) * (0.80 + .25 * lens_h) * inside
    emit = np.stack([e, e * .86, e * .66], -1)
    write("Run_SealedBeamLens", np.clip(col, 0, 1), normal_from_height(blur(height, .8), 3.0), mask(.88 - .2 * edge, np.ones_like(rr)), np.clip(emit / max(1e-6, e.max()), 0, 1))

    # cookie. URP decodes the cookie atlas as sRGB (measured 2026-10-03: a stored 0.37
    # lit the floor at 0.11 of the no-cookie level), so the wanted LINEAR profile is
    # authored here and stored sRGB-encoded. Profile of a PAR-36 sealed beam: a hot
    # filament core, a broad even plateau, one faint brighter ring where the reflector
    # and the lens flutes meet, a soft rim, a slight horizontal flute ripple.
    def beam(rr, x):
        core = np.exp(-(rr / .22) ** 2)
        plateau = np.clip((.97 - rr) / .30, 0, 1) ** 1.2
        ring = np.exp(-((rr - .70) / .05) ** 2)
        flutes = 1 + .06 * np.sin(x * np.pi * 14) * np.clip(1 - rr, 0, 1)
        return (0.45 * core + .62 * plateau + .10 * ring) * flutes

    def store(lin, name):
        lin = blur(np.clip(lin, 0, 1), 1.0)
        enc = np.where(lin <= .0031308, lin * 12.92, 1.055 * np.power(np.clip(lin, 0, 1), 1 / 2.4) - .055)
        save_rgb(f"{OUT}/{name}.png", np.repeat(np.clip(enc, 0, 1)[..., None], 3, -1))
        print("wrote", name)

    C = 256
    yy, xx = np.mgrid[0:C, 0:C].astype(np.float32)
    x = (xx + .5) / C * 2 - 1; y = (yy + .5) / C * 2 - 1
    # (1) Core only, for a spot whose outer angle IS the beam (50 deg flood, or the goal
    #     head's 30 deg spot lamp): the profile fills the cookie.
    rr = np.sqrt(x * x + (y * 1.08) ** 2)                 # slightly wide (flutes spread sideways)
    lin = beam(rr, x)
    store(np.clip(lin / lin.max(), 0, 1) * (rr < .995), "Cookie_SealedBeamCore_S")
    # (2) Beam + spill, for a 100 deg spot (inner 96): the same 50 deg beam in the middle of
    #     the cookie (cookie radius = tan(theta) / tan(50 deg)), then the lamp's stray light
    #     past the beam edge: the lens flutes and the reflector lip throw ~6 % of the peak
    #     out to ~45 deg off axis (a sealed beam's "field" beyond its 10 % beam angle).
    #     It lights the deck, the main and the wall around each unit, so the heads sit in a
    #     dim glow instead of floating in black.
    rc = np.sqrt(x * x + (y * 1.0) ** 2)
    theta = np.degrees(np.arctan(rc * np.tan(np.radians(50.0))))
    rr_old = np.tan(np.radians(theta)) / np.tan(np.radians(25.0))
    xs = x * np.tan(np.radians(50.0)) / np.tan(np.radians(25.0))
    core = beam(np.sqrt((rr_old * np.cos(np.arctan2(y, x))) ** 2 + (rr_old * np.sin(np.arctan2(y, x)) * 1.08) ** 2), xs)
    core = np.where(rr_old < .995, core, 0) / beam(np.zeros(1), np.zeros(1))[0]
    spill = SPILL * (1 - np.clip((theta - 24.0) / 24.0, 0, 1)) ** 1.5 * (1 + .25 * np.exp(-((theta - 31.0) / 3.0) ** 2))
    lin = np.maximum(np.clip(core, 0, None), spill) * (rc < .995)
    store(np.clip(lin / lin.max(), 0, 1), "Cookie_SealedBeam_S")


def wrap_lens():
    W, H = 1024, 512
    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
    u = (xx + .5) / W; v = 1 - (yy + .5) / H
    top = v >= .5
    lv = np.where(top, (v - .5) * 2, v * 2)        # 0..1 within each half
    # prism field: small pyramids on a 4 mm pitch (u spans 1.2 m, lv spans 0.19 / 0.066 m)
    pu = (u * 1200 / 4.0) % 1; pv = (lv * np.where(top, 190, 66) / 4.0) % 1
    pyr = 1 - np.maximum(np.abs(2 * pu - 1), np.abs(2 * pv - 1))
    ends = np.clip(np.minimum(u, 1 - u) / .03, 0, 1)
    # two T12 tubes 90 mm apart seen through the bottom panel (bands across lv)
    bands = np.exp(-((lv - .27) / .10) ** 2) + np.exp(-((lv - .73) / .10) ** 2)
    e_top = (.55 + .45 * bands) * (0.9 + .1 * pyr) * ends
    e_side = (.42 + .25 * np.exp(-((lv - .85) / .25) ** 2)) * (0.9 + .1 * pyr) * ends
    e = np.where(top, e_top, e_side)
    emit = np.stack([e * .92, e * .96, e], -1)
    alb = (.86 + .06 * pyr) * np.where(top, 1.0, .95)
    col = np.stack([alb, alb, alb * .98], -1)
    write("Run_WrapLens", np.clip(col, 0, 1), normal_from_height(pyr * 1.0, 1.2), None, np.clip(emit, 0, 1))


def wire_grid():
    N = 512                       # 100 mm tile, 12.5 mm squares, 0.6 mm wire
    yy, xx = np.mgrid[0:N, 0:N].astype(np.float32)
    p = N / 8.0
    d = np.minimum(np.minimum(xx % p, p - xx % p), np.minimum(yy % p, p - yy % p))
    al = np.clip(1.6 - d / (0.3 / 100 * N), 0, 1)
    rgba = np.zeros((N, N, 4), np.float32)
    rgba[..., :3] = hexc("#6E6E6A"); rgba[..., 3] = al
    Image.fromarray((rgba * 255 + .5).astype(np.uint8), "RGBA").save(f"{OUT}/Run_WireGrid_A.png")
    print("wrote Run_WireGrid")
