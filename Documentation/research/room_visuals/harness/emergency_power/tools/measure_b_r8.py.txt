"""Run! direction B (EMERGENCY POWER) measurements + derived frames + JPG export.
Usage: /usr/bin/python3 measure_b.py <render dir> <out images dir> <prefix> <json out>
Brief: Documentation/research/room_visuals/10_run_directions.md §7.4-7.7.
"""
import sys, os, json, math
import numpy as np
from PIL import Image, ImageDraw, ImageFilter, ImageFont

SRC, OUT, PREFIX, JOUT = sys.argv[1:5]
os.makedirs(OUT, exist_ok=True)

def load(name):
    return np.asarray(Image.open(os.path.join(SRC, name + ".png")).convert("RGB")).astype(np.float32) / 255.0

def lin(c):
    return np.where(c <= 0.04045, c / 12.92, ((c + 0.055) / 1.055) ** 2.4)

def enc(c):
    c = np.clip(c, 0, 1)
    return np.where(c <= 0.0031308, c * 12.92, 1.055 * np.power(c, 1 / 2.4) - 0.055)

def Y(img):
    return lin(img) @ np.array([.2126, .7152, .0722], np.float32)

def stats(img):
    y = Y(img)
    p = np.percentile(y, [5, 50, 95, 99.5])
    im = img * 255
    s = im.sum(-1) + 1e-6
    redfrac = im[..., 0] / s
    return dict(Y_p5=round(float(p[0]), 4), Y_p50=round(float(p[1]), 4), Y_p95=round(float(p[2]), 4), Y_p995=round(float(p[3]), 4),
                contrast_p95_p5=round(float((p[2] + .05) / (p[0] + .05)), 2),
                sat_red_pct=round(float(((redfrac >= 0.8) & (s > 30)).mean() * 100), 2),
                red_dominant_pct=round(float(((redfrac > 0.6) & (s > 30)).mean() * 100), 2))

# ID palette (sRGB-encoded unlit colours from FrontRoomsRunLookdev.IdColour)
PAL = {
    "Background": (0, 0, 0), "Relay": (1, 1, 1), "Goal": (1, 0, 0), "Dolly": (0, 1, 0), "Cart": (0, 0, 1),
    "Truck": (1, 1, 0), "Units": (1, 0, 1), "Main": (0, 1, 1), "Emitter": (.5, .5, .5), "Door": (1, .5, 0), "Sign": (.5, 0, 1),
}
def idmap(name):
    im = load(name + "_id")
    keys = list(PAL.keys())
    cols = np.array([enc(np.array(PAL[k], np.float32)) for k in keys], np.float32)
    d = ((im[:, :, None, :] - cols[None, None, :, :]) ** 2).sum(-1)
    idx = d.argmin(-1)
    return {k: idx == i for i, k in enumerate(keys)}

def ring(mask, px=12):
    m = Image.fromarray((mask * 255).astype(np.uint8))
    dil = np.asarray(m.filter(ImageFilter.MaxFilter(2 * px + 1))) > 127
    return dil & ~mask

def ratio(a, b):
    hi, lo = max(a, b), min(a, b)
    return round(hi / max(lo, 1e-6), 2)

def protan(img):
    M = np.array([[0.152286, 1.052583, -0.204868], [0.114503, 0.786281, 0.099216], [-0.003882, -0.048116, 1.051998]], np.float32)
    return enc(np.clip(lin(img) @ M.T, 0, 1))

def grey(img):
    y = Y(img)
    return np.repeat(enc(y)[..., None], 3, -1)

def save_jpg(arr, name):
    im = Image.fromarray((np.clip(arr, 0, 1) * 255 + .5).astype(np.uint8))
    if im.width > 1920:
        im = im.resize((1920, round(im.height * 1920 / im.width)), Image.LANCZOS)
    path = os.path.join(OUT, f"{PREFIX}_{name}.jpg")
    im.save(path, quality=85, optimize=True)
    return path

# ------------------------------------------------------------------ camera model (pinhole, like Unity)
def project(eye, look, fov, p, w=1920, h=1080):
    eye, look, p = map(lambda v: np.array(v, np.float64), (eye, look, p))
    f = look - eye; f /= np.linalg.norm(f)
    r = np.cross([0, 1, 0], f); r /= np.linalg.norm(r)
    u = np.cross(f, r)
    d = p - eye
    x, y, z = d @ r, d @ u, d @ f
    t = math.tan(math.radians(fov) / 2)
    sx = w / 2 + (x / (z * t * w / h)) * w / 2
    sy = h / 2 - (y / (z * t)) * h / 2
    return sx, sy, z

for yy in range(0, 1080 - hh, 8):
    for xx in range(0, 1920 - ww, 8):
        if not (xx + ww < x0 - 4 or xx > x1 + 4 or yy + hh < y0 - 4 or yy > y1 + 4):
            continue
        s = ii[yy + hh, xx + ww] - ii[yy, xx + ww] - ii[yy + hh, xx] + ii[yy, xx]
        m = s / (hh * ww)
        if m > best:
            best, where = float(m), (xx, yy)
# r3: the goal AREA as the eye reads it at a sprint -- the door, its frame, the EXIT sign and the
# lit end-wall ring around them: the door bbox grown by one door width each side and half a
# door height above (stops at the floor line), vs the brightest same-size window elsewhere.

    for xx in range(0, 1920 - AW, 8):
        if not (xx + AW < ax0 or xx > ax1 or yy + AH < ay0 or yy > ay1):

        s_ = ii[yy + AH, xx + AW] - ii[yy, xx + AW] - ii[yy + AH, xx] + ii[yy, xx]
        m_ = s_ / (AH * AW)
        if m_ > best2:
            best2, where2 = float(m_), (xx, yy)

                       goal_area_meanY=round(am, 4), goal_area_bbox=[int(ax0), int(ay0), int(AW), int(AH)],
                       brightest_other_area_meanY=round(best2, 4), area_at=where2)

# ---- r4: goal vs the brightest NON-FIXTURE window (brief §7.5.2: "highest region, or second after
# a fixture"). Fixture = the twin-head units and every emitter face (lamp lenses, wrap lenses, EXIT
# faces), dilated 24 px so a head's own glare halo counts as the fixture.
fix = ring(ids["Units"] | ids["Emitter"], 24) | ids["Units"] | ids["Emitter"]
fix_ii = np.pad(fix.astype(np.float64), ((1, 0), (1, 0))).cumsum(0).cumsum(1)
def brightest_nonfixture(h_, w_, ex):
    bst, at = 0.0, None
    for yy in range(0, 1080 - h_, 8):
        for xx in range(0, 1920 - w_, 8):
            if not (xx + w_ < ex[0] or xx > ex[1] or yy + h_ < ex[2] or yy > ex[3]):
                continue
            f_ = fix_ii[yy + h_, xx + w_] - fix_ii[yy, xx + w_] - fix_ii[yy + h_, xx] + fix_ii[yy, xx]
            if f_ > .05 * h_ * w_:
                continue
            s_ = ii[yy + h_, xx + w_] - ii[yy, xx + w_] - ii[yy + h_, xx] + ii[yy, xx]
            if s_ / (h_ * w_) > bst:
                bst, at = float(s_ / (h_ * w_)), (xx, yy)
    return round(bst, 4), at

# ---- r8: OUTLINE contrast. The brief's bar compares the whole obstacle's mean with a 12 px ring;
# a two-tone object (a light cart with a dark vinyl bag) can average to its background and still
# separate along its outline. For each outline pixel: mean Y of the object pixels vs the background
# pixels inside a 7 x 7 window around it; report the median ratio and the share of the outline at
# >= 2 : 1. Reported next to the brief's number, not instead of it.
def boxsum(a, r=3):
    c = np.pad(a, ((r + 1, r), (r + 1, r))).cumsum(0).cumsum(1)
    return c[2 * r + 1:, 2 * r + 1:] - c[:-2 * r - 1, 2 * r + 1:] - c[2 * r + 1:, :-2 * r - 1] + c[:-2 * r - 1, :-2 * r - 1]
def outline(yy, m, bg):
    mi = m.astype(np.float64); bi = bg.astype(np.float64)
    sin, nin = boxsum(yy * mi), boxsum(mi); sout, nout = boxsum(yy * bi), boxsum(bi)
    edge = m & (nout > 3)
    if edge.sum() < 10:
        return None
    a_ = sin[edge] / np.maximum(nin[edge], 1); b_ = sout[edge] / np.maximum(nout[edge], 1)
    r_ = np.maximum(a_, b_) / np.maximum(np.minimum(a_, b_), 1e-5)
    return dict(edge_px=int(edge.sum()), median=round(float(np.median(r_)), 2), share_ge2=round(float((r_ >= 2).mean()), 3))
