/// Spec: Documentation/research/room_visuals/10_run_directions.md §1, §3, §7.
/// Builds, in an empty scene at world (4992, 0, 4992): a Level 0 anteroom (3 x 2 cells),

/// photosafety timeline and a stats file to Verification/run_lookdev.

    static float headIntensity = 10f, beamDensity = .05f, beamGain = 16f;
    // Heads: a 100 deg spot whose cookie carries the 50/20 beam plus ~2 % stray light out to
    // ~45 deg (Cookie_SealedBeam_S). The goal head (unit 21, +Z) is the same 50/20 flood lamp
    // (brief §3.1), core-only cookie, aimed at the leaf (pitch -13). aimCross: the flood heads
    // aim at the floor at x = -side * aimCross (across the corridor) instead of the axis, so the

    // goalInner 48: the core cookie already carries the 50/20 beam profile (a 20 deg inner

    // At 6 m the spot's pool is ~2.5 m across: the leaf, the frame, the floor in front, a ring
    // of end wall. It makes the goal the hottest pool of the corridor (brief §3.1).
    static float headOuter = 100f, headInner = 96f, goalGain = 5f, goalOuter = 26f, goalInner = 22f, aimCross = .55f;
    // Floor bounce (r2): one up-facing, unshadowed spot 0.6 m UNDER each pool. The floor faces

    static float startProbeMul = 1f;

    // first render rendered identical to probes on (0 px over 6 levels). So each state renders in
    // its own Unity run, with its own probe set fixed for the whole run.
    static string runState = "all";   // r6: -runProbeMul v sets the probe multiplier BEFORE the first render (sweeps after it do not re-sync probes reliably in batch)
    static bool matte; static int probeRes = 256; static string probeDir = "Assets/RunLookdevProbes";
    static bool dbgNoCookie, dbgNoHeadShadows; static float dbgAmbientMul = 1f; static string dbgProbe = "custom";

    [MenuItem("FrontRooms/Rendering/Run look-dev (B emergency power)")]
    public static void Capture() => Run(null, true, true);

    public static void RunBatch()
    {
        var args = Environment.GetCommandLineArgs();
        string shots = null;
        var timeline = true; var idPass = true;
        for (var i = 0; i < args.Length; i++)
        {
            if (args[i] == "-runShots" && i + 1 < args.Length) shots = args[i + 1];
            if (args[i] == "-runOut" && i + 1 < args.Length) outDir = args[i + 1];
            if (args[i] == "-runNoTimeline") timeline = false;
            if (args[i] == "-runNoId") idPass = false;
            if (args[i] == "-runHead" && i + 1 < args.Length) headIntensity = float.Parse(args[i + 1], CultureInfo.InvariantCulture);
            if (args[i] == "-runBeam" && i + 1 < args.Length) beamDensity = float.Parse(args[i + 1], CultureInfo.InvariantCulture);
            if (args[i] == "-runNoCookie") dbgNoCookie = true;
            if (args[i] == "-runNoHalo") noHalo = true;
            if (args[i] == "-runHunter") useHunter = true;
            if (args[i] == "-runWrap" && i + 1 < args.Length) wrapIntensity = float.Parse(args[i + 1], CultureInfo.InvariantCulture);
            if (args[i] == "-runWrapCone" && i + 2 < args.Length) { wrapOuter = float.Parse(args[i + 1], CultureInfo.InvariantCulture); wrapInner = float.Parse(args[i + 2], CultureInfo.InvariantCulture); }
            if (args[i] == "-runBeamGain" && i + 1 < args.Length) beamGain = float.Parse(args[i + 1], CultureInfo.InvariantCulture);
            if (args[i] == "-runBriefAmbient") briefAmbient = true;
            if (args[i] == "-runLift" && i + 1 < args.Length) liftW = float.Parse(args[i + 1], CultureInfo.InvariantCulture);
            if (args[i] == "-runNoHeadShadows") dbgNoHeadShadows = true;
            if (args[i] == "-runAmbientMul" && i + 1 < args.Length) dbgAmbientMul = float.Parse(args[i + 1], CultureInfo.InvariantCulture);
            if (args[i] == "-runProbe" && i + 1 < args.Length) dbgProbe = args[i + 1];
            if (args[i] == "-runGoalGain" && i + 1 < args.Length) goalGain = float.Parse(args[i + 1], CultureInfo.InvariantCulture);
            if (args[i] == "-runGoalCone" && i + 2 < args.Length) { goalOuter = float.Parse(args[i + 1], CultureInfo.InvariantCulture); goalInner = float.Parse(args[i + 2], CultureInfo.InvariantCulture); }
            if (args[i] == "-runAimCross" && i + 1 < args.Length) aimCross = float.Parse(args[i + 1], CultureInfo.InvariantCulture);
            if (args[i] == "-runNoSpill") noSpill = true;
            if (args[i] == "-runDiagSH") diagSH = true;
            if (args[i] == "-runSweep" && i + 1 < args.Length) sweepSpec = args[i + 1];
            if (args[i] == "-runBounce" && i + 1 < args.Length) bounceIntensity = float.Parse(args[i + 1], CultureInfo.InvariantCulture);
            if (args[i] == "-runExitE" && i + 1 < args.Length) exitEmission = float.Parse(args[i + 1], CultureInfo.InvariantCulture);
            if (args[i] == "-runMatte") matte = true;
            if (args[i] == "-runProbeRes" && i + 1 < args.Length) probeRes = int.Parse(args[i + 1], CultureInfo.InvariantCulture);
            if (args[i] == "-runState" && i + 1 < args.Length) runState = args[i + 1];
            if (args[i] == "-runProbeMul" && i + 1 < args.Length) startProbeMul = float.Parse(args[i + 1], CultureInfo.InvariantCulture);
        }
        Run(shots, timeline, idPass);
    }

    static void Log(string s) { log.Add(s); Debug.Log("[RunLookdev] " + s); }

    static void Run(string onlyShots, bool timeline, bool idPass)
    {
        if (string.IsNullOrEmpty(outDir)) outDir = Path.Combine(Directory.GetParent(Application.dataPath).FullName, "Verification", "run_lookdev");
        Directory.CreateDirectory(outDir);
        EnsureAssets();
        var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
        EditorSceneManager.SetActiveScene(scene);
        ids.Clear(); mats.Clear(); log.Clear();

        root = new GameObject("RUN LOOKDEV B").transform;
        root.position = RootWorld;
        FrontRoomsLook.ApplyAmbient();
        FrontRoomsPostStack.Ensure(root);
        Log("ambient probe L0 rgb = " + RenderSettings.ambientProbe[0, 0].ToString("0.000") + ", " + RenderSettings.ambientProbe[1, 0].ToString("0.000") + ", " + RenderSettings.ambientProbe[2, 0].ToString("0.000"));
        SetupReflections();

        ante = Group(root, "Level 0 anteroom");
        corridor = Group(root, "Run corridor B");
        stub = Group(root, "Level 0 stub");
        rig = corridor.gameObject.AddComponent<FrontRoomsRunLookdevRig>();
        rig.globalAmbient = RenderSettings.ambientProbe;
        rig.headIntensity = headIntensity;
        rig.wrapIntensity = wrapIntensity;
        rig.beamDensity = beamDensity;
        rig.bounceIntensity = bounceIntensity;
        rig.armedAmbient *= dbgAmbientMul; rig.trippedAmbient *= dbgAmbientMul;
        rig.bounceAmbient = !briefAmbient;

        BuildAnteroom();
        BuildStub();
        BuildShell();
        BuildServices();
        BuildWraps();
        BuildUnits();
        BuildBounce();
        BuildSigns();
        BuildFireEquipment();
        BuildDoors();
        BuildObstacles();
        BuildRelay();

        var vol = FrontRoomsPostStack.EnsureZoneVolume(corridor, "Run_B", new Bounds(new Vector3(0f, H * .5f, Len * .5f), new Vector3(3.0f, H, Len)), 1.0f);
        if (vol != null) vol.priority = 2f; else Log("WARNING: FrontRoomsPost_Run_B missing");

        // Corridor ambient (brief §7.3): LightProbeUsage.CustomProvided + an SH set per renderer
        // through a MaterialPropertyBlock (CopySHCoefficientArraysFrom). Verified 2026-10-04 in
        // this clone: it reaches both FrontRooms/Surface and URP Lit (x0 / x1 / x10 renders differ;
        // x1 equals the earlier shader-override path to 4 decimals), so no shader change is needed.
        // Corridor reflections: a custom box probe whose cube and intensity follow the state.
        rig.ambientMode = FrontRoomsRunLookdevRig.AmbientMode.CustomSH;
        if (matte) BuildProbe(); else BuildBoxProbes();
        if (!matte && runState == "none")
        {
            // No box probes in this run (the blackout, and the "reflections off" comparison): the
            // default reflection becomes a near-black cube, so glossy surfaces reflect nothing.
            RenderSettings.customReflectionTexture = FlatCube("Run_NoReflectionCube", new Color(.002f, .002f, .002f), new Color(.002f, .002f, .002f), new Color(.002f, .002f, .002f), new Color(.002f, .002f, .002f));
            Log("no box probes; default reflection = near-black cube (.002)");
        }
        var nonSurface = new SortedSet<string>();
        foreach (var r in corridor.GetComponentsInChildren<Renderer>(true))
        {
            if (IsFx(r)) { r.lightProbeUsage = LightProbeUsage.Off; r.reflectionProbeUsage = ReflectionProbeUsage.Off; }
            else
            {
                r.lightProbeUsage = LightProbeUsage.CustomProvided;
                r.reflectionProbeUsage = ReflectionProbeUsage.BlendProbes;
                foreach (var m in r.sharedMaterials) if (m != null && m.shader.name != "FrontRooms/Surface") nonSurface.Add(m.name + " [" + m.shader.name + "]");
            }
            rig.corridorRenderers.Add(r);
            if (ids.TryGetValue(r, out var rid) && rid == Id.Units) rig.unitRenderers.Add(r);
        }
        Log("corridor ambient = CustomProvided SH via property blocks; materials not on the Surface shader (URP Lit/Unlit, same SH path): " + string.Join("; ", nonSurface));
        Physics.SyncTransforms();
        Log("SH global probe: " + FrontRoomsRunLookdevRig.Describe(RenderSettings.ambientProbe));
        Log("SH corridor armed: " + FrontRoomsRunLookdevRig.Describe(rig.CorridorSH(-1f)) + " | tripped: " + FrontRoomsRunLookdevRig.Describe(rig.CorridorSH(3f)) + " (bounce model " + rig.bounceAmbient + ")");
        Log("floor = " + (matte ? "MATTE VCT (mask Run_ServiceVCT_S: smoothness .21-.80, lane ~.72; one flat custom cube, no box projection)" : "GLOSSY waxed VCT (mask Run_ServiceVCTGloss_S: field ~.80, lane ~.85, seams ~.38; " + rig.probes.Count + " box-projected probes, " + probeRes + " px, baked per state)"));
        if (!matte && runState != "none") BakeBoxProbes();
        rig.probeMul = startProbeMul;
        if (startProbeMul != 1f) Log("probe multiplier " + startProbeMul + " from the first render");
        if (Environment.GetCommandLineArgs().Contains("-runProbeDebug"))
        {
            // Debug (r6): which reflection does each pixel sample? Box probes -> solid magenta cubes,
            // default reflection -> solid green. Set before the first render (the Forward+ probe
            // atlas keeps the first texture it sees per probe).
            Cubemap Solid(string n, Color c)
            {
                var cube = new Cubemap(32, TextureFormat.RGBAHalf, true) { name = n };
                foreach (CubemapFace f in Enum.GetValues(typeof(CubemapFace)))
                {
                    if (f == CubemapFace.Unknown) continue;
                    var px = new Color[32 * 32];
                    for (var i = 0; i < px.Length; i++) px[i] = c;
                    cube.SetPixels(px, f);
                }
                cube.Apply(true);
                return cube;
            }
            var mag = Solid("dbg magenta", new Color(3f, 0f, 3f)); var grn = Solid("dbg green", new Color(0f, 3f, 0f));
            foreach (var p in rig.probes.Concat(rig.probesTripped)) p.customBakedTexture = mag;
            RenderSettings.customReflectionTexture = grn;
            Log("PROBE DEBUG: box probes magenta, default reflection green");
        }

        var shots = Shots();
        if (Environment.GetCommandLineArgs().Contains("-runDebugTop"))
        {
            var cam = MakeCamera();
            var drt = new RenderTexture(1920, 400, 24, RenderTextureFormat.ARGB32) { antiAliasing = 4 };
            cam.targetTexture = drt;
            cam.orthographic = true; cam.orthographicSize = 2.8f;
            cam.transform.SetPositionAndRotation(root.TransformPoint(new Vector3(0f, 2.85f, 13.5f)), Quaternion.LookRotation(Vector3.down, Vector3.left));
            cam.nearClipPlane = .01f; cam.farClipPlane = 4f;
            relay.gameObject.SetActive(false);
            rig.Evaluate(3f);
            foreach (var h in rig.heads) Log("head " + h.light.name + " enabled " + h.light.enabled + " int " + h.light.intensity + " range " + h.light.range + " angle " + h.light.spotAngle + "/" + h.light.innerSpotAngle + " pos " + corridor.InverseTransformPoint(h.light.transform.position).ToString("F2") + " fwd " + corridor.InverseTransformDirection(h.light.transform.forward).ToString("F2") + " type " + h.light.type + " cull " + h.light.cullingMask + " layers " + h.light.renderingLayerMask);
            cam.Render(); cam.Render(); Save(drt, Path.Combine(outDir, "dbg_top_all.png"));
            for (var i = 0; i < rig.heads.Count; i++)
            {
                for (var j = 0; j < rig.heads.Count; j++) rig.heads[j].light.enabled = i == j;
                cam.Render(); cam.Render(); Save(drt, Path.Combine(outDir, "dbg_top_head" + i + ".png"));
            }
            cam.targetTexture = null; drt.Release(); UnityEngine.Object.DestroyImmediate(cam.gameObject);
            File.WriteAllLines(Path.Combine(outDir, "log.txt"), log);
            return;
        }
        if (Environment.GetCommandLineArgs().Contains("-runDebugSoffit"))
        {
            var cam = MakeCamera();
            var drt = new RenderTexture(960, 540, 24, RenderTextureFormat.ARGB32) { antiAliasing = 4 };
            cam.targetTexture = drt;
            relay.gameObject.SetActive(false);
            rig.Evaluate(-1f);
            Pose(cam, new Shot { eye = new Vector3(0f, 1.62f, 10f), look = new Vector3(0f, 2.9f, 12f), fov = 72f });
            cam.Render(); cam.Render(); Save(drt, Path.Combine(outDir, "dbg_up_surface.png"));
            var soffitR = corridor.Find("soffit").GetComponent<Renderer>();
            Log("soffit mat " + soffitR.sharedMaterial.name + " shader " + soffitR.sharedMaterial.shader.name + " enabled " + soffitR.enabled + " bounds " + soffitR.bounds);
            soffitR.sharedMaterial = Lit("dbg grey", Color.gray, .2f);
            cam.Render(); cam.Render(); Save(drt, Path.Combine(outDir, "dbg_up_lit.png"));
            soffitR.enabled = false;
            cam.Render(); cam.Render(); Save(drt, Path.Combine(outDir, "dbg_up_none.png"));
            cam.targetTexture = null; drt.Release(); UnityEngine.Object.DestroyImmediate(cam.gameObject);
            File.WriteAllLines(Path.Combine(outDir, "log.txt"), log);
            return;
        }
        if (diagSH) DiagSH(shots);
        Render(shots, onlyShots, idPass);
        if (!string.IsNullOrEmpty(sweepSpec)) Sweep(shots, sweepSpec);
        if (timeline && (onlyShots == null || onlyShots.Contains("timeline"))) Timeline();
        Stats();
        File.WriteAllLines(Path.Combine(outDir, "log.txt"), log);
        Debug.Log("[RunLookdev] done -> " + outDir);
    }

    static Camera MakeCamera()
    {
        var go = new GameObject("RUN LOOKDEV camera");
        var cam = go.AddComponent<Camera>();
        cam.nearClipPlane = .06f;
        cam.farClipPlane = 80f;
        cam.clearFlags = CameraClearFlags.SolidColor;
        cam.backgroundColor = new Color(0x22 / 255f, 0x23 / 255f, 0x1C / 255f);
        FrontRoomsPostStack.ConfigureCamera(cam);
        // Main's ConfigureCamera now opts every camera into the G14 Metal RT glass. This scene has
        // no RT glass (the wired lite is the brief's URP stand-in), and the first render after the
        // opt-in stalled > 20 min in batch mode (plugin "waiting: layout self-test"). The clone's
        // renderer has the RT feature off; drop the per-camera state too.
        var rtCam = cam.GetComponent("FrontRoomsGlassRTCamera");
        if (rtCam != null) UnityEngine.Object.DestroyImmediate(rtCam);
        return cam;
    }

    static void Pose(Camera cam, Shot s)
    {
        var eye = root.TransformPoint(s.eye);
        cam.transform.SetPositionAndRotation(eye, Quaternion.LookRotation(root.TransformPoint(s.look) - eye, Vector3.up));
        cam.fieldOfView = s.fov;
    }

    static void Render(List<Shot> shots, string onlyShots, bool idPass)
    {
        var cam = MakeCamera();
        var rt = new RenderTexture(1920, 1080, 24, RenderTextureFormat.ARGB32) { antiAliasing = 4 };
        cam.targetTexture = rt;
        var filter = string.IsNullOrEmpty(onlyShots) ? null : new HashSet<string>(onlyShots.Split(','));
        var samples = new StringBuilder();
        samples.Append("{\n");
        var firstSample = true;
        foreach (var s in shots)
        {
            if (filter != null && !filter.Contains(s.name)) continue;
            relay.gameObject.SetActive(s.relay);
            rig.Evaluate(s.t);
            var shotState = s.t < 0f ? "armed" : rig.MeanHeadLevel(s.t) > 0f ? "tripped" : "none";
            if (runState != "all" && runState != shotState) Log("WARNING: shot " + s.name + " is " + shotState + " but this run's probes are " + runState);
            Pose(cam, s);
            cam.Render();
            cam.Render();
            Save(rt, Path.Combine(outDir, s.name + ".png"));
            Log("shot " + s.name + " t=" + s.t.ToString("0.00") + " lights on " + LightsOn());
            // Floor axis samples and key points (pixel coords, origin top-left).
            if (!firstSample) samples.Append(",\n");
            firstSample = false;
            samples.Append("\"" + s.name + "\": {\"t\": " + F(s.t) + ", \"axis\": [");
            var first = true;
            for (var z = .5f; z <= 26.5f; z += .5f)
            {
                var p = cam.WorldToScreenPoint(root.TransformPoint(new Vector3(0f, .003f, z)));
                if (p.z <= 0f || p.x < 0 || p.x >= 1920 || p.y < 0 || p.y >= 1080) continue;
                samples.Append((first ? "" : ", ") + "[" + F(z) + ", " + F(p.x) + ", " + F(1080f - p.y) + "]");
                first = false;
            }
            samples.Append("], \"points\": {");
            var ptList = new List<(string, Vector3)> { ("pool12", new Vector3(0f, .003f, 12f)), ("pool6", new Vector3(0f, .003f, 6f)), ("pool18", new Vector3(0f, .003f, 18f)), ("goal_leaf", new Vector3(0f, 1.0f, GoalFace)), ("relay_head", new Vector3(0f, 2.3f, 11.5f)) };
            for (var hi = 0; hi < rig.heads.Count; hi++) ptList.Add(("head" + hi, root.InverseTransformPoint(rig.heads[hi].light.transform.position)));
            var pts = ptList.ToArray();
            var fp = true;
            foreach (var (n, v) in pts)
            {
                var p = cam.WorldToScreenPoint(root.TransformPoint(v));
                samples.Append((fp ? "" : ", ") + "\"" + n + "\": [" + F(p.x) + ", " + F(1080f - p.y) + ", " + F(p.z) + "]");
                fp = false;
            }
            samples.Append("}}");
            if (idPass && s.id) RenderId(cam, s);
        }
        samples.Append("\n}\n");
        File.WriteAllText(Path.Combine(outDir, "samples.json"), samples.ToString());
        cam.targetTexture = null;
        rt.Release();
        UnityEngine.Object.DestroyImmediate(cam.gameObject);
    }

    static bool IsFx(Renderer r) => r.sharedMaterial != null && (r.sharedMaterial.shader.name.Contains("Beam") || r.sharedMaterial.shader.name.Contains("Halo"));

    static string F(float v) => v.ToString("0.###", CultureInfo.InvariantCulture);

    static string LightsOn()
    {
        var on = 0; var shadowed = 0;
        foreach (var l in root.GetComponentsInChildren<Light>(true))
            if (l.enabled && l.intensity > 0f) { on++; if (l.shadows != LightShadows.None) shadowed++; }
        return on + " (shadowed " + shadowed + ")";
    }

    /// <summary>Flat unlit ID colours, no post, no MSAA, no fog: masks for the measurements.</summary>
    static void RenderId(Camera cam, Shot s)
    {
        var unlit = Shader.Find("Universal Render Pipeline/Unlit");
        var idMats = new Dictionary<Id, Material>();
        foreach (var kv in IdColour) { var m = new Material(unlit); m.SetColor("_BaseColor", kv.Value); idMats[kv.Key] = m; }
        var saved = new Dictionary<Renderer, Material[]>();
        var hidden = new List<Renderer>();
        foreach (var r in root.GetComponentsInChildren<Renderer>(false))
        {
            if (IsFx(r)) { if (r.enabled) { hidden.Add(r); r.enabled = false; } continue; }
            saved[r] = r.sharedMaterials;
            var id = ids.TryGetValue(r, out var v) ? v : Id.Background;
            var arr = new Material[r.sharedMaterials.Length];
            for (var i = 0; i < arr.Length; i++) arr[i] = idMats[id];
            r.sharedMaterials = arr;
        }
        var fog = RenderSettings.fog;
        RenderSettings.fog = false;
        var data = cam.GetUniversalAdditionalCameraData();
        data.renderPostProcessing = false;
        var prevRt = cam.targetTexture;
        var rt = new RenderTexture(1920, 1080, 24, RenderTextureFormat.ARGB32) { antiAliasing = 1 };
        cam.targetTexture = rt;
        cam.allowMSAA = false;
        cam.Render();
        Save(rt, Path.Combine(outDir, s.name + "_id.png"));
        cam.targetTexture = prevRt;
        cam.allowMSAA = true;
        rt.Release();
        data.renderPostProcessing = true;
        RenderSettings.fog = fog;
        foreach (var kv in saved) kv.Key.sharedMaterials = kv.Value;
        foreach (var r in hidden) r.enabled = true;
    }

    /// <summary>30 fps from t = -0.5 to 2.0 s at the C2 pose: mean frame luminance (linear Y) and the
    /// saturated-red fraction, for the photosafety table.</summary>
    static void Timeline()
    {
        var cam = MakeCamera();
        var rt = new RenderTexture(480, 270, 24, RenderTextureFormat.ARGB32) { antiAliasing = 4 };
        cam.targetTexture = rt;
        relay.gameObject.SetActive(false);
        var shot = new Shot { eye = new Vector3(0f, 1.62f, 1.5f), look = new Vector3(0f, 1.30f, 27f), fov = 72f };
        Pose(cam, shot);
        var csv = new StringBuilder("t,meanY,p95Y,red_sat_frac\n");
        var tex = new Texture2D(480, 270, TextureFormat.RGB24, false);
        for (var f = 0; f <= 75; f++)
        {
            var t = -.5f + f / 30f;
            rig.Evaluate(t);
            cam.Render(); cam.Render();
            var prev = RenderTexture.active; RenderTexture.active = rt;
            tex.ReadPixels(new Rect(0, 0, 480, 270), 0, 0); tex.Apply();
            RenderTexture.active = prev;
            var px = tex.GetPixels32();
            var ys = new float[px.Length]; var sum = 0.0; var red = 0;
            for (var i = 0; i < px.Length; i++)
            {
                var c = px[i];
                var y = .2126f * Lin(c.r) + .7152f * Lin(c.g) + .0722f * Lin(c.b);
                ys[i] = y; sum += y;
                var s = c.r + c.g + c.b;
                if (s > 30 && c.r >= .8f * s) red++;
            }
            Array.Sort(ys);
            csv.Append(F(t) + "," + (sum / px.Length).ToString("0.00000", CultureInfo.InvariantCulture) + "," + ys[(int)(ys.Length * .95f)].ToString("0.0000", CultureInfo.InvariantCulture) + "," + ((float)red / px.Length).ToString("0.0000", CultureInfo.InvariantCulture) + "\n");
        }
        File.WriteAllText(Path.Combine(outDir, "timeline.csv"), csv.ToString());
        var changes = new StringBuilder("time_s,lamp,from,to\n");
        foreach (var c in rig.Changes(0f, 2f)) changes.Append(F(c.time) + "," + c.lamp + "," + F(c.from) + "," + F(c.to) + "\n");
        File.WriteAllText(Path.Combine(outDir, "changes.csv"), changes.ToString());
        cam.targetTexture = null; rt.Release();
        UnityEngine.Object.DestroyImmediate(cam.gameObject);
        UnityEngine.Object.DestroyImmediate(tex);
    }

    static float Lin(byte b) { var c = b / 255f; return c <= .04045f ? c / 12.92f : Mathf.Pow((c + .055f) / 1.055f, 2.4f); }

    // ------------------------------------------------------------ probe, diag, sweep
    static Cubemap FlatCube(string name, Color top, Color upper, Color lower, Color bottom)
    {
        var cube = new Cubemap(16, TextureFormat.RGBAHalf, false) { name = name };
        foreach (CubemapFace f in Enum.GetValues(typeof(CubemapFace)))
        {
            if (f == CubemapFace.Unknown) continue;
            var px = new Color[16 * 16];
            for (var i = 0; i < px.Length; i++)
            {
                var row = i / 16;   // row 0 = the top of a side face
                px[i] = f == CubemapFace.PositiveY ? top : f == CubemapFace.NegativeY ? bottom
                      : row < 3 ? top : row < 9 ? upper : row < 12 ? lower : bottom;
            }
            cube.SetPixels(px, f);
        }
        cube.Apply();
        return cube;
    }

    static void BuildProbe()
    {
        var go = new GameObject("Run corridor reflection");
        go.transform.SetParent(corridor, false);
        go.transform.localPosition = new Vector3(0f, H * .5f, Len * .5f);
        var p = go.AddComponent<ReflectionProbe>();
        p.mode = ReflectionProbeMode.Custom;
        p.size = new Vector3(3.0f, H + .1f, Len + .2f);
        p.boxProjection = false;
        p.importance = 10;
        p.blendDistance = 0f;
        // linear radiance seen from the corridor centre (rough, from the S1a / S1b frames)
        rig.armedCube = FlatCube("Run_ArmedCube", new Color(.03f, .03f, .03f), new Color(.36f, .38f, .39f), new Color(.12f, .14f, .13f), new Color(.30f, .30f, .29f));
        rig.trippedCube = FlatCube("Run_TrippedCube", new Color(.002f, .002f, .002f), new Color(.010f, .008f, .006f), new Color(.004f, .0035f, .003f), new Color(.035f, .028f, .018f));
        p.customBakedTexture = rig.armedCube;
        rig.probe = p;
    }

    /// <summary>r3: three box-projected probes, one per 3 cells (z 0-9, 9-18, 18-27), the box on the
    /// wall faces, floor and soffit, captured at 1.2 m. Baked per state in BakeBoxProbes.</summary>
    static void BuildBoxProbes()
    {
        RenderSettings.reflectionBounces = 1;
        foreach (var state in new[] { "armed", "tripped" })
            for (var i = 0; i < 3; i++)
            {
                if (runState != "all" && runState != state) continue;
            {
                var z0 = i * 9f; var z1 = Mathf.Min(GoalFace, z0 + 9f);
                var zc = (z0 + z1) * .5f;
                var go = new GameObject("Run box probe " + i + " " + state);
                go.transform.SetParent(corridor, false);
                go.transform.localPosition = new Vector3(0f, 1.2f, zc);
                var p = go.AddComponent<ReflectionProbe>();
                p.mode = ReflectionProbeMode.Baked;
                p.resolution = probeRes;
                p.hdr = true;
                p.boxProjection = true;
                // r4 fix: the influence box must reach PAST the shell by more than the blend distance.
                // r3 sized it to the shell exactly (x +-1.42, y 0..2.9) with a .3 blend, and URP weights a
                // probe by min(distance to the box faces) / blend: every floor, wall and soffit pixel sat
                // ON a face, so its weight was 0 and it fell back to the flat default cube (measured r4:
                // probes x0 vs x1 vs x3 changed S1b by at most 7-9 of 765 levels). The box now overhangs
                // the shell by .16 m on x and y (blend .15), so shell pixels weigh 1; box projection
                // lands .16 m behind the real faces (a 6 % parallax error across the 2.84 m width).
                const float m = .16f;
                p.center = new Vector3(0f, H * .5f - 1.2f, 0f);
                p.size = new Vector3(2 * Face + 2 * m, H + 2 * m, (z1 - z0) + .6f);   // .3 m overlap each side along z
                p.blendDistance = .15f;
                p.importance = 10;
                p.nearClipPlane = .05f; p.farClipPlane = 40f;
                p.clearFlags = ReflectionProbeClearFlags.SolidColor;
                p.backgroundColor = Color.black;
                p.shadowDistance = 40f;
                p.cullingMask = ~0;
                // r5: one fixed probe set per state (see FrontRoomsRunLookdevRig.probesTripped).
                (state == "armed" ? rig.probes : rig.probesTripped).Add(p);
            }
            }
        Log("probe state for this run: " + runState + " (" + rig.probes.Count + " armed + " + rig.probesTripped.Count + " tripped box probes)");
    }

    static void BakeBoxProbes()
    {
        var full = Path.Combine(Directory.GetParent(Application.dataPath).FullName, probeDir);
        Directory.CreateDirectory(full);
        AssetDatabase.Refresh();
        relay.gameObject.SetActive(false);
        // Baked probes only render objects flagged Reflection Probe Static (r3 first try: all six
        // cubes came out black). Flag the whole set; the Relay stays dynamic (and hidden).
        var flagged = 0;
        foreach (var r in root.GetComponentsInChildren<Renderer>(true))
        {
            if (r.transform.IsChildOf(relay)) continue;
            var go = r.gameObject;
            GameObjectUtility.SetStaticEditorFlags(go, GameObjectUtility.GetStaticEditorFlags(go) | StaticEditorFlags.ReflectionProbeStatic);
            flagged++;
        }
        Log("probe bake: " + flagged + " renderers flagged ReflectionProbeStatic");
        var all = rig.probes.Concat(rig.probesTripped).ToList();
        foreach (var p in all) { p.mode = ReflectionProbeMode.Baked; p.customBakedTexture = null; }
        foreach (var (state, t) in new[] { ("armed", -1f), ("tripped", 3f) })
        {
            var set = state == "armed" ? rig.probes : rig.probesTripped;
            if (set.Count == 0) continue;
            rig.Evaluate(t);
            // the probes must not see each other (or the other state) while baking
            foreach (var p in all) p.enabled = false;
            for (var i = 0; i < set.Count; i++)
            {
                var p = set[i];
                p.enabled = true;
                p.intensity = 1f;
                var path = probeDir + "/RunB_probe" + i + "_" + state + ".exr";
                var ok = Lightmapping.BakeReflectionProbe(p, path);
                AssetDatabase.ImportAsset(path, ImportAssetOptions.ForceUpdate);
                var cube = AssetDatabase.LoadAssetAtPath<Texture>(path);
                var bytes = new FileInfo(Path.Combine(Directory.GetParent(Application.dataPath).FullName, path)).Length;
                Log("probe " + i + " " + state + " bake " + ok + " -> " + path + " " + (cube != null ? cube.width + " px " + cube.dimension : "MISSING") + ", exr " + (bytes / 1024) + " KB" + (bytes < 40000 ? " (SUSPECT: nearly constant)" : ""));
                (state == "armed" ? rig.armedCubes : rig.trippedCubes).Add(cube);
                p.mode = ReflectionProbeMode.Custom;
                p.customBakedTexture = cube;
                p.enabled = false;
            }
        }
        rig.Evaluate(-1f);
    }

    /// <summary>Parameter sweep: "name:key=v,key=v;name2:..." rendered for the shots in "shots=a+b"
    /// (default s1b). Keys: head, goal, amb (ambient mul), beam (density), spill (0/1), exit.</summary>
    static void Sweep(List<Shot> shots, string spec)
    {
        var cam = MakeCamera();
        var rt = new RenderTexture(1920, 1080, 24, RenderTextureFormat.ARGB32) { antiAliasing = 4 };
        cam.targetTexture = rt;
        var cookieSpill = AssetDatabase.LoadAssetAtPath<Texture2D>(TexDir + "Cookie_SealedBeam_S.png");
        var cookieCore = AssetDatabase.LoadAssetAtPath<Texture2D>(TexDir + "Cookie_SealedBeamCore_S.png");
        var baseHead = rig.headIntensity; var baseBeam = rig.beamDensity;
        var baseGain = rig.heads.Select(h => h.gain).ToList();
        foreach (var part in spec.Split(';'))
        {
            if (string.IsNullOrWhiteSpace(part)) continue;
            var name = part.Split(':')[0].Trim();
            var kv = part.Contains(":") ? part.Substring(part.IndexOf(':') + 1) : "";
            var shotList = new List<string> { "s1b" };
            rig.headIntensity = baseHead; rig.beamDensity = baseBeam; rig.ambientMul = 1f; rig.bounceIntensity = bounceIntensity; rig.probeMul = 1f;
            foreach (var m in mats.Values) if (m.name == "Run exit face B") m.SetColor("_EmissionColor", Color.white * exitEmission);
            for (var i = 0; i < rig.heads.Count; i++) rig.heads[i].gain = baseGain[i];
            var spill = !noSpill;
            foreach (var pair in kv.Split(','))
            {
                var e = pair.Split('=');
                if (e.Length != 2) continue;
                var key = e[0].Trim(); var val = e[1].Trim();
                if (key == "shots") { shotList = val.Split('+').ToList(); continue; }
                var v = float.Parse(val, CultureInfo.InvariantCulture);
                if (key == "head") rig.headIntensity = v;
                if (key == "goal") for (var i = 0; i < rig.heads.Count; i++) if (rig.heads[i].goal) rig.heads[i].gain = v;
                if (key == "amb") rig.ambientMul = v;
                if (key == "beam") rig.beamDensity = v;
                if (key == "spill") spill = v > .5f;
                if (key == "bounce") rig.bounceIntensity = v;
                if (key == "probe") rig.probeMul = v;
                if (key == "exitE") foreach (var m in mats.Values) if (m.name == "Run exit face B") m.SetColor("_EmissionColor", Color.white * v);
            }
            foreach (var h in rig.heads)
            {
                if (h.goal) continue;
                h.light.spotAngle = spill ? headOuter : 50f; h.light.innerSpotAngle = spill ? headInner : 20f;
                h.light.cookie = spill ? cookieSpill : cookieCore;
            }
            foreach (var sn in shotList)
            {
                var s = shots.Find(x => x.name == sn);
                if (s.name == null) continue;
                relay.gameObject.SetActive(s.relay);
                rig.Evaluate(s.t);
                Pose(cam, s);
                cam.Render(); cam.Render();
                Save(rt, Path.Combine(outDir, "sweep_" + name + "_" + sn + ".png"));
            }
            Log("sweep " + name + ": head " + rig.headIntensity + " goalGain " + rig.heads.Max(h => h.gain) + " amb " + rig.ambientMul + " beam " + rig.beamDensity + " spill " + spill + " bounce " + rig.bounceIntensity + " probeMul " + rig.probeMul);
        }
        // restore
        rig.headIntensity = baseHead; rig.beamDensity = baseBeam; rig.ambientMul = 1f; rig.bounceIntensity = bounceIntensity; rig.probeMul = 1f;
        foreach (var m in mats.Values) if (m.name == "Run exit face B") m.SetColor("_EmissionColor", Color.white * exitEmission);
        for (var i = 0; i < rig.heads.Count; i++) rig.heads[i].gain = baseGain[i];
        foreach (var h in rig.heads)
        {
            if (h.goal) continue;
            h.light.spotAngle = noSpill ? 50f : headOuter; h.light.innerSpotAngle = noSpill ? 20f : headInner;
            h.light.cookie = noSpill ? cookieCore : cookieSpill;
        }
        cam.targetTexture = null; rt.Release(); UnityEngine.Object.DestroyImmediate(cam.gameObject);
    }

    // ---------------------------------------------------------------- assets
    static void EnsureAssets()
    {
        // Post profile Run_B (§3.1): lift -0.01, contrast +12, saturation -10, WB temp -6,
        // post-exposure -0.1, bloom threshold 1.0 / intensity 0.6, vignette 0.32.
        var profile = AssetDatabase.LoadAssetAtPath<VolumeProfile>(PostPath);
        if (profile == null) { profile = ScriptableObject.CreateInstance<VolumeProfile>(); AssetDatabase.CreateAsset(profile, PostPath); }
        // Brief: lift w -0.01. Measured 2026-10-03: a -0.01 offset clips every pre-tonemap value
        // under 0.01 to black (the soffit, the red main, every gap), which breaks the 40:1 rule
        // by construction. 0.0 still sits 0.035 under the global film lift (+0.035).
        Comp<LiftGammaGain>(profile).lift.Override(new Vector4(1f, 1f, 1f, liftW));
        var ca = Comp<ColorAdjustments>(profile);
        ca.contrast.Override(12f); ca.saturation.Override(-10f); ca.postExposure.Override(-.1f);
        Comp<WhiteBalance>(profile).temperature.Override(-6f);
        var bl = Comp<Bloom>(profile); bl.threshold.Override(1.0f); bl.intensity.Override(.6f);
        Comp<Vignette>(profile).intensity.Override(.32f);
        EditorUtility.SetDirty(profile);

        // Kit slot materials the new Blender assets use (FrontRoomsKitLibrary maps slots by name).
        SurfaceAsset("Run_SealedBeamLens", "Run_SealedBeamLens", Vector2.one, Color.white, true, smooth: 1f, bump: .8f, emission: "Run_SealedBeamLens", emissionColor: Color.black);
        SurfaceAsset("Run_NeonPilot", null, Vector2.one, new Color(.45f, .05f, .035f), true, smooth: .85f, emission: "", emissionColor: Color.black);
        SurfaceAsset("Run_UnitLegend", "Run_UnitLegend", Vector2.one, Color.white, true, smooth: .45f);
        SurfaceAsset("Run_WrapLens", "Run_WrapLens", Vector2.one, Color.white, true, smooth: .55f, bump: .6f, emission: "Run_WrapLens", emissionColor: Color.black);
        AssetDatabase.SaveAssets();
        foreach (var a in new[] { "Kit_EmergencyTwinHead", "Kit_EmergencyTwinHead_Yoke", "Kit_EmergencyTwinHead_Lamp", "Kit_WrapFixture_4ft" })
        {
            AssetDatabase.ImportAsset("Assets/Resources/Props/Models/" + a + ".json", ImportAssetOptions.ForceUpdate);
            AssetDatabase.ImportAsset("Assets/Resources/Props/Models/" + a + ".fbx", ImportAssetOptions.ForceUpdate);
        }
        FrontRoomsKitLibrary.ClearCache();
    }

    static void ConfigureSurface(Material m, string tex, Vector2 tile, Color tint, bool meshUV, float smooth, float bump, float macroTone, float macroDirt, float floorGrime, float ceilingGrime, string emission, Color emissionColor)
    {
        m.SetTexture("_BaseMap", tex != null ? T(tex + "_A") : null);
        m.SetTexture("_BumpMap", tex != null ? T(tex + "_N") : null);
        m.SetTexture("_MaskMap", tex != null ? T(tex + "_S") : null);
        m.SetColor("_BaseColor", tint);
        m.SetTextureScale("_BaseMap", Vector2.one); m.SetTextureOffset("_BaseMap", Vector2.zero);
        m.SetVector("_TileSize", new Vector4(tile.x, tile.y, 0, 0));
        m.SetFloat("_BumpScale", bump);
        m.SetFloat("_Smoothness", smooth);
        m.SetFloat("_Metallic", 0f);
        m.SetFloat("_OcclusionStrength", .8f);
        m.SetTexture("_MacroMap", T("MacroWear_M"));
        m.SetFloat("_MacroTone", macroTone);
        m.SetFloat("_MacroDirt", macroDirt);
        m.SetColor("_StainColor", new Color(.42f, .38f, .30f));
        m.SetFloat("_StainStrength", 0f);
        m.SetFloat("_WetStrength", 0f);
        m.SetFloat("_FloorGrime", floorGrime);
        m.SetFloat("_CeilingGrime", ceilingGrime);
        m.SetFloat("_CeilingHeight", H);
        m.SetFloat("_MeshUV", meshUV ? 1f : 0f);
        if (meshUV) m.EnableKeyword("_FR_MESH_UV"); else m.DisableKeyword("_FR_MESH_UV");
        if (emission != null)
        {
            m.SetTexture("_EmissionMap", emission.Length == 0 ? null : T(emission + "_E"));
            m.SetColor("_EmissionColor", emissionColor);
            m.SetFloat("_UseEmission", 1f);
            m.EnableKeyword("_EMISSION");
            m.globalIlluminationFlags = MaterialGlobalIlluminationFlags.RealtimeEmissive;
        }
        else { m.SetFloat("_UseEmission", 0f); m.DisableKeyword("_EMISSION"); }
    }

    static void SurfaceAsset(string name, string tex, Vector2 tile, Color tint, bool meshUV, float smooth = 1f, float bump = 1f, string emission = null, Color emissionColor = default)
    {
        var path = SurfDir + name + ".mat";
        var m = AssetDatabase.LoadAssetAtPath<Material>(path);
        var shader = Shader.Find("FrontRooms/Surface");
        if (m == null) { m = new Material(shader) { name = name }; AssetDatabase.CreateAsset(m, path); }
        m.shader = shader;
        ConfigureSurface(m, tex, tile, tint, meshUV, smooth, bump, 0f, 0f, 0f, 0f, emission, emissionColor);
        EditorUtility.SetDirty(m);
    }

    static Material Surface(string name, string tex, Vector2 tile, Color tint, bool meshUV, float smooth = 1f, float bump = 1f, float macroTone = .2f, float macroDirt = .2f, float floorGrime = 0f, float ceilingGrime = 0f)
    {
        if (mats.TryGetValue(name, out var c)) return c;
        var m = new Material(Shader.Find("FrontRooms/Surface")) { name = name };
        ConfigureSurface(m, tex, tile, tint, meshUV, smooth, bump, macroTone, macroDirt, floorGrime, ceilingGrime, null, Color.black);
        mats[name] = m;
        return m;
    }

    static Material Lit(string name, Color c, float smooth = .3f, float metal = 0f, Texture tex = null, Texture emiTex = null, Color? emi = null, bool clip = false, bool transparent = false, Vector2? tiling = null)
    {
        if (mats.TryGetValue(name, out var cached)) return cached;
        if (!emi.HasValue && !clip && !transparent && tiling == null)
        {
            // Opaque, not emissive: the FrontRooms Surface shader (mesh UVs, flat tint, the
            // texture as a decal over the quad's 0..1 UVs), so props shade exactly like the shell
            // (same macro wear, same BRDF, same corridor SH).
            var sm = new Material(Shader.Find("FrontRooms/Surface")) { name = name };
            ConfigureSurface(sm, null, Vector2.one, c, true, smooth, 0f, 0f, 0f, 0f, 0f, null, Color.black);
            if (tex != null) sm.SetTexture("_BaseMap", tex);
            sm.SetFloat("_Metallic", metal);
            mats[name] = sm;
            return sm;
        }
        var m = new Material(Shader.Find("Universal Render Pipeline/Lit")) { name = name };
        m.SetColor("_BaseColor", c);
        m.SetFloat("_Smoothness", smooth);
        m.SetFloat("_Metallic", metal);
        if (tex != null) { m.SetTexture("_BaseMap", tex); if (tiling.HasValue) m.SetTextureScale("_BaseMap", tiling.Value); }
        if (emi.HasValue)
        {
            m.EnableKeyword("_EMISSION");
            m.SetColor("_EmissionColor", emi.Value);
            if (emiTex != null) m.SetTexture("_EmissionMap", emiTex);
            m.globalIlluminationFlags = MaterialGlobalIlluminationFlags.RealtimeEmissive;
        }
        if (clip)
        {
            m.SetFloat("_AlphaClip", 1f); m.SetFloat("_Cutoff", .5f); m.EnableKeyword("_ALPHATEST_ON");
            m.renderQueue = (int)RenderQueue.AlphaTest;
        }
        if (transparent)
        {
            m.SetFloat("_Surface", 1f); m.SetFloat("_Blend", 0f);
            m.SetFloat("_SrcBlend", (float)BlendMode.SrcAlpha); m.SetFloat("_DstBlend", (float)BlendMode.OneMinusSrcAlpha);
            m.SetFloat("_ZWrite", 0f); m.EnableKeyword("_SURFACE_TYPE_TRANSPARENT");
            m.renderQueue = (int)RenderQueue.Transparent;
            m.SetShaderPassEnabled("DepthOnly", false);
        }
        mats[name] = m;
        return m;
    }

    static Cubemap flatCube;
    static void SetupReflections()
    {
        // The empty scene would reflect Unity's default procedural sky (blue) on every
        // glossy surface. Replace it with a flat cube in the trilight ambient colours; the
        // corridor gets a box probe whose intensity the rig scales with the ambient.
        flatCube = new Cubemap(16, TextureFormat.RGBAHalf, false) { name = "Run_FlatAmbientCube" };
        var sky = FrontRoomsLook.AmbientSky.linear; var eq = FrontRoomsLook.AmbientEquator.linear; var gr = FrontRoomsLook.AmbientGround.linear;
        foreach (CubemapFace f in Enum.GetValues(typeof(CubemapFace)))
        {
            if (f == CubemapFace.Unknown) continue;
            var col = f == CubemapFace.PositiveY ? sky : f == CubemapFace.NegativeY ? gr : eq;
            var px = new Color[16 * 16];
            for (var i = 0; i < px.Length; i++)
            {
                var row = i / 16;
                // side faces: ceiling colour at the top rows, floor colour at the bottom rows
                var c = col;
                if (f != CubemapFace.PositiveY && f != CubemapFace.NegativeY) c = row < 8 ? Color.Lerp(sky, eq, row / 8f) : Color.Lerp(eq, gr, (row - 8) / 8f);
                px[i] = c;
            }
            flatCube.SetPixels(px, f);
        }
        flatCube.Apply();
        RenderSettings.defaultReflectionMode = DefaultReflectionMode.Custom;
        RenderSettings.customReflectionTexture = flatCube;
    }

    /// <summary>A box whose UVs are in metres on every face (for mesh-UV materials: wood, steel, cardboard).</summary>
    static Renderer MBox(Transform parent, string name, Vector3 centre, Vector3 size, Material m, Id id = Id.Background, Quaternion? rot = null)
    {
        var key = size.x.ToString("0.0000") + "_" + size.y.ToString("0.0000") + "_" + size.z.ToString("0.0000");
        if (!meshes.TryGetValue(key, out var mesh))
        {
            mesh = new Mesh { name = "metre box " + key };
            var v = new List<Vector3>(); var n = new List<Vector3>(); var uv = new List<Vector2>(); var tri = new List<int>();
            var h = size * .5f;
            void Face(Vector3 normal, Vector3 uAxis, Vector3 vAxis)
            {
                var b = v.Count;
                var c = Vector3.Scale(normal, h);
                var hu = Mathf.Abs(Vector3.Dot(h, uAxis)); var hv = Mathf.Abs(Vector3.Dot(h, vAxis));
                var corners = new[] { c - uAxis * hu - vAxis * hv, c + uAxis * hu - vAxis * hv, c + uAxis * hu + vAxis * hv, c - uAxis * hu + vAxis * hv };
                foreach (var p in corners) { v.Add(p); n.Add(normal); uv.Add(new Vector2(Vector3.Dot(p, uAxis), Vector3.Dot(p, vAxis))); }
                // winding: Unity is clockwise-front; normal = cross(vAxis, uAxis) side faces outward when (u, v) is (right, up) seen from outside
                tri.AddRange(new[] { b, b + 2, b + 1, b, b + 3, b + 2 });
            }
            Face(Vector3.back, Vector3.right, Vector3.up);
            Face(Vector3.forward, Vector3.left, Vector3.up);
            Face(Vector3.right, Vector3.forward, Vector3.up);
            Face(Vector3.left, Vector3.back, Vector3.up);
            Face(Vector3.up, Vector3.right, Vector3.forward);
            Face(Vector3.down, Vector3.right, Vector3.back);
            mesh.SetVertices(v); mesh.SetNormals(n); mesh.SetUVs(0, uv); mesh.SetTriangles(tri, 0);
            mesh.RecalculateTangents(); mesh.RecalculateBounds();
            meshes[key] = mesh;
        }
        var go = new GameObject(name);
        go.transform.SetParent(parent, false);
        go.transform.localPosition = centre;
        go.transform.localRotation = rot ?? Quaternion.identity;
        go.AddComponent<MeshFilter>().sharedMesh = mesh;
        var r = go.AddComponent<MeshRenderer>();
        r.sharedMaterial = m;
        ids[r] = id;
        return r;
    }

    /// <summary>A decal quad whose readable face looks along <paramref name="normal"/> (±X or ±Z).</summary>
    static Renderer Decal(Transform parent, string name, Vector3 c, Vector2 size, Vector3 normal, Material m, Id id = Id.Background)
    {
        var go = GameObject.CreatePrimitive(PrimitiveType.Quad);
        go.name = name;
        go.transform.SetParent(parent, false);
        go.transform.localPosition = c;
        // Unity's quad is visible from -Z (its normal is -Z) with U along +X.
        go.transform.localRotation = Quaternion.FromToRotation(Vector3.back, normal);
        if (Vector3.Dot(normal, Vector3.forward) > .9f) go.transform.localRotation = Quaternion.Euler(0f, 180f, 0f);
        go.transform.localScale = new Vector3(size.x, size.y, 1f);
        UnityEngine.Object.DestroyImmediate(go.GetComponent<Collider>());
        var r = go.GetComponent<MeshRenderer>();
        r.sharedMaterial = m;
        r.shadowCastingMode = ShadowCastingMode.Off;
        ids[r] = id;
        return r;
    }

    static void L0Walls(Transform parent, float x0, float x1, float z0, float z1, bool south, bool north, bool west, bool east)
    {
        var paper = FrontRoomsSurfaces.Room(RoomRule.Lobby, FrontRoomsSurfaces.Slot.Wall);

        if (east) W(new Vector3(x1, H * .5f, (z0 + z1) * .5f), new Vector3(t, H, z1 - z0 + t), false);

    /// <summary>Wall-run segments along z between [z0, z1], cut at this side's doors (± half) and extra gaps.</summary>
    static List<(float, float)> Runs(int side, float z0, float z1, float half, (float z, float h)[] extra = null)
    {
        var cuts = new List<(float, float)>();
        foreach (var d in Doors(side)) cuts.Add((d - half, d + half));
        if (extra != null) foreach (var e in extra) cuts.Add((e.z - e.h, e.z + e.h));
        cuts.Sort();
        var list = new List<(float, float)>();
        var cur = z0;
        foreach (var (a, b) in cuts) { if (a > cur) list.Add((cur, Mathf.Min(a, z1))); cur = Mathf.Max(cur, b); }
        if (cur < z1) list.Add((cur, z1));
        return list;
    }

    static float[] Doors(int side) => side > 0 ? new[] { 5f, 23f } : new[] { 17f };

    static void SideWall(int side, float[] doors)
    {
        var x = side * 1.5f;
        float z0 = 0f, z1 = Len;          // the end walls (x +-1.58) close the corners; no overlap with the L0 face
        var cuts = doors.Select(d => (d - .55f, d + .55f)).OrderBy(c => c.Item1).ToList();
        var cur = z0;
        void Span(float a, float b)
        {
            if (b - a < .005f) return;
            Box(corridor, "block lower", new Vector3(x, .6f, (a + b) * .5f), new Vector3(.16f, 1.2f, b - a), blockLo);
            Box(corridor, "block upper", new Vector3(x, (1.2f + H) * .5f, (a + b) * .5f), new Vector3(.16f, H - 1.2f, b - a), blockHi);
        }
        foreach (var (a, b) in cuts)
        {
            Span(cur, a);
            Box(corridor, "door header", new Vector3(x, (2.15f + H) * .5f, (a + b) * .5f), new Vector3(.16f, H - 2.15f, b - a), blockHi);
            cur = b;
        }
        Span(cur, z1);
    }

    static void EndWall(float zWall, float zc, float halfGap, float gapH)
    {
        // corridor half: z in [zWall, zWall + .08] at the entrance, [26.92, 27] at the goal
        float x0 = -1.58f, x1 = 1.58f;
        foreach (var (a, b) in new[] { (x0, -halfGap), (halfGap, x1) })
        {
            Box(corridor, "end block lower", new Vector3((a + b) * .5f, .6f, zc), new Vector3(b - a, 1.2f, .08f), blockLo);
            Box(corridor, "end block upper", new Vector3((a + b) * .5f, (1.2f + H) * .5f, zc), new Vector3(b - a, H - 1.2f, .08f), blockHi);
        }
        Box(corridor, "end header", new Vector3(0f, (gapH + H) * .5f, zc), new Vector3(halfGap * 2f, H - gapH, .08f), gapH < 1.2f ? blockLo : blockHi);
    }

    static Mesh ConeMesh(float length, float r0, float slope, int seg)
    {
        var m = new Mesh { name = "Run beam cone" };
        var v = new List<Vector3>(); var n = new List<Vector3>(); var uv = new List<Vector2>(); var tri = new List<int>();
        const int rings = 6;
        for (var j = 0; j <= rings; j++)
        {
            var s = j / (float)rings; var zc = s * length; var r = r0 + slope * zc;
            for (var i = 0; i <= seg; i++)
            {
                var a = i / (float)seg * Mathf.PI * 2f;
                var d = new Vector3(Mathf.Cos(a), Mathf.Sin(a), 0f);
                v.Add(d * r + Vector3.forward * zc);
                n.Add((d - Vector3.forward * slope).normalized);
                uv.Add(new Vector2(i / (float)seg, s));
            }
        }
        for (var j = 0; j < rings; j++)
            for (var i = 0; i < seg; i++)
            {
                var a = j * (seg + 1) + i; var b = a + seg + 1;
                tri.AddRange(new[] { a, b, a + 1, a + 1, b, b + 1 });
            }
        m.SetVertices(v); m.SetNormals(n); m.SetUVs(0, uv); m.SetTriangles(tri, 0); m.RecalculateBounds();
        return m;
    }

"the game's Hunter (FrontRooms3DGame.CreateHunter, prototype FrontRoomsRelayRig, run pose)" : "brief proxy (capsule r .24 top 1.73, shoulders .78 x .24 x .30 at 1.72, head 0.28 at 1.60 +.12)") + "; bounds top " + b.max.y.ToString("0.00") + " m, width " + b.size.x.ToString("0.00"));
        relay.gameObject.SetActive(false);
    }

    static void Save(RenderTexture rt, string path)
    {
        var prev = RenderTexture.active;
        RenderTexture.active = rt;
        var image = new Texture2D(rt.width, rt.height, TextureFormat.RGB24, false);
        image.ReadPixels(new Rect(0, 0, rt.width, rt.height), 0, 0);
        image.Apply();
        RenderTexture.active = prev;
        File.WriteAllBytes(path, image.EncodeToPNG());
        UnityEngine.Object.DestroyImmediate(image);
    }
