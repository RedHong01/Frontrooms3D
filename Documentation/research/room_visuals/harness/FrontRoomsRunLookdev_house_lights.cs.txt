/// <summary>
/// CLONE-ONLY pre-render of Run! direction C · HOUSE LIGHTS ("the doors go"),
/// Documentation/research/room_visuals/10_run_directions.md §1, §4, §7.
/// Builds the shared layout (Level 0 anteroom, a 1 x 9 cell corridor, the RN-X
/// goal door and a lit stub beyond it) with C's finishes, strips, held door
/// pairs, flashers and props, drives it with FrontRoomsRunLookdevRig.Evaluate(t)
/// and renders the §7.4 shots plus ID masks for the §7.5 measurements to

    public static void RunBatch()
    {
        var args = Environment.GetCommandLineArgs();
        string shots = null;
        var masks = true;
        for (var i = 0; i < args.Length; i++)
        {
            if (args[i] == "-runQuick") quick = true;
            if (args[i] == "-runDump") dump = true;
            if (args[i] == "-runShots" && i + 1 < args.Length) shots = args[i + 1];
            if (args[i] == "-runNoMasks") masks = false;
        }
        Run(shots, masks);
    }

    static float Env(string key, float fallback)
    {
        var v = Environment.GetEnvironmentVariable(key);
        return v != null && float.TryParse(v, System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture, out var f) ? f : fallback;
    }

    [MenuItem("FrontRooms/Rendering/Run C look-dev (clone only)")]
    public static void Capture() => Run(null, true);

    // ====================================================================== assets
    static Material SurfaceMat(string name, string tex, Vector2 tile, Color tint, float smooth = 1f, float bump = 1f,
        float macroTone = .2f, float macroDirt = .15f, float floorGrime = 0f, float ceilingGrime = 0f, bool meshUV = false,
        string emission = null, Color? emissionColor = null, Vector4? st = null, float stain = 0f)
    {
        var path = SurfDir + "/" + name + ".mat";
        var shader = Shader.Find("FrontRooms/Surface");
        var m = AssetDatabase.LoadAssetAtPath<Material>(path);
        if (m == null) { m = new Material(shader) { name = name }; AssetDatabase.CreateAsset(m, path); }
        m.shader = shader;
        Texture2D T(string stem) => stem == null ? null : AssetDatabase.LoadAssetAtPath<Texture2D>(SurfDir + "/Textures/" + stem + ".png");
        m.SetTexture("_BaseMap", T(tex == null ? null : tex + "_A"));
        m.SetTexture("_BumpMap", T(tex == null ? null : tex + "_N"));
        m.SetTexture("_MaskMap", T(tex == null ? null : tex + "_S"));
        m.SetColor("_BaseColor", tint);
        var s = st ?? new Vector4(1, 1, 0, 0);
        m.SetTextureScale("_BaseMap", new Vector2(s.x, s.y));
        m.SetTextureOffset("_BaseMap", new Vector2(s.z, s.w));
        m.SetVector("_TileSize", new Vector4(tile.x, tile.y, 0, 0));
        m.SetFloat("_BumpScale", bump);
        m.SetFloat("_Smoothness", smooth);
        m.SetFloat("_Metallic", 0f);
        m.SetFloat("_OcclusionStrength", .8f);
        m.SetTexture("_MacroMap", T("MacroWear_M"));
        m.SetFloat("_MacroTone", macroTone);
        m.SetFloat("_MacroDirt", macroDirt);
        m.SetFloat("_StainStrength", stain);
        m.SetFloat("_WetStrength", 0f);
        m.SetFloat("_FloorGrime", floorGrime);
        m.SetFloat("_CeilingGrime", ceilingGrime);
        m.SetFloat("_CeilingHeight", H);
        m.SetFloat("_MeshUV", meshUV ? 1f : 0f);
        if (meshUV) m.EnableKeyword("_FR_MESH_UV"); else m.DisableKeyword("_FR_MESH_UV");
        if (Env("RUNC_SOFTAO", 1f) > .5f) m.EnableKeyword("_FR_SOFT_AO"); else m.DisableKeyword("_FR_SOFT_AO");
        if (emission != null)
        {
            m.SetTexture("_EmissionMap", emission.Length == 0 ? null : T(emission));
            m.SetColor("_EmissionColor", emissionColor ?? Color.white);
            m.SetFloat("_UseEmission", 1f);
            m.EnableKeyword("_EMISSION");
        }
        else { m.SetFloat("_UseEmission", 0f); m.DisableKeyword("_EMISSION"); }
        EditorUtility.SetDirty(m);
        return m;
    }

    static Material LitMat(string name, Color color, float smooth, float metallic = 0f, bool emissive = false, Color? emission = null,
        bool asset = true, bool transparent = false, Texture2D tex = null, Vector2? texScale = null, bool alphaClip = false, bool twoSided = false)
    {
        var shader = Shader.Find("Universal Render Pipeline/Lit");
        Material m = null;
        var path = SurfDir + "/" + name + ".mat";
        if (asset)
        {
            m = AssetDatabase.LoadAssetAtPath<Material>(path);
            if (m == null) { m = new Material(shader) { name = name }; AssetDatabase.CreateAsset(m, path); }
        }
        else m = new Material(shader) { name = name, hideFlags = HideFlags.DontUnloadUnusedAsset };  // NewScene(Single) would destroy it
        m.shader = shader;
        m.SetColor("_BaseColor", color);
        m.SetFloat("_Smoothness", smooth);
        m.SetFloat("_Metallic", metallic);
        if (tex != null)
        {
            m.SetTexture("_BaseMap", tex);
            m.SetTextureScale("_BaseMap", texScale ?? Vector2.one);
        }
        if (emissive)
        {
            m.EnableKeyword("_EMISSION");
            m.SetColor("_EmissionColor", emission ?? Color.black);
            m.globalIlluminationFlags = MaterialGlobalIlluminationFlags.None;
        }
        else m.DisableKeyword("_EMISSION");
        if (transparent)
        {
            m.SetFloat("_Surface", 1f);
            m.SetFloat("_Blend", 0f);
            m.SetFloat("_SrcBlend", (float)BlendMode.SrcAlpha);
            m.SetFloat("_DstBlend", (float)BlendMode.OneMinusSrcAlpha);
            m.SetFloat("_SrcBlendAlpha", (float)BlendMode.One);
            m.SetFloat("_DstBlendAlpha", (float)BlendMode.OneMinusSrcAlpha);
            m.SetFloat("_ZWrite", 0f);
            m.SetOverrideTag("RenderType", "Transparent");
            m.EnableKeyword("_SURFACE_TYPE_TRANSPARENT");
            m.renderQueue = (int)RenderQueue.Transparent;
        }
        if (alphaClip)
        {
            m.SetFloat("_AlphaClip", 1f);
            m.SetFloat("_Cutoff", .5f);
            m.EnableKeyword("_ALPHATEST_ON");
            m.renderQueue = (int)RenderQueue.AlphaTest;
        }
        if (twoSided) m.SetFloat("_Cull", 0f);
        if (asset) EditorUtility.SetDirty(m);
        return m;
    }

    static void PrepareAssets()
    {
        AssetDatabase.Refresh();
        if (!quick)
            foreach (var stem in new[] { "Run_GlazedTile", "Run_Plaster", "Run_CeilingGyp", "Run_DoorLaminate" })
                foreach (var suf in new[] { "_A", "_N", "_S" })
                    AssetDatabase.ImportAsset(SurfDir + "/Textures/" + stem + suf + ".png", ImportAssetOptions.ForceUpdate);

        var vctTint = new Color(217f / 218f, 213f / 215f, 200f / 204f);
        var charcoal = new Color(.34f, .335f, .333f);
        double Off(double x0) { var o = -x0 / P4; o -= Math.Floor(o * 4.0) / 4.0; return o; }
        double OffV(double z0) { var o = z0 / P4; o -= Math.Floor(o * 4.0) / 4.0; return o; }
        var vz = (float)OffV(RootPos.z);
        M["tile"] = SurfaceMat("Run_C_GlazedTile", "Run_GlazedTile", new Vector2(TileW, TileH), Color.white, smooth: 1f, bump: 1f, macroTone: .12f, macroDirt: .18f, floorGrime: .55f);
        M["trim"] = SurfaceMat("Run_C_GlazedTrim", "Run_GlazedTile", new Vector2(TileW, TileH), Color.white, smooth: 1f, bump: 1f, macroTone: .12f, macroDirt: .18f, meshUV: true);
        // r5: no ceiling-grime band on the plaster (a scrubbed link corridor, not Level 0); with SSAO it drew a dark crown line.
        M["plaster"] = SurfaceMat("Run_C_Plaster", "Run_Plaster", Vector2.one, Color.white, smooth: 1f, bump: .8f, macroTone: .22f, macroDirt: .14f, ceilingGrime: Env("RUNC_CGRIME", 0f));
        M["ceiling"] = SurfaceMat("Run_C_Ceiling", "Run_CeilingGyp", new Vector2(P4, P4), Color.white, smooth: 1f, bump: .8f, macroTone: .2f, macroDirt: .1f, stain: .08f);
        // Waxed VCT (Red, 2026-10-03 14:1x: glossy, not matte). Run_VCT_S R averages 0.525 (seams lower),
        // so the scales give a burnished centre lane of ~0.82, bands ~0.80 and wall-side strips ~0.74.
        var waxLane = Env("RUNC_WAX", 1.56f);
        M["vct"] = SurfaceMat("Run_C_VCT", "Run_VCT", new Vector2(P4, P4), vctTint, smooth: waxLane, macroTone: .18f, macroDirt: .3f, st: new Vector4(1, 1, (float)Off(RootPos.x), vz));
        M["vctEdge"] = SurfaceMat("Run_C_VCTEdge", "Run_VCT", new Vector2(P4, P4), vctTint, smooth: waxLane * .905f, macroTone: .18f, macroDirt: .38f, st: new Vector4(1, 1, (float)Off(RootPos.x), vz));
        M["bandR"] = SurfaceMat("Run_C_VCTBand_R", "Run_VCT", new Vector2(P4, P4), charcoal, smooth: waxLane * .975f, macroTone: .1f, macroDirt: .2f, st: new Vector4(1, 1, (float)Off(RootPos.x + .82), vz));
        M["bandL"] = SurfaceMat("Run_C_VCTBand_L", "Run_VCT", new Vector2(P4, P4), charcoal, smooth: waxLane * .975f, macroTone: .1f, macroDirt: .2f, st: new Vector4(1, 1, (float)Off(RootPos.x - 1.12), vz));
        M["Run_Enamel"] = SurfaceMat("Run_Enamel", "PaintedMetal", Vector2.one, new Color(1.07f, 1.09f, 1.15f), smooth: 1.35f, bump: .35f, macroTone: .05f, macroDirt: .05f, meshUV: true, emission: "", emissionColor: Color.black);   // emission = light from its own bare tubes (rig)
        // r4: hammertone grey at ~.30 linear (sRGB tint .72 x PaintedMetal .83); the r1-r3 tint .46 gave ~.12, which read black in every shot.
        M["Run_MagGrey"] = SurfaceMat("Run_MagGrey", "PaintedMetal", new Vector2(.25f, .25f), new Color(.72f, .73f, .73f), smooth: .95f, bump: 1.6f, macroTone: .05f, macroDirt: .1f, meshUV: true);

        M["Run_ExitFace"] = SurfaceMat("Run_ExitFace", "Run_C_ExitFace", Vector2.one, Color.white, smooth: Env("RUNC_EXITSMOOTH", .25f), macroTone: 0f, macroDirt: 0f, meshUV: true, emission: "Run_C_ExitFace_E", emissionColor: ExitGlow * Env("RUNC_EXITEM", 1.6f));   // r8: red-held glow (as A's D3): white x the red letters read salmon through the cool post

        M["Run_TubeGlass"] = SurfaceMat("Run_TubeGlass", null, Vector2.one, new Color(.93f, .94f, .95f), smooth: .9f, macroTone: 0f, macroDirt: 0f, meshUV: true, emission: "", emissionColor: Color.black);
        M["tubeOff"] = SurfaceMat("Run_TubeGlassOff", null, Vector2.one, new Color(.78f, .79f, .78f), smooth: .88f, macroTone: 0f, macroDirt: 0f, meshUV: true);
        M["Run_TubeEnd"] = SurfaceMat("Run_TubeEnd", null, Vector2.one, new Color(.30f, .29f, .27f), smooth: .85f, macroTone: 0f, macroDirt: 0f, meshUV: true, emission: "", emissionColor: Color.black);
        M["tubeEndDead"] = SurfaceMat("Run_TubeEndDead", null, Vector2.one, new Color(.30f, .29f, .27f), smooth: .85f, macroTone: 0f, macroDirt: 0f, meshUV: true, emission: "", emissionColor: DeadGlow * Env("RUNC_DEADEM", .7f));   // dim orange cathode glow (2.2 read yellow through ACES; (1, .36, .08) x 1.1 read peach after the -22 saturation; E2 swept x 1.3 / .7 / .35 on (1, .28, .05): .7)
        M["Run_FlasherDome"] = SurfaceMat("Run_FlasherDome", null, Vector2.one, new Color(.50f, .035f, .025f), smooth: .92f, macroTone: 0f, macroDirt: 0f, meshUV: true, emission: "", emissionColor: Color.black);
        // Safety orange, not fire red: (.78, .30, .07) read red through the cool post (-22 sat, -10 temp) and broke
        // C's palette rule (red only in the flashers and the pull station).
        M["Run_Orange"] = SurfaceMat("Run_Orange", null, Vector2.one, new Color(.88f, .45f, .10f), smooth: .45f, macroTone: .05f, macroDirt: .1f, meshUV: true);
        // r6: the lites use a copy of the game's own window glass (Glass_Window, FrontRooms/Glass: face-on alpha .11,
        // pane F0 .08, grime maps) instead of the brief's URP Lit stand-in (.80/.85/.86, alpha .25). The stand-in's 25 %
        // lit white veil lifted the Relay's dark body behind the z18 lites to the corridor's own value (r3: 1.3 : 1).
        // RUNC_BRIEFGLASS=1 restores the stand-in. Interior door lites are cleaner than exterior windows: less dust and smear.
        var gw = Env("RUNC_BRIEFGLASS", 0f) < .5f ? FrontRoomsSurfaces.TryGet("Glass_Window") : null;
        if (gw != null)
        {
            var gm = new Material(gw) { name = "RunC lite glass (Glass_Window copy)", hideFlags = HideFlags.DontUnloadUnusedAsset };

            if (gm.HasProperty("_Dust")) gm.SetFloat("_Dust", .45f);
            if (gm.HasProperty("_Smear")) gm.SetFloat("_Smear", .5f);
            if (gm.HasProperty("_Prints")) gm.SetFloat("_Prints", .8f);
            M["glass"] = gm;
            glassKind = "game window glass (copy of Glass_Window, " + gw.shader.name + ", dust .45, smear .5, prints .8, RT receive off)";
        }
        else
        {
            M["glass"] = LitMat("RunC wired glass", new Color(.80f, .85f, .86f, .25f), .9f, asset: false, transparent: true);
            glassKind = "the brief's stand-in (URP Lit transparent .80/.85/.86, alpha .25, smoothness .9)";
        }
        M["wire"] = LitMat("RunC wire grid", new Color(1f, 1f, 1f, 1f), .5f, metallic: .6f, asset: false,
            tex: AssetDatabase.LoadAssetAtPath<Texture2D>(SurfDir + "/Textures/Run_C_WireGrid_A.png"), texScale: new Vector2(5f, 15f), alphaClip: true, twoSided: true);

        M["relayHead"] = LitMat("RunC relay head", HexColor("D8D4C8"), .35f, asset: false);
        M["aluminium"] = FrontRoomsSurfaces.Get("Prop_Aluminium");
        AssetDatabase.SaveAssets();

        // Post: only what C changes over the global film look (§4.1).
        var profile = AssetDatabase.LoadAssetAtPath<VolumeProfile>(PostPath);
        if (profile == null) { profile = ScriptableObject.CreateInstance<VolumeProfile>(); AssetDatabase.CreateAsset(profile, PostPath); }
        var color = Get<ColorAdjustments>(profile);
        color.postExposure.Override(.25f);
        color.contrast.Override(10f);
        color.saturation.Override(-22f);
        var white = Get<WhiteBalance>(profile);
        white.temperature.Override(-10f);
        white.tint.Override(4f);
        var lgg = Get<LiftGammaGain>(profile);
        lgg.lift.Override(new Vector4(1f, 1f, 1f, 0f));
        var smh = Get<ShadowsMidtonesHighlights>(profile);
        smh.highlights.Override(new Vector4(1f, 1f, 1f, -.05f));
        var bloom = Get<Bloom>(profile);
        bloom.threshold.Override(1.1f);
        bloom.intensity.Override(.35f);
        var grain = Get<FilmGrain>(profile);
        grain.intensity.Override(.20f);
        EditorUtility.SetDirty(profile);
        AssetDatabase.SaveAssets();

        // Re-import the Run kit so slot remaps pick up the materials above.
        if (!quick) foreach (var path in Directory.GetFiles(ModelDir, "Kit_*.fbx"))
        {
            var n = Path.GetFileNameWithoutExtension(path);
            if (KitNames.Contains(n)) AssetDatabase.ImportAsset(path.Replace('\\', '/'), ImportAssetOptions.ForceUpdate);
        }
    }

    static T Get<T>(VolumeProfile profile) where T : VolumeComponent
    {
        if (!profile.TryGet<T>(out var c))
        {
            c = profile.Add<T>(true);
            c.name = typeof(T).Name;
            AssetDatabase.AddObjectToAsset(c, profile);
        }
        c.active = true;
        return c;
    }

    static void Run(string onlyShots, bool masks)
    {
        log.Clear();
        // New scene FIRST: NewScene(Single) unloads in-memory materials made before it (they rendered magenta).
        var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
        EditorSceneManager.SetActiveScene(scene);
        PrepareAssets();
        root = new GameObject("RUN C LOOKDEV").transform;
        root.position = RootPos;
        FrontRoomsLook.ApplyAmbient();
        // As the map does in a Level 0 zone (main G6): the anteroom and the stub reflect the Level 0 cube.
        // Inside the corridor the three box probes below win (blending on in the clone's URP asset).
        FrontRoomsLook.SetZoneReflection(FrontRoomsLook.ReflectionZone.Level0, 0f);
        Log("zone reflection: Level0 cube (" + FrontRoomsZoneReflection.Mode + "), intensity " + RenderSettings.reflectionIntensity.ToString("0.000") + " (slider)");
        FrontRoomsPostStack.Ensure(root);
        groups.Clear(); relayRenderers.Clear(); glassRenderers.Clear(); allLights.Clear();
        var rigGo = new GameObject("rig");
        rigGo.transform.SetParent(root, false);
        rig = rigGo.AddComponent<FrontRoomsRunLookdevRig>();
        rig.CaptureAmbient();
        rig.stripIntensity = Env("RUNC_STRIP", rig.stripIntensity);
        rig.bounceUp = Env("RUNC_BOUNCE_UP", rig.bounceUp);
        rig.bounceAll = Env("RUNC_BOUNCE_ALL", rig.bounceAll);
        rig.armedLevel = Env("RUNC_ARMED", rig.armedLevel);
        SpotTilt = Env("RUNC_TILT", 0f);
        SpotY = Env("RUNC_SPOTY", 2.86f);
        SpotOuter = Env("RUNC_OUTER", 179f);
        SpotInner = Env("RUNC_INNER", 170f);
        ProbeRes = (int)Env("RUNC_PROBERES", 512f);
        rig.flasherEmission = Env("RUNC_FLASH", rig.flasherEmission);
        // The brief's flasher point (0.3) was set against strip spots of 4.5. The strips were tuned down to 1.3 to hit
        // the §4.1 Y targets, so the point keeps the brief's ratio (0.3 x 1.3 / 4.5 = 0.087). At 0.3 its 0.8 m disc
        // on the white wall was brighter than the dome itself, and the flashing red area grew 3x.
        rig.flasherLight = Env("RUNC_FLASHLIGHT", .3f * rig.stripIntensity / 4.5f);
        var rd = AssetDatabase.LoadAssetAtPath<UniversalRendererData>("Assets/Settings/FrontRooms_URP_Renderer.asset");
        if (rd != null)
            foreach (var f in rd.rendererFeatures)
                if (f != null && f.name.Contains("Ambient Occlusion"))
                {
                    var on = Env("RUNC_NOSSAO", 0f) < .5f;
                    f.SetActive(on);
                    Log("SSAO feature " + (on ? "on (game setting)" : "OFF (diagnostic)"));
                }
        Log("tuning: strip intensity " + rig.stripIntensity + ", bounce up " + rig.bounceUp + ", bounce all " + rig.bounceAll + ", armed level " + rig.armedLevel + ", spot tilt " + SpotTilt + " deg, spot y " + SpotY + ", cone " + SpotOuter + "/" + SpotInner + " deg, flasher emission " + rig.flasherEmission + ", flasher point " + rig.flasherLight.ToString("0.000") + ", probe res " + ProbeRes);
        Log("ambient probe L0 r=" + RenderSettings.ambientProbe[0, 0].ToString("0.000") + " (non-zero = trilight SH present)");
        Log("lite glass: " + glassKind);

        BuildAnteroom();
        BuildCorridor();
        BuildPairs();
        BuildGoal();
        BuildStub();
        BuildStrips();
        BuildProps();
        BuildRelay();
        BuildProbes();

        var vol = FrontRoomsPostStack.EnsureZoneVolume(root, "Run_C", new Bounds(new Vector3(0f, H * .5f, 13.5f), new Vector3(3f, H, 27f)), 1f);
        if (vol != null) { vol.priority = 2f; Log("local post FrontRoomsPost_Run_C, priority 2, blend 1.0 m"); }
        else Log("WARNING: FrontRoomsPost_Run_C not found");

        var shots = Shots();
        // RUNC_SWEEP = "1.2,1.5,...": render the selected shots once per strip intensity (suffix _s1p2 ...),
        // so one Unity launch tunes the level. Masks only on the first pass (geometry does not change).
        var sweep = Environment.GetEnvironmentVariable("RUNC_SWEEP");
        if (!string.IsNullOrEmpty(sweep))
        {
            var first = true;
            foreach (var v in sweep.Split(','))
            {
                var f = float.Parse(v, System.Globalization.CultureInfo.InvariantCulture);
                rig.stripIntensity = f;
                rig.flasherLight = .3f * f / 4.5f;
                variantSuffix = "_s" + v.Replace(".", "p");
                bakedKey = null;
                Log("sweep pass: strip intensity " + f);
                Render(shots, onlyShots, masks && first);
                first = false;
            }
            variantSuffix = "";
        }
        else if (!string.IsNullOrEmpty(Environment.GetEnvironmentVariable("RUNC_SIGNSWEEP")))
        {
            // RUNC_SIGNSWEEP = "exit:dead,exit:dead,...": the EXIT face glow and the dead tube's cathode glow per pass
            foreach (var pair in Environment.GetEnvironmentVariable("RUNC_SIGNSWEEP").Split(','))
            {
                var kv = pair.Split(':');
                var ex = float.Parse(kv[0], System.Globalization.CultureInfo.InvariantCulture);
                var de = float.Parse(kv[1], System.Globalization.CultureInfo.InvariantCulture);
                M["Run_ExitFace"].SetColor("_EmissionColor", ExitGlow * ex);
                M["tubeEndDead"].SetColor("_EmissionColor", DeadGlow * de);
                variantSuffix = "_e" + kv[0].Replace(".", "p") + "_d" + kv[1].Replace(".", "p");
                bakedKey = null;
                Log("sign sweep pass: exit glow " + ex + ", dead-end glow " + de);
                Render(shots, onlyShots, false);
            }
            variantSuffix = "";
        }
        else Render(shots, onlyShots, masks);
        WritePhotosafety();
        WriteCosts();
        var outDir = OutDir();
        File.WriteAllLines(Path.Combine(outDir, "lookdev_log.txt"), log);
    }

    static GameObject Kit(string name, Transform parent, Vector3 pos, Quaternion rot, uint layers, bool corridor = true, Vector3? scale = null)
    {
        var model = AssetDatabase.LoadAssetAtPath<GameObject>(ModelDir + name + ".fbx");
        if (model == null) { Log("MISSING kit model " + name); return new GameObject(name + " (missing)"); }
        var go = (GameObject)UnityEngine.Object.Instantiate(model);
        go.name = name;
        go.transform.SetParent(parent, false);
        go.transform.localPosition = pos;
        go.transform.localRotation = rot;
        if (scale.HasValue) go.transform.localScale = scale.Value;
        foreach (var r in go.GetComponentsInChildren<Renderer>(true))
        {
            var mats = r.sharedMaterials;
            for (var i = 0; i < mats.Length; i++)
            {
                if (mats[i] == null) continue;
                var key = mats[i].name.Replace(" (Instance)", "");
                if (M.TryGetValue(key, out var mm)) mats[i] = mm;
                else
                {
                    var fs = FrontRoomsSurfaces.TryGet(key);
                    if (fs != null) mats[i] = fs;
                }
            }
            r.sharedMaterials = mats;
            r.renderingLayerMask = layers;
            r.shadowCastingMode = ShadowCastingMode.On;
            if (corridor) rig.corridorRenderers.Add(r);
        }
        var lod = go.GetComponent<LODGroup>();
        if (lod != null) lod.ForceLOD(0);
        return go;
    }

    /// <summary>A glazed-tile wainscot on a wall face: field slab (0.02 proud), bullnose cap and cove base.
    /// start = the run's start on the wall face at floor level; along = run direction; n = the face's outward normal.</summary>
    static void WainscotRun(Transform parent, Vector3 start, Vector3 along, float length, Vector3 n)
    {
        var mid = start + along * (length * .5f) + n * (TileProud * .5f);
        var size = new Vector3(Mathf.Abs(along.x) * length + Mathf.Abs(n.x) * TileProud, WainTop - CapR, Mathf.Abs(along.z) * length + Mathf.Abs(n.z) * TileProud);
        Box(parent, "tile wainscot", mid + Vector3.up * ((WainTop - CapR) * .5f), size, M["tile"], LCorr, true);
        // bullnose: quarter round from the tile face (n * 0.02) up to the plaster face at the cap height
        var cap = new List<Vector2>();
        for (var i = 0; i <= 8; i++)
        {
            var a = Mathf.PI * .5f * i / 8f;
            cap.Add(new Vector2(CapR * Mathf.Cos(a), WainTop - CapR + CapR * Mathf.Sin(a)));
        }
        Trim(parent, start, along, length, n, cap, WainTop - CapR);
        // cove: concave fillet from the tile face down to the floor
        var cove = new List<Vector2>();
        for (var i = 0; i <= 8; i++)
        {
            var a = Mathf.PI * .5f * i / 8f;
            cove.Add(new Vector2(TileProud + CoveR - CoveR * Mathf.Cos(a), CoveR - CoveR * Mathf.Sin(a)));
        }
        Trim(parent, start, along, length, n, cove, CoveR);
    }

    /// <summary>Extrude a profile (x = distance out from the wall face along n, y = height) along a straight run.
    /// Mesh UVs reproduce the wall's world projection (u along the wall, v = height), so joints line up.</summary>
    static void Trim(Transform parent, Vector3 start, Vector3 along, float length, Vector3 n, List<Vector2> prof, float v0)
    {
        var verts = new List<Vector3>(); var norms = new List<Vector3>(); var uvs = new List<Vector2>(); var tris = new List<int>();
        var t = Vector3.Cross(Vector3.up, n).normalized;  // the Surface shader's wall tangent
        var arc = 0f;
        for (var i = 0; i < prof.Count; i++)
        {
            if (i > 0) arc += (prof[i] - prof[i - 1]).magnitude;
            Vector2 tan;
            if (i == 0) tan = prof[1] - prof[0];
            else if (i == prof.Count - 1) tan = prof[i] - prof[i - 1];
            else tan = prof[i + 1] - prof[i - 1];
            tan.Normalize();
            var pn = new Vector2(tan.y, -tan.x);           // rotate -90
            var nn = (n * pn.x + Vector3.up * pn.y).normalized;
            // make the normal face away from the wall (into the corridor) or up
            if (Vector3.Dot(nn, n) + nn.y < 0f) nn = -nn;
            for (var e = 0; e < 2; e++)
            {
                var p = start + along * (e * length) + n * prof[i].x + Vector3.up * prof[i].y;
                verts.Add(p);
                norms.Add(nn);
                var w = root.TransformPoint(p);
                uvs.Add(new Vector2(Vector3.Dot(w, t), v0 + (prof[i].y >= v0 ? arc : -arc)));
            }
        }
        for (var i = 0; i < prof.Count - 1; i++)
        {
            int a = i * 2, b = i * 2 + 1, c = i * 2 + 2, d = i * 2 + 3;
            tris.AddRange(new[] { a, c, b, b, c, d });
        }
        var mesh = new Mesh { name = "trim" };
        mesh.SetVertices(verts); mesh.SetNormals(norms); mesh.SetUVs(0, uvs); mesh.SetTriangles(tris, 0);
        // wind toward the normal
        var tr = mesh.triangles;
        var first = (verts[tr[1]] - verts[tr[0]]); var second = (verts[tr[2]] - verts[tr[0]]);
        if (Vector3.Dot(Vector3.Cross(first, second), norms[tr[0]]) < 0f)
        {
            for (var k = 0; k < tr.Length; k += 3) { var x = tr[k + 1]; tr[k + 1] = tr[k + 2]; tr[k + 2] = x; }
            mesh.triangles = tr;
        }
        mesh.RecalculateTangents();
        mesh.RecalculateBounds();
        var go = new GameObject("tile trim");
        go.transform.SetParent(parent, false);
        go.AddComponent<MeshFilter>().sharedMesh = mesh;
        var r = go.AddComponent<MeshRenderer>();
        r.sharedMaterial = M["trim"];
        r.renderingLayerMask = LCorr;
        rig.corridorRenderers.Add(r);
    }

    static GameObject Plate(Transform parent, string name, Vector3 centre, Vector2 size, Vector3 facing, Material face, uint layers, float depth = .004f)
    {
        var go = new GameObject(name);
        go.transform.SetParent(parent, false);
        var right = Vector3.Cross(Vector3.up, -facing).normalized;
        var p = centre + facing * (depth + .0005f);
        var hw = size.x * .5f; var hh = size.y * .5f;
        var mesh = new Mesh { name = name };
        mesh.SetVertices(new List<Vector3> { p - right * hw - Vector3.up * hh, p + right * hw - Vector3.up * hh, p + right * hw + Vector3.up * hh, p - right * hw + Vector3.up * hh });
        mesh.SetNormals(new List<Vector3> { facing, facing, facing, facing });
        mesh.SetUVs(0, new List<Vector2> { new Vector2(0, 0), new Vector2(1, 0), new Vector2(1, 1), new Vector2(0, 1) });
        var tr = new[] { 0, 2, 1, 0, 3, 2 };
        if (Vector3.Dot(Vector3.Cross(mesh.vertices[2] - mesh.vertices[0], mesh.vertices[1] - mesh.vertices[0]), facing) < 0f) tr = new[] { 0, 1, 2, 0, 2, 3 };
        mesh.SetTriangles(tr, 0);
        mesh.RecalculateTangents();
        mesh.RecalculateBounds();
        var q = new GameObject("face");
        q.transform.SetParent(go.transform, false);
        q.AddComponent<MeshFilter>().sharedMesh = mesh;
        var r = q.AddComponent<MeshRenderer>();
        r.sharedMaterial = face;
        r.renderingLayerMask = layers;
        rig.corridorRenderers.Add(r);
        var back = Box(go.transform, "backing", centre + facing * (depth * .5f), new Vector3(Mathf.Abs(right.x) * size.x + Mathf.Abs(facing.x) * depth, size.y, Mathf.Abs(right.z) * size.x + Mathf.Abs(facing.z) * depth),
            FrontRoomsSurfaces.Get("Prop_PlasticWhite"), layers, true);
        back.transform.localScale += new Vector3(Mathf.Abs(right.x), 0f, Mathf.Abs(right.z)) * .006f + Vector3.up * .006f;
        return go;
    }

    // ---------------------------------------------------------------- 9 bare 8 ft strips, 3 per compartment
    static void BuildStrips()
    {
        var st = new GameObject("strips").transform;
        st.SetParent(root, false);
        rig.enamel = M["Run_Enamel"];
        rig.tubeOn = M["Run_TubeGlass"]; rig.tubeOff = M["tubeOff"]; rig.tubeEnd = M["Run_TubeEnd"]; rig.tubeEndDead = M["tubeEndDead"];
        var spot = 0;
        for (var i = 1; i <= 9; i++)
        {
            var comp = (i - 1) / 3; var k = (i - 1) % 3;
            var z = 9f * comp + 4.5f + (k - 1) * 2.44f;
            var fix = Kit("Kit_StripFixture_8ft", st, new Vector3(0f, H, z), Quaternion.Euler(0f, 90f, 0f), LCorr);
            fix.name = "strip " + i;
            // r4: the spots sit inside the channel line (y 2.86, between the tubes and the ceiling) so the walls are lit
            // right up to the ceiling as a bare tube lights them; at 2.76 the top 0.14 m of every wall fell outside the
            // cone and read as a dark band under the ceiling. The channel then must not shadow its own lamps.
            foreach (var r in fix.GetComponentsInChildren<Renderer>(true)) r.shadowCastingMode = ShadowCastingMode.Off;
            Group("strips", fix);
            var s = new FrontRoomsRunLookdevRig.Strip { number = i, z = z, economyOff = i % 2 == 0, deadTube = i == 7 };
            s.body = fix.GetComponentsInChildren<Renderer>(true);   // before the tubes are parented
            for (var t = 0; t < 2; t++)
            {
                var tube = Kit("Kit_T12Tube_8ft", fix.transform, new Vector3(0f, -.078f, t == 0 ? .0255f : -.0255f), Quaternion.identity, LCorr);
                foreach (var r in tube.GetComponentsInChildren<Renderer>()) r.shadowCastingMode = ShadowCastingMode.Off;
                s.tubes[t] = tube.GetComponentInChildren<Renderer>();
                Group("strips", tube);
            }
            for (var e = 0; e < 2; e++)
            {
                var shadow = spot % 3 == 0;
                // r2: a bare T12 strip radiates sideways as well as down, so the upper walls face the tubes from
                // 1.4 m and are the brightest walls in a real bare-strip corridor. r1 leaned 165/120 deg spots 30 deg
                // toward each wall; their cone edges then crossed the door leaves and cross walls as hard V wedges.
                // r2 points them straight down from just under the tubes with a near-hemisphere cone (179/140 deg):
                // walls are lit up to the tube line, the cone edge lies almost level with the source (no wedges),
                // and the ceiling above the tubes gets nothing direct (the channel shields it), only the bounce lobe.
                var l = Spot(st, "strip " + i + (e == 0 ? "a" : "b"), new Vector3(0f, SpotY, z + (e == 0 ? -.6f : .6f)), SpotOuter, SpotInner, FrontRoomsRunLookdevRig.StripColor, 4.5f, 9f, shadow, shadow ? (LCorr | LAnte) : LCorr);
                l.transform.localRotation = Quaternion.AngleAxis(e == 0 ? -SpotTilt : SpotTilt, Vector3.forward) * Quaternion.Euler(90f, 0f, 0f);
                s.lights[e] = l;
                spot++;
            }
            rig.strips.Add(s);
        }
    }

    static string bakedKey;

    static void BakeProbes(string key)
    {
        if (!useProbes || key == bakedKey) return;
        bakedKey = key;
        var sw = System.Diagnostics.Stopwatch.StartNew();
        for (var k = 0; k < probes.Count; k++)
        {
            var path = ProbeDir + "/RunC_Refl_c" + k + "_" + key + ".exr";
            var ok = false;
            try { ok = Lightmapping.BakeReflectionProbe(probes[k], path); }
            catch (Exception e) { Log("probe bake threw: " + e.Message); }
            if (!ok) { Log("probe bake FAILED " + path); continue; }
            var importer = AssetImporter.GetAtPath(path) as TextureImporter;
            if (importer != null)
            {
                importer.textureShape = TextureImporterShape.TextureCube;
                var settings = new TextureImporterSettings();
                importer.ReadTextureSettings(settings);
                settings.cubemapConvolution = TextureImporterCubemapConvolution.Specular;
                importer.SetTextureSettings(settings);
                importer.mipmapEnabled = true;
                importer.sRGBTexture = false;
                importer.filterMode = FilterMode.Trilinear;
                importer.wrapMode = TextureWrapMode.Clamp;
                importer.maxTextureSize = ProbeRes;
                importer.textureCompression = TextureImporterCompression.CompressedHQ;
                importer.SaveAndReimport();
            }
            probes[k].customBakedTexture = AssetDatabase.LoadAssetAtPath<Cubemap>(path);
        }
        Log("probes baked for state " + key + " in " + (sw.ElapsedMilliseconds / 1000f).ToString("0.0") + " s");
    }

    static void Render(List<Shot> shots, string onlyShots, bool masks)
    {
        var outDir = OutDir();
        var camGo = new GameObject("RUN C camera");
        var cam = camGo.AddComponent<Camera>();
        cam.nearClipPlane = .06f;
        cam.farClipPlane = 80f;
        cam.clearFlags = CameraClearFlags.SolidColor;
        cam.backgroundColor = HexColor("22231C");
        FrontRoomsPostStack.ConfigureCamera(cam);
        var rt = new RenderTexture(1920, 1080, 24, RenderTextureFormat.ARGB32) { antiAliasing = 4 };
        cam.targetTexture = rt;
        var filter = string.IsNullOrEmpty(onlyShots) ? null : new HashSet<string>(onlyShots.Split(','));
        foreach (var s in shots)
        {
            if (filter != null && !filter.Contains(s.name)) continue;
            relay.gameObject.SetActive(s.relay > 0);
            if (s.relay > 0)
            {
                relay.localPosition = new Vector3(s.relay == 2 ? -.65f : 0f, 0f, 11.5f);
                relay.localRotation = Quaternion.identity; // faces +Z, toward the camera at 23.5
            }
            rig.flasherForceOff = s.flashOff;
            rig.pairAngleOverride[1] = s.push ? FrontRoomsRunLookdevRig.PushAngle(s.pushAge) : float.NaN;
            foreach (var g in glassRenderers) if (g != null) g.enabled = !s.noGlass;
            rig.Evaluate(s.t);
            BakeProbes("t" + s.t.ToString("+0.00;-0.00", System.Globalization.CultureInfo.InvariantCulture).Replace(".", "p") + (s.relay > 0 ? "_r" + s.relay : "") + (s.flashOff ? "_fo" : "") + (s.push ? "_push" : "") + (s.noGlass ? "_ng" : "") + variantSuffix);
            if (dump)
            {
                foreach (var r in UnityEngine.Object.FindObjectsByType<Renderer>(FindObjectsSortMode.None))
                {
                    if (!(r.name.Contains("glass") || r.name.Contains("wire") || r.name.Contains("Tube") || r.name.Contains("Flasher") || (r.transform.parent && r.transform.parent.name.Contains("Tube")))) continue;
                    var sb = new StringBuilder("DUMP " + s.name + " " + r.GetType().Name + " " + r.name + " / " + (r.transform.parent ? r.transform.parent.name : "") + " en " + r.enabled + " mats " + r.sharedMaterials.Length);
                    var mf = r.GetComponent<MeshFilter>();
                    if (mf && mf.sharedMesh) sb.Append(" submeshes " + mf.sharedMesh.subMeshCount);
                    foreach (var m in r.sharedMaterials)
                        sb.Append(" | " + (m == null ? "NULL" : m.name + " [" + (m.shader ? m.shader.name : "noshader") + " err " + (m.shader && ShaderUtil.ShaderHasError(m.shader)) + " passes " + m.passCount + " kw " + string.Join(" ", m.shaderKeywords) + "]"));
                    sb.Append(" hasMPB " + r.HasPropertyBlock());
                    Log(sb.ToString());
                }
                dump = false;
            }
            if (s.name == "dbg_amb0") { rig.ambientArmed = 0f; rig.Evaluate(s.t); }
            cam.fieldOfView = s.fov;
            var eye = root.TransformPoint(s.eye);
            cam.transform.SetPositionAndRotation(eye, Quaternion.LookRotation(root.TransformPoint(s.look) - eye, Vector3.up));
            cam.Render();
            cam.Render();
            Save(rt, Path.Combine(outDir, s.name + variantSuffix + ".png"));
            Log("shot " + s.name + variantSuffix + " t=" + s.t.ToString("0.00") + " eye " + s.eye + " look " + s.look + " fov " + s.fov + " lights on " + allLights.Count(l => l.enabled)
                + " (shadowed " + allLights.Count(l => l.enabled && l.shadows != LightShadows.None) + ") ambient: " + rig.ambientMode
                + " " + DrawEstimate(cam));
            if (s.name == "dbg_amb0") { rig.ambientArmed = 1f; rig.Evaluate(s.t); }
            if (masks && !string.IsNullOrEmpty(s.masks))
                foreach (var g in s.masks.Split(','))
                    RenderMask(cam, rt, g, Path.Combine(outDir, s.name + variantSuffix + "_mask_" + g + ".png"));
        }
        cam.targetTexture = null;
        rt.Release();
        relay.gameObject.SetActive(false);
        rig.flasherForceOff = false;
        rig.pairAngleOverride[1] = float.NaN;
        foreach (var g in glassRenderers) if (g != null) g.enabled = true;
    }

    /// <summary>UnityStats stays 0 in batch mode, so count what the frame submits: renderers whose bounds
    /// pass the camera frustum (one draw per submesh in the opaque/transparent pass; the depth-normals
    /// prepass that SSAO needs repeats the opaque ones), plus one shadow-caster draw per submesh inside the
    /// range of each shadowed light. An upper bound before SRP-batcher/instancing savings.</summary>
    static string DrawEstimate(Camera cam)
    {
        var planes = GeometryUtility.CalculateFrustumPlanes(cam);
        int visR = 0, visSub = 0, transSub = 0, shadowSub = 0; long visTris = 0;
        var all = UnityEngine.Object.FindObjectsByType<MeshRenderer>(FindObjectsSortMode.None);
        foreach (var r in all)
        {
            if (!r.enabled || !r.gameObject.activeInHierarchy) continue;
            var mf = r.GetComponent<MeshFilter>();
            if (mf == null || mf.sharedMesh == null) continue;
            if (!GeometryUtility.TestPlanesAABB(planes, r.bounds)) continue;
            visR++;
            for (var k = 0; k < mf.sharedMesh.subMeshCount; k++)
            {
                visSub++;
                visTris += (long)mf.sharedMesh.GetIndexCount(k) / 3;
                var m = k < r.sharedMaterials.Length ? r.sharedMaterials[k] : null;
                if (m != null && m.renderQueue >= (int)RenderQueue.Transparent) transSub++;
            }
        }
        var shadowViews = 0;
        foreach (var l in allLights)
        {
            if (!l.enabled || l.shadows == LightShadows.None) continue;
            shadowViews++;
            var sphere = new Bounds(l.transform.position, Vector3.one * (2f * l.range));
            foreach (var r in all)
            {
                if (!r.enabled || !r.gameObject.activeInHierarchy || r.shadowCastingMode == ShadowCastingMode.Off) continue;
                var mf = r.GetComponent<MeshFilter>();
                if (mf == null || mf.sharedMesh == null || !sphere.Intersects(r.bounds)) continue;
                shadowSub += mf.sharedMesh.subMeshCount;
            }
        }
        var opaque = visSub - transSub;
        return "draws est: visible renderers " + visR + ", submeshes " + visSub + " (transparent " + transSub + "), tris " + visTris
            + ", depth-normals prepass " + opaque + ", shadow views " + shadowViews + " with " + shadowSub + " caster draws; total ~" + (visSub + opaque + shadowSub);
    }

    static void RenderMask(Camera cam, RenderTexture rt, string group, string path)
    {
        if (!groups.TryGetValue(group, out var target)) { Log("no group " + group); return; }
        var black = new Material(Shader.Find("Universal Render Pipeline/Unlit")); black.SetColor("_BaseColor", Color.black);
        var white = new Material(Shader.Find("Universal Render Pipeline/Unlit")); white.SetColor("_BaseColor", Color.white);
        var targetSet = new HashSet<Renderer>(target);
        var saved = new Dictionary<Renderer, Material[]>();
        var hidden = new List<Renderer>();
        var forced = new List<Renderer>();
        foreach (var r in UnityEngine.Object.FindObjectsByType<Renderer>(FindObjectsSortMode.None))
        {
            saved[r] = r.sharedMaterials;
            var isGlass = glassRenderers.Contains(r) || r.name == "wire grid";
            if (isGlass && !targetSet.Contains(r)) { if (r.enabled) { r.enabled = false; hidden.Add(r); } continue; }
            var m = targetSet.Contains(r) ? white : black;
            r.sharedMaterials = Enumerable.Repeat(m, r.sharedMaterials.Length).ToArray();
            if (isGlass && !r.enabled) { r.enabled = true; forced.Add(r); }   // a no-glass shot still needs its lite apertures
        }
        var data = cam.GetUniversalAdditionalCameraData();
        var fog = RenderSettings.fog;
        RenderSettings.fog = false;
        data.renderPostProcessing = false;
        cam.backgroundColor = Color.black;
        cam.Render();
        Save(rt, path);
        data.renderPostProcessing = true;
        cam.backgroundColor = HexColor("22231C");
        RenderSettings.fog = fog;
        foreach (var kv in saved) kv.Key.sharedMaterials = kv.Value;
        foreach (var r in hidden) r.enabled = true;
        foreach (var r in forced) r.enabled = false;
        UnityEngine.Object.DestroyImmediate(black); UnityEngine.Object.DestroyImmediate(white);
    }

    static void Save(RenderTexture rt, string path)
    {
        var prev = RenderTexture.active;
        RenderTexture.active = rt;
        var img = new Texture2D(rt.width, rt.height, TextureFormat.RGB24, false);
        img.ReadPixels(new Rect(0, 0, rt.width, rt.height), 0, 0);
        img.Apply();
        RenderTexture.active = prev;
        File.WriteAllBytes(path, img.EncodeToPNG());
        UnityEngine.Object.DestroyImmediate(img);
    }

    static void WritePhotosafety()
    {
        var sb = new StringBuilder();
        sb.AppendLine("lamp\tchanges in 0..4.5 s\ttimes (s)\tmax changes in any 1 s window\tchanges in 0..2 s");
        foreach (var (name, changes) in rig.ChangeTable(4.5f))
        {
            var max = 0;
            foreach (var t0 in changes) max = Mathf.Max(max, changes.Count(t => t >= t0 - 1e-4f && t < t0 + 1f - 1e-3f));   // half-open 1 s window (1 ms sampling)
            sb.AppendLine(name + "\t" + changes.Count + "\t" + string.Join(", ", changes.Select(t => t.ToString("0.000"))) + "\t" + max + "\t" + changes.Count(t => t <= 2f));
        }
        var doors = new StringBuilder();
        foreach (var t in new[] { -1f, 0f, .5f, .9f, 1f, 1.4f, 2f, 3f, 3.5f, 4f })
            doors.Append("t=" + t.ToString("0.0") + " " + FrontRoomsRunLookdevRig.DoorAngle(t).ToString("0.0") + " deg; ");
        sb.AppendLine("door leaf angle: " + doors);
        File.WriteAllText(Path.Combine(OutDir(), "photosafety.txt"), sb.ToString());
        Log("photosafety table written");
    }

    static void WriteCosts()
    {
        var sb = new StringBuilder();
        var corridor = rig.corridorRenderers.Where(r => r != null).ToList();
        var tris = 0L; var subs = 0;
        var texs = new HashSet<Texture>(); var mats = new HashSet<Material>();
        foreach (var r in corridor)
        {
            var mf = r.GetComponent<MeshFilter>();
            if (mf != null && mf.sharedMesh != null) { tris += mf.sharedMesh.triangles.LongLength / 3; subs += mf.sharedMesh.subMeshCount; }
            foreach (var m in r.sharedMaterials)
            {
                if (m == null) continue;
                mats.Add(m);
                foreach (var id in m.GetTexturePropertyNameIDs()) { var t = m.GetTexture(id); if (t != null) texs.Add(t); }
            }
        }
        sb.AppendLine("corridor renderers " + corridor.Count + ", submeshes (draws before batching) " + subs + ", triangles " + tris);
        sb.AppendLine("materials " + mats.Count + ": " + string.Join(", ", mats.Select(m => m.name).OrderBy(n => n)));
        var bytes = 0L;
        foreach (var t in texs) bytes += UnityEngine.Profiling.Profiler.GetRuntimeMemorySizeLong(t);
        sb.AppendLine("textures " + texs.Count + " (" + (bytes / (1024f * 1024f)).ToString("0.0") + " MB runtime): " + string.Join(", ", texs.Select(t => t.name + " " + t.width + "x" + t.height).OrderBy(n => n)));
        foreach (var name in KitNames)
        {
            var info = AssetDatabase.LoadAssetAtPath<GameObject>(ModelDir + name + ".fbx");
            if (info == null) continue;
            var tc = info.GetComponentsInChildren<MeshFilter>().Sum(f => f.sharedMesh != null ? f.sharedMesh.triangles.Length / 3 : 0);
            sb.AppendLine("kit " + name + ": " + tc + " tris");
        }
        rig.Evaluate(-1f);
        sb.AppendLine("lights armed: " + allLights.Count(l => l.enabled) + " (shadowed " + allLights.Count(l => l.enabled && l.shadows != LightShadows.None) + ")");
        rig.Evaluate(3f);
        sb.AppendLine("lights tripped: " + allLights.Count(l => l.enabled) + " (shadowed " + allLights.Count(l => l.enabled && l.shadows != LightShadows.None) + ")");
        sb.AppendLine("of which anteroom 6 + stub 1 map lamps");
        File.WriteAllText(Path.Combine(OutDir(), "costs.txt"), sb.ToString());
        Log("costs written");
    }
