"""Armature (keeper) plate for Kit_MagHolder, on the leaf's pull face near
its top free corner. CLONE-ONLY look-dev. Zinc-plated mounting plate, a
swivel stud, and the 70 mm steel armature disc 20 mm off the leaf.
ORIGIN = the leaf face; the disc faces -Y (Unity +Z)."""
import run_c_common as rc  # noqa: F401

NAME = "Kit_MagArmature"


def build(kit):
    kit.box((0.060, 0.004, 0.060), (0, -0.002, 0), "Prop_Aluminium", bevel=0.0015, segments=2, name="mount plate")
    for sz in (-1, 1):
        kit.cylinder(0.0035, 0.002, (0, -0.0045, sz * 0.022), "Prop_Chrome", verts=10, rot=(90, 0, 0), bevel=0.0007, segments=1, name="screw")
    kit.cylinder(0.009, 0.010, (0, -0.009, 0), "Run_MagGrey", verts=16, rot=(90, 0, 0), bevel=0.002, name="swivel")
    kit.cylinder(0.035, 0.006, (0, -0.017, 0), "Prop_Aluminium", verts=32, rot=(90, 0, 0), bevel=0.0012, segments=2, name="armature disc")
    kit.no_collider()
