"""Pressed-steel double-rabbet frame, dark bronze, for Run! C. NOT AN ASSET.

Frame-local Unity metres: opening x -W/2 .. W/2 at the wall centre z = 0
(wall faces +-0.08), floor y = 0, head soffit y = 2.1. Leaves sit in the
swing-side rabbet (z 0.035 .. 0.08) against a 16 mm stop; frame faces are
2 in (0.0508) wide, 12 mm proud of the wall faces.
"""
import run_c_common as rc  # noqa: F401  (slots)

FACE = 0.0508
# Section (u into the opening from the soffit plane, z across the wall).
SECTION = [(-FACE, -0.092), (0.0, -0.092), (0.0, -0.055), (0.016, -0.055), (0.016, 0.032), (0.0, 0.032),
           (0.0, 0.092), (-FACE, 0.092), (-FACE, 0.080), (-0.040, 0.080), (-0.040, -0.080), (-FACE, -0.080)]
HEAD = 2.1


def frame(kit, W, silencers=True):
    top = HEAD + FACE
    # Jambs: outline in Blender X/Y, extruded up Z. Blender X = -x_u, Y = -z_u.
    left = [(W / 2 - u, -z) for u, z in SECTION]        # x_u = -W/2 + u
    right = [(-W / 2 + u, -z) for u, z in reversed(SECTION)]   # x_u = W/2 - u (mirrored, keep winding)
    kit.extrude(left, top, (0, 0, top / 2), "Prop_SteelBrown", plane="xy", bevel=0.0025, segments=2, name="jamb L")
    kit.extrude(right, top, (0, 0, top / 2), "Prop_SteelBrown", plane="xy", bevel=0.0025, segments=2, name="jamb R")
    # Head: outline in Blender Y/Z, extruded along X between the soffits.
    head = [(-z, HEAD - u) for u, z in SECTION]
    kit.extrude(head, W, (0, 0, 0), "Prop_SteelBrown", plane="yz", bevel=0.0025, segments=2, name="head")
    if silencers:
        for x in (-W / 2 + 0.006, W / 2 - 0.006):
            for y in (0.45, 1.05, 1.65):
                kit.cylinder(0.005, 0.003, (-x, -0.0335, y), "Prop_Rubber", verts=10, rot=(90, 0, 0), bevel=0.001, segments=1, name="silencer")
    kit.no_collider()
