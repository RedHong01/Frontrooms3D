"""8 ft two-lamp bare strip (slimline, F96T12) for Run! C. CLONE-ONLY look-dev.

A 2.40 m white baked-enamel steel channel (0.10 wide, 0.055 deep) with end
plates, a two-piece ballast cover (the 1.5 mm seam shows dark), embossed
knockouts, and a white phenolic lampholder at each end. No lens, no louver:
the tubes are bare (Kit_T12Tube_8ft, placed in Unity at the 'tube_a' /
'tube_b' anchors). 2.44 m overall. ORIGIN = top centre (the ceiling plane);
the fixture hangs below z = 0. Length along Blender X (Unity X).
"""
import run_c_common as rc  # noqa: F401

NAME = "Kit_StripFixture_8ft"
SMOOTH_ANGLE = 40.0

L = 2.40
D = 0.055
TUBE_Y, TUBE_Z = 0.0255, -0.078


def build(kit):
    for s in (-1, 1):
        kit.box((L / 2 - 0.0008, 0.100, D), (s * (L / 4 + 0.0004), 0, -D / 2), "Run_Enamel", bevel=0.007, segments=3, name="channel")
        kit.box((0.010, 0.103, D + 0.004), (s * (L / 2 + 0.005), 0, -D / 2 - 0.001), "Run_Enamel", bevel=0.002, segments=2, name="end plate")
        for sy in (-1, 1):
            kit.cylinder(0.0028, 0.002, (s * (L / 2 + 0.0105), sy * 0.030, -0.030), "Prop_Chrome", verts=8, rot=(0, 90, 0), bevel=0.0006, segments=1, name="end screw")
        # lampholder: bracket + phenolic block with the socket face toward the centre
        hx = s * (L / 2 - 0.022)
        kit.box((0.030, 0.090, 0.010), (hx, 0, -D - 0.005), "Run_Enamel", bevel=0.002, segments=1, name="holder bracket")
        kit.box((0.026, 0.088, 0.040), (hx, 0, -D - 0.010 - 0.019), "Prop_PlasticWhite", bevel=0.008, segments=3, name="lampholder")
        for sy in (-1, 1):
            kit.cylinder(0.011, 0.004, (hx - s * 0.013, sy * TUBE_Y, TUBE_Z), "Prop_PlasticWhite", verts=16, rot=(0, 90, 0), bevel=0.001, segments=1, name="socket ring")
    # embossed knockouts along both sides and the cover
    for x in (-0.9, -0.3, 0.3, 0.9):
        for sy in (-1, 1):
            kit.cylinder(0.011, 0.0016, (x, sy * 0.0505, -0.028), "Run_Enamel", verts=16, rot=(90, 0, 0), bevel=0.0005, segments=1, name="knockout")
    for x in (-0.6, 0.6):
        kit.cylinder(0.0035, 0.002, (x, 0, -D - 0.0005), "Prop_Chrome", verts=8, bevel=0.0006, segments=1, name="cover screw")
    kit.anchor("tube_a", (0, TUBE_Y, TUBE_Z))
    kit.anchor("tube_b", (0, -TUBE_Y, TUBE_Z))
    kit.no_collider()
    kit.tag("run", "ceiling")
