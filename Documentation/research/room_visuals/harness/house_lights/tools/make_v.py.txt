"""Verification images for Run! C (HOUSE LIGHTS), final round r9. JPG q85, 1920 wide."""
import os, sys
from PIL import Image, ImageDraw, ImageFont
H = "/Users/redwang/FrontRoomsVisualWork/hl_work/runs"
OUT = sys.argv[1]
FONT = "/Users/redwang/FrontRoomsVisualWork/proj_run_house_lights/Assets/Fonts/Period1990/TeXGyre/texgyreheros-bold.otf"
f = ImageFont.truetype(FONT, 24)
fs = ImageFont.truetype(FONT, 20)
BG = (34, 35, 28)

def label(d, x, y, text, sub=None):
    w = d.textlength(text, font=f) + 16
    d.rectangle((x, y, x + w, y + 34), fill=(244, 223, 59))
    d.text((x + 8, y + 5), text, font=f, fill=(10, 10, 10))
    if sub:
        w2 = d.textlength(sub, font=fs) + 16
        d.rectangle((x, y + 34, x + w2, y + 62), fill=(10, 10, 10))
        d.text((x + 8, y + 37), sub, font=fs, fill=(240, 238, 230))

def panel(path, crop=None, size=None):
    im = Image.open(path).convert("RGB")
    if crop: im = im.crop(crop)
    if size: im = im.resize(size, Image.LANCZOS)
    return im

def grid(items, cols, tile, name, pad=0):
    rows = (len(items) + cols - 1) // cols
    sh = Image.new("RGB", (cols * tile[0] + (cols - 1) * pad, rows * tile[1] + (rows - 1) * pad), BG)
    d = ImageDraw.Draw(sh)
    for i, (path, crop, text, sub) in enumerate(items):
        x = (i % cols) * (tile[0] + pad); y = (i // cols) * (tile[1] + pad)
        sh.paste(panel(path, crop, tile), (x, y))
        label(d, x + 12, y + 12, text, sub)
    if sh.width > 1920:
        sh = sh.resize((1920, round(sh.height * 1920 / sh.width)), Image.LANCZOS)
    sh.save(os.path.join(OUT, "run_dir_house_lights_" + name + ".jpg"), quality=85)
    print(name, sh.size)
