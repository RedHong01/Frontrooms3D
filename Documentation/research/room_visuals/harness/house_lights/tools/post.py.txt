"""Derive, measure and export the Run! C (HOUSE LIGHTS) frames.

/usr/bin/python3 post.py <frames dir> <out images dir> <out measurements json> [--export]
Derived frames: greyscale = linear Rec.709 Y re-encoded to sRGB; protan = Machado 2009
severity 1.0 in linear RGB. Region contrast = mean linear Y inside an ID mask vs a 12 px ring.
"""
import json, os, sys
import numpy as np
from PIL import Image, ImageDraw, ImageFont, ImageFilter

SRC, OUT, MJSON = sys.argv[1], sys.argv[2], sys.argv[3]
EXPORT = "--export" in sys.argv
PREFIX = "run_dir_house_lights_"
FONT = "/Users/redwang/FrontRoomsVisualWork/proj_run_house_lights/Assets/Fonts/Period1990/TeXGyre/texgyreheros-bold.otf"

MACHADO_PROTAN = np.array([[0.152286, 1.052583, -0.204868],
                           [0.114503, 0.786281, 0.099216],
                           [-0.003882, -0.048116, 1.051998]], np.float32)


def lin(c):
    c = c / 255.0
    return np.where(c <= 0.04045, c / 12.92, ((c + 0.055) / 1.055) ** 2.4)


def srgb(l):
    l = np.clip(l, 0, 1)
    return np.where(l <= 0.0031308, l * 12.92, 1.055 * l ** (1 / 2.4) - 0.055) * 255


def load(name):
    return np.asarray(Image.open(os.path.join(SRC, name + ".png")).convert("RGB")).astype(np.float32)


def Y(im):
    return lin(im) @ np.array([.2126, .7152, .0722], np.float32)


def grey(im):
    g = srgb(Y(im))
    return np.stack([g, g, g], -1)


def protan(im):
    return srgb(lin(im) @ MACHADO_PROTAN.T)


def stats(im):
    y = Y(im)
    r, g, b = im[..., 0], im[..., 1], im[..., 2]
    s = r + g + b + 1e-6
    redfrac = r / s
    p = np.percentile(y, [5, 50, 95, 99.5])
    return dict(Y_p5=round(float(p[0]), 4), Y_p50=round(float(p[1]), 4), Y_p95=round(float(p[2]), 4), Y_p995=round(float(p[3]), 4),
                p95_over_p5=round(float(p[2] / max(p[0], 1e-4)), 1),
                contrast_p95_p5=round(float((p[2] + .05) / (p[0] + .05)), 2),
                sat_red_pct=round(float(((redfrac >= 0.8) & (s > 30)).mean() * 100), 3),
                red_dominant_pct=round(float(((redfrac > 0.6) & (s > 30)).mean() * 100), 2),
                mean_rgb=[round(float(x), 1) for x in im.reshape(-1, 3).mean(0)])


def mask(name):
    p = os.path.join(SRC, name + ".png")
    if not os.path.exists(p):
        return None
    return (np.asarray(Image.open(p).convert("L")).astype(np.float32) / 255) > .5


def dilate(m, px):
    """Square dilation by px (separable 1-D max filters; PIL's MaxFilter(25) took minutes per mask)."""
    out = m.copy()
    for axis in (0, 1):
        src = out.copy()
        for k in range(1, px + 1):
            out[(slice(k, None),) if axis == 0 else (slice(None), slice(k, None))] |= src[(slice(None, -k),) if axis == 0 else (slice(None), slice(None, -k))]
            out[(slice(None, -k),) if axis == 0 else (slice(None), slice(None, -k))] |= src[(slice(k, None),) if axis == 0 else (slice(None), slice(k, None))]
    return out


def ring(m, px):
    return dilate(m, px) & ~m


def region(y, m, px=12):
    if m is None or m.sum() == 0:
        return dict(px=0)
    rg = ring(m, px)
    yi, yo = float(y[m].mean()), float(y[rg].mean())
    ys, xs = np.nonzero(m)
    return dict(px=int(m.sum()), bbox=[int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())], bbox_h=int(ys.max() - ys.min() + 1),
                Y_inside=round(yi, 4), Y_ring=round(yo, 4),
                ratio_plain=round(max(yi, yo) / max(min(yi, yo), 1e-4), 2),
                ratio_wcag=round((max(yi, yo) + .05) / (min(yi, yo) + .05), 2))


def region_in(y, m, aperture, px=12):
    """Mask vs a ring that is kept inside an aperture (the pair's lites): through a 0.25 m lite the plain ring
    lands on the dark leaf around the glass, so it would measure the Relay against the door, not the corridor."""
    if m is None or m.sum() == 0 or aperture is None:
        return dict(px=0)
    rg = ring(m, px) & aperture & ~m
    if rg.sum() == 0:
        return dict(px=int(m.sum()), ring_px=0)
    yi, yo = float(y[m].mean()), float(y[rg].mean())
    return dict(px=int(m.sum()), ring_px=int(rg.sum()), Y_inside=round(yi, 4), Y_ring_in_lite=round(yo, 4),
                ratio_plain=round(max(yi, yo) / max(min(yi, yo), 1e-4), 2),
                ratio_wcag=round((max(yi, yo) + .05) / (min(yi, yo) + .05), 2))


def save_jpg(arr, name, prefix=None):
    im = Image.fromarray(np.clip(arr, 0, 255).astype(np.uint8))
    if im.width > 1920:
        im = im.resize((1920, int(im.height * 1920 / im.width)), Image.LANCZOS)
    im.save(os.path.join(OUT, (prefix or PREFIX) + name + ".jpg"), quality=85)


SHOTS = ["s1a", "s1b", "s2", "s3a", "s3b", "s3b2", "s3c", "s3e", "s3d", "c1", "c2", "c3", "c4", "c4b", "c4p", "c4p_ng", "c4r", "p1", "p1off", "x1", "x1b", "x2", "dbg_amb0"]


def main():
    os.makedirs(OUT, exist_ok=True)
    res = {"frames": {}, "regions": {}}
    imgs = {s: load(s) for s in SHOTS if os.path.exists(os.path.join(SRC, s + ".png"))}
    for s, im in imgs.items():
        res["frames"][s] = stats(im)
    der = {}
    for s in ("s1a", "s1b", "s2", "c4", "c4b"):
        if s in imgs:
            der[s + "_grey"] = grey(imgs[s])
    for s in ("s1b", "c4", "c4b"):
        if s in imgs:
            der[s + "_protan"] = protan(imgs[s])
    for k, v in der.items():
        res["frames"][k] = stats(v)
    R = res["regions"]
    for shot, groups in (("s1a", ("goal", "doors", "burnisher", "wheelchair")),
                         ("s1b", ("doors", "goal", "burnisher", "wheelchair", "flashers")), ("s2", ("goal", "holders", "strips", "doors")),
                         ("c3", ("goal", "wheelchair", "doors")), ("c4", ("wheelchair", "doors")), ("c4b", ("doors",)),
                         ("s3c", ("holders",)), ("s3e", ("holders",)), ("s3d", ("flashers",))):
        if shot not in imgs:
            continue
        y = Y(imgs[shot])
        for g in groups:
            R[shot + "_" + g] = region(y, mask(shot + "_mask_" + g))
    if "s1b" in imgs:
        R["s1b_doors_ring48"] = region(Y(imgs["s1b"]), mask("s1b_mask_doors"), 48)
        R["s1b_grey_doors"] = region(Y(der["s1b_grey"]), mask("s1b_mask_doors"))
    for s in ("c4", "c4b", "c4p", "c4p_ng", "c4r"):
        if s in imgs:
            m = mask(s + "_mask_relay")
            mb = mask(s + "_mask_relayBody")
            lites = mask(s + "_mask_lites")
            R[s + "_relay"] = region(Y(imgs[s]), m)
            R[s + "_relayBody"] = region(Y(imgs[s]), mb)
            if s + "_protan" not in der:
                der[s + "_protan"] = protan(imgs[s])
            R[s + "_relay_protan"] = region(Y(der[s + "_protan"]), m)
            if lites is not None:
                R[s + "_relay_in_lites"] = region_in(Y(imgs[s]), m, lites)
                R[s + "_relayBody_in_lites"] = region_in(Y(imgs[s]), mb, lites)
                R[s + "_relayBody_in_lites_protan"] = region_in(Y(der[s + "_protan"]), mb, lites)
                R[s + "_lites_px"] = int(lites.sum())
    if "p1" in imgs:
        m = mask("p1_mask_flashers")
        im = imgs["p1"]
        r, g, b = im[..., 0], im[..., 1], im[..., 2]
        sred = (r / (r + g + b + 1e-6) >= .8) & (r + g + b > 30)
        R["p1_flasher_dome_px"] = int(m.sum()) if m is not None else None
        R["p1_saturated_red_px"] = int(sred.sum())
        R["p1_saturated_red_pct"] = round(float(sred.mean() * 100), 3)
        # WCAG 2 red-flash area: 25 % of any 10 deg field, the 341 x 256 px box at 1024 x 768 = 2.78 % of the screen
        R["wcag_red_area_limit_px_1080p"] = int(round(341 * 256 / (1024 * 768) * im.shape[0] * im.shape[1]))
        red_dom = (r / (r + g + b + 1e-6) >= .6) & (r + g + b > 30)
        R["p1_red_dominant_px"] = int(red_dom.sum())
    if "p1" in imgs and "p1off" in imgs:
        on, off = imgs["p1"], imgs["p1off"]
        def redd(im, k):
            r, g, b = im[..., 0], im[..., 1], im[..., 2]
            return (r / (r + g + b + 1e-6) >= k) & (r + g + b > 30)
        yon, yoff = Y(on), Y(off)
        dy = np.abs(yon - yoff)
        # WCAG 2.x: a flash is a pair of opposing changes of >= 10 % relative luminance where the darker is < 0.80;
        # a red flash is a change involving saturated red. Count the pixels that change that much.
        flash = (dy >= .10) & (np.minimum(yon, yoff) < .80)
        R["p1_flash_px_dY_ge_0p10"] = int(flash.sum())
        R["p1_flash_pct"] = round(float(flash.mean() * 100), 3)
        R["p1_red_flash_px"] = int((flash & (redd(on, .8) | redd(off, .8))).sum())
        R["p1_red_dominant_change_px"] = int((redd(on, .6) ^ redd(off, .6)).sum())
        R["p1_any_change_px_dY_ge_0p02"] = int((dy >= .02).sum())
        R["p1_saturated_red_px_on_off"] = [int(redd(on, .8).sum()), int(redd(off, .8).sum())]
    if "dbg_amb0" in imgs and "s1a" in imgs:
        R["ambient_test_s1a_minus_amb0_meanY"] = round(float(Y(imgs["s1a"]).mean() - Y(imgs["dbg_amb0"]).mean()), 4)
    with open(MJSON, "w") as f:
        json.dump(res, f, indent=1)
    print(json.dumps(res, indent=1))
    if EXPORT:
        for s in ("s1a", "s1b", "s2", "s3a", "s3b", "s3b2", "s3c", "s3e", "s3d", "c1", "c2", "c3", "c4", "c4b", "c4p", "c4p_ng", "c4r", "x1", "x1b", "x2", "p1", "p1off"):
            if s in imgs:
                save_jpg(imgs[s], s)
                if s in ("s1a", "s1b", "s2", "s3a", "s3b", "c1", "c2", "c3", "c4"):
                    save_jpg(imgs[s], s, "run_house_lights_")   # the workflow's step-3 name for the fixed shots, same pixels
        for k, v in der.items():
            save_jpg(v, k)
        sheet(imgs, der)


def sheet(imgs, der):
    order = [("s2", "S2  armed, from Level 0"), ("s1a", "S1a  armed (economy switching)"), ("c1", "C1  t -0.5 s"), ("c2", "C2  t +0.9 s, front at 6.4 m"),
             ("s1b", "S1b  tripped, t +3 s"), ("c3", "C3  t +2.0 s"), ("c4", "C4  t +4 s, Relay on axis"), ("c4b", "C4b  Relay 0.65 m off axis"),
             ("s3a", "S3a  held leaf, armed"), ("s3b", "S3b  released, t +1.0 s"), ("s3c", "S3c  magnetic holder (hero)"), ("s3d", "S3d  flasher on, t +0.5 s"),
             ("c4r", "C4r  Relay, inside the z18 pair"), ("x2", "X2  goal RN-X, t +3 s"), ("s1b_grey", "S1b  greyscale"), ("c4b_protan", "C4b  protan")]
    W, H, pad, cap = 480, 270, 8, 26
    cols = 4
    rows = (len(order) + cols - 1) // cols
    sh = Image.new("RGB", (cols * (W + pad) + pad, rows * (H + cap + pad) + pad + 50), (34, 35, 28))
    d = ImageDraw.Draw(sh)
    f = ImageFont.truetype(FONT, 18)
    ft = ImageFont.truetype(FONT, 26)
    d.text((pad, 12), "RUN! C  HOUSE LIGHTS  the doors go   clone pre-render (main of 2026-10-07 16:34 + C look-dev), final round r9, 2026-10-08", font=ft, fill=(236, 234, 228))
    for i, (k, label) in enumerate(order):
        src = imgs.get(k)
        if src is None:
            src = der.get(k)
        if src is None:
            continue
        im = Image.fromarray(np.clip(src, 0, 255).astype(np.uint8)).resize((W, H), Image.LANCZOS)
        x = pad + (i % cols) * (W + pad)
        y = 50 + pad + (i // cols) * (H + cap + pad)
        sh.paste(im, (x, y))
        d.text((x + 2, y + H + 3), label, font=f, fill=(236, 234, 228))
    if sh.width > 1920:
        sh = sh.resize((1920, round(sh.height * 1920 / sh.width)), Image.LANCZOS)
    sh.save(os.path.join(OUT, PREFIX + "sheet.jpg"), quality=85)


main()
