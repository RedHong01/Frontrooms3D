"""Direction A (RED WARD) textures for the clone only.

  RunA_ExitFace_A.png / _E.png   period stencil EXIT face (TeX Gyre Heros Bold
                                 Condensed, 152 mm caps, no arrows) for the
                                 0.306 x 0.176 m visible face of Kit_ExitSign.
  RunA_Plate_<id>_A.png           engraved phenolic door plates (Heros Bold).
  RunA_WireGrid_A.png             12.5 mm square wire mesh, 0.6 mm wire, RGBA.

Usage: python3 make_textures.py <clone>
"""
import math
import os
import sys

import numpy as np
from PIL import Image, ImageDraw, ImageFilter, ImageFont

CLONE = sys.argv[1]
OUT = os.path.join(CLONE, "Assets", "Resources", "Surfaces", "Textures")
FONTS = os.path.join(CLONE, "Assets", "Fonts", "Period1990", "TeXGyre")
CN_BOLD = os.path.join(FONTS, "texgyreheroscn-bold.otf")
BOLD = os.path.join(FONTS, "texgyreheros-bold.otf")
rng = np.random.default_rng(1990)


def glyph_masks(text, font_path, cap_px, x_scale):
    """Each glyph as a tight L mask, cap height = cap_px, squeezed by x_scale."""
    probe = ImageFont.truetype(font_path, 1000)
    im = Image.new("L", (1400, 1400))
    ImageDraw.Draw(im).text((100, 0), "H", font=probe, fill=255)
    cap_1000 = im.getbbox()[3] - im.getbbox()[1]
    size = int(round(1000 * cap_px / cap_1000))
    font = ImageFont.truetype(font_path, size)
    # Common vertical box: cap top to baseline of "H" (keeps hyphens mid-height).
    hb = Image.new("L", (size * 2, size * 2))
    ImageDraw.Draw(hb).text((size // 2, 0), "H", font=font, fill=255)
    top, bottom = hb.getbbox()[1], hb.getbbox()[3]
    out = []
    for ch in text:
        if ch == " ":
            out.append(None)
            continue
        g = Image.new("L", (size * 2, size * 2))
        ImageDraw.Draw(g).text((size // 2, 0), ch, font=font, fill=255)
        bb = g.getbbox()
        g = g.crop((bb[0], top, bb[2], bottom))
        w = max(1, int(round(g.width * x_scale)))
        out.append(g.resize((w, g.height), Image.LANCZOS))
    return out, size


def set_line(canvas_size, masks, gap_px, space_px, centre):
    """Compose glyph masks into one line mask centred at `centre`."""
    W, H = canvas_size
    total = sum((m.width if m is not None else space_px) for m in masks) + gap_px * (len(masks) - 1)
    hmax = max(m.height for m in masks if m is not None)
    line = Image.new("L", (W, H))
    x = int(round(centre[0] - total / 2))
    y = int(round(centre[1] - hmax / 2))
    for m in masks:
        if m is None:
            x += space_px + gap_px
            continue
        line.paste(m, (x, y + hmax - m.height), m)
        x += m.width + gap_px
    return line, total


def exit_face():
    # Visible face 0.306 x 0.176 m at 4 px/mm.
    W, H = 1224, 704
    ppm = W / 306.0
    cap = int(round(152 * ppm))
    masks, _ = glyph_masks("EXIT", CN_BOLD, cap, 0.75)
    gap = int(round(8 * ppm))
    letters, total = set_line((W, H), masks, gap, 0, (W / 2, H / 2))
    print("EXIT legend width %.0f mm, cap %.0f mm, I stroke %.1f mm" % (total / ppm, cap / ppm, masks[2].width / ppm))
    L = np.asarray(letters).astype(np.float32) / 255.0

    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
    # Off-white stamped steel face, enamel, a little heat yellowing above the
    # lamps (two incandescent lamps behind the face at 1/3 and 2/3 width).
    base = np.array([0.905, 0.890, 0.848])
    noise = rng.normal(0, 1, (H // 8 + 1, W // 8 + 1)).astype(np.float32)
    noise = np.asarray(Image.fromarray(noise).resize((W, H), Image.BICUBIC))
    fine = rng.normal(0, 1, (H, W)).astype(np.float32)
    heat = np.zeros((H, W), np.float32)
    for cx in (W / 3, 2 * W / 3):
        heat += np.exp(-(((xx - cx) / (W * 0.16)) ** 2 + ((yy - H * 0.05) / (H * 0.30)) ** 2))
    alb = np.ones((H, W, 3), np.float32) * base
    alb *= (1 + 0.012 * noise[..., None] + 0.006 * fine[..., None])
    alb[..., 2] -= 0.045 * heat
    alb[..., 1] -= 0.018 * heat
    # Face edge shading where it tucks under the bezel lip.
    edge = np.minimum.reduce([xx, W - 1 - xx, yy, H - 1 - yy])
    alb *= (0.90 + 0.10 * np.clip(edge / 10.0, 0, 1))[..., None]
    # Stencil letters: the red diffuser seen through the cut steel, with the
    # 1 mm cut edge as a darker rim.
    rim = np.asarray(letters.filter(ImageFilter.MaxFilter(9))).astype(np.float32) / 255.0 - L
    rim = np.clip(rim, 0, 1)
    red = np.array([0.70, 0.13, 0.10])
    alb = alb * (1 - L[..., None]) + red * L[..., None]
    alb *= (1 - 0.35 * rim[..., None])
    Image.fromarray((np.clip(alb, 0, 1) ** (1 / 1.0) * 255).astype(np.uint8)).save(os.path.join(OUT, "RunA_ExitFace_A.png"))

    # Emission: only the letters glow. Two lamps behind -> two hot spots, the
    # letter ends a touch dimmer and warmer; 1 px soft edge from the diffuser.
    hot = np.zeros((H, W), np.float32)
    for cx in (W / 3, 2 * W / 3):
        hot += np.exp(-(((xx - cx) / (W * 0.22)) ** 2 + ((yy - H * 0.5) / (H * 0.55)) ** 2))
    hot = 0.74 + 0.26 * np.clip(hot / hot.max(), 0, 1)
    soft = np.asarray(letters.filter(ImageFilter.GaussianBlur(1.2))).astype(np.float32) / 255.0
    col = np.stack([np.ones_like(hot), 0.10 + 0.05 * (1 - hot), 0.045 + 0.02 * (1 - hot)], -1)  # sRGB-ish red, warmer at the ends
    em = col * (hot * soft)[..., None]
    Image.fromarray((np.clip(em, 0, 1) * 255).astype(np.uint8)).save(os.path.join(OUT, "RunA_ExitFace_E.png"))


def plate(name, text, w_mm, h_mm, cap_mm):
    ppm = 4.0
    W, H = int(w_mm * ppm), int(h_mm * ppm)
    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
    # Engraved phenolic: dark brown face, the cream core shows in the cuts.
    face = np.array([0.255, 0.200, 0.160])
    core = np.array([0.900, 0.875, 0.800])
    noise = rng.normal(0, 1, (H, W)).astype(np.float32)
    alb = np.ones((H, W, 3), np.float32) * face * (1 + 0.02 * noise[..., None])
    masks, _ = glyph_masks(text, BOLD, int(round(cap_mm * ppm)), 1.0)
    letters, total = set_line((W, H), masks, int(round(1.2 * ppm)), int(round(cap_mm * 0.45 * ppm)), (W / 2, H / 2))
    L = np.asarray(letters).astype(np.float32) / 255.0
    # Engraved border 3 mm in, 0.6 mm line.
    b = 3 * ppm
    border = ((np.abs(xx - b) < 1.2) | (np.abs(xx - (W - 1 - b)) < 1.2) | (np.abs(yy - b) < 1.2) | (np.abs(yy - (H - 1 - b)) < 1.2))
    border &= (xx >= b - 1.2) & (xx <= W - b + 0.2) & (yy >= b - 1.2) & (yy <= H - b + 0.2)
    L = np.maximum(L, border.astype(np.float32))
    alb = alb * (1 - L[..., None]) + core * L[..., None]
    edge = np.minimum.reduce([xx, W - 1 - xx, yy, H - 1 - yy])
    alb *= (0.82 + 0.18 * np.clip(edge / 4.0, 0, 1))[..., None]
    Image.fromarray((np.clip(alb, 0, 1) * 255).astype(np.uint8)).save(os.path.join(OUT, "RunA_Plate_%s_A.png" % name))
    print("plate %s: %d x %d px, legend %.0f mm" % (name, W, H, total / ppm))


def wire_grid():
    # One 12.5 mm cell = 128 px; 0.6 mm wire = 6 px, split across the wrap.
    N = 128
    im = np.zeros((N, N, 4), np.uint8)
    col = (110, 110, 106)
    for i in list(range(0, 3)) + list(range(N - 3, N)):
        im[i, :, :3] = col
        im[i, :, 3] = 255
        im[:, i, :3] = col
        im[:, i, 3] = 255
    Image.fromarray(im, "RGBA").save(os.path.join(OUT, "RunA_WireGrid_A.png"))


if __name__ == "__main__":
    os.makedirs(OUT, exist_ok=True)
    exit_face()
    plate("4-11", "4-11", 200, 75, 25)
    plate("4-13", "4-13", 200, 75, 25)
    plate("STAFF", "STAFF ONLY", 250, 75, 20)
    plate("LINEN", "LINEN", 200, 75, 22)
    plate("QUIET", "QUIET PLEASE", 300, 90, 22)
    wire_grid()
