import sys, numpy as np
from PIL import Image
def lin(c):
    c=c/255.0; return np.where(c<=0.04045,c/12.92,((c+0.055)/1.055)**2.4)
def load(p): return np.asarray(Image.open(p).convert('RGB')).astype(np.float32)
def Y(im): return lin(im)@np.array([.2126,.7152,.0722])
def mask(p,c):
    im=load(p)/255; q=np.where(im<.25,0,np.where(im<.85,.5,1)); return np.all(np.abs(q-np.array(c))<.01,-1)
def dil(m,n):
    m=m.copy()
    for _ in range(n):
        g=m.copy(); g[1:]|=m[:-1]; g[:-1]|=m[1:]; g[:,1:]|=m[:,:-1]; g[:,:-1]|=m[:,1:]; m=g
    return m
ID='/Users/redwang/FrontRoomsVisualWork/wf/red_ward_v5/r4_renders/'
for d in sys.argv[1:]:
    out=[]
    for n in ['s1b','c3','c4']:
        im=load(d+'/'+n+'.png'); y=Y(im); r,g,b=im[...,0],im[...,1],im[...,2]; s=r+g+b+1e-6
        p=np.percentile(y,[5,50,95])
        out.append(f"{n}: p5 {p[0]:.4f} p50 {p[1]:.4f} p95 {p[2]:.4f} mean {y.mean():.4f} c {(p[2]+.05)/(p[0]+.05):.2f} sat {(((r/s)>=.8)&(s>30)).mean()*100:.1f}")
    im=load(d+'/c4.png'); y=Y(im)
    body=mask(ID+'c4_id.png',(1,.5,0)); whole=body|mask(ID+'c4_id.png',(.5,0,1)); ring=dil(whole,12)&~whole
    out.append(f"relay {y[ring].mean()/y[body].mean():.2f}")
    print(d.split('/')[-1]); print('\n'.join(out))
