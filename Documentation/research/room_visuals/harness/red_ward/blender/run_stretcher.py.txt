"""1980s hydraulic transport stretcher (Run! direction A props; clone-only).

Reference: the hospital transport stretcher of the 1980s: a putty-enamelled
base shroud on four 6 in swivel casters, twin hydraulic columns in black
rubber bellows, a chrome tubular litter 0.70 x 2.0 m at about 0.85 m, a white
vinyl pad, side rails folded down, a push handle at the head end. A folded
cotton blanket at the foot. No brand plates.
Long axis along Y (Unity Z); the head end is +Y. Stands on z = 0.
"""

import run_props_common as c

NAME = "Kit_Stretcher"
SMOOTH_ANGLE = 40.0
LOD1 = 0.45


def build(kit):
    L, W, TOP = 1.96, 0.66, 0.80          # litter frame (tube centre line), height
    # Casters and base.
    for x in (-0.22, 0.22):
        for y in (-0.62, 0.62):
            c.caster(kit, x, y, 0.075, 0.042, stem_top=0.22, name="caster")
    kit.box((0.56, 1.46, 0.10), (0, 0, 0.27), c.PUTTY, bevel=0.025, segments=3, name="base shroud")
    kit.box((0.50, 1.40, 0.02), (0, 0, 0.215), c.BLACK, bevel=0.004, segments=1, name="base skirt")
    kit.box((0.06, 0.20, 0.025), (0.31, -0.35, 0.25), c.BLACK, bevel=0.006, segments=2, name="brake pedal")
    kit.box((0.06, 0.20, 0.025), (-0.31, 0.35, 0.25), c.BLACK, bevel=0.006, segments=2, name="pump pedal")
    for y in (-0.40, 0.40):
        kit.cylinder(0.065, 0.36, (0, y, 0.50), c.RUBBER, verts=24, bevel=0.01, segments=2, name="bellows")
        for k in range(5):
            kit.cylinder(0.071, 0.012, (0, y, 0.35 + k * 0.075), c.RUBBER, verts=24, bevel=0.005, segments=1, name="bellows rib")
    # Litter deck and frame.
    kit.box((W - 0.04, L - 0.06, 0.02), (0, 0, TOP - 0.025), c.PUTTY, bevel=0.006, segments=1, name="deck")
    kit.box((0.18, 1.20, 0.06), (0, 0, TOP - 0.065), c.PUTTY, bevel=0.01, segments=2, name="deck spine")
    hw, hl = W / 2, L / 2
    kit.tube([(-hw, -hl + 0.05, TOP), (-hw, hl - 0.05, TOP), (-hw + 0.05, hl, TOP), (hw - 0.05, hl, TOP),
              (hw, hl - 0.05, TOP), (hw, -hl + 0.05, TOP), (hw - 0.05, -hl, TOP), (-hw + 0.05, -hl, TOP), (-hw, -hl + 0.05, TOP)],
             0.016, c.CHROME, verts=14, name="litter frame")
    # Pad (white vinyl) with a folded blanket at the foot.
    kit.soft_box((W - 0.02, L - 0.04, 0.085), (0, 0, TOP + 0.045), c.WHITE_VINYL, radius=0.035, segments=28, rings=12, sag=0.004, puff=0.006, name="pad")
    kit.soft_box((0.52, 0.34, 0.07), (0.02, -0.66, TOP + 0.12), c.LINEN_BLUE, radius=0.03, segments=20, rings=10, puff=0.008, name="blanket")
    kit.soft_box((0.46, 0.30, 0.10), (0, 0.80, TOP + 0.13), c.LINEN, radius=0.045, segments=20, rings=10, puff=0.01, sag=0.01, name="pillow")
    # Side rails folded down along both sides.
    for sx in (-1, 1):
        x = sx * (hw + 0.035)
        kit.tube([(x, -0.58, TOP - 0.06), (x, 0.58, TOP - 0.06)], 0.012, c.CHROME, verts=12, name="side rail top")
        kit.tube([(x, -0.58, TOP - 0.24), (x, 0.58, TOP - 0.24)], 0.012, c.CHROME, verts=12, name="side rail bottom")
        for y in (-0.45, -0.15, 0.15, 0.45):
            kit.tube([(x, y, TOP - 0.24), (x, y, TOP - 0.06)], 0.008, c.CHROME, verts=10, name="side rail bar")
        for y in (-0.58, 0.58):
            kit.tube([(x, y, TOP - 0.24), (x, y, TOP - 0.06), (sx * (hw - 0.01), y, TOP - 0.02)], 0.011, c.CHROME, verts=10, name="side rail hinge arm")
    # Push handle at the head end, IV socket at the head corner.
    kit.tube([(-0.26, hl - 0.02, TOP), (-0.26, hl + 0.06, TOP + 0.20), (0.26, hl + 0.06, TOP + 0.20), (0.26, hl - 0.02, TOP)],
             0.014, c.CHROME, verts=12, name="push handle")
    kit.cylinder(0.014, 0.08, (hw - 0.03, hl - 0.03, TOP + 0.02), c.CHROME, verts=12, name="IV socket")
    kit.collider((0, 0, 0.5), (0.74, 2.02, 1.0))
    kit.tag("run", "hospital", "floor")
