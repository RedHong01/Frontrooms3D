"""Folding hospital wheelchair, 1980s (clone-only, Run! direction A props).

Reference: the standard chrome X-brace folding chair of US hospitals: 24 in
rear wheels with grey solid tyres, chrome hand rims and wire spokes, 8 in
front casters, black vinyl sling seat and back, padded armrests, push
handles with black grips, swing-away footrests. No brand.
The front (footrests) faces -Y (Unity +Z). Stands on z = 0.
"""

import math

import run_props_common as c

NAME = "Kit_Wheelchair"
SMOOTH_ANGLE = 40.0
LOD1 = 0.4

HX = 0.235            # side frame x
REAR_Y, REAR_Z, REAR_R = 0.16, 0.305, 0.295
WHEEL_X = 0.300


def wheel(kit, sx):
    x = sx * WHEEL_X
    c.torus(kit, REAR_R, 0.016, (x, REAR_Y, REAR_Z), c.CASTER, seg=48, ring=10, name="tyre")
    c.torus(kit, REAR_R - 0.022, 0.007, (x, REAR_Y, REAR_Z), c.CHROME, seg=48, ring=6, name="rim")
    c.torus(kit, 0.265, 0.0085, (x + sx * 0.035, REAR_Y, REAR_Z), c.CHROME, seg=48, ring=8, name="hand rim")
    for k in range(6):
        a = 2 * math.pi * k / 6 + 0.3
        p0 = (x + sx * 0.03, REAR_Y + math.cos(a) * 0.255, REAR_Z + math.sin(a) * 0.255)
        p1 = (x + sx * 0.012, REAR_Y + math.cos(a) * 0.265, REAR_Z + math.sin(a) * 0.265)
        kit.tube([p0, p1], 0.005, c.CHROME, verts=6, name="hand rim stud")
    for k in range(18):
        a = 2 * math.pi * k / 18
        side = 0.022 if k % 2 else -0.022
        p0 = (x + side, REAR_Y, REAR_Z)
        p1 = (x, REAR_Y + math.cos(a) * (REAR_R - 0.028), REAR_Z + math.sin(a) * (REAR_R - 0.028))
        kit.lod1_drop(kit.tube([p0, p1], 0.0016, c.CHROME, verts=5, caps=False, name="spoke"))
    kit.cylinder(0.032, 0.07, (x, REAR_Y, REAR_Z), c.CHROME, verts=16, rot=(0, 90, 0), bevel=0.004, segments=2, name="hub")


def build(kit):
    for sx in (-1, 1):
        wheel(kit, sx)
        x = sx * HX
        # Side frame: lower rail, front post, seat rail, back post up to the push handle.
        kit.tube([(x, -0.30, 0.20), (x, 0.17, 0.20), (x, 0.20, 0.26), (x, 0.21, 0.52), (x, 0.24, 0.92), (x, 0.33, 0.95)],
                 0.0115, c.CHROME, verts=12, name="back post")
        kit.tube([(x, -0.30, 0.20), (x, -0.29, 0.52), (x, 0.20, 0.50)], 0.0115, c.CHROME, verts=12, name="front post + seat rail")
        # Armrest frame and pad.
        kit.tube([(x, -0.12, 0.50), (x, -0.13, 0.70), (x, 0.17, 0.70), (x, 0.19, 0.60)], 0.010, c.CHROME, verts=10, name="arm frame")
        kit.box((0.055, 0.27, 0.032), (x, 0.02, 0.722), c.SLING, bevel=0.012, segments=2, name="arm pad")
        # Push handle grip.
        kit.cylinder(0.016, 0.11, (x, 0.30, 0.948), c.RUBBER, verts=14, rot=(90, 0, 0), bevel=0.004, segments=2, name="grip")
        # Swing-away footrest hanger, plate and heel loop.
        kit.tube([(x, -0.29, 0.50), (x * 0.95, -0.36, 0.42), (x * 0.85, -0.44, 0.13)], 0.010, c.CHROME, verts=10, name="footrest hanger")
        kit.box((0.15, 0.13, 0.012), (x * 0.55, -0.48, 0.12), c.BLACK, bevel=0.003, segments=1, rot=(8, 0, 0), name="footplate")
        # Front caster under the front post.
        c.caster(kit, sx * 0.225, -0.31, 0.095, 0.03, slot_wheel=c.CASTER, stem_top=0.20, name="front caster")
        # Brake lever by the rear wheel.
        kit.tube([(x, 0.02, 0.27), (x + sx * 0.03, -0.04, 0.37)], 0.007, c.CHROME, verts=8, name="brake lever")
        kit.cylinder(0.011, 0.03, (x + sx * 0.03, -0.04, 0.38), c.BLACK, verts=10, name="brake knob")
    # Folding X brace under the seat.
    kit.tube([(-HX, 0.0, 0.21), (HX, 0.0, 0.49)], 0.010, c.CHROME, verts=10, name="x brace")
    kit.tube([(HX, 0.02, 0.21), (-HX, 0.02, 0.49)], 0.010, c.CHROME, verts=10, name="x brace")
    # Sling seat and back.
    kit.soft_box((2 * HX - 0.01, 0.42, 0.018), (0, -0.04, 0.505), c.SLING, radius=0.008, segments=20, rings=8, sag=0.025, name="seat sling")
    kit.soft_box((2 * HX - 0.01, 0.016, 0.33), (0, 0.215, 0.73), c.SLING, radius=0.006, segments=20, rings=10, puff=0.004, rot=(-8, 0, 0), name="back sling")
    kit.collider((0, -0.05, 0.47), (0.66, 1.05, 0.95))
    kit.tag("run", "hospital", "floor")
