def torus(kit, R, r, loc, slot, axis="x", seg=40, ring=10, name="torus"):
    """A ring (tyre, hand rim) of major radius R and tube radius r, its axis
    along X (a wheel rolling along Y) or Z."""
    prof = []
    for i in range(ring + 1):
        a = 2 * math.pi * i / ring
        prof.append((R + r * math.cos(a), r * math.sin(a)))
    rot = (0, 90, 0) if axis == "x" else (0, 0, 0)
    return kit.lathe(prof, loc, slot, verts=seg, rot=rot, name=name, close_top=False, close_bottom=False)


def caster(kit, x, y, wheel_r, width, slot_wheel=CASTER, fork=CHROME, stem_top=None, name="caster"):
    """A swivel caster: wheel on a fork, the wheel axle along X, centred at (x, y)."""
    kit.cylinder(wheel_r, width, (x, y, wheel_r), slot_wheel, verts=20, rot=(0, 90, 0), bevel=0.004, segments=2, name=name + " wheel")
    kit.cylinder(wheel_r * 0.35, width + 0.012, (x, y, wheel_r), fork, verts=12, rot=(0, 90, 0), bevel=0.001, segments=1, name=name + " hub")
    top = wheel_r * 2 + 0.02
    for sx in (-1, 1):
        kit.box((0.006, wheel_r * 0.9, wheel_r * 1.2 + 0.02), (x + sx * (width / 2 + 0.008), y - wheel_r * 0.15, wheel_r * 1.1 + 0.01), fork,
                bevel=0.002, segments=1, name=name + " fork")
    kit.box((width + 0.03, wheel_r * 0.9, 0.012), (x, y - wheel_r * 0.15, top), fork, bevel=0.002, segments=1, name=name + " yoke")
    if stem_top is not None and stem_top > top:
        kit.cylinder(0.009, stem_top - top, (x, y - wheel_r * 0.15, (top + stem_top) / 2), fork, verts=12, bevel=0.0, name=name + " stem")
