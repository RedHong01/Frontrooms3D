// G10 GLASS SIGN-OFF HARNESS (private clone proj_glass only; never copied into the real project).
// Built from the interaction audit's harness (Documentation/research/interaction_audit/harness):
// same scene, same seed (Random.InitState(4242) before Space), same title handoff, same target
// search, same game camera (1920x1080, 4x MSAA, full post). For every view it renders, in one
// frozen game moment (Time.timeScale 0, so lamp flicker cannot differ between states):
//   A  BEFORE      what the game draws today: the map's runtime TransparentGlass (3 cm cube, casts
//                  shadows), the scene's ambient, Unity's default sky reflection at 0.3, and the
//                  shipped Prop_Glass / Prop_BottleBlue (copies of the real project's .mat files).
//   B  AFTER-GLASS Glass_Window on every built pane (6 mm, shadows off, as the map contract in
//                  10_implementation.md §8 asks), the zone cube via FrontRoomsZoneReflection.SetImmediate,
//                  the new prop materials; scene ambient untouched (attribution only).
//   C  AFTER       B + FrontRoomsLook.ApplyAmbient(): what the game will draw once the map chat calls
//                  ApplyAmbient + SetZoneReflection at run start.
//   N  no pane     for window views only: the held pane hidden, per lighting state (reference for metrics).
// FrontRoomsMapWorld.cs is NOT edited: panes are swapped from here, at runtime, and swapped back.
// Run 3 (fix pass, 2026-10-03): FrontRoomsLook carries main's F5 ambient (= the scene's), so B and C must match
// (logged); a structure metric (high-pass |Δ| and 1-SSIM of luminance inside the pane quad) is added next to the mean
// ΔY; the diagnostics compare the new pane with the previous material values; a STAGED Glass_Shard view (shards on
// Level 0 carpet at 1 m and 2 m, three shard material options) is added.
// Run: Unity -batchmode -projectPath <clone> -executeMethod FrontRoomsGlassG10Capture.RunBatch -logFile <f>

    static FrontRoomsGlassG10Capture()
    {
        if (SessionState.GetBool(ActiveKey, false)) EditorApplication.update += Tick;
    }

    public static void RunBatch()
    {
        EditorSceneManager.OpenScene(ScenePath, OpenSceneMode.Single);
        SessionState.SetBool(ActiveKey, true);
        SessionState.SetFloat(DeadlineKey, (float)EditorApplication.timeSinceStartup + 1500f);
        EditorApplication.update -= Tick;
        EditorApplication.update += Tick;
        EditorApplication.EnterPlaymode();
    }

    /// <summary>G14 hook proof (weight 0 = bit-identical; positive, orientation and depth-tag tests). Writes Verification/glass_g10_proof.</summary>
    public static void RunProofBatch()
    {
        SessionState.SetBool(ProofKey, true);
        RunBatch();
    }

    static bool ProofMode => SessionState.GetBool(ProofKey, false);

    static string OutDir
    {
        get
        {
            if (outDir == null)
            {
                outDir = Path.Combine(Directory.GetParent(Application.dataPath).FullName, "Verification", ProofMode ? "glass_g10_proof" : "glass_g10");
                Directory.CreateDirectory(outDir);
            }
            return outDir;
        }
    }

    static void Log(string s)
    {
        var line = "[" + Time.frameCount + " t=" + Time.time.ToString("0.000", CultureInfo.InvariantCulture) + "] " + s;
        log.AppendLine(line);
        Debug.Log("[G10] " + line);
        try { File.WriteAllText(Path.Combine(OutDir, "g10_log.txt"), log.ToString()); } catch { }
    }

    static void Finish()
    {
        finished = true;
        Time.captureDeltaTime = 0f;
        Time.timeScale = 1f;
        SessionState.SetBool(DoneKey, true);
        Flush();
    }

    static void Tick()
    {
        var timedOut = EditorApplication.timeSinceStartup > SessionState.GetFloat(DeadlineKey, float.MaxValue);
        if (finished || timedOut || SessionState.GetBool(DoneKey, false))
        {
            if (timedOut && !finished && !SessionState.GetBool(DoneKey, false)) { Log("TIMEOUT"); Finish(); }
            if (EditorApplication.isPlaying) { EditorApplication.ExitPlaymode(); return; }
            if (EditorApplication.isPlayingOrWillChangePlaymode) return;
            EditorApplication.update -= Tick;
            SessionState.EraseBool(ActiveKey);
            SessionState.EraseBool(DoneKey);
            SessionState.EraseBool(ProofKey);
            if (Application.isBatchMode) EditorApplication.Exit(0);
            return;
        }
        if (!EditorApplication.isPlaying) return;
        if (stack.Count == 0)
        {
            stack.Push(Main());
            Log("routine created");
        }
        if (Time.frameCount < waitUntilFrame) return;
        try
        {
            for (var guard = 0; guard < 1000; guard++)
            {
                var top = stack.Peek();
                if (!top.MoveNext())
                {
                    stack.Pop();
                    if (stack.Count == 0) { Finish(); return; }
                    continue;
                }
                var cur = top.Current;
                if (cur is IEnumerator sub) { stack.Push(sub); continue; }
                waitUntilFrame = Time.frameCount + (cur is int n ? Mathf.Max(1, n) : 1);
                return;
            }
        }
        catch (Exception e)
        {
            Log("EXCEPTION " + e);
            Finish();
        }
    }

    static void Flush()
    {
        try
        {
            File.WriteAllText(Path.Combine(OutDir, "g10_log.txt"), log.ToString());
            File.WriteAllText(Path.Combine(OutDir, "g10_frames.txt"), frames.ToString());
            File.WriteAllText(Path.Combine(OutDir, "g10_metrics.txt"), metrics.ToString());
        }
        catch (Exception e) { Debug.LogError(e); }
    }

    // ---------------- reflection helpers (private game/map members; brittle by design) ----------------
    static FieldInfo FI(Type t, string n)
    {
        for (; t != null; t = t.BaseType)
        {
            var f = t.GetField(n, BF);
            if (f != null) return f;
        }
        return null;
    }
    static object Get(object o, string n) => FI(o.GetType(), n)?.GetValue(o);
    static T Get<T>(object o, string n) => (T)Get(o, n);
    static void Set(object o, string n, object v) => FI(o.GetType(), n).SetValue(o, v);
    static object Call(object o, string n, params object[] a) => o.GetType().GetMethod(n, BF).Invoke(o, a);

    static string Phase => Get(game, "phase").ToString();
    static Transform PlayerRoot => Get<Transform>(game, "playerRoot");

    static string V(Vector3 v) => "(" + v.x.ToString("0.00", CultureInfo.InvariantCulture) + ", " + v.y.ToString("0.00", CultureInfo.InvariantCulture) + ", " + v.z.ToString("0.00", CultureInfo.InvariantCulture) + ")";
    static string F2(float f) => f.ToString("0.###", CultureInfo.InvariantCulture);
    static string F(double f, string fmt = "0.00") => f.ToString(fmt, CultureInfo.InvariantCulture);

    // ---------------- render ----------------
    static void EnsureRT()
    {
        if (rt != null) return;
        // 4 samples: with a target texture URP takes MSAA from the texture (= the pipeline's 4x).
        rt = new RenderTexture(W, H, 24, RenderTextureFormat.ARGB32, RenderTextureReadWrite.sRGB) { antiAliasing = 4, name = "G10 capture" };
        rt.Create();
        readTex = new Texture2D(W, H, TextureFormat.RGB24, false);
    }

    static Color32[] RenderPixels(string saveAs = null)
    {
        EnsureRT();
        var prev = cam.targetTexture;
        cam.targetTexture = rt;
        cam.Render();
        cam.Render();
        cam.targetTexture = prev;
        var active = RenderTexture.active;
        RenderTexture.active = rt;
        readTex.ReadPixels(new Rect(0, 0, W, H), 0, 0);
        readTex.Apply(false);
        RenderTexture.active = active;
        if (saveAs != null) File.WriteAllBytes(Path.Combine(OutDir, saveAs), readTex.EncodeToJPG(85));
        return readTex.GetPixels32();
    }

    static double MedianRenderMs(int n)
    {
        EnsureRT();
        var prev = cam.targetTexture;
        cam.targetTexture = rt;
        var px = new Texture2D(1, 1, TextureFormat.RGBA32, false);
        var list = new List<double>();
        var sw = new Stopwatch();
        for (var i = 0; i < n + 3; i++)
        {
            sw.Restart();
            cam.Render();
            var active = RenderTexture.active;
            RenderTexture.active = rt;
            px.ReadPixels(new Rect(0, 0, 1, 1), 0, 0);
            px.Apply(false);
            RenderTexture.active = active;
            sw.Stop();
            if (i >= 3) list.Add(sw.Elapsed.TotalMilliseconds);
        }
        cam.targetTexture = prev;
        UnityEngine.Object.Destroy(px);
        list.Sort();
        return list[list.Count / 2];
    }

    static Material glassWindow, propGlassNew, propBottleNew, propGlassOld, propBottleOld;
    // What ApplyState(after) assigns (the assets, or temporary copies in the proof and diagnostics).
    static Material windowAfter, propGlassAfter, propBottleAfter;
    static readonly HashSet<Material> propGlassFamily = new HashSet<Material>(), propBottleFamily = new HashSet<Material>();

    sealed class PaneOrig { public Material mat; public Vector3 scale; public ShadowCastingMode shadows; }
    static readonly Dictionary<Renderer, PaneOrig> paneOrig = new Dictionary<Renderer, PaneOrig>();

    static List<Renderer> AllPanes()
    {
        var list = new List<Renderer>();
        if (Get(map, "built") is IDictionary built)
            foreach (DictionaryEntry e in built)
                if (Get(e.Value, "windows") is IEnumerable windows)
                    foreach (var w in windows)
                        if (Get(w, "pane") is GameObject pane && pane != null) list.Add(pane.GetComponent<Renderer>());
        return list;
    }

    static int ThinAxis(Vector3 s)
    {
        s = new Vector3(Mathf.Abs(s.x), Mathf.Abs(s.y), Mathf.Abs(s.z));
        return s.x <= s.y && s.x <= s.z ? 0 : (s.y <= s.z ? 1 : 2);
    }

    static int ApplyPanes(bool after)
    {
        foreach (var k in paneOrig.Keys.Where(k => k == null).ToList()) paneOrig.Remove(k);
        var n = 0;
        foreach (var r in AllPanes())
        {
            if (!paneOrig.TryGetValue(r, out var o))
            {
                o = new PaneOrig { mat = r.sharedMaterial, scale = r.transform.localScale, shadows = r.shadowCastingMode };
                paneOrig[r] = o;
            }
            if (after)
            {
                r.sharedMaterial = windowAfter != null ? windowAfter : glassWindow;
                var s = o.scale;
                s[ThinAxis(s)] = PaneThicknessAfter;
                r.transform.localScale = s;
                r.shadowCastingMode = ShadowCastingMode.Off;
            }
            else
            {
                r.sharedMaterial = o.mat;
                r.transform.localScale = o.scale;
                r.shadowCastingMode = o.shadows;
            }
            n++;
        }
        return n;
    }

    static int ApplyProps(bool after)
    {
        var n = 0;
        foreach (var r in UnityEngine.Object.FindObjectsByType<MeshRenderer>(FindObjectsInactive.Include, FindObjectsSortMode.None))
        {
            var mats = r.sharedMaterials;
            var changed = false;
            for (var i = 0; i < mats.Length; i++)
            {
                var m = mats[i];
                if (m == null) continue;
                if (propGlassFamily.Contains(m)) { mats[i] = after ? propGlassAfter : propGlassOld; changed = true; }
                else if (propBottleFamily.Contains(m)) { mats[i] = after ? propBottleAfter : propBottleOld; changed = true; }
            }
            if (changed) { r.sharedMaterials = mats; n++; }
        }
        return n;
    }

    static string ApplyState(State s, FrontRoomsLook.ReflectionZone zone)
    {
        Restore(env0);
        var panes = ApplyPanes(s != State.Before);
        var props = ApplyProps(s != State.Before);
        if (s != State.Before)
        {
            FrontRoomsZoneReflection.SetImmediate(zone);
            if (s == State.AfterFull) FrontRoomsLook.ApplyAmbient();
        }
        return s + ": panes " + panes + ", prop renderers " + props;
    }

    // ---------------- zone rule (the harness's; the map chat may pick differently) ----------------
    static object FixtureOf(GridCoord cell)
    {
        if (!(Get(map, "built") is IDictionary built)) return null;
        foreach (DictionaryEntry e in built)
            foreach (var f in (IEnumerable)Get(e.Value, "fixtures"))
            {
                var l = Get<Light>(f, "light");
                if (l != null && map.CellOf(l.transform.position) == cell) return f;
            }
        return null;
    }

    static FrontRoomsLook.ReflectionZone ZoneFor(Vector3 feet, out string why)
    {
        var cell = map.CellOf(feet);
        var z = map.ZoneOf(cell);
        var fx = FixtureOf(cell);
        string lamp;
        var dead = false;
        if (fx == null) lamp = "no lamp in the cell";
        else
        {
            var mode = (int)Get(fx, "mode");
            var level = (float)Get(fx, "level");
            var l = Get<Light>(fx, "light");
            lamp = "lamp mode " + mode + " level " + F2(level) + " enabled " + l.enabled;
            dead = mode == 3 || !l.enabled;
        }
        FrontRoomsLook.ReflectionZone r;
        if (dead) r = FrontRoomsLook.ReflectionZone.DeadLamp;
        else if (z.theme == ZoneTheme.Office) r = FrontRoomsLook.ReflectionZone.Office;
        else if (z.height == ZoneHeight.Tall) r = FrontRoomsLook.ReflectionZone.Tall;
        else r = FrontRoomsLook.ReflectionZone.Level0;
        why = "cell " + cell + " " + z.theme + "/" + z.height + ", " + lamp;
        return r;
    }

    // ---------------- pane mask + metrics ----------------
    static Vector2? ToPixel(Vector3 world)
    {
        var p = Matrix4x4.Perspective(cam.fieldOfView, (float)W / H, cam.nearClipPlane, cam.farClipPlane);
        var c = p * cam.worldToCameraMatrix * new Vector4(world.x, world.y, world.z, 1f);
        if (c.w <= 1e-4f) return null;
        return new Vector2((c.x / c.w * .5f + .5f) * W, (c.y / c.w * .5f + .5f) * H);
    }

    static List<Vector2> PaneQuad(Renderer pane)
    {
        var t = pane.transform;
        var s = paneOrig.TryGetValue(pane, out var o) ? o.scale : t.localScale;
        var thin = ThinAxis(s);
        var v = thin == 1 ? 2 : 1;
        var u = 3 - thin - v;
        var quad = new List<Vector2>();
        foreach (var (su, sv) in new[] { (-.5f, -.5f), (.5f, -.5f), (.5f, .5f), (-.5f, .5f) })
        {
            var local = Vector3.zero;
            local[u] = su; local[v] = sv;
            var px = ToPixel(t.TransformPoint(local));
            if (!px.HasValue) return null;
            quad.Add(px.Value);
        }
        return quad;
    }

    static bool[] Mask(List<Vector2> quad, float inset, out int count, out RectInt box)
    {
        var centre = new Vector2(quad.Average(q => q.x), quad.Average(q => q.y));
        var q2 = quad.Select(q => Vector2.Lerp(q, centre, inset)).ToList();
        var mask = new bool[W * H];
        count = 0;
        var x0 = Mathf.Clamp(Mathf.FloorToInt(q2.Min(q => q.x)), 0, W - 1);
        var x1 = Mathf.Clamp(Mathf.CeilToInt(q2.Max(q => q.x)), 0, W - 1);
        var y0 = Mathf.Clamp(Mathf.FloorToInt(q2.Min(q => q.y)), 0, H - 1);
        var y1 = Mathf.Clamp(Mathf.CeilToInt(q2.Max(q => q.y)), 0, H - 1);
        box = new RectInt(x0, y0, x1 - x0 + 1, y1 - y0 + 1);
        float Cross(Vector2 a, Vector2 b, Vector2 p) => (b.x - a.x) * (p.y - a.y) - (b.y - a.y) * (p.x - a.x);
        for (var y = y0; y <= y1; y++)
        for (var x = x0; x <= x1; x++)
        {
            var p = new Vector2(x + .5f, y + .5f);
            int pos = 0, neg = 0;
            for (var i = 0; i < 4; i++)
            {
                var cr = Cross(q2[i], q2[(i + 1) % 4], p);
                if (cr > 0) pos++; else if (cr < 0) neg++;
            }
            if (pos == 0 || neg == 0) { mask[y * W + x] = true; count++; }
        }
        return mask;
    }

    static float Y(Color32 c) => .2126f * c.r + .7152f * c.g + .0722f * c.b;

    static string Stats(string label, Color32[] with, Color32[] without, bool[] mask)
    {
        double n = 0, sd = 0, sdd = 0, sy = 0, syy = 0, sn = 0, snn = 0, sabs = 0, dr = 0, dg = 0, db = 0;
        for (var i = 0; i < mask.Length; i++)
        {
            if (!mask[i]) continue;
            var a = with[i]; var b = without[i];
            var ya = Y(a); var yb = Y(b);
            var d = ya - yb;
            n++; sd += d; sdd += d * d; sy += ya; syy += ya * ya; sn += yb; snn += yb * yb;
            sabs += (Mathf.Abs(a.r - b.r) + Mathf.Abs(a.g - b.g) + Mathf.Abs(a.b - b.b)) / 3.0;
            dr += a.r - b.r; dg += a.g - b.g; db += a.b - b.b;
        }
        if (n < 1) return label + ": empty mask";
        var md = sd / n; var vd = Math.Max(0, sdd / n - md * md);
        var my = sy / n; var vy = Math.Max(0, syy / n - my * my);
        var mn = sn / n; var vn = Math.Max(0, snn / n - mn * mn);
        return label + ": pixels " + n + " · meanY with " + F(my, "0.0") + " / without " + F(mn, "0.0")
            + " · ΔY mean " + F(md, "+0.00;-0.00") + " · ΔY std " + F(Math.Sqrt(vd)) + " · contrast kept σ(with)/σ(without) " + F(vn > 1e-6 ? Math.Sqrt(vy) / Math.Sqrt(vn) : 0, "0.000")
            + " · mean|ΔRGB| " + F(sabs / n) + " · ΔR/ΔG/ΔB " + F(dr / n, "+0.0;-0.0") + "/" + F(dg / n, "+0.0;-0.0") + "/" + F(db / n, "+0.0;-0.0");
    }

    // Structure inside the pane quad (run 3). Mean ΔY cancels by construction when the reflected room and the room
    // behind are equally bright, so also measure what the pane does to the image's structure:
    //   HP|Δ|   mean |hp(with) − hp(without)| of luminance (/255), hp = Y − its 9×9 box mean;
    //   1−SSIM  1 − mean luminance SSIM (7×7 window, 8-bit constants); 0 = the same structure as no pane.
    static string Structure(Color32[] with, Color32[] without, bool[] mask)
    {
        int x0 = W, y0 = H, x1 = -1, y1 = -1;
        for (var i = 0; i < mask.Length; i++)
        {
            if (!mask[i]) continue;
            int x = i % W, y = i / W;
            if (x < x0) x0 = x; if (x > x1) x1 = x; if (y < y0) y0 = y; if (y > y1) y1 = y;
        }
        if (x1 < 0) return "structure: empty mask";
        int bx0 = Math.Max(0, x0 - 4), by0 = Math.Max(0, y0 - 4), bx1 = Math.Min(W - 1, x1 + 4), by1 = Math.Min(H - 1, y1 + 4);
        int bw = bx1 - bx0 + 1, bh = by1 - by0 + 1, sw = bw + 1;
        var ia = new double[sw * (bh + 1)]; var ib = new double[ia.Length]; var iaa = new double[ia.Length]; var ibb = new double[ia.Length]; var iab = new double[ia.Length];
        for (var y = 0; y < bh; y++)
        for (var x = 0; x < bw; x++)
        {
            var src = (by0 + y) * W + bx0 + x;
            double a = Y(with[src]), b = Y(without[src]);
            int k = (y + 1) * sw + x + 1, up = y * sw + x + 1, left = (y + 1) * sw + x, diag = y * sw + x;
            ia[k] = a + ia[up] + ia[left] - ia[diag];
            ib[k] = b + ib[up] + ib[left] - ib[diag];
            iaa[k] = a * a + iaa[up] + iaa[left] - iaa[diag];
            ibb[k] = b * b + ibb[up] + ibb[left] - ibb[diag];
            iab[k] = a * b + iab[up] + iab[left] - iab[diag];
        }
        double Box(double[] I, int cx, int cy, int r, out int n)
        {
            int xa = Math.Max(0, cx - r), xb = Math.Min(bw - 1, cx + r), ya = Math.Max(0, cy - r), yb = Math.Min(bh - 1, cy + r);
            n = (xb - xa + 1) * (yb - ya + 1);
            return I[(yb + 1) * sw + xb + 1] - I[ya * sw + xb + 1] - I[(yb + 1) * sw + xa] + I[ya * sw + xa];
        }
        const double C1 = 6.5025, C2 = 58.5225;
        double hp = 0, ssim = 0; long cnt = 0;
        for (var i = 0; i < mask.Length; i++)
        {
            if (!mask[i]) continue;
            int lx = i % W - bx0, ly = i / W - by0;
            double a = Y(with[i]), b = Y(without[i]);
            var ma4 = Box(ia, lx, ly, 4, out var n4) / n4; var mb4 = Box(ib, lx, ly, 4, out _) / n4;
            hp += Math.Abs((a - ma4) - (b - mb4));
            var ma = Box(ia, lx, ly, 3, out var n3) / n3; var mb = Box(ib, lx, ly, 3, out _) / n3;
            var va = Math.Max(0, Box(iaa, lx, ly, 3, out _) / n3 - ma * ma);
            var vb = Math.Max(0, Box(ibb, lx, ly, 3, out _) / n3 - mb * mb);
            var cab = Box(iab, lx, ly, 3, out _) / n3 - ma * mb;
            ssim += ((2 * ma * mb + C1) * (2 * cab + C2)) / ((ma * ma + mb * mb + C1) * (va + vb + C2));
            cnt++;
        }
        return "structure: HP|Δ| " + F(hp / cnt, "0.000") + " · 1−SSIM " + F(1.0 - ssim / cnt, "0.0000");
    }

    static string Diff(Color32[] a, Color32[] b)
    {
        double s = 0; var peak = 0;
        for (var i = 0; i < a.Length; i++)
        {
            var e = (Mathf.Abs(a[i].r - b[i].r) + Mathf.Abs(a[i].g - b[i].g) + Mathf.Abs(a[i].b - b[i].b)) / 3;
            s += e; if (e > peak) peak = e;
        }
        return "mean " + F(s / a.Length, "0.000") + " /255, max " + peak + " /255";
    }

    // ---------------- one view, all states ----------------
    // Run 2: amplified |AFTER - no pane| image and three single-knob variants of Glass_Window (temporary material
    // copies, never the asset) so the fix pass can see which term makes the pane visible.
    static void SaveDiff(string name, Color32[] a, Color32[] b, float gain)
    {
        var px = new Color32[a.Length];
        for (var i = 0; i < a.Length; i++)
        {
            byte D(byte x, byte y) => (byte)Mathf.Clamp(Mathf.Abs(x - y) * gain, 0f, 255f);
            px[i] = new Color32(D(a[i].r, b[i].r), D(a[i].g, b[i].g), D(a[i].b, b[i].b), 255);
        }
        var t = new Texture2D(W, H, TextureFormat.RGB24, false);
        t.SetPixels32(px);
        t.Apply(false);
        File.WriteAllBytes(Path.Combine(OutDir, name), t.EncodeToJPG(85));
        UnityEngine.Object.Destroy(t);
    }

    static void Place(Vector3 feet, Vector3 lookAt)
    {
        var root = PlayerRoot;
        var body = Get<CharacterController>(game, "playerBody");
        body.enabled = false;
        root.position = feet;
        body.enabled = true;
        Physics.SyncTransforms();
        Look(lookAt);
    }

    static void Look(Vector3 lookAt)
    {
        var root = PlayerRoot;
        var eye = root.position + Vector3.up * ModuleUnits.PlayerEye;
        var d = lookAt - eye;
        var yaw = Mathf.Atan2(d.x, d.z) * Mathf.Rad2Deg;
        var pitch = -Mathf.Asin(Mathf.Clamp(d.y / Mathf.Max(1e-4f, d.magnitude), -1f, 1f)) * Mathf.Rad2Deg;
        Set(game, "yaw", yaw);
        Set(game, "pitch", pitch);
        root.rotation = Quaternion.Euler(0f, yaw, 0f);
        cam.transform.localRotation = Quaternion.Euler(pitch, 0f, 0f);
    }

    static IEnumerator Settle(int extra = 20)
    {

    }

    // ---------------- map queries (as the audit harness) ----------------
    static readonly GridCoord[] Steps = { new GridCoord(1, 0), new GridCoord(-1, 0), new GridCoord(0, 1), new GridCoord(0, -1) };

    static IEnumerable<GridCoord> CellsAround(GridCoord center, int radiusCells)
    {
        for (var r = 0; r <= radiusCells; r++)
        for (var dy = -r; dy <= r; dy++)
        for (var dx = -r; dx <= r; dx++)
        {
            if (Mathf.Max(Mathf.Abs(dx), Mathf.Abs(dy)) != r) continue;
            yield return new GridCoord(center.x + dx, center.y + dy);
        }
    }

    static Vector3 Dir(GridCoord a, GridCoord b)
    {
        var d = map.CellCenter(b) - map.CellCenter(a);
        d.y = 0f;
        return d.normalized;
    }

    static bool FreeStanding(Vector3 feet)
    {
        var r = ModuleUnits.PlayerRadius;
        var root = PlayerRoot;
        foreach (var c in Physics.OverlapCapsule(feet + Vector3.up * (r + .15f), feet + Vector3.up * (ModuleUnits.PlayerHeight - r), r, ~0, QueryTriggerInteraction.Ignore))
            if (root == null || !c.transform.IsChildOf(root)) return false;
        return true;
    }

    static Collider WindowCollider(GridCoord a, GridCoord b)
    {
        var dict = Get(map, "windowByCollider") as IDictionary;
        var p = map.CrossingPoint(a, b);
        foreach (DictionaryEntry e in dict)
        {
            var pos = (Vector3)Get(e.Value, "position");
            if ((new Vector2(pos.x - p.x, pos.z - p.z)).sqrMagnitude < .01f) return (Collider)e.Key;
        }
        return null;
    }

    static int StraightRun(GridCoord c, GridCoord step, int max = 12)
    {
        var n = 0;
        var cur = c;
        while (n < max)
        {
            var next = cur + step;
            if (!map.IsBuilt(next) || map.PassageBetween(cur, next) != FrontRoomsMapWorld.Passage.Open) break;
            cur = next;
            n++;
        }
        return n;
    }

    static List<object> AllFixtures()
    {
        var list = new List<object>();
        var built = Get(map, "built") as IDictionary;
        foreach (DictionaryEntry e in built)
            foreach (var f in (IEnumerable)Get(e.Value, "fixtures")) list.Add(f);
        return list;
    }

    static bool DressingNear(Vector3 point, float radius)
    {
        foreach (Transform chunk in map.transform)
        foreach (Transform t in chunk)
            if (t.name == "office dressing")
                foreach (var r in t.GetComponentsInChildren<Renderer>())
                    if ((r.bounds.center - point).sqrMagnitude < radius * radius) return true;
        return false;
    }

    // ---------------- representative frames (audit scoring) ----------------
    static IEnumerator Representative(string id, string kind, GridCoord origin)
    {
        GridCoord? coarse = null;
        foreach (var c in CellsAround(origin, 50))
        {
            if (map.InStartArea(c)) continue;
            var z = map.Cache.ZoneOf(c);
            if (kind == "L0" && z.theme == ZoneTheme.Level0 && z.height == ZoneHeight.Standard) { coarse = c; break; }
            if (kind == "Office" && z.theme == ZoneTheme.Office) { coarse = c; break; }
            if (kind == "Dark") { coarse = origin; break; }
        }
        if (!coarse.HasValue) { Log(kind + ": no cell found"); yield break; }
        Place(map.CellCenter(coarse.Value), map.CellCenter(coarse.Value) + Vector3.forward * 5f + Vector3.up * 1.5f);
        yield return Settle(30);
        Resume();
        var here = map.CellOf(PlayerRoot.position);
        GridCoord best = here; GridCoord bestStep = Steps[0]; float bestScore = float.MinValue;
        var fixtures = AllFixtures();
        foreach (var c in CellsAround(here, 14))
        {
            if (!map.IsBuilt(c) || map.InStartArea(c)) continue;
            var z = map.ZoneOf(c);
            if (kind == "L0" && (z.theme != ZoneTheme.Level0 || z.height != ZoneHeight.Standard)) continue;
            if (kind == "Office" && z.theme != ZoneTheme.Office) continue;
            if (!FreeStanding(map.CellCenter(c))) continue;
            foreach (var s in Steps)
            {
                var run = StraightRun(c, s);
                float score;
                if (kind == "Dark")
                {
                    if (run < 2) continue;
                    var light = 0f;
                    var p = map.CellCenter(c);
                    foreach (var f in fixtures)
                    {
                        var mode = (int)Get(f, "mode");
                        var lt = Get<Light>(f, "light");
                        if (lt == null) continue;
                        var d = Vector3.Distance(new Vector3(lt.transform.position.x, 0f, lt.transform.position.z), p);
                        var bi = (float)Get(f, "baseIntensity");
                        light += (mode == 3 ? .02f : mode == 4 ? .3f : mode == 2 ? .6f : 1f) * bi / (1f + d * d);
                    }
                    score = -light + run * .001f;
                }
                else if (kind == "Office")
                {
                    if (run < 2) continue;
                    score = run + (DressingNear(map.CellCenter(c), 5f) ? 20f : 0f);
                }
                else score = run;
                if (score > bestScore) { bestScore = score; best = c; bestStep = s; }
            }
        }
        lastRepCell = best;
        var feet = map.CellCenter(best);
        Place(feet, map.CellCenter(best + bestStep + bestStep + bestStep) + Vector3.up * 1.45f);
        yield return Settle(20);
        Resume();
        var z2 = map.ZoneOf(best);
        yield return Cycle(id, "representative " + kind + " · cell " + best + " " + z2.theme + "/" + z2.height + " · looking " + bestStep + " (run " + StraightRun(best, bestStep) + ") · score " + F2(bestScore) + " (audit procedure)", null, kind == "L0");
        if (kind == "Office")
        {
            var side = new GridCoord(bestStep.y, -bestStep.x);
            Look(map.CellCenter(best + side + side) + Vector3.up * 1.3f);
            yield return 3;
            yield return Cycle("35_rep_Office_b", "Office, quarter turn (audit frame 35)");
        }
    }

        foreach (var c in CellsAround(here, 14))

            if (z.theme != ZoneTheme.Level0 || z.height == ZoneHeight.Tall) continue;

            if (fx == null || (int)Get(fx, "mode") != 3) continue;
            if (!FreeStanding(map.CellCenter(c))) continue;
            GridCoord? step = null; var bestRun = 1;
            foreach (var s in Steps) { var r = StraightRun(c, s); if (r > bestRun) { bestRun = r; step = s; } }
            if (!step.HasValue) continue;
            Place(map.CellCenter(c), map.CellCenter(c + step.Value + step.Value + step.Value) + Vector3.up * 1.45f);
            yield return Settle(20);
            Resume();
            yield return Cycle("38_deadlamp_cell", "standing under a dead lamp (mode 3) · cell " + c + " " + z.theme + "/" + z.height + " · looking " + step.Value + " (run " + bestRun + ")");
            yield break;
        }
        Log("dead-lamp cell: none within 14 cells");
    }

    // ---------------- windows (audit viewpoints) ----------------
    static IEnumerator WindowSequence(string label, List<(GridCoord a, GridCoord b)> candidates)
    {
        foreach (var (a0, b0) in candidates)
        {
            var a = a0; var b = b0;
            var cross = map.CrossingPoint(a, b);
            var dir = Dir(a, b);
            var center = cross + Vector3.up * ((ModuleUnits.WindowSill + ModuleUnits.WindowTop) * .5f);
            Place(cross - dir * 1.5f, center);
            yield return Settle(20);
            Resume();
            var col = WindowCollider(a, b);
            if (col == null) { Log(label + ": window " + a + "-" + b + " not built; next"); continue; }
            if (!FreeStanding(cross - dir * 1.5f))
            {
                var t = a; a = b; b = t; dir = -dir;
                if (!FreeStanding(cross - dir * 1.5f)) { Log(label + ": both sides blocked; next"); continue; }
            }
            var pane = col.GetComponent<Renderer>();
            var za = map.ZoneOf(a); var zb = map.ZoneOf(b);
            var where = "window " + a + "→" + b + " · camera side " + za.theme + "/" + za.height + " · far side " + zb.theme + "/" + zb.height;
            var side = Vector3.Cross(Vector3.up, dir).normalized;

            Place(cross - dir * 1.5f, center);
            yield return 5;
            Resume();
            yield return Cycle(label + "_1.5m", "head-on from 1.5 m · " + where, pane, label == "window_L0", true);
            Place(cross - dir * 1.2f + side * 1.4f, center);
            yield return 4;
            yield return Cycle(label + "_oblique50", "~50° (1.2 m back, 1.4 m to the side; audit frame " + (label == "window_L0" ? "04" : "23") + ") · " + where, pane, false, true);
            Place(cross - dir * .7f, center);
            yield return 4;
            yield return Cycle(label + "_0.7m", "head-on from 0.7 m · " + where, pane, label == "window_L0", true);

            var steepOk = false; var grazeFeet = Vector3.zero;
            foreach (var off in new[] { .6f, .7f, .8f })
            foreach (var along in new[] { 1.15f, -1.15f, 1.0f, -1.0f })
            {
                if (steepOk) break;
                var f = cross - dir * off + side * along;
                if (!FreeStanding(f)) continue;
                var eye = f + Vector3.up * ModuleUnits.PlayerEye;
                var blocked = false;
                foreach (var h in Physics.RaycastAll(eye, (center - eye).normalized, (center - eye).magnitude - .02f, ~0, QueryTriggerInteraction.Ignore))
                    if (h.collider != col && !h.collider.transform.IsChildOf(PlayerRoot)) { blocked = true; break; }
                if (blocked) continue;
                grazeFeet = f; steepOk = true;
            }
            if (steepOk)
            {
                Place(grazeFeet, center);
                yield return 4;
                var dist = Vector3.Distance(new Vector3(grazeFeet.x, 0f, grazeFeet.z), new Vector3(cross.x, 0f, cross.z));
                var inc = Vector3.Angle(-dir, (grazeFeet + Vector3.up * ModuleUnits.PlayerEye - center).normalized);
                yield return Cycle(label + "_steep", "steep, " + F(inc, "0") + "° from the pane normal (" + F2(dist) + " m from the opening centre, same cell) · " + where, pane, false, true);
            }
            else Log(label + ": no clear steep viewpoint");
            yield break;
        }
        Log(label + ": no usable window among candidates");
    }

    // ---------------- dead lamp: a window seen from under a dead lamp, and a lit window with a dead room beyond ----------------
    static IEnumerator DeadLampWindow(GridCoord origin)
    {
        var cands = new List<(GridCoord a, GridCoord b)>();
        foreach (var c in CellsAround(origin, 70))
        {
            if (map.InStartArea(c)) continue;
            foreach (var s in new[] { new GridCoord(1, 0), new GridCoord(0, 1) })
            {
                var n = c + s;
                if (map.InStartArea(n) || map.Cache.Edge(c, n) != EdgeKind.Window) continue;
                cands.Add((c, n));
            }
            if (cands.Count >= 80) break;
        }
        Log("dead-lamp window search: " + cands.Count + " window edges within 70 cells");
        (GridCoord a, GridCoord b)? under = null, beyond = null;
        for (var pass = 0; pass < 2 && !(under.HasValue && beyond.HasValue); pass++)
        {
            foreach (var (c0, n0) in cands)
            {
                if (under.HasValue && beyond.HasValue) break;
                if (pass == 1 && !(map.IsBuilt(c0) && map.IsBuilt(n0)))
                {
                    Place(map.CellCenter(c0), map.CellCenter(n0) + Vector3.up * 1.2f);
                    yield return Settle(3);
                    Resume();
                }
                if (!map.IsBuilt(c0) || !map.IsBuilt(n0) || WindowCollider(c0, n0) == null) continue;
                foreach (var (a, b) in new[] { (c0, n0), (n0, c0) })
                {
                    var fa = FixtureOf(a); var fb = FixtureOf(b);
                    var deadA = fa != null && (int)Get(fa, "mode") == 3;
                    var deadB = fb != null && (int)Get(fb, "mode") == 3;
                    if (!FreeStanding(map.CellCenter(a))) continue;
                    if (!under.HasValue && deadA) under = (a, b);
                    else if (!beyond.HasValue && !deadA && fa != null && deadB) beyond = (a, b);
                }
            }
        }
        Log("dead-lamp windows: camera under a dead lamp " + (under.HasValue ? under.Value.a + "→" + under.Value.b : "none")
            + " · lit camera side, dead lamp beyond " + (beyond.HasValue ? beyond.Value.a + "→" + beyond.Value.b : "none"));
        if (under.HasValue) yield return DeadWindowViews("38_deadlamp_window", under.Value.a, under.Value.b, "camera under a dead lamp (mode 3)");
        else yield return DeadLampCell();
        if (beyond.HasValue) yield return DeadWindowViews("39_dark_beyond_window", beyond.Value.a, beyond.Value.b, "camera lit, dead lamp in the far cell");
    }

    /// <summary>World bounds of the sub-meshes that use a glass material (falls back to the renderer bounds).</summary>
    static Bounds GlassBounds(Renderer r)
    {
        var mf = r.GetComponent<MeshFilter>();
        var mats = r.sharedMaterials;
        if (mf != null && mf.sharedMesh != null)
        {
            Bounds? acc = null;
            for (var i = 0; i < mats.Length && i < mf.sharedMesh.subMeshCount; i++)
            {
                if (!IsGlassMat(mats[i])) continue;
                var lb = mf.sharedMesh.GetSubMesh(i).bounds;
                for (var k = 0; k < 8; k++)
                {
                    var corner = lb.center + Vector3.Scale(lb.extents, new Vector3((k & 1) == 0 ? -1 : 1, (k & 2) == 0 ? -1 : 1, (k & 4) == 0 ? -1 : 1));
                    var w = r.transform.TransformPoint(corner);
                    if (acc.HasValue) { var b = acc.Value; b.Encapsulate(w); acc = b; } else acc = new Bounds(w, Vector3.zero);
                }
            }
            if (acc.HasValue && acc.Value.size.sqrMagnitude > 1e-6f) return acc.Value;
        }
        return r.bounds;
    }

    static bool TryView(Vector3 target, Transform root, Vector3 front, float[] dists, float[] angs, out Vector3 feet)
    {
        feet = Vector3.zero;
        front.y = 0f;
        if (front.sqrMagnitude < 1e-4f) front = Vector3.forward;
        front.Normalize();
        var floorY = map.transform.position.y;
        foreach (var dist in dists)
        foreach (var ang in angs)
        {
            var dir = Quaternion.Euler(0f, ang, 0f) * front;
            var f = new Vector3(target.x, floorY, target.z) + dir * dist;
            if (!FreeStanding(f)) continue;
            var eye = f + Vector3.up * ModuleUnits.PlayerEye;
            var to = target - eye;
            var blocked = false;
            foreach (var h in Physics.RaycastAll(eye, to.normalized, to.magnitude, ~0, QueryTriggerInteraction.Ignore))
            {
                if (h.collider.transform.IsChildOf(PlayerRoot)) continue;
                if (root != null && h.collider.transform.IsChildOf(root)) continue;
                if (to.magnitude - h.distance < .2f) continue;
                blocked = true; break;
            }
            if (blocked) continue;
            feet = f;
            return true;
        }
        return false;
    }

    // ---------------- STAGED: Glass_Shard on Level 0 carpet (20 §8: untested until run 3) ----------------
    // Fourteen procedural shards (3-7 cm convex pieces, 6 mm thick, lying flat or propped up to 12°) in a 0.5 m patch
    // 1.0 m and 2.0 m ahead on Level 0 carpet. Faces take the variant's material, the thin side faces Glass_Edge:
    //   S1 Glass_Shard as set up (URP Lit opaque, near-black base, smoothness .95);
    //   S2 opaque, base = 0.85 × the carpet's mean albedo (Carpet_LoopPile_A.png mean sRGB .539/.473/.326), smoothness .95;
    //   S3 transparent FrontRooms/Glass (a copy of Prop_Glass, _DustFilm 0).
    // Mask: the same shards drawn unlit magenta, against no shards. Metric: mean Y of the shard pixels with and
    // without shards, and the fraction of shard pixels darker than 0.6 × the carpet under them ("dark chips").
    static void AddTri(List<Vector3> v, List<Vector3> n, List<int> t, Vector3 a, Vector3 b, Vector3 c, Vector3 want)
    {
        var nn = Vector3.Cross(b - a, c - a);
        if (Vector3.Dot(nn, want) < 0f) { var x = b; b = c; c = x; }
        var normal = want.normalized;
        var i = v.Count;
        v.Add(a); v.Add(b); v.Add(c);
        n.Add(normal); n.Add(normal); n.Add(normal);
        t.Add(i); t.Add(i + 1); t.Add(i + 2);
    }

    static Mesh ShardMesh(System.Random rng)
    {
        var count = rng.Next(4, 7);
        var size = .03f + (float)rng.NextDouble() * .04f;
        var stretch = 1f + (float)rng.NextDouble();
        var pts = new List<Vector3>();
        for (var i = 0; i < count; i++)
        {
            var a = (i + .7f * (float)rng.NextDouble()) / count * Mathf.PI * 2f;
            var r = size * (.55f + .45f * (float)rng.NextDouble());
            pts.Add(new Vector3(Mathf.Cos(a) * r * stretch, 0f, Mathf.Sin(a) * r));
        }
        const float thick = .006f;
        var up = Vector3.up * thick;
        var c0 = Vector3.zero;
        foreach (var q in pts) c0 += q;
        c0 /= count;
        var v = new List<Vector3>(); var n = new List<Vector3>(); var faces = new List<int>(); var sides = new List<int>();
        for (var i = 0; i < count; i++)
        {
            var p0 = pts[i]; var p1 = pts[(i + 1) % count];
            AddTri(v, n, faces, c0 + up, p0 + up, p1 + up, Vector3.up);
            AddTri(v, n, faces, c0, p0, p1, Vector3.down);
            var outward = Vector3.Cross(p1 - p0, Vector3.up);
            if (Vector3.Dot(outward, (p0 + p1) * .5f - c0) < 0f) outward = -outward;
            AddTri(v, n, sides, p0, p1, p1 + up, outward);
            AddTri(v, n, sides, p0, p1 + up, p0 + up, outward);
        }
        var m = new Mesh { name = "G10 shard" };
        m.SetVertices(v);
        m.SetNormals(n);
        m.subMeshCount = 2;
        m.SetTriangles(faces, 0);
        m.SetTriangles(sides, 1);
        m.RecalculateBounds();
        return m;
    }

    static List<MeshRenderer> SpawnShards(Transform parent, Vector3 centre, float floorY, System.Random rng)
    {
        var list = new List<MeshRenderer>();
        for (var i = 0; i < 14; i++)
        {
            var go = new GameObject("G10 shard " + i);
            go.transform.SetParent(parent, false);
            var ang = (float)rng.NextDouble() * Mathf.PI * 2f;
            var rad = .25f * Mathf.Sqrt((float)rng.NextDouble());
            var tilt = rng.NextDouble() < .3 ? 5f + 7f * (float)rng.NextDouble() : 0f;
            go.transform.position = new Vector3(centre.x + Mathf.Cos(ang) * rad, floorY + .001f + (tilt > 0f ? .006f : 0f), centre.z + Mathf.Sin(ang) * rad);
            go.transform.rotation = Quaternion.Euler(tilt, (float)rng.NextDouble() * 360f, 0f);
            go.AddComponent<MeshFilter>().sharedMesh = ShardMesh(rng);
            list.Add(go.AddComponent<MeshRenderer>());
        }
        return list;
    }

    static string ShardStats(string label, Color32[] with, Color32[] without, bool[] mask)
    {
        double n = 0, sw = 0, so = 0, dark = 0;
        for (var i = 0; i < mask.Length; i++)
        {
            if (!mask[i]) continue;
            var a = Y(with[i]); var b = Y(without[i]);
            n++; sw += a; so += b;
            if (a < .6f * b) dark++;
        }
        if (n < 1) return label + ": empty mask";
        return label + ": shard pixels " + n + " · meanY shards " + F(sw / n, "0.0") + " / carpet without shards " + F(so / n, "0.0")
            + " · ratio " + F(sw / Math.Max(1e-3, so), "0.00") + " · dark chips (< 0.6 × carpet) " + F(100.0 * dark / n, "0.0") + " %";

    // ================= G14 hook proof (RunProofBatch) =================
    // Pass 1 visits every view and renders, with post off into an ARGBFloat 4x MSAA target (HDR buffer 64 bit):
    //   B1 baseline (Hidden/FrontRooms/GlassBaseline = the shader without the hook), (i) the hook shader with
    //   _FR_GlassRTWeight / _FR_GlassRTReflection never set in this process, B2 baseline again (determinism).
    // Pass 2 replays the same poses: B baseline, (ii) weight 0 + 1x1 red A=1, (iii) weight 1 + A=0,
    //   P1 weight 1, RT = (2,0,0,1) (glass must change) and the same with every window pane hidden (props must not),
    //   P2 orientation (only the top-left texel of a 2x2 RT is covered), P3 depth tag far away (no change),
    //   P4 depth tag = the pane's own eye depth per pixel (the pane changes), P5 tag + 0.5 m (no change),
    //   P6 = P4 at render scale 0.75.
    sealed class ProofView { public string id, note; public Vector3 feet, paneAt; public float yaw, pitch; public Renderer pane; public FrontRoomsLook.ReflectionZone zone; }

    {
        foreach (var r in AllPanes())
            if (r != null && (r.transform.position - p).sqrMagnitude < 1e-4f) return r;
        return null;
    }

    static float[] RenderFloat(string jpg = null)
    {
        if (prt == null)
        {
            var desc = new RenderTextureDescriptor(W, H, RenderTextureFormat.ARGBFloat, 24) { msaaSamples = 4, sRGB = false };
            var supported = SystemInfo.GetRenderTextureSupportedMSAASampleCount(desc);
            prt = new RenderTexture(desc) { name = "G10 proof (ARGBFloat)" };
            prt.Create();
            pread = new Texture2D(W, H, TextureFormat.RGBAFloat, false, true);
            Log("proof target ARGBFloat " + W + "x" + H + " · MSAA requested 4, supported " + supported + ", texture " + prt.antiAliasing);
        }
        var prev = cam.targetTexture;
        cam.targetTexture = prt;
        cam.Render();
        cam.Render();
        cam.targetTexture = prev;
        var active = RenderTexture.active;
        RenderTexture.active = prt;
        pread.ReadPixels(new Rect(0, 0, W, H), 0, 0);
        pread.Apply(false);
        RenderTexture.active = active;
        var px = pread.GetRawTextureData<float>().ToArray();
        if (jpg != null) SaveFloatJpg(jpg, px);
        return px;
    }

    static void SaveFloatJpg(string name, float[] px)
    {
        if (pjpg == null) pjpg = new Texture2D(W, H, TextureFormat.RGB24, false);
        var c = new Color32[W * H];
        byte B(float v) => (byte)Mathf.Clamp(Mathf.LinearToGammaSpace(Mathf.Max(0f, v)) * 255f + .5f, 0f, 255f);
        for (int p = 0, i = 0; p < c.Length; p++, i += 4) c[p] = new Color32(B(px[i]), B(px[i + 1]), B(px[i + 2]), 255);
        pjpg.SetPixels32(c);
        pjpg.Apply(false);
        File.WriteAllBytes(Path.Combine(OutDir, name), pjpg.EncodeToJPG(90));
    }

    sealed class FDiff { public long pixels, inMask, maskPixels; public double maxAbs; public long[] quad = new long[4], maskQuad = new long[4]; }

    // Quadrants: 0 top-left, 1 top-right, 2 bottom-left, 3 bottom-right (image as displayed; rows bottom-up).
    static int Quadrant(int p) { int x = p % W, y = p / W; return (y >= H / 2 ? 0 : 2) + (x >= W / 2 ? 1 : 0); }

    static FDiff FloatDiff(float[] a, float[] b, bool[] mask = null)
    {
        var d = new FDiff();
        for (int p = 0, i = 0; p < W * H; p++, i += 4)
        {
            if (mask != null && mask[p]) { d.maskPixels++; d.maskQuad[Quadrant(p)]++; }
            var diff = false;
            for (var c = 0; c < 4; c++)
            {
                if (BitConverter.SingleToInt32Bits(a[i + c]) == BitConverter.SingleToInt32Bits(b[i + c])) continue;
                diff = true;
                var e = Math.Abs((double)a[i + c] - b[i + c]);
                if (!double.IsNaN(e) && e > d.maxAbs) d.maxAbs = e;
            }
            if (!diff) continue;
            d.pixels++;
            d.quad[Quadrant(p)]++;
            if (mask != null && mask[p]) d.inMask++;
        }
        return d;
    }

    static string DiffText(FDiff d) => "differing pixels " + d.pixels + " · max|Δ| " + d.maxAbs.ToString("G6", CultureInfo.InvariantCulture)
        + (d.maskPixels > 0 ? " · in the pane mask " + d.inMask + " of " + d.maskPixels + " (" + F(100.0 * d.inMask / d.maskPixels, "0.0") + " %) · outside " + (d.pixels - d.inMask) : "")
        + " · by quadrant TL/TR/BL/BR " + string.Join("/", d.quad);

    static void Check(string id, string what, bool ok, string detail)
    {
        proofChecks++;
        if (!ok) proofFails.Add(id + " " + what);
        metrics.AppendLine("  " + (ok ? "PASS " : "FAIL ") + what + ": " + detail);
        Flush();
    }

    static Texture2D TexF(int w, int h, float[] rgba)
    {
        var t = new Texture2D(w, h, TextureFormat.RGBAFloat, false, true) { filterMode = FilterMode.Point, wrapMode = TextureWrapMode.Clamp, name = "G10 proof RT input" };
        t.SetPixelData(rgba, 0);
        t.Apply(false);
        return t;
    }

    static void SetRT(Texture t, float weight)
    {
        if (t != null) Shader.SetGlobalTexture(RTTexId, t);
        Shader.SetGlobalFloat(RTWeightId, weight);
    }

    /// <summary>RGB (2,0,0) and A = 1 + eye depth (+ extra) of the pane's camera-side face, per pixel; 0 elsewhere.</summary>
    static float[] DepthTag(Renderer pane, float extra, out int count)
    {
        var data = new float[W * H * 4];
        count = 0;
        var t = pane.transform;
        var s = t.lossyScale;
        var thin = ThinAxis(s);
        var vAx = thin == 1 ? 2 : 1;
        var uAx = 3 - thin - vAx;
        var axes = new[] { t.right, t.up, t.forward };
        var n = axes[thin].normalized;
        var camPos = cam.transform.position;
        var fwd = cam.transform.forward;
        var side = Mathf.Sign(Vector3.Dot(camPos - t.position, n));
        var planePoint = t.position + n * (Mathf.Abs(s[thin]) * .5f * side);
        var inv = (Matrix4x4.Perspective(cam.fieldOfView, (float)W / H, cam.nearClipPlane, cam.farClipPlane) * cam.worldToCameraMatrix).inverse;
        for (var y = 0; y < H; y++)
        for (var x = 0; x < W; x++)
        {
            var farP = inv * new Vector4((x + .5f) / W * 2f - 1f, (y + .5f) / H * 2f - 1f, 1f, 1f);
            var dir = ((Vector3)farP / farP.w - camPos).normalized;
            var den = Vector3.Dot(dir, n);
            if (Mathf.Abs(den) < 1e-6f) continue;
            var tt = Vector3.Dot(planePoint - camPos, n) / den;
            if (tt <= 0f) continue;
            var hit = camPos + dir * tt;
            var local = t.InverseTransformPoint(hit);
            if (Mathf.Abs(local[uAx]) > .5f || Mathf.Abs(local[vAx]) > .5f) continue;
            var i = (y * W + x) * 4;
            data[i] = 2f;
            data[i + 3] = 1f + Vector3.Dot(hit - camPos, fwd) + extra;
            count++;
        }
        return data;
    }

    static IEnumerator ProofMain(GridCoord origin, List<(GridCoord a, GridCoord b)> l0, List<(GridCoord a, GridCoord b)> office)
    {
        baselineShader = AssetDatabase.LoadAssetAtPath<Shader>("Assets/Editor/Audit/G10/FrontRoomsGlassBaseline.shader");
        var hookShader = glassWindow.shader;
        Log("proof: hook shader " + hookShader.name + " (error " + ShaderUtil.ShaderHasError(hookShader) + ") · baseline " + (baselineShader != null ? baselineShader.name + " (error " + ShaderUtil.ShaderHasError(baselineShader) + ")" : "MISSING")
            + " · Glass_Window _RTReceive " + (glassWindow.HasProperty("_RTReceive") ? F2(glassWindow.GetFloat("_RTReceive")) : "n/a")
            + " · Prop_Glass _RTReceive " + (propGlassNew.HasProperty("_RTReceive") ? F2(propGlassNew.GetFloat("_RTReceive")) : "n/a"));
        if (baselineShader == null || ShaderUtil.ShaderHasError(baselineShader) || ShaderUtil.ShaderHasError(hookShader)) { Log("FAILED: shader missing or has errors"); yield break; }
        baseWin = ProofCopy(glassWindow, baselineShader, "baseline"); hookWin = ProofCopy(glassWindow, null, "hook");
        baseProp = ProofCopy(propGlassNew, baselineShader, "baseline"); hookProp = ProofCopy(propGlassNew, null, "hook");
        baseBottle = ProofCopy(propBottleNew, baselineShader, "baseline"); hookBottle = ProofCopy(propBottleNew, null, "hook");
        foreach (var m in new[] { baseProp, hookProp }) propGlassFamily.Add(m);
        foreach (var m in new[] { baseBottle, hookBottle }) propBottleFamily.Add(m);
        Log("proof copies: window keywords base [" + string.Join(" ", baseWin.shaderKeywords) + "] hook [" + string.Join(" ", hookWin.shaderKeywords) + "]");

        var urp = UniversalRenderPipeline.asset;
        var hdrWas = urp.hdrColorBufferPrecision;
        var scaleWas = urp.renderScale;
        urp.hdrColorBufferPrecision = HDRColorBufferPrecision._64Bits;
        var camData = cam.GetUniversalAdditionalCameraData();
        var postWas = camData.renderPostProcessing;
        camData.renderPostProcessing = false;
        Log("proof: post off · HDR buffer " + hdrWas + " -> " + urp.hdrColorBufferPrecision + " · render scale " + F2(scaleWas) + " · camera HDR " + cam.allowHDR
            + " · before any set: _FR_GlassRTWeight " + F2(Shader.GetGlobalFloat(RTWeightId)) + ", _FR_GlassRTReflection " + (Shader.GetGlobalTexture(RTTexId) != null ? Shader.GetGlobalTexture(RTTexId).name : "null"));

        metrics.AppendLine("# G14 hook proof — " + DateTime.Now.ToString("yyyy-MM-dd HH:mm", CultureInfo.InvariantCulture));
        metrics.AppendLine("post off, ARGBFloat target 4x MSAA, URP HDR buffer 64 bit, exact float comparison (bit patterns)");
        Shader.DisableKeyword(RTKeyword);
        metrics.AppendLine("## Pass 1 (globals never set, keyword " + RTKeyword + " off)");
        proofPass = 1;
        yield return Representative("01_rep_L0", "L0", origin);
        if (l0.Count > 0) yield return WindowSequence("window_L0", l0);
        if (office.Count > 0) yield return WindowSequence("window_Office", office);
        yield return Representative("34_rep_Office", "Office", origin);
        yield return DeadLampWindow(origin);
        yield return Props();
        yield return HutchStaged(origin);

        metrics.AppendLine("## Pass 2 (globals set)");
        proofPass = 2;
        foreach (var v in proofViews) yield return ProofPass2(v);

        SetRT(Texture2D.blackTexture, 0f);
        Shader.DisableKeyword(RTKeyword);
        urp.hdrColorBufferPrecision = hdrWas;
        urp.renderScale = scaleWas;
        camData.renderPostProcessing = postWas;
        metrics.AppendLine("## Summary: " + proofChecks + " checks, " + proofFails.Count + " failed" + (proofFails.Count > 0 ? ": " + string.Join("; ", proofFails) : ""));
        Log("proof summary: " + proofChecks + " checks, " + proofFails.Count + " failed");
        Flush();
    }

    static IEnumerator ProofCapture(string id, string note, Renderer pane)
    {
        if (proofPass != 1) yield break;
        var feet = PlayerRoot.position;
        var zone = ZoneFor(feet, out var why);
        var v = new ProofView { id = id, note = note, feet = feet, yaw = Get<float>(game, "yaw"), pitch = Get<float>(game, "pitch"), pane = pane, zone = zone, paneAt = pane != null ? pane.transform.position : Vector3.zero };
        proofViews.Add(v);
        Time.timeScale = 0f;
        yield return 2;
        UseProof(true);
        ApplyState(State.AfterFull, zone);
        yield return 3;
        var b1 = RenderFloat(id + "_B_baseline.jpg");
        UseProof(false);
        var h1 = RenderFloat();
        UseProof(true);
        var b2 = RenderFloat();
        Time.timeScale = 1f;
        metrics.AppendLine("### " + id + " (pass 1) — " + note + " · zone " + zone + " (" + why + ")");
        var det = FloatDiff(b1, b2);
        Check(id, "determinism: baseline vs baseline again", det.pixels == 0, DiffText(det));
        var d = FloatDiff(b1, h1);
        Check(id, "(i) hook shader (keyword " + RTKeyword + " off), globals never set, vs baseline", d.pixels == 0, DiffText(d));
        Log("proof pass 1 " + id + ": determinism " + det.pixels + " px, (i) " + d.pixels + " px");
    }

    static IEnumerator ProofPass2(ProofView v)
    {
        var body = Get<CharacterController>(game, "playerBody");
        body.enabled = false;
        PlayerRoot.position = v.feet;
        body.enabled = true;
        Physics.SyncTransforms();
        Set(game, "yaw", v.yaw);
        Set(game, "pitch", v.pitch);
        PlayerRoot.rotation = Quaternion.Euler(0f, v.yaw, 0f);
        cam.transform.localRotation = Quaternion.Euler(v.pitch, 0f, 0f);
        yield return Settle(10);
        Resume();
        Time.timeScale = 0f;
        yield return 2;
        UseProof(true);
        ApplyState(State.AfterFull, v.zone);
        yield return 3;
        metrics.AppendLine("### " + v.id + " (pass 2)");
        if (v.pane == null && v.paneAt != Vector3.zero)
        {
            v.pane = PaneAt(v.paneAt);
            if (v.pane != null) { UseProof(true); ApplyState(State.AfterFull, v.zone); yield return 3; }
            metrics.AppendLine("  pane was rebuilt by streaming; found again by position: " + (v.pane != null));
        }
        bool[] mask = null;
        if (v.pane != null)
        {
            var q = PaneQuad(v.pane);
            if (q != null) mask = Mask(q, 0f, out _, out _);
        }
        var red = TexF(1, 1, new[] { 1f, 0f, 0f, 1f });
        var zeroA = TexF(1, 1, new[] { 1f, 0f, 0f, 0f });
        var hot = TexF(1, 1, new[] { 2f, 0f, 0f, 1f });
        var far = TexF(1, 1, new[] { 2f, 0f, 0f, 1001f });
        var quadTex = TexF(2, 2, new[] { 0f, 0f, 0f, 0f, 0f, 0f, 0f, 0f, 2f, 0f, 0f, 1f, 0f, 0f, 0f, 0f });

        var b = RenderFloat();
        UseProof(false);
        // Keyword off (the default): the hook is not compiled in, whatever the globals say.
        Shader.DisableKeyword(RTKeyword);
        SetRT(red, 0f);
        var h2 = RenderFloat();
        SetRT(zeroA, 1f);
        var h3 = RenderFloat();
        SetRT(hot, 1f);
        var h4 = RenderFloat();
        var dh2 = FloatDiff(b, h2, mask);
        Check(v.id, "(ii) keyword off, weight 0, RT = 1x1 red A=1, vs baseline", dh2.pixels == 0, DiffText(dh2));
        var dh3 = FloatDiff(b, h3, mask);
        Check(v.id, "(iii) keyword off, weight 1, RT A=0, vs baseline", dh3.pixels == 0, DiffText(dh3));
        var dh4 = FloatDiff(b, h4, mask);
        Check(v.id, "(iv) keyword off, weight 1, RT (2,0,0,1), vs baseline", dh4.pixels == 0, DiffText(dh4));

        // Keyword on: what the RT pass turns on for the camera it feeds.
        Shader.EnableKeyword(RTKeyword);
        SetRT(red, 0f);
        var k0 = RenderFloat();
        var dk0 = FloatDiff(b, k0, mask);
        metrics.AppendLine("  info keyword on, weight 0, vs baseline (code differs, so the last bit may): " + DiffText(dk0));
        SetRT(zeroA, 1f);
        var k3 = RenderFloat();
        var dk3 = FloatDiff(k0, k3, mask);
        Check(v.id, "keyword on: weight 1 with A=0 equals weight 0", dk3.pixels == 0, DiffText(dk3));
        SetRT(hot, 1f);
        var p1 = RenderFloat(v.pane != null ? v.id + "_P1_rt_red.jpg" : null);
        var dp1 = FloatDiff(k0, p1, mask);
        if (v.pane != null) Check(v.id, "P1 keyword on, weight 1, RT (2,0,0,1): the pane changes", dp1.inMask > 0, DiffText(dp1));
        else metrics.AppendLine("  info P1 keyword on, weight 1, RT (2,0,0,1), no held pane in this view: " + DiffText(dp1));
        SetRT(far, 1f);
        var p3 = RenderFloat();
        var dp3 = FloatDiff(k0, p3, mask);
        Check(v.id, "P3 depth tag 1000 m (no glass at that depth) equals weight 0", dp3.pixels == 0, DiffText(dp3));
        if (v.pane != null && dp1.inMask > 0)
        {
            SetRT(quadTex, 1f);
            var p2 = RenderFloat();
            var dp2 = FloatDiff(k0, p2, mask);
            var onlyTL = dp2.quad[1] == 0 && dp2.quad[2] == 0 && dp2.quad[3] == 0;
            Check(v.id, "P2 orientation: only the RT's top-left texel is covered -> changes only in the screen's top-left quadrant", onlyTL && (dp2.quad[0] > 0 || dp2.maskQuad[0] == 0),
                DiffText(dp2) + " · pane pixels by quadrant " + string.Join("/", dp2.maskQuad));
        }

        // Props must not take the RT: hide every window pane, keyword on, weight 1 vs weight 0.
        var panes = AllPanes();
        foreach (var r in panes) r.enabled = false;
        SetRT(red, 0f);
        var bn = RenderFloat();
        SetRT(hot, 1f);
        var pn = RenderFloat();
        foreach (var r in panes) r.enabled = true;
        var dpn = FloatDiff(bn, pn);
        Check(v.id, "props (_RTReceive 0), every pane hidden, keyword on: weight 1 + RT (2,0,0,1) equals weight 0", dpn.pixels == 0, DiffText(dpn));

        if (v.pane != null && dp1.inMask > 0)
        {
            var tagData = DepthTag(v.pane, 0f, out var tagged);
            var tag = TexF(W, H, tagData);
            SetRT(tag, 1f);
            var p4 = RenderFloat(v.id + "_P4_depthtag.jpg");
            var dp4 = FloatDiff(k0, p4, mask);
            Check(v.id, "P4 depth tag = the pane's own eye depth (" + tagged + " tagged px): the pane changes like P1, nothing outside", dp4.inMask >= dp1.inMask * .95 && dp4.pixels - dp4.inMask <= Math.Max(50, dp4.inMask / 50),
                DiffText(dp4) + " · P1 changed " + dp1.inMask + " in the mask");
            UnityEngine.Object.Destroy(tag);
            var offData = DepthTag(v.pane, .5f, out _);
            var off = TexF(W, H, offData);
            SetRT(off, 1f);
            var p5 = RenderFloat();
            var dp5 = FloatDiff(k0, p5, mask);
            Check(v.id, "P5 depth tag + 0.5 m (as for a pane behind): equals weight 0", dp5.pixels == 0, DiffText(dp5));
            UnityEngine.Object.Destroy(off);

            var urp = UniversalRenderPipeline.asset;
            var scaleWas = urp.renderScale;
            urp.renderScale = .75f;
            SetRT(red, 0f);
            var b6 = RenderFloat();
            var tag6 = TexF(W, H, tagData);
            SetRT(tag6, 1f);
            var p6 = RenderFloat();
            urp.renderScale = scaleWas;
            var dp6 = FloatDiff(b6, p6, mask);
            Check(v.id, "P6 = P4 at render scale 0.75: the pane changes, (almost) nothing outside it", dp6.inMask >= dp1.inMask * .9 && dp6.pixels - dp6.inMask <= Math.Max(200, dp6.inMask / 20), DiffText(dp6));
            UnityEngine.Object.Destroy(tag6);
        }
        Shader.DisableKeyword(RTKeyword);
        SetRT(red, 0f);
        UseProof(true);
        foreach (var t in new[] { red, zeroA, hot, far, quadTex }) UnityEngine.Object.Destroy(t);
        Time.timeScale = 1f;
        Log("proof pass 2 " + v.id + " done");
        yield return 2;
    }
