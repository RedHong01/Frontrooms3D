// RT RUNTIME PROBE — private clone (proj_rt) only. NEVER copy into the real project.
// Plays Assets/Scenes/FrontRooms3D.unity with the same start as the interaction-audit and
// G11 harnesses (Random.InitState(4242) before the title start), finds a Level 0 window and
// probes the Metal hardware-RT glass prototype (FrontRoomsMetalGlassRT*):
//   - native capability vs SystemInfo.supportsRayTracing, plugin import settings;
//   - whether the game itself creates the RT controller;
//   - four views of one pane (head-on level gaze, head-on centred, 50 deg, steep), each with the
//     RT feature OFF and ON, the RT result texture read back (raw float RGBA), and ray-cast masks
//     of where the pane really is (correct projection) and where the prototype's projection puts it;
//   - RebuildScene CPU time, BLAS/TLAS build counts and times, trace GPU time (instrumented clone
//     plugin: FRGlassRT_Stat), frame time ON vs OFF;
//   - first-frame-after-rescan blank, one-frame lag, broken-pane ghost, Relay in front of / behind
//     the pane, second camera re-allocation, a planar mirror reference, a free-running 1 Hz rescan.
// Output: <clone>/Verification/rt_probe (PNG frames, .f32 raw RT readbacks, .u8 masks, probe_log.txt).
// Run: Unity -batchmode -projectPath <proj_rt> -executeMethod FrontRoomsGlassRTProbe.RunBatch -logFile <f>

    public static void RunBatch()
    {
        try { File.Delete(Path.Combine(OutDir, "probe_log_editor.txt")); } catch { }
        Caps("editor, before play mode");
        PluginInfo();
        EditorSceneManager.OpenScene(ScenePath, OpenSceneMode.Single);
        SessionState.SetBool(ActiveKey, true);
        SessionState.SetFloat(DeadlineKey, (float)EditorApplication.timeSinceStartup + 1100f);
        EditorApplication.update -= Tick;
        EditorApplication.update += Tick;
        EditorApplication.EnterPlaymode();
    }

    static string OutDir
    {
        get
        {
            if (outDir == null)
            {
                outDir = Path.Combine(Directory.GetParent(Application.dataPath).FullName, "Verification", "rt_probe");
                Directory.CreateDirectory(outDir);
            }
            return outDir;
        }
    }

    static void Log(string s)
    {
        var line = "[" + Time.frameCount + "] " + s;
        log.AppendLine(line);
        Debug.Log("[RTProbe] " + line);
        try { File.WriteAllText(Path.Combine(OutDir, EditorApplication.isPlaying ? "probe_log.txt" : "probe_log_editor.txt"), log.ToString()); } catch { }
    }

    static void Finish()
    {
        finished = true;
        Time.captureDeltaTime = 0f;
        SessionState.SetBool(DoneKey, true);
        try
        {
            File.WriteAllText(Path.Combine(OutDir, "probe_log.txt"), log.ToString());
            File.WriteAllText(Path.Combine(OutDir, "frames.txt"), frames.ToString());
        }
        catch { }
    }

    static void Tick()
    {
        var timedOut = EditorApplication.timeSinceStartup > SessionState.GetFloat(DeadlineKey, float.MaxValue);
        if (finished || timedOut || SessionState.GetBool(DoneKey, false))
        {
            if (timedOut && !finished && !SessionState.GetBool(DoneKey, false)) { Log("TIMEOUT"); Finish(); }
            if (EditorApplication.isPlaying) { EditorApplication.ExitPlaymode(); return; }
            if (EditorApplication.isPlayingOrWillChangePlaymode) return;
            EditorApplication.update -= Tick;
            SessionState.EraseBool(ActiveKey);
            SessionState.EraseBool(DoneKey);
            if (Application.isBatchMode) EditorApplication.Exit(0);
            return;
        }
        if (!EditorApplication.isPlaying) return;
        if (stack.Count == 0) { stack.Push(Main()); Log("routine created"); }
        if (Time.frameCount < waitUntilFrame) return;
        try
        {
            for (var guard = 0; guard < 1000; guard++)
            {
                var top = stack.Peek();
                if (!top.MoveNext())
                {
                    stack.Pop();
                    if (stack.Count == 0) { Finish(); return; }
                    continue;
                }
                var cur = top.Current;
                if (cur is IEnumerator sub) { stack.Push(sub); continue; }
                waitUntilFrame = Time.frameCount + (cur is int n ? Mathf.Max(1, n) : 1);
                return;
            }
        }
        catch (Exception e)
        {
            Log("EXCEPTION " + e);
            Finish();
        }
    }

    // ---------------- reflection helpers ----------------
    static FieldInfo FI(Type t, string n)
    {
        for (; t != null; t = t.BaseType)
        {
            var f = t.GetField(n, BF);
            if (f != null) return f;
        }
        return null;
    }
    static object Get(object o, string n) => FI(o.GetType(), n)?.GetValue(o);
    static T Get<T>(object o, string n) => (T)Get(o, n);
    static void Set(object o, string n, object v) => FI(o.GetType(), n).SetValue(o, v);
    static object Call(object o, string n, params object[] a) => o.GetType().GetMethod(n, BF).Invoke(o, a);
    static string Phase => Get(game, "phase").ToString();
    static Transform PlayerRoot => Get<Transform>(game, "playerRoot");
    static string V(Vector3 v) => "(" + v.x.ToString("0.00", CultureInfo.InvariantCulture) + ", " + v.y.ToString("0.00", CultureInfo.InvariantCulture) + ", " + v.z.ToString("0.00", CultureInfo.InvariantCulture) + ")";
    static string F(double f) => f.ToString("0.00", CultureInfo.InvariantCulture);
    static string F3(double f) => f.ToString("0.000", CultureInfo.InvariantCulture);

    // ---------------- capability / import ----------------
    static void Caps(string when)
    {
        string native, loadSeen, err;
        try { native = FRGlassRT_DeviceSupportsRaytracing().ToString(); }
        catch (Exception e) { native = "EXCEPTION " + e.GetType().Name + ": " + e.Message; }
        try { loadSeen = FRGlassRT_PluginLoadSeen().ToString(); }
        catch (Exception e) { loadSeen = "EXCEPTION " + e.GetType().Name; }
        err = NativeError();
        Log("CAPS (" + when + "): gfx " + SystemInfo.graphicsDeviceType + " · " + SystemInfo.graphicsDeviceName + " · " + SystemInfo.graphicsDeviceVersion
            + " · OS " + SystemInfo.operatingSystem
            + " · Unity SystemInfo.supportsRayTracing " + SystemInfo.supportsRayTracing
            + " · supportsRayTracingShaders " + SystemInfo.supportsRayTracingShaders
            + " · supportsInlineRayTracing " + SystemInfo.supportsInlineRayTracing
            + " · NATIVE FRGlassRT_DeviceSupportsRaytracing " + native
            + " · UnityPluginLoad seen (IUnityGraphicsMetalV2 present) " + loadSeen
            + " · trace pipeline compiled " + S(St.PipelineValid) + " (MSL compile+pipeline " + F(S(St.CompileMs)) + " ms)"
            + " · native last error \"" + err + "\""
            + " · build target " + EditorUserBuildSettings.activeBuildTarget);
    }

            + " · editor " + Safe(() => imp.GetCompatibleWithEditor().ToString())

            + " · StandaloneWindows64 " + Safe(() => imp.GetCompatibleWithPlatform(BuildTarget.StandaloneWindows64).ToString())
            + " · WebGL " + Safe(() => imp.GetCompatibleWithPlatform(BuildTarget.WebGL).ToString())
            + " · editor CPU \"" + Safe(() => imp.GetEditorData("CPU")) + "\" OS \"" + Safe(() => imp.GetEditorData("OS")) + "\""
            + " · OSX CPU \"" + Safe(() => imp.GetPlatformData(BuildTarget.StandaloneOSX, "CPU")) + "\""

    // ---------------- capture helpers ----------------
    static void RenderTo(Camera c, RenderTexture target)
    {
        var prev = c.targetTexture;
        c.targetTexture = target;
        c.Render();
        c.targetTexture = prev;
    }

    static void Sync(RenderTexture target)
    {
        var active = RenderTexture.active;
        RenderTexture.active = target;
        tiny.ReadPixels(new Rect(0, 0, 1, 1), 0, 0);
        tiny.Apply();
        RenderTexture.active = active;
    }

    static void SaveFrame(string file, string note)
    {
        var active = RenderTexture.active;
        RenderTexture.active = rt;
        frameTex.ReadPixels(new Rect(0, 0, W, H), 0, 0);
        frameTex.Apply();
        RenderTexture.active = active;
        File.WriteAllBytes(Path.Combine(OutDir, file), frameTex.EncodeToPNG());
        var t = cam.transform;
        frames.AppendLine(file + " | frame " + Time.frameCount + " | cam " + V(t.position) + " fwd " + V(t.forward) + " up " + V(t.up) + " fov " + F(cam.fieldOfView) + " | " + note);
        try { File.WriteAllText(Path.Combine(OutDir, "frames.txt"), frames.ToString()); } catch { }
    }

    static void Shot(string file, string note)
    {
        RenderTo(cam, rt);
        SaveFrame(file, note);
    }

    static RenderTexture Result => ctrl != null ? Get<RenderTexture>(ctrl, "result") : null;

    static float[] ReadResult(out int w, out int h, out int covered)
    {
        w = h = covered = 0;
        var r = Result;
        if (r == null) return null;
        w = r.width; h = r.height;
        var t = new Texture2D(w, h, TextureFormat.RGBAFloat, false, true);
        var prev = RenderTexture.active;
        RenderTexture.active = r;
        t.ReadPixels(new Rect(0, 0, w, h), 0, 0);
        t.Apply(false);
        RenderTexture.active = prev;
        var data = t.GetRawTextureData<float>().ToArray();
        Object.DestroyImmediate(t);
        for (var i = 3; i < data.Length; i += 4) if (data[i] > .5f) covered++;
        return data;
    }

    static void SaveF32(float[] d, string file)
    {
        if (d == null) return;
        var bytes = new byte[d.Length * 4];
        Buffer.BlockCopy(d, 0, bytes, 0, bytes.Length);
        File.WriteAllBytes(Path.Combine(OutDir, file), bytes);
    }

    static string Stats(List<double> v)
    {
        if (v.Count == 0) return "no samples";
        var s = v.OrderBy(x => x).ToList();
        return "median " + F(s[s.Count / 2]) + " ms, min " + F(s[0]) + ", mean " + F(s.Average()) + ", max " + F(s[s.Count - 1]) + " (" + s.Count + ")";
    }

    static double Median(List<double> v) => v.Count == 0 ? double.NaN : v.OrderBy(x => x).ElementAt(v.Count / 2);

    // The controller's own focus rule: Camera.main, else the first FrontRoomsMetalGlassTarget found.
    static string FocusInfo()
    {
        var targets = Object.FindObjectsByType<FrontRoomsMetalGlassTarget>(FindObjectsSortMode.None);
        if (targets.Length == 0) return "no targets";
        var main = Camera.main;
        var focus = main != null ? main.transform.position : targets[0].transform.position;
        FrontRoomsMetalGlassTarget nearest = null; var best = float.PositiveInfinity;
        foreach (var tg in targets) { if (!tg.isActiveAndEnabled) continue; var d = (tg.transform.position - focus).sqrMagnitude; if (d < best) { best = d; nearest = tg; } }
        return "Camera.main " + (main != null ? main.name : "NULL") + " · registration centre = " + (nearest != null ? nearest.name + " " + V(nearest.transform.position) + ", " + F(Vector3.Distance(nearest.transform.position, cam.transform.position)) + " m from the game camera" : "none");
    }

    static double Rebuild(string why)
    {
        FRGlassRT_ResetStats();
        var sw = Stopwatch.StartNew();
        Call(ctrl, "RebuildScene");
        sw.Stop();
        var regs = Get<List<Renderer>>(ctrl, "registeredRenderers");
        var meshes = Get(ctrl, "meshIndices") as IDictionary;
        var mats = Get(ctrl, "materialIndices") as IDictionary;
        var blasCpu = S(St.BlasCpuMs);
        var tlasCpu = S(St.TlasCpuMs);
        Log(string.Format(CultureInfo.InvariantCulture,
            "REBUILD ({0}) [ablation flags " + diagBase + " · " + FocusInfo() + "]: RebuildScene main-thread wall {1} ms · native BLAS builds {2} (CPU incl. waitUntilCompleted {3} ms, GPU {4} ms, {5} MB, {6} tris) · TLAS {7} instances CPU {8} ms GPU {9} ms ({10} KB) · managed part {11} ms · registered renderers {12} · unique meshes {13} · materials {14} · ready {15} · AddMesh on main thread {16} · native error \"{17}\"",
            why, F(sw.Elapsed.TotalMilliseconds), S(St.BlasCount), F(blasCpu), F(S(St.BlasGpuMs)), F(S(St.BlasBytes) / 1048576.0), S(St.BlasTriangles),
            S(St.TlasInstances), F(tlasCpu), F(S(St.TlasGpuMs)), F(S(St.TlasBytes) / 1024.0), F(sw.Elapsed.TotalMilliseconds - blasCpu - tlasCpu),
            regs?.Count ?? -1, meshes?.Count ?? -1, mats?.Count ?? -1, Get(ctrl, "ready"), S(St.AddMeshMainThread), NativeError()));
        return sw.Elapsed.TotalMilliseconds;
    }

    // ---------------- grain (analysis renders only) ----------------
    static readonly Dictionary<FilmGrain, bool> grainOriginal = new Dictionary<FilmGrain, bool>();
    static void Grain(bool on)
    {
        foreach (var v in Object.FindObjectsByType<Volume>(FindObjectsInactive.Include, FindObjectsSortMode.None))
        {
            var p = v.sharedProfile;
            if (p == null || !p.TryGet<FilmGrain>(out var g)) continue;
            if (!grainOriginal.ContainsKey(g)) grainOriginal[g] = g.active;
            g.active = on && grainOriginal[g];
        }
    }

    // ---------------- placement ----------------
    static void Place(Vector3 feet, Vector3 lookAt)
    {
        var root = PlayerRoot;
        var body = Get<CharacterController>(game, "playerBody");
        body.enabled = false;
        root.position = feet;
        body.enabled = true;
        Physics.SyncTransforms();
        Look(lookAt);
    }

    static void Look(Vector3 lookAt)
    {
        var root = PlayerRoot;
        var eye = root.position + Vector3.up * ModuleUnits.PlayerEye;
        var d = lookAt - eye;
        var yaw = Mathf.Atan2(d.x, d.z) * Mathf.Rad2Deg;
        var pitch = -Mathf.Asin(Mathf.Clamp(d.y / Mathf.Max(1e-4f, d.magnitude), -1f, 1f)) * Mathf.Rad2Deg;
        Set(game, "yaw", yaw);
        Set(game, "pitch", pitch);
        root.rotation = Quaternion.Euler(0f, yaw, 0f);
        cam.transform.localRotation = Quaternion.Euler(pitch, 0f, 0f);
    }

    static void Resume()
    {
        if (Phase != "Paused") return;
        var phaseType = typeof(FrontRooms3DGame).GetNestedType("Phase", BindingFlags.NonPublic);
        Call(game, "SetPhase", Enum.Parse(phaseType, "Playing"));
    }

    static IEnumerator Settle(int extra = 20)
    {
        yield return 3;
        var start = Time.frameCount;
        while (!map.Settled && Time.frameCount - start < 1500) yield return 1;
        yield return extra;
    }

    static bool FreeStanding(Vector3 feet)
    {
        var r = ModuleUnits.PlayerRadius;
        var root = PlayerRoot;
        foreach (var c in Physics.OverlapCapsule(feet + Vector3.up * (r + .15f), feet + Vector3.up * (ModuleUnits.PlayerHeight - r), r, ~0, QueryTriggerInteraction.Ignore))
            if (root == null || !c.transform.IsChildOf(root)) return false;
        return true;
    }

    static bool SeesPane(Vector3 feet, Collider pane)
    {
        var eye = feet + Vector3.up * ModuleUnits.PlayerEye;
        var b = pane.bounds;
        var pts = new List<Vector3> { b.center };
        // inset corners of the pane's broad face
        var t = pane.transform;
        foreach (var sx in new[] { -.42f, .42f })
        foreach (var sy in new[] { -.42f, .42f })
            pts.Add(t.TransformPoint(new Vector3(sx, sy, 0f)));
        foreach (var p in pts)
        {
            var d = p - eye;
            if (!Physics.Raycast(eye, d.normalized, out var hit, d.magnitude + .2f, ~0, QueryTriggerInteraction.Ignore)) return false;
            if (hit.collider != pane) return false;
        }
        return true;
    }

    static Collider WindowCollider(GridCoord a, GridCoord b)
    {
        var dict = Get(map, "windowByCollider") as IDictionary;
        var p = map.CrossingPoint(a, b);
        foreach (DictionaryEntry e in dict)
        {
            var pos = (Vector3)Get(e.Value, "position");
            if ((new Vector2(pos.x - p.x, pos.z - p.z)).sqrMagnitude < .01f) return (Collider)e.Key;
        }
        return null;
    }

    static IEnumerable<GridCoord> CellsAround(GridCoord center, int radiusCells)
    {
        for (var r = 0; r <= radiusCells; r++)
        for (var dy = -r; dy <= r; dy++)
        for (var dx = -r; dx <= r; dx++)
        {
            if (Mathf.Max(Mathf.Abs(dx), Mathf.Abs(dy)) != r) continue;
            yield return new GridCoord(center.x + dx, center.y + dy);
        }
    }

    // ---------------- ray-cast masks ----------------
    // Pixel (x, y) with y = 0 at the BOTTOM (Unity texture/ReadPixels convention). A pixel is 255
    // when the first collider along the ray through its centre is the pane. tanHalf is the value used
    // for the vertical extent: Mathf.Tan(fov/2) is correct; fov in radians is what the prototype sends.
    static byte[] RayMask(Collider target, float tanHalf, out int count)
    {
        var t = cam.transform;
        var o = t.position; var r = t.right; var u = t.up; var f = t.forward;
        var aspect = (float)W / H;
        var m = new byte[MW * MH];
        count = 0;
        // The player's own colliders are not part of what the camera sees.
        var own = PlayerRoot.GetComponentsInChildren<Collider>(true).Where(c => c.enabled).ToList();
        foreach (var c in own) c.enabled = false;
        Physics.SyncTransforms();
        for (var y = 0; y < MH; y++)
        {
            var ndcY = ((y + .5f) / MH) * 2f - 1f;
            for (var x = 0; x < MW; x++)
            {
                var ndcX = ((x + .5f) / MW) * 2f - 1f;
                var dir = (f + r * (ndcX * tanHalf * aspect) + u * (ndcY * tanHalf)).normalized;
                if (Physics.Raycast(o, dir, out var hit, 200f, ~0, QueryTriggerInteraction.Ignore) && hit.collider == target)
                {
                    m[y * MW + x] = 255;
                    count++;
                }
            }
        }
        foreach (var c in own) c.enabled = true;
        Physics.SyncTransforms();
        return m;
    }

    // ---------------- view probe ----------------
    struct View
    {
        public string name;
        public Vector3 feet, look;
        public string note;
    }

    static readonly List<string> summary = new List<string>();

    static IEnumerator ProbeView(View v, Collider pane, Vector3 paneCenter, Vector3 dir)
    {
        Place(v.feet, v.look);
        yield return 3;
        Resume();
        Place(v.feet, v.look);
        yield return 2;
        var t = cam.transform;
        var toPane = paneCenter - t.position;
        var incidence = Vector3.Angle(-toPane.normalized, -dir); // angle between view ray to pane centre and pane normal
        Log("VIEW " + v.name + " — " + v.note + " · eye " + V(t.position) + " fwd " + V(t.forward) + " · distance to pane centre " + F(toPane.magnitude) + " m · incidence at pane centre " + F(incidence) + " deg");

        // Ray-cast masks: where the pane is (correct projection) and where the prototype's projection puts it.
        var tanTrue = Mathf.Tan(cam.fieldOfView * .5f * Mathf.Deg2Rad);
        var tanSent = cam.fieldOfView * Mathf.Deg2Rad;
        var sw = Stopwatch.StartNew();
        var truth = RayMask(pane, tanTrue, out var nTruth);
        var pred = RayMask(pane, tanSent, out var nPred);
        File.WriteAllBytes(Path.Combine(OutDir, v.name + "_mask_truth.u8"), truth);
        File.WriteAllBytes(Path.Combine(OutDir, v.name + "_mask_predicted_bugfov.u8"), pred);
        Log("MASK " + v.name + ": camera fov " + F(cam.fieldOfView) + " deg · correct tan(fov/2) " + F3(tanTrue) + " · prototype sends fov in radians " + F3(tanSent)
            + " (x" + F3(tanSent / tanTrue) + ") · pane pixels at " + MW + "x" + MH + ": correct " + nTruth + ", predicted-with-prototype-projection " + nPred + " (ray-casts " + F(sw.Elapsed.TotalMilliseconds) + " ms)");

        // --- analysis renders (film grain off so ON/OFF differences are only the composite) ---
        Grain(false);
        SetRT(false);
        Shot(v.name + "_A_off.png", "analysis, grain off, RT OFF");
        var rebuildMs = Rebuild(v.name);
        var regs = Get<List<Renderer>>(ctrl, "registeredRenderers");
        var paneRegistered = regs != null && regs.Contains(pane.GetComponent<Renderer>());
        FRGlassRT_ResetStats();
        Shot(v.name + "_A_on_first_after_rescan.png", "analysis, grain off, RT ON, first render after RebuildScene");
        var d0 = ReadResult(out var rw, out var rh, out var cov0);
        Shot(v.name + "_A_on.png", "analysis, grain off, RT ON, second render");

    static void Coverage(Collider pane, string tag)
    {
        var regs = new HashSet<Renderer>(Get<List<Renderer>>(ctrl, "registeredRenderers"));
        var radius = Get<float>(ctrl, "registrationRadius");
        var cap = Get<int>(ctrl, "maxInstances");
        var targets = Object.FindObjectsByType<FrontRoomsMetalGlassTarget>(FindObjectsSortMode.None);
        var focus = Camera.main != null ? Camera.main.transform.position : (targets.Length > 0 ? targets[0].transform.position : cam.transform.position);
        FrontRoomsMetalGlassTarget nearest = null; var best = float.PositiveInfinity;
        foreach (var tg in targets) { var d = (tg.transform.position - focus).sqrMagnitude; if (tg.isActiveAndEnabled && d < best) { best = d; nearest = tg; } }
        var all = Object.FindObjectsByType<MeshRenderer>(FindObjectsSortMode.InstanceID);
        int within = 0, withinDropped = 0, glassFar = 0, multiSub = 0, badFormat = 0, noMat = 0;
        long trisSkippedSub = 0;
        var center = nearest != null ? nearest.transform.position : focus;
        foreach (var r in all)
        {
            if (!r.enabled || r.sharedMaterial == null) { noMat++; continue; }
            var mf = r.GetComponent<MeshFilter>();
            if (mf == null || mf.sharedMesh == null) continue;
            var isGlass = r.GetComponent<FrontRoomsMetalGlassTarget>() != null;
            var inR = (r.bounds.center - center).sqrMagnitude <= radius * radius;
            if (inR) { within++; if (!regs.Contains(r)) withinDropped++; }
            if (isGlass && !inR) glassFar++;
            if (!regs.Contains(r)) continue;
            var mesh = mf.sharedMesh;
            if (mesh.subMeshCount > 1)
            {
                multiSub++;
                for (var s = 1; s < mesh.subMeshCount; s++) trisSkippedSub += mesh.GetIndexCount(s) / 3;
            }
            if (mesh.GetVertexAttributeFormat(VertexAttribute.Position) != VertexAttributeFormat.Float32) badFormat++;
        }
        var skinned = Object.FindObjectsByType<SkinnedMeshRenderer>(FindObjectsSortMode.None);
        var skinnedNear = skinned.Count(s => s.enabled && (s.bounds.center - center).sqrMagnitude <= radius * radius);
        var particles = Object.FindObjectsByType<ParticleSystemRenderer>(FindObjectsSortMode.None).Count(p => p.enabled && (p.bounds.center - center).sqrMagnitude <= radius * radius);
        var lights = Object.FindObjectsByType<Light>(FindObjectsSortMode.None);
        var litNear = lights.Count(l => l.enabled && l.intensity > 0f && (l.transform.position - center).sqrMagnitude <= radius * radius);
        var paneReg = pane != null && regs.Contains(pane.GetComponent<Renderer>());
        Log("COVERAGE (" + tag + "): nearest target " + (nearest != null ? nearest.name + " at " + V(nearest.transform.position) + (pane != null && nearest.gameObject == pane.gameObject ? " (= probed pane)" : " (NOT the probed pane)") : "none")
            + " · probed pane registered " + paneReg
            + " · MeshRenderers in scene " + all.Length + " · within " + radius + " m of it " + within + " · of those NOT registered " + withinDropped
            + " · registered " + regs.Count + " (cap " + cap + ")"
            + " · glass panes in scene " + targets.Length + " (registered regardless of distance; beyond radius " + glassFar + ")"
            + " · registered multi-submesh renderers " + multiSub + " (" + trisSkippedSub + " triangles in submeshes 1+ never enter the BLAS)"
            + " · non-Float32 positions " + badFormat
            + " · SkinnedMeshRenderers within radius " + skinnedNear + " (of " + skinned.Length + ", never registered)"
            + " · ParticleSystemRenderers within radius " + particles
            + " · enabled lights within radius " + litNear + " (of " + lights.Length + " Light components; the trace uses none)");

        // What the camera sees that the trace does not know about.
        var planes = GeometryUtility.CalculateFrustumPlanes(cam);
        var missing = new List<string>();
        var visible = 0;
        foreach (var r in Object.FindObjectsByType<Renderer>(FindObjectsSortMode.None))
        {
            if (!r.enabled || !r.gameObject.activeInHierarchy) continue;
            if ((cam.cullingMask & (1 << r.gameObject.layer)) == 0) continue;
            if (!GeometryUtility.TestPlanesAABB(planes, r.bounds)) continue;
            visible++;
            if (r is MeshRenderer mr && regs.Contains(mr)) continue;
            missing.Add(r.GetType().Name + ":" + r.name);
        }
        Log("COVERAGE (" + tag + ") camera frustum: " + visible + " renderers, NOT in the RT scene " + missing.Count + ": " + string.Join(", ", missing.GroupBy(s => s).Select(g => g.Key + (g.Count() > 1 ? " x" + g.Count() : "")).Take(30)));
    }

    // ---------------- hit-identity diagnostic (clone plugin flag 32, run 7) ----------------
    // The kernel writes R = reflected hit instance + 1 (0 = miss -> sky), G = reflected hit distance,
    // B = first-hit instance * 16 + primitive, instead of colour. Always with the FOV corrected (flag 16).
    static Dictionary<int, bool> paneBroadFace;
    static void ClassifyPaneTriangles(Collider pane)
    {
        paneBroadFace = new Dictionary<int, bool>();
        try
        {
            var mesh = pane.GetComponent<MeshFilter>().sharedMesh;
            var tris = mesh.GetTriangles(0);
            var verts = mesh.vertices;
            var scale = pane.transform.lossyScale;
            var thin = scale.x < scale.y && scale.x < scale.z ? 0 : scale.y < scale.z ? 1 : 2;
            var sb = new StringBuilder();
            for (var t = 0; t < tris.Length / 3; t++)
            {
                var n = Vector3.Cross(verts[tris[t * 3 + 1]] - verts[tris[t * 3]], verts[tris[t * 3 + 2]] - verts[tris[t * 3]]).normalized;
                var broad = Mathf.Abs(n[thin]) > .9f;
                paneBroadFace[t] = broad;
                sb.Append(t).Append(broad ? ":broad " : ":edge ");
            }
            Log("PANE triangles (" + mesh.name + ", thin local axis " + "xyz"[thin] + " = " + F3(scale[thin]) + " m): " + sb);
        }
        catch (Exception e) { Log("PANE triangles: could not read the mesh (" + e.GetType().Name + ": " + e.Message + ")"); }
    }

    static void WriteInstances(string tag)
    {
        var regs = Get<List<Renderer>>(ctrl, "registeredRenderers");
        var sb = new StringBuilder("index\trenderer\tmaterial\tshader\tglass\tlayer\tposition\n");

    static string Category(Renderer r)

    }

    static float[] DiagRender(string tag, Collider pane, int extraFlags = 16)
    {
        FRGlassRT_SetDiagnostics(diagBase | extraFlags | 32);
        RenderTo(cam, rt);
        RenderTo(cam, rt);
        var d = ReadResult(out _, out _, out var cov);
        FRGlassRT_SetDiagnostics(diagBase);
        RenderTo(cam, rt);
        if (d == null) { Log("DIAG " + tag + ": no result texture"); return null; }
        SaveF32(d, tag + "_diag_rt.f32");
        WriteInstances(tag);
        var regs = Get<List<Renderer>>(ctrl, "registeredRenderers");
        var paneIdx = pane != null ? regs.IndexOf(pane.GetComponent<Renderer>()) : -1;
        int glass = 0, miss = 0, onPane = 0, broad = 0, edge = 0, otherGlass = 0;
        var hitCat = new Dictionary<string, int>();
        var dist = new List<float>();
        for (var i = 0; i < d.Length; i += 4)
        {
            if (d[i + 3] <= .5f) continue;
            glass++;
            var h = (int)Math.Round(d[i]);
            if (h == 0) miss++;
            else
            {
                var r = h - 1 < regs.Count ? regs[h - 1] : null;
                var c = Category(r);
                hitCat[c] = hitCat.TryGetValue(c, out var k) ? k + 1 : 1;
                dist.Add(d[i + 1]);
            }
            var b = (int)Math.Round(d[i + 2]);
            var inst = b / 16; var prim = b % 16;
            if (inst == paneIdx)
            {
                onPane++;
                if (paneBroadFace != null && paneBroadFace.TryGetValue(prim, out var isBroad)) { if (isBroad) broad++; else edge++; }
            }
            else otherGlass++;
        }
        dist.Sort();
        var top = string.Join(", ", hitCat.OrderByDescending(kv => kv.Value).Take(8).Select(kv => kv.Key + " " + (100.0 * kv.Value / Math.Max(1, glass)).ToString("0.0", CultureInfo.InvariantCulture) + " %"));
        Log("DIAG " + tag + " (FOV corrected, hit identities): glass px " + glass + " · reflection ray MISSED the RT scene (sky gradient) " + miss + " px = " + (100.0 * miss / Math.Max(1, glass)).ToString("0.0", CultureInfo.InvariantCulture) + " %"
            + " · reflected surfaces: " + top
            + " · reflected hit distance median " + (dist.Count > 0 ? F(dist[dist.Count / 2]) : "-") + " m (10th pct " + (dist.Count > 0 ? F(dist[dist.Count / 10]) : "-") + ", 90th " + (dist.Count > 0 ? F(dist[dist.Count * 9 / 10]) : "-") + ")"
            + " · first hit on the probed pane " + onPane + " px (broad faces " + broad + ", 30 mm edge faces " + edge + ") · on other glass instances " + otherGlass + " px");
        return d;
    }

    // Run 8: why do reflection rays miss (sky) inside a closed office? Walk the corridor behind the camera
    // (the mirror axis) and report which renderer owns the ceiling / end wall there and whether it is registered.
    static void MissCause(Vector3 paneCenter, Vector3 dir)
    {
        var regs = new HashSet<Renderer>(Get<List<Renderer>>(ctrl, "registeredRenderers"));
        var radius = Get<float>(ctrl, "registrationRadius");
        var eye = cam.transform.position;
        var back = -new Vector3(dir.x, 0f, dir.z).normalized;
        var all = Object.FindObjectsByType<MeshRenderer>(FindObjectsSortMode.None).Where(r => r.enabled && r.sharedMaterial != null).ToList();
        var own = PlayerRoot.GetComponentsInChildren<Collider>(true).Where(c => c.enabled).ToList();
        foreach (var c in own) c.enabled = false;
        Physics.SyncTransforms();
        var len = Physics.Raycast(eye, back, out var endHit, 80f, ~0, QueryTriggerInteraction.Ignore) ? endHit.distance : 80f;
        var sb = new StringBuilder("MISS CAUSE (registration radius " + F(radius) + " m around the pane centre; rule: renderer.bounds.center within the radius): free space behind the eye along the mirror axis " + F(len) + " m");
        string Owner(Vector3 pt, Func<Renderer, bool> filter)
        {
            var owners = all.Where(r => filter(r) && r.bounds.SqrDistance(pt) < .0025f).ToList();
            if (owners.Count == 0) return "no renderer bounds contain it";
            return string.Join(" + ", owners.Select(r => r.name + " [" + r.sharedMaterial.name + "] bounds centre " + V(r.bounds.center) + " size " + V(r.bounds.size)
                + ", centre " + F(Vector3.Distance(r.bounds.center, paneCenter)) + " m from the pane, registered " + regs.Contains(r)));
        }
        foreach (var s in new[] { 1f, 3f, 5f, 8f, 11f, 14f, 18f, 22f, 26f, 30f })
        {
            if (s > len - .2f) break;
            var floorPt = eye + back * s;
            if (!Physics.Raycast(floorPt, Vector3.up, out var up, 10f, ~0, QueryTriggerInteraction.Ignore)) { sb.Append(" | " + F(s) + " m: no ceiling collider above"); continue; }
            sb.Append(" | ceiling " + F(s) + " m behind the eye (y " + F(up.point.y) + ", " + F(Vector3.Distance(up.point, paneCenter)) + " m from the pane): " + Owner(up.point, r => r.sharedMaterial.name.Contains("Ceiling") || r.name.Contains("Ceiling")));
        }
        if (len < 80f) sb.Append(" | END WALL at " + F(len) + " m (" + F(Vector3.Distance(endHit.point, paneCenter)) + " m from the pane): " + Owner(endHit.point, r => true));
        foreach (var c in own) c.enabled = true;
        Physics.SyncTransforms();
        Log(sb.ToString());
    }

    static int CountHits(float[] d, int instIndex, out double meanRow)
    {
        meanRow = double.NaN;
        if (d == null || instIndex < 0) return 0;
        var n = 0; double sy = 0;
        for (var i = 0; i < d.Length; i += 4)
        {
            if (d[i + 3] <= .5f) continue;
            if ((int)Math.Round(d[i]) - 1 != instIndex) continue;
            n++; sy += (i / 4) / W;
        }
        if (n > 0) meanRow = sy / n;
        return n;
    }

            Log("SKINNED occluder at " + V(occ.transform.position) + " · registered " + regs2.Contains(smr) + " · material " + wallMat.name + " · game phase " + Phase);
            SetRT(false);
            Shot("skinned_A_off.png", "SkinnedMeshRenderer between camera and pane, RT OFF, grain off");
            SetRT(true);
            RenderTo(cam, rt);
            Shot("skinned_A_on.png", "SkinnedMeshRenderer between camera and pane, RT ON, grain off");
            SaveF32(ReadResult(out _, out _, out var covSk), "skinned_rt.f32");
            FRGlassRT_SetDiagnostics(diagBase | 16);
            RenderTo(cam, rt);
            Shot("skinned_fovfix_A_on.png", "SkinnedMeshRenderer between camera and pane, RT ON, FOV corrected, grain off");
            SaveF32(ReadResult(out _, out _, out var covSkFix), "skinned_fovfix_rt.f32");
            FRGlassRT_SetDiagnostics(diagBase);
            Grain(true);
            Log("SKINNED occluder: RT covered px " + covSk + " (FOV corrected " + covSkFix + ")");
            Object.Destroy(occ);
            yield return 2;
        }

        // ---- free-running controller (1 s rescans, as in the standalone map) ----
        Place(headCenter.feet, headCenter.look);
        yield return 3;
        Resume();
        Place(headCenter.feet, headCenter.look);
        Rebuild("before free run");
        RenderTo(cam, rt);
        {
            Set(ctrl, "nextScan", 0f);
            var small = new RenderTexture(240, 135, 0, RenderTextureFormat.ARGBFloat) { name = "RTProbe coverage" };
            small.Create();
            var smallTex = new Texture2D(240, 135, TextureFormat.RGBAFloat, false, true);
            var sw = Stopwatch.StartNew();
            double last = -1;
            var lastBlas = S(St.BlasCount);
            FRGlassRT_ResetStats();
            lastBlas = 0;
            var lines = new StringBuilder();
            var prevCov = -1.0;
            var savedBlink = 0;
            var intervals = new List<double>();
            var rebuildIntervals = new List<double>();
            var blankFrames = 0; var rebuildFrames = 0;
            var u0 = Time.unscaledTime;
            for (var i = 0; i < 200; i++)
            {
                RenderTo(cam, rt);
                var r = Result;
                double cov = -1;
                if (r != null)
                {
                    Graphics.Blit(r, small);
                    var prevA = RenderTexture.active;
                    RenderTexture.active = small;
                    smallTex.ReadPixels(new Rect(0, 0, 240, 135), 0, 0);
                    smallTex.Apply(false);
                    RenderTexture.active = prevA;
                    var px = smallTex.GetRawTextureData<float>();
                    double sa = 0;
                    for (var k = 3; k < px.Length; k += 4) sa += px[k];
                    cov = sa / (240 * 135);
                }
                var now = sw.Elapsed.TotalMilliseconds;
                var blas = S(St.BlasCount);
                var rebuilt = blas > lastBlas;
                if (rebuilt) rebuildFrames++;
                lastBlas = blas;
                if (last >= 0) { intervals.Add(now - last); if (rebuilt) rebuildIntervals.Add(now - last); }
                if (cov == 0 && prevCov > 0)
                {
                    blankFrames++;
                    if (savedBlink < 1) { SaveFrame("free_run_blank_frame.png", "free run frame " + i + ": first render after the 1 s rescan — RT coverage 0"); savedBlink++; }
                }
                lines.AppendLine(i + "\tframe " + Time.frameCount + "\tunscaled " + F3(Time.unscaledTime) + "\tinterval_ms " + (last >= 0 ? F(now - last) : "-") + "\tcoverage " + F3(cov) + "\trebuiltThisFrame " + rebuilt + "\tblasTotal " + blas);
                prevCov = cov;
                last = now;
                yield return 1;
            }
            File.WriteAllText(Path.Combine(OutDir, "free_run_frames.tsv"), lines.ToString());
            Log("FREE RUN 200 frames (one camera render per frame, rescan every " + Get(ctrl, "rescanSeconds") + " s of unscaled time; unscaled time advanced " + F3(Time.unscaledTime - u0) + " s): rescans " + rebuildFrames
                + " · frames whose RT output was blank right after a rescan " + blankFrames + " · frame interval all " + Stats(intervals) + " · frames with a rescan " + Stats(rebuildIntervals)
                + " · BLAS builds " + S(St.BlasCount) + " · BLAS CPU total " + F(S(St.BlasCpuMs)) + " ms");
            Set(ctrl, "nextScan", float.MaxValue);
            small.Release();
        }

        // ---- streamed mesh (suspect 6): a runtime Mesh like a chunk shell, edited in place, then destroyed,
        // with and without a rescan. Geometry is identified with the hit-identity diagnostic (flag 32). ----
        {
            Place(headCenter.feet, headCenter.look);
            yield return 2;
            Resume();
            Place(headCenter.feet, headCenter.look);
            yield return 1;
            var eye = cam.transform.position;
            var back = -new Vector3(dir.x, 0f, dir.z).normalized;
            var room = Physics.Raycast(eye, back, out var behindHit, 10f, ~0, QueryTriggerInteraction.Ignore) ? behindHit.distance : 10f;
            var gap = Mathf.Clamp(room * .5f, .35f, 1.0f);
            var src = Resources.GetBuiltinResource<Mesh>("Cube.fbx");
            var sm = new Mesh { name = "RTProbe streamed mesh" };
            sm.SetVertices(src.vertices);
            sm.SetNormals(src.normals);
            sm.SetUVs(0, src.uv);
            sm.SetTriangles(src.triangles, 0);
            sm.RecalculateBounds();
            var sgo = new GameObject("RTProbe / streamed slab");
            sgo.AddComponent<MeshFilter>().sharedMesh = sm;
            var smr2 = sgo.AddComponent<MeshRenderer>();
            var smat = new Material(Shader.Find("Universal Render Pipeline/Lit")) { name = "RTProbe red slab" };
            smat.SetColor("_BaseColor", new Color(1f, .12f, .08f));
            smr2.sharedMaterial = smat;
            sgo.transform.SetPositionAndRotation(new Vector3(eye.x, 0f, eye.z) + back * gap + Vector3.up * 1.25f, Quaternion.LookRotation(back));
            sgo.transform.localScale = new Vector3(.8f, .6f, .04f);
            yield return 2;
            Rebuild("streamed slab registered");
            var regsS = Get<List<Renderer>>(ctrl, "registeredRenderers");
            var slabIdx = regsS.IndexOf(smr2);
            var ptr0 = sm.GetNativeVertexBufferPtr(0);
            Log("STREAM: slab " + V(sgo.transform.position) + " (" + F(gap) + " m behind the eye; free space behind the eye " + F(room) + " m) · registered index " + slabIdx + " · native vertex buffer " + ptr0);
            var d0 = DiagRender("stream_0_registered", pane);
            var n0 = CountHits(d0, slabIdx, out var y0);
            FRGlassRT_SetDiagnostics(diagBase | 16);
            Grain(false);
            Shot("stream_0_registered_A_on.png", "streamed slab behind the camera, registered, RT ON (FOV corrected), grain off");
            SaveF32(ReadResult(out _, out _, out _), "stream_0_rt.f32");
            FRGlassRT_SetDiagnostics(diagBase);

            // (1) edit the vertices in place, as a live rebuild or a moving fracture piece would, without a rescan
            var vs = sm.vertices;
            for (var i = 0; i < vs.Length; i++) vs[i] += new Vector3(0f, .5f / .6f, 0f); // +0.5 m in world space
            sm.SetVertices(vs);
            sm.RecalculateBounds();
            sm.UploadMeshData(false);
            yield return 1;
            var ptr1 = sm.GetNativeVertexBufferPtr(0);
            var d1 = DiagRender("stream_1_edited_norescan", pane);
            var n1 = CountHits(d1, slabIdx, out var y1);
            Rebuild("streamed slab edited, rescanned");
            regsS = Get<List<Renderer>>(ctrl, "registeredRenderers");
            var slabIdx2 = regsS.IndexOf(smr2);
            var d2 = DiagRender("stream_2_edited_rescanned", pane);
            var n2 = CountHits(d2, slabIdx2, out var y2);
            Log("STREAM edit: vertex buffer " + ptr0 + " → " + ptr1 + (ptr0 == ptr1 ? " (same buffer, written in place)" : " (Unity REALLOCATED the buffer)")
                + " · slab pixels in the reflection: registered " + n0 + " (mean row " + F(y0) + ") · moved +0.5 m, no rescan " + n1 + " (mean row " + F(y1) + ") · after the rescan " + n2 + " (mean row " + F(y2) + ")"
                + " · trace errors " + S(St.TraceErrors) + " · native error \"" + NativeError() + "\"");

            // (2) destroy the mesh and its object (what FreeMeshes/Kill do when a chunk streams out), no rescan
            FRGlassRT_ResetStats();
            Object.Destroy(sgo);
            Object.Destroy(sm);
            yield return 2;
            var alive = sm != null;
            var d3 = DiagRender("stream_3_destroyed_norescan", pane);
            var n3 = CountHits(d3, slabIdx2, out var y3);
            FRGlassRT_SetDiagnostics(diagBase | 16);
            Shot("stream_3_destroyed_A_on.png", "streamed slab destroyed, no rescan yet, RT ON (FOV corrected), grain off");
            SaveF32(ReadResult(out _, out _, out _), "stream_3_rt.f32");
            FRGlassRT_SetDiagnostics(diagBase);
            Log("STREAM destroy: Mesh object alive " + alive + " · slab pixels still in the reflection with no rescan " + n3 + " (mean row " + F(y3) + ") · traces completed " + S(St.TraceCompleted) + "/" + S(St.TraceSubmitted) + " · trace errors " + S(St.TraceErrors) + " · native error \"" + NativeError() + "\"");
            Rebuild("after the streamed slab was destroyed");
            var d4 = DiagRender("stream_4_destroyed_rescanned", pane);
            Grain(true);
            Object.Destroy(smat);
        }

        // ---- broken pane: the trace keeps painting glass until the next rescan ----
        Place(headCenter.feet, headCenter.look);
        yield return 2;
        Resume();
        Place(headCenter.feet, headCenter.look);
        yield return 1;
        Rebuild("before breaking the pane");
        RenderTo(cam, rt);
        Shot("ghost_0_intact_on.png", "pane intact, RT ON");
        ReadResult(out _, out _, out var covIntact);
        var broke = map.Hold(pane, 1.1f, out _);
        yield return 2;
        Resume();
        Place(headCenter.feet, headCenter.look);
        yield return 1;
        Log("GHOST: Hold() broke the pane " + broke + " · pane object alive " + (pane != null));
        Shot("ghost_1_broken_on.png", "pane broken (object destroyed), no rescan yet, RT ON");
        var g1 = ReadResult(out _, out _, out var covBroken);
        SaveF32(g1, "ghost_1_rt.f32");
        SetRT(false);
        Shot("ghost_1_broken_off.png", "pane broken, RT OFF (what the game shows)");
        SetRT(true);
        Rebuild("after the pane broke");
        Coverage(null, "after break");
        RenderTo(cam, rt);
        Shot("ghost_2_rescanned_on.png", "after the next rescan, RT ON");
        ReadResult(out _, out _, out var covAfter);
        Log("GHOST coverage: intact " + covIntact + " px · broken, before rescan " + covBroken + " px · after rescan " + covAfter + " px");

        // ---- summary ----
        Log("SUMMARY\n  " + string.Join("\n  ", summary));
        Log("DONE");
    }
