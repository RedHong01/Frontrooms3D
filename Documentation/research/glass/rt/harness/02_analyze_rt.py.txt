"""Analyse the RT probe output (proj_rt/Verification/rt_probe) and build report images.

Conventions: Unity readbacks (.f32, .u8) have row 0 at the BOTTOM; PNG frames have row 0 at the TOP.
Everything is converted to top-first here.
"""

import numpy as np
from PIL import Image, ImageDraw, ImageFont

FONT = None
for f in ["/System/Library/Fonts/Supplemental/Arial Bold.ttf", "/System/Library/Fonts/Helvetica.ttc"]:
    if os.path.exists(f):
        FONT = f
        break


def font(sz):
    return ImageFont.truetype(FONT, sz) if FONT else ImageFont.load_default()


def p(name):
    return os.path.join(SRC, name)


def load_png(name):
    return np.asarray(Image.open(p(name)).convert("RGB")).astype(np.int16)


def load_f32(name, w=W, h=H):
    a = np.fromfile(p(name), dtype=np.float32)
    if a.size != w * h * 4:
        raise ValueError(f"{name}: {a.size} floats, expected {w*h*4}")
    return np.flipud(a.reshape(h, w, 4))


def load_u8(name, w=MW, h=MH):
    return np.flipud(np.fromfile(p(name), dtype=np.uint8).reshape(h, w)) > 127


def half(mask):
    m = mask.reshape(MH, 2, MW, 2).mean(axis=(1, 3))
    return m >= 0.5


def iou(a, b):
    u = np.logical_or(a, b).sum()
    return float(np.logical_and(a, b).sum()) / float(u) if u else float("nan")


def painted(on, off, thr=10):
    return (np.abs(on - off).max(axis=2) > thr)


def bbox(m):
    ys, xs = np.nonzero(m)
    if len(xs) == 0:
        return None
    return [int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())]


def centroid(m):
    ys, xs = np.nonzero(m)
    if len(xs) == 0:
        return None
    return [round(float(xs.mean()), 1), round(float(ys.mean()), 1)]


def edge(m):
    e = np.zeros_like(m)
    e[1:-1, 1:-1] = m[1:-1, 1:-1] & ~(m[:-2, 1:-1] & m[2:, 1:-1] & m[1:-1, :-2] & m[1:-1, 2:])
    return e


def thick(e, r=2):
    out = e.copy()
    for dy in range(-r, r + 1):
        for dx in range(-r, r + 1):
            out |= np.roll(np.roll(e, dy, axis=0), dx, axis=1)
    return out


def up2(m):
    return np.repeat(np.repeat(m, 2, axis=0), 2, axis=1)


def to_srgb(lin):
    lin = np.clip(lin, 0, 1)
    return np.where(lin <= 0.0031308, lin * 12.92, 1.055 * np.power(lin, 1 / 2.4) - 0.055)


def rt_image(rt, gain=1.0):
    rgb = to_srgb(rt[..., :3] * gain)
    a = rt[..., 3:4] > 0.5
    img = np.where(a, rgb, 0.0)
    return Image.fromarray((img * 255 + 0.5).astype(np.uint8))


def label(img, text, sub=None):
    img = img.copy()
    d = ImageDraw.Draw(img)
    fs = max(18, img.width // 40)
    pad = fs // 3
    tw = d.textlength(text, font=font(fs))
    h = fs + 2 * pad + (int(fs * 0.8) + pad if sub else 0)
    w2 = max(tw, d.textlength(sub, font=font(int(fs * 0.8))) if sub else 0)
    d.rectangle([0, 0, w2 + 2 * pad, h], fill=(0, 0, 0))
    d.text((pad, pad), text, fill=(255, 255, 255), font=font(fs))
    if sub:
        d.text((pad, pad + fs + pad // 2), sub, fill=(220, 220, 220), font=font(int(fs * 0.8)))
    return img


def overlay(base, layers):
    """layers: list of (mask_fullres, rgb)"""
    img = base.astype(np.uint8).copy()
    for m, c in layers:
        e = thick(edge(m), 2)
        img[e] = c
    return Image.fromarray(img)


def sheet(images, cols, width=1600, gap=6):
    rows = (len(images) + cols - 1) // cols
    cw = (width - gap * (cols - 1)) // cols
    ch = int(cw * images[0].height / images[0].width)
    out = Image.new("RGB", (width, rows * ch + gap * (rows - 1)), (24, 24, 24))
    for i, im in enumerate(images):
        r, c = divmod(i, cols)
        out.paste(im.resize((cw, ch), Image.LANCZOS), (c * (cw + gap), r * (ch + gap)))
    return out


def save_jpg(img, name):
    if img.width > 1600:
        img = img.resize((1600, int(img.height * 1600 / img.width)), Image.LANCZOS)
    img.save(os.path.join(OUT, name), "JPEG", quality=85)
    return name

def erode(m, n=1):
    e = m.copy()
    for _ in range(n):
        e2 = np.zeros_like(e)
        e2[1:-1, 1:-1] = e[1:-1, 1:-1] & e[:-2, 1:-1] & e[2:, 1:-1] & e[1:-1, :-2] & e[1:-1, 2:]
        e = e2
    return e


def lum(im):
    return 0.2126 * im[..., 0] + 0.7152 * im[..., 1] + 0.0722 * im[..., 2]

def read_instances(tag):
    rows = {}
    fn = p(tag + "_instances.tsv")
    if not os.path.exists(fn):
        return rows
    with open(fn) as fh:
        rd = csv.reader(fh, delimiter="\t")
        next(rd, None)
        for row in rd:
            if not row or not row[0].isdigit():
                continue
            rows[int(row[0])] = {"renderer": row[1], "material": row[2] if len(row) > 2 else "", "glass": row[4] if len(row) > 4 else ""}
    return rows


def diag_analyse(tag, base_png=None, crop_mask=None, highlight=None):
    fn = tag + "_diag_rt.f32"
    if not os.path.exists(p(fn)):
        return None, None
    d = load_f32(fn)
    inst = read_instances(tag)
    g = d[..., 3] > 0.5
    hit = np.rint(d[..., 0]).astype(np.int64)
    first = np.rint(d[..., 2]).astype(np.int64)
    dist = d[..., 1]
    mats = {}
    for i, row in inst.items():
        mats[i + 1] = row["material"] or row["renderer"]
    cat = np.full(hit.shape, "", dtype=object)
    r = {"glass_px": int(g.sum()), "miss_px": int((g & (hit == 0)).sum())}
    shares = {}
    for h in np.unique(hit[g]):
        name = "MISS (sky gradient)" if h == 0 else mats.get(int(h), f"instance {h-1}")
        n = int((g & (hit == h)).sum())
        shares[name] = shares.get(name, 0) + n
    r["reflected_surface_share"] = {k: round(100.0 * v / max(1, r["glass_px"]), 2) for k, v in sorted(shares.items(), key=lambda kv: -kv[1])}
    dd = dist[g & (hit > 0)]
    if dd.size:
        r["reflected_distance_m"] = {"p10": round(float(np.percentile(dd, 10)), 2), "median": round(float(np.median(dd)), 2), "p90": round(float(np.percentile(dd, 90)), 2), "max": round(float(dd.max()), 2)}
    prim = first % 16
    finst = first // 16
    if broad:
        edge = g & np.vectorize(lambda q: not broad.get(int(q), True))(prim)
        r["first_hit_edge_face_px"] = int(edge.sum())
    else:
        edge = np.zeros_like(g)
    r["first_hit_instances"] = {inst.get(int(k), {}).get("renderer", str(k)) + f" #{int(k)}": int((g & (finst == k)).sum()) for k in np.unique(finst[g])}
    # image
    base = load_png(base_png).astype(np.float32) * 0.45 if base_png else np.zeros((H, W, 3), np.float32)
    img = base.copy()
    order = [k for k in r["reflected_surface_share"].keys() if not k.startswith("MISS")]
    colours = {}
    for i, k in enumerate(order):
        colours[k] = PALETTE[i % len(PALETTE)]
    colours["MISS (sky gradient)"] = MISS
    for h in np.unique(hit[g]):
        name = "MISS (sky gradient)" if h == 0 else mats.get(int(h), f"instance {h-1}")
        img[g & (hit == h)] = colours[name]
    if highlight is not None:
        img[highlight & g] = (255, 30, 30)
    img[edge] = (255, 255, 0)
    im = Image.fromarray(np.clip(img, 0, 255).astype(np.uint8))
    if crop_mask is not None and crop_mask.any():
        ys, xs = np.nonzero(crop_mask)
        x0, x1 = max(0, xs.min() - 80), min(W, xs.max() + 80)
        y0, y1 = max(0, ys.min() - 60), min(H, ys.max() + 60)
        im = im.crop((x0, y0, x1, y1))
    return r, (im, colours)


def legend_tile(width, height, title, r, colours):
    im = Image.new("RGB", (width, height), (24, 24, 24))
    dr = ImageDraw.Draw(im)
    fs = 22
    y = 12
    dr.text((14, y), title, fill=(255, 255, 255), font=font(fs + 2)); y += fs + 16
    for k, v in list(r["reflected_surface_share"].items())[:9]:
        c = colours.get(k, (128, 128, 128))
        dr.rectangle([14, y + 3, 34, y + 21], fill=c)
        dr.text((44, y), f"{k}: {v:.1f} %", fill=(235, 235, 235), font=font(fs)); y += fs + 8
    if "first_hit_edge_face_px" in r:
        dr.rectangle([14, y + 3, 34, y + 21], fill=(255, 255, 0))
        dr.text((44, y), f"first hit on a 30 mm edge face: {r['first_hit_edge_face_px']} px", fill=(235, 235, 235), font=font(fs)); y += fs + 8
    if "reflected_distance_m" in r:
        q = r["reflected_distance_m"]
        dr.text((14, y + 6), f"reflected hit distance: median {q['median']} m (10-90 %: {q['p10']}-{q['p90']} m)", fill=(200, 200, 200), font=font(fs - 2))
    return im
