"""patch5c.py - G14b revision 2 (2026-10-07): rt_bench5b.mm -> rt_bench5c.mm. Not part of the game.

Adds, without changing any TREE=0 result (TREE=0 is bench 5b bit for bit):
  TREE=1      the DESIGN tree of report 11 r2 s2.5 instead of bench 5's first layout:
              - see-through continuation in registers (G14's FRTraceReflection loop), not pushed;
              - only reflection branches pushed, on a stack of maxBounces - 1 entries (High 1, Ultra 2);
              - G14's pane terms: Fr = 0.08 + 0.92 F^5, transmission 1 - (0.11 + 0.89 F^5) (main .metal:467-468)
                instead of 11's (1 - F) 0.9;
              - G14's layer limit: past the layer budget the pane shows the zone cube along the ray (main :456-462);
              - facing rule: an exiting glass face passes with no Fresnel and no layer (the bench's triNormal is the
                unflipped world normal of positive-scale boxes, so 'entering' is exact here);
              - soft gate and cut-off as before; the cube takes the (1 - g) share when a branch is pushed.
              Reference caches get a '_t1' key.
  ARMEDCUBE=1 the tripped corridor (scene 1) uses the ARMED corridor capture (scene 3) (check r2's case);
  TRIPCUBE=1  the armed corridor (scene 3) uses the TRIPPED capture (scene 1). Both add a key to the caches.
  mode 'ramp' issue 2 (B12): Ultra floor ray-count switches 1->2 (High's mirror path -> 2 GGX), 2->3, 3->4,
              instant vs a linear ramp between the nested estimators over N frames, optionally staggered by a
              hashed per-16x16-tile phase of 0..P-1 frames. Floor px, display /255, worst frame of the ramp.
"""
import sys
src = open("rt_bench5b.mm").read()

def rep(old, new, count=1):
    global src
    n = src.count(old)
    if n != count:
        sys.exit(f"patch5c: expected {count} x {old[:70]!r}, found {n}")
    src = src.replace(old, new)

DESIGN = r'''
// ---- r2: the DESIGN tree (report 11 r2 s2.5): G14's see-through loop in registers + a reflection-only stack of
// maxBounces - 1 entries; G14's pane terms and layer limit; exiting glass faces pass (facing rule).
struct RB { float3 o; float3 d; float w; float trav; uint depth; uint layers; };
static float3 tracePathD(constant Frame& f, constant GpuInst* insts, instance_acceleration_structure scene,
                         texture2d_array<half> alb, texturecube<half> env, float3 o, float3 d, float importance,
                         float coneStart, thread uint& rays, thread uint& srays, thread float& firstT, thread uint& maxDepth)
{
  RB st[2]; uint sp = 0u; uint cap = f.maxBounces > 1u ? min(f.maxBounces - 1u, 2u) : 0u;
  float3 L = 0; firstT = -1;
  float w = 1; float trav = coneStart; uint depth = 1u, layers = 0u;
  intersector<triangle_data, instancing> isect; isect.assume_geometry_type(geometry_type::triangle);
  for (uint guard = 0u; guard < 24u; guard++) {
    rays++; maxDepth = max(maxDepth, depth);
    ray r; r.origin = o; r.direction = d; r.min_distance = 0.0005f; r.max_distance = f.maxDist;
    auto hit = isect.intersect(r, scene, f.rayMask);
    bool endB = false;
    if (hit.type != intersection_type::triangle) { L += w * envS(f, env, d, 0); if (firstT < 0) firstT = f.maxDist; endB = true; }
    else {
      if (firstT < 0) firstT = hit.distance;
      constant GpuInst& g = insts[hit.instance_id];
      float3 n = triNormal(g, hit.primitive_id); bool entering = dot(n, d) < 0; if (!entering) n = -n;
      float3 P = o + d * hit.distance; float travH = trav + hit.distance;
      float c = saturate(-dot(n, d));
      if ((g.flags & F_GLASS) && !entering) { o = P + d * 0.002f; trav = travH; continue; }   // exiting face: pass, no layer
      if (g.flags & F_GLASS) {
        if (layers >= f.maxLayers) { L += w * envS(f, env, d, 0); endB = true; }               // G14: the pane shows the cube
        else {
          float m = 1.0f - c; float f5 = m * m * m * m * m;
          float Fp = 0.08f + 0.92f * f5, tr = 1.0f - (0.11f + 0.89f * f5);                       // G14 .metal:467-468
          float3 R = reflect(d, n); float wr = w * Fp;
          if (depth < f.maxBounces && sp < cap && importance * wr > f.cutoff) { RB b; b.o = P + n * 0.002f; b.d = R; b.w = wr; b.trav = travH; b.depth = depth + 1u; b.layers = layers; st[sp++] = b; }
          else L += wr * envS(f, env, R, 0);
          w *= tr; layers++; o = P + d * 0.002f; trav = travH; continue;
        }
      } else {
        bool glossy = (g.flags & F_FLOOR) != 0 && f.floorGlossy != 0 && n.y > 0.9f;
        bool metal = (g.flags & F_METAL) != 0;
        float rough = glossy ? f.floorRough : g.emis.w;
        float3 Ld = shadeHit(f, insts, scene, alb, hit.instance_id, P, n, d, travH * f.pixelAngle,
                             depth == 1u ? f.shadow1 : f.shadow2, rough, srays);
        if (glossy || metal) {
          float Fs = metal ? schlick(0.7f, c) : schlick(0.04f, c); float3 R = reflect(d, n);
          L += w * (metal ? 1.f : (1 - Fs)) * Ld;
          float wr = w * Fs;
          float gg = (f.pad0 & 1u) ? (1.0f - smoothstep(0.10f, 0.25f, rough)) : 1.0f;
          if (gg > 0.0f && depth < f.maxBounces && rough <= f.recurseMaxRough && importance * wr * gg > f.cutoff && sp < cap) {
            RB b; b.o = P + n * 0.002f; b.d = R; b.w = wr * gg; b.trav = travH; b.depth = depth + 1u; b.layers = layers; st[sp++] = b;
            L += wr * (1.0f - gg) * envS(f, env, R, 4.0f * rough);                              // the cube keeps the (1 - g) share
          } else L += wr * envS(f, env, R, 4.0f * rough);
        } else L += w * Ld;
        endB = true;
      }
    }
    if (endB) {
      if (sp == 0u) break;
      RB b = st[--sp]; o = b.o; d = b.d; w = b.w; trav = b.trav; depth = b.depth; layers = b.layers;
    }
  }
  return L;
}
static float3 tracePath(constant Frame& f, constant GpuInst* insts, instance_acceleration_structure scene,
                        texture2d_array<half> alb, texturecube<half> env, float3 o, float3 d, float importance,
                        float coneStart, thread uint& rays, thread uint& srays, thread float& firstT, thread uint& maxDepth)
{
  if (f.tree == 1u) return tracePathD(f, insts, scene, alb, env, o, d, importance, coneStart, rays, srays, firstT, maxDepth);
  return tracePathF(f, insts, scene, alb, env, o, d, importance, coneStart, rays, srays, firstT, maxDepth);
}

// Stackless depth-1 path'''
rep("\n// Stackless depth-1 path", DESIGN)

RAMP = r'''    if (what == "ramp") {

      // The estimators are nested (sample s hashes (cell, s) only), so n+1 rays reuse the n rays' samples; a ramp
      // blends the two filtered results linearly (the mip blur is linear) over N frames; with P > 0 each 16x16 tile
      // starts its ramp after a hashed phase of 0..P-1 frames. n = 1 is High's mirror path (k 2) in Ultra's tree.
      printf("# --- ramp 1920x1080, footprint jitter, floor px, display /255, worst frame of each ramp  %s\n", loadStr().c_str());
      struct RC { const char* key; const char* ul; };
      std::vector<RC> rcs = {{"A20", "A_U3"}, {"B20", "B_U2"}, {"C20", "C_U2"}, {"D20", "D_U2"}, {"F20", "F_U3"}, {"G15", "G_U2"}};
      for (auto& rc : rcs) {

        std::vector<std::vector<float>> lin(5);
        uint32_t sj = jitterMode; jitterMode = 2;
        for (uint32_t n = 1; n <= 4; n++) {
