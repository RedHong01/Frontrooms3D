"""r2_b6_summary.py - G14b r2, bench 6 with the facing rule: per view/tree, median over runs of each variant's ratio to main (min and median)."""
import re, glob, statistics as st
d = {}; loads = []
for f in sorted(glob.glob("r2_f_*_[12].txt")):
    for line in open(f):
        m = re.search(r"b6 (\w) (\w+) (\S+) \d+x\d+ (\w+)\s+min ([\d.]+) p25 ([\d.]+) med ([\d.]+) ms \| ratio to main: min ([\d.]+) med ([\d.]+) \| load (\S+)", line)
        if m:
            key = (m.group(1) + " " + m.group(2), m.group(3)); d.setdefault(key, {}).setdefault(m.group(4), []).append((float(m.group(5)), float(m.group(7)), float(m.group(8)), float(m.group(9)))); loads.append(m.group(10))
print(f"# loads {loads[0]} .. {loads[-1]}; ratio = variant / main of the same process (interleaved, 36 reps); min-ratio [median-ratio], median over runs")
names = ["main", "s1", "s2", "s3", "x1", "x2", "x3", "y1", "y2", "y3"]
print(f"{'view':12} {'tree':10} {'main ms min/med':>16} | " + " ".join(f"{n:>13}" for n in names[1:]))
for k in sorted(d):
    v = d[k]; mm = st.median(x[0] for x in v["main"]); md = st.median(x[1] for x in v["main"])
    print(f"{k[0]:12} {k[1]:10} {mm:7.3f}/{md:7.3f} | " + " ".join(f"{st.median(x[2] for x in v[n]):5.3f} [{st.median(x[3] for x in v[n]):5.3f}]" for n in names[1:]))
