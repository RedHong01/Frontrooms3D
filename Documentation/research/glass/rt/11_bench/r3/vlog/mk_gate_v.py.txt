# G14b record stage (2026-10-07): soft gate on/off at view G (2.5 m from the goal, armed), depth 2 vs 1, per RT-off cube.
# Data: r2/b5 r2_t1_diff2_{l0exr,flat,own05}{,_gate}.txt (rows 'G_h1 (1) vs G_h2 (1)'); the l0exr_gate row is re-rendered
# in vlog_g14b/v_l0g_* (same numbers). Two stacked panels, one measure each, sized for a 654 x 657 slide slot (x2).
from PIL import Image, ImageDraw, ImageFont
FD = '/Users/redwang/Library/Fonts/'; S = 2
f20 = ImageFont.truetype(FD + 'IBMPlexMono-Regular.otf', 20 * S); fb = ImageFont.truetype(FD + 'IBMPlexMono-Bold.otf', 22 * S)
INK, MUTED, GRID = '#0A0A0A', '#52514E', '#E4E2DC'; C = ['#2a78d6', '#eb6834']
W, H = 654 * S, 657 * S
img = Image.new('RGB', (W, H), '#FFFFFF'); d = ImageDraw.Draw(img)
x = 0
for c, t in [(C[0], 'soft gate on (design)'), (C[1], 'gate off')]:
    d.rounded_rectangle([x, 4 * S, x + 22 * S, 26 * S], 3 * S, fill=c); d.text((x + 30 * S, 0), t, font=f20, fill=INK); x += int(d.textlength(t, font=f20)) + 60 * S
cubes = ["main's cube", 'flat cube', 'own capture']
panels = [('LITE SHOWS THE FLOOR', 'max change /255', [(73, 85), (71, 83), (63, 74)], 100, 40, 'R2: >= 40', lambda v: '%d' % v),
          ('FLOOR SHOWS THE LITE', 'px >= 8/255 (1920x1080)', [(359, 509), (331, 479), (345, 489)], 600, 200, 'R2: >= 200 px', lambda v: '%d' % v)]
lab = 170 * S; x0 = lab; x1 = W - 60 * S
ph = 300 * S
for k, (title, unit, vals, vmax, bar_at, bar_txt, fmt) in enumerate(panels):
    py = 50 * S + k * (ph + 12 * S)
    d.text((0, py), title, font=fb, fill=INK); d.text((0, py + 30 * S), unit, font=f20, fill=MUTED)
    X = lambda v: x0 + (x1 - x0) * v / vmax
    top = py + 70 * S; rh = 60 * S; bh = 24 * S; yb = top + 3 * rh
    for t in range(0, vmax + 1, vmax // 4):
        d.line([X(t), top - 4 * S, X(t), yb], fill=GRID, width=S); tw = d.textlength(str(t), font=f20); d.text((X(t) - tw / 2, yb + 6 * S), str(t), font=f20, fill=MUTED)
    for i, (cn, (on, off)) in enumerate(zip(cubes, vals)):
        y = top + i * rh + 4 * S; d.text((0, y + 12 * S), cn, font=f20, fill=INK)
        for j, v in enumerate((on, off)):
            yy = y + j * (bh + 2 * S); xe = X(v)
            d.rounded_rectangle([x0, yy, xe, yy + bh], 4 * S, fill=C[j]); d.rectangle([x0, yy, x0 + 4 * S, yy + bh], fill=C[j])
            d.text((xe + 8 * S, yy - 3 * S), fmt(v), font=f20, fill=INK)
    xl = int(X(bar_at))
    for yy in range(top - 8 * S, yb, 12 * S): d.line([xl, yy, xl, min(yy + 7 * S, yb)], fill=INK, width=2 * S)
    d.text((xl + 8 * S, top - 30 * S), bar_txt, font=f20, fill=INK)
img.save('gate_ab_chart_v.png'); print(img.size)
