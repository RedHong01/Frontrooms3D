#!/usr/bin/env python3
"""patch5w.py - G14b record stage (2026-10-07 23:xx): rt_bench5v.mm -> rt_bench5w.mm. Not part of the game.
Adds mode 'img4': views A/B/D/F (S1 tripped, near goal tripped, stream room, S1 armed), each rendered RT off (comp 0)
and High with the floor receiver (comp 1) through the same render path as 'diff2', plus the pixel-class mask
(R = floor, G = glass), as 1920x1080 PNGs. Every other mode unchanged."""
import sys
src = open("rt_bench5v.mm").read()
old = '    if (what == "img3") {'
if src.count(old) != 1: sys.exit("patch5w: anchor not found once")
new = '''    if (what == "img4") {
      for (const char* cid : {"A_h2", "B_h2", "D_h2", "F_h2"}) {
        std::vector<float> cls;
        for (uint32_t m : {0u, 1u}) { auto px = render(cid, m, &cls); std::string p = std::string("w_") + tag + "_" + cid + (m ? "_rt" : "_off") + ".png";
          writePng(p, px, ti.W, ti.H); printf("wrote %s | %s\\n", p.c_str(), loadStr().c_str()); }
        std::vector<uint8_t> mk((size_t)ti.W * ti.H * 4, 0);
        for (size_t i = 0; i < (size_t)ti.W * ti.H; i++) { int k = (int)(cls[i * 4 + 3] + .5f); mk[i * 4] = k == 2 ? 255 : 0; mk[i * 4 + 1] = k == 1 ? 255 : 0; mk[i * 4 + 3] = 255; }
        std::string p = std::string("w_") + tag + "_" + cid + "_mask.png"; writePng(p, mk, ti.W, ti.H); printf("wrote %s\\n", p.c_str());
      }
    }
'''
open("rt_bench5w.mm", "w").write(src.replace(old, new + old))
print("wrote rt_bench5w.mm")
