static Mesh makeMesh(const std::vector<float>& pos, const std::vector<uint32_t>& idx) {
  Mesh m; m.vcount = (uint32_t)pos.size() / 3; m.tcount = (uint32_t)idx.size() / 3;
  m.vb = [dev newBufferWithBytes:pos.data() length:pos.size() * 4 options:MTLResourceStorageModeShared];
  m.ib = [dev newBufferWithBytes:idx.data() length:idx.size() * 4 options:MTLResourceStorageModeShared];
  return m;
}
static Mesh boxMesh() {
  std::vector<float> p = {-.5f,-.5f,-.5f, .5f,-.5f,-.5f, .5f,.5f,-.5f, -.5f,.5f,-.5f, -.5f,-.5f,.5f, .5f,-.5f,.5f, .5f,.5f,.5f, -.5f,.5f,.5f};
  std::vector<uint32_t> i = {0,2,1,0,3,2, 4,5,6,4,6,7, 0,1,5,0,5,4, 3,7,6,3,6,2, 0,4,7,0,7,3, 1,2,6,1,6,5};
  return makeMesh(p, i);
}
static Mesh blobMesh(int lat, int lon, uint32_t seed) {
  std::mt19937 rng(seed); std::uniform_real_distribution<float> u(.85f, 1.15f);
  std::vector<float> p; std::vector<uint32_t> i;
  for (int a = 0; a <= lat; a++) for (int b = 0; b <= lon; b++) {
    float th = M_PI * a / lat, ph = 2 * M_PI * b / lon, r = .5f * u(rng);
    p.push_back(r * sinf(th) * cosf(ph)); p.push_back(r * cosf(th)); p.push_back(r * sinf(th) * sinf(ph));
  }
  for (int a = 0; a < lat; a++) for (int b = 0; b < lon; b++) {
    uint32_t v0 = a * (lon + 1) + b, v1 = v0 + 1, v2 = v0 + lon + 1, v3 = v2 + 1;
    i.insert(i.end(), {v0, v2, v1, v1, v2, v3});
  }
  return makeMesh(p, i);
}
static void buildBlas(std::vector<Mesh>& meshes) {
  auto cb = [q commandBuffer]; auto enc = [cb accelerationStructureCommandEncoder];
  std::vector<id<MTLBuffer>> scratch;
  for (auto& m : meshes) {
    auto g = [MTLAccelerationStructureTriangleGeometryDescriptor descriptor];
    g.vertexBuffer = m.vb; g.vertexStride = 12; g.indexBuffer = m.ib; g.indexType = MTLIndexTypeUInt32; g.triangleCount = m.tcount; g.opaque = YES;
    auto d = [MTLPrimitiveAccelerationStructureDescriptor descriptor]; d.geometryDescriptors = @[g];
    auto s = [dev accelerationStructureSizesWithDescriptor:d];
    m.blas = [dev newAccelerationStructureWithSize:s.accelerationStructureSize];
    id<MTLBuffer> sc = [dev newBufferWithLength:std::max<NSUInteger>(s.buildScratchBufferSize, 16) options:MTLResourceStorageModePrivate];
    scratch.push_back(sc);
    [enc buildAccelerationStructure:m.blas descriptor:d scratchBuffer:sc scratchBufferOffset:0];
  }
  [enc endEncoding]; [cb commit]; [cb waitUntilCompleted];
}

static float3 xf(constant GpuInst& g, float3 p) { float4 h = float4(p, 1); return float3(dot(g.m0, h), dot(g.m1, h), dot(g.m2, h)); }
static float3 triNormal(constant GpuInst& g, uint p) {
  float3 a = xf(g, float3(g.vb[g.ib[p*3]*3], g.vb[g.ib[p*3]*3+1], g.vb[g.ib[p*3]*3+2]));

static float schlick(float f0, float c) { float m = 1 - saturate(c); float m2 = m * m; return f0 + (1 - f0) * m2 * m2 * m; }
static float paneF(float c) { return schlick(0.08f, c); }       // two-surface 6 mm pane (03 s5.1)
static float G1(float nl, float a) { float a2 = a * a; return 2 * nl / (nl + sqrt(a2 + (1 - a2) * nl * nl)); }
static float3 envS(constant Frame& f, texturecube<half> env, float3 d, float lod) {
  constexpr sampler cs(filter::linear, mip_filter::linear);
  if (f.pad1 & 1u) { float r = lod * 0.25f; lod = min(6.0f * r * (1.7f - 0.7f * r), float(env.get_num_mip_levels() - 1)); }   // URP PerceptualRoughnessToMipmapLevel (callers pass 4 r)

static float3 shadeHit(constant Frame& f, constant GpuInst* insts, instance_acceleration_structure scene,
                       texture2d_array<half> alb, uint inst, float3 P, float3 n, float3 rd, float cone,
                       uint nShadow, float rough, thread uint& srays)
{
  constexpr sampler smp(filter::linear, mip_filter::linear, address::repeat);
  constant GpuInst& g = insts[inst];
  float3 an = abs(n);
  float2 uv = an.y > .5f ? P.xz : (an.x > .5f ? P.zy : P.xy);
  uv /= 0.6f;
  float lod = log2(max(cone / (0.6f / 1024.f), 1e-4f) / max(abs(dot(n, rd)), 0.2f));
  uint layer = g.tint.w >= 0 ? uint(g.tint.w) : ((g.flags & F_FLOOR) ? 0u : (an.y > .5f ? 1u : 2u + (inst & 5u)));
  float3 albedo = float3(alb.sample(smp, uv, layer, level(lod)).rgb) * g.tint.rgb;
  bool metal = (g.flags & F_METAL) != 0;
  float f0 = metal ? 0.7f : 0.04f;
  float a = max(rough * rough, 0.02f), a2 = a * a;
  float3 V = -rd; float NV = saturate(dot(n, V)) + 1e-4f;
  float3 L = 0; uint shadowed = 0;
  intersector<instancing> sh; sh.accept_any_intersection(true); sh.assume_geometry_type(geometry_type::triangle);
  for (uint k = 0; k < f.lightCount; k++) {
    Light lt = f.lights[k];
    float3 tl = lt.pr.xyz - P; float d2 = dot(tl, tl); float3 ld = tl * rsqrt(d2);
    float ndl = dot(n, ld); float r2 = lt.pr.w * lt.pr.w;
    if (ndl <= 0 || d2 > r2) continue;
    float fr = d2 / r2; float sm = saturate(1 - fr * fr); float att = sm * sm / max(d2, 1e-4f);
    float sa = saturate(dot(lt.dir.xyz, ld) * lt.dir.w + lt.col.w); att *= sa * sa;       // URP AngleAttenuation (point: w 0, add 1)
    if (att <= 1e-5f) continue;
    float vis = 1;
    if (shadowed < nShadow) {
      shadowed++; srays++;
      ray s; s.origin = P + n * 0.002f; s.direction = ld; s.min_distance = 0.001f; s.max_distance = sqrt(d2) - 0.02f;
      vis = sh.intersect(s, scene, 0x01u).type == intersection_type::none ? 1.f : 0.f;   // glass casts no shadow
    }
    float3 H = normalize(ld + V); float NH = saturate(dot(n, H));
    float dd = NH * NH * (a2 - 1) + 1; float D = a2 / (M_PI_F * dd * dd);
    float k2 = a * 0.5f; float G = (ndl / (ndl * (1 - k2) + k2)) * (NV / (NV * (1 - k2) + k2));
    float spec = D * G * f0 / (4 * NV * ndl + 1e-4f) * M_PI_F;                          // URP folds pi into light units
    L += ndl * vis * att * lt.col.rgb * ((metal ? 0.f : 1.f) * albedo + spec * (metal ? albedo : float3(1)));
  }
  bool emits = all(g.aux.xyz == 0) || dot(n, g.aux.xyz) > g.aux.w;
  return L + (metal ? 0.f : 1.f) * albedo * f.ambient + (emits ? g.emis.rgb : float3(0));
}
struct PE { float3 o; float3 d; float w; float trav; uint depth; uint layers; };
// Explicit-stack ray tree. Returns radiance arriving along d at o (no receiver Fresnel).
static float3 tracePath(constant Frame& f, constant GpuInst* insts, instance_acceleration_structure scene,
                        texture2d_array<half> alb, texturecube<half> env, float3 o, float3 d, float importance,
                        float coneStart, thread uint& rays, thread uint& srays, thread float& firstT, thread uint& maxDepth)
{
  PE st[8]; int sp = 0;
  PE e0; e0.o = o; e0.d = d; e0.w = 1; e0.trav = coneStart; e0.depth = 1; e0.layers = 0; st[sp++] = e0;
  float3 L = 0; firstT = -1;
  intersector<triangle_data, instancing> isect; isect.assume_geometry_type(geometry_type::triangle);
  uint budget = 16;
  while (sp > 0 && budget > 0) {
    PE e = st[--sp]; budget--; rays++; maxDepth = max(maxDepth, e.depth);
    ray r; r.origin = e.o; r.direction = e.d; r.min_distance = 0.0005f; r.max_distance = f.maxDist;
    auto hit = isect.intersect(r, scene, f.rayMask);
    if (hit.type != intersection_type::triangle) { L += e.w * envS(f, env, e.d, 0); if (firstT < 0) firstT = f.maxDist; continue; }
    if (firstT < 0) firstT = hit.distance;
    constant GpuInst& g = insts[hit.instance_id];
    float3 n = triNormal(g, hit.primitive_id); bool entering = dot(n, e.d) < 0; if (!entering) n = -n;
    float3 P = e.o + e.d * hit.distance;
    float trav = e.trav + hit.distance;
    float c = saturate(-dot(n, e.d));
    if ((g.flags & F_GLASS) && !entering) {               // back face of the same 6 mm pane: the pane Fresnel covers both surfaces
      if (sp < 8) { PE t = e; t.o = P + e.d * 0.002f; t.trav = trav; st[sp++] = t; } else L += e.w * envS(f, env, e.d, 0);
      continue;
    }
    if (g.flags & F_GLASS) {
      float Fp = paneF(c); float3 R = reflect(e.d, n);
      float wt = e.w * (1 - Fp) * 0.9f;                    // see-through continuation (not a bounce)
      if (e.layers < f.maxLayers && sp < 8) { PE t; t.o = P + e.d * 0.002f; t.d = e.d; t.w = wt; t.trav = trav; t.depth = e.depth; t.layers = e.layers + 1; st[sp++] = t; }
      else L += wt * envS(f, env, e.d, 0);
      float wr = e.w * Fp;                                  // reflection of the pane: a bounce
      if (e.depth < f.maxBounces && importance * wr > f.cutoff && sp < 8) { PE t; t.o = P + n * 0.002f; t.d = R; t.w = wr; t.trav = trav; t.depth = e.depth + 1; t.layers = e.layers; st[sp++] = t; }
      else L += wr * envS(f, env, R, 0);
      continue;
    }
    bool glossy = (g.flags & F_FLOOR) != 0 && f.floorGlossy != 0 && n.y > 0.9f;
    bool metal = (g.flags & F_METAL) != 0;
    float rough = glossy ? f.floorRough : g.emis.w;
    float3 Ld = shadeHit(f, insts, scene, alb, hit.instance_id, P, n, e.d, trav * f.pixelAngle,
                         e.depth == 1 ? f.shadow1 : f.shadow2, rough, srays);
    if (glossy || metal) {
      float Fs = metal ? schlick(0.7f, c) : schlick(0.04f, c); float3 R = reflect(e.d, n);
      L += e.w * (metal ? 1.f : (1 - Fs)) * Ld;
      float wr = e.w * Fs;
      float gg = (f.pad0 & 1u) ? (1.0f - smoothstep(0.10f, 0.25f, rough)) : 1.0f;
      if (gg > 0.0f && e.depth < f.maxBounces && rough <= f.recurseMaxRough && importance * wr * gg > f.cutoff && sp < 8) { PE t; t.o = P + n * 0.002f; t.d = R; t.w = wr * gg; t.trav = trav; t.depth = e.depth + 1; t.layers = e.layers; st[sp++] = t; L += wr * (1.0f - gg) * envS(f, env, R, 4.0f * rough); }
      else L += wr * envS(f, env, R, 4.0f * rough);
    } else L += e.w * Ld;
  }
  return L;
}

// Stackless depth-1 path: G14's FRTraceReflection shape (see-through layers in a loop, first opaque hit shaded, cube otherwise).
static float3 tracePathNS(constant Frame& f, constant GpuInst* insts, instance_acceleration_structure scene,
                        texture2d_array<half> alb, texturecube<half> env, float3 o, float3 d, float importance,
                        float coneStart, thread uint& rays, thread uint& srays, thread float& firstT, thread uint& maxDepth)
{
  float3 L = 0; firstT = -1; float w = 1; float trav = coneStart; uint layers = 0; maxDepth = max(maxDepth, 1u);
  intersector<triangle_data, instancing> isect; isect.assume_geometry_type(geometry_type::triangle);
  for (uint guard = 0; guard < 16u; guard++) {
    rays++;
    ray r; r.origin = o; r.direction = d; r.min_distance = 0.0005f; r.max_distance = f.maxDist;
    auto hit = isect.intersect(r, scene, f.rayMask);
    if (hit.type != intersection_type::triangle) { L += w * envS(f, env, d, 0); if (firstT < 0) firstT = f.maxDist; break; }
    if (firstT < 0) firstT = hit.distance;
    constant GpuInst& g = insts[hit.instance_id];
    float3 n = triNormal(g, hit.primitive_id); bool entering = dot(n, d) < 0; if (!entering) n = -n;
    float3 P = o + d * hit.distance; trav += hit.distance;
    float c = saturate(-dot(n, d));
    if ((g.flags & F_GLASS) && !entering) { o = P + d * 0.002f; continue; }
    if (g.flags & F_GLASS) {
      float Fp = paneF(c); float3 R = reflect(d, n);
      L += w * Fp * envS(f, env, R, 0);
      float wt = w * (1 - Fp) * 0.9f;
      if (layers < f.maxLayers) { layers++; w = wt; o = P + d * 0.002f; continue; }
      L += wt * envS(f, env, d, 0); break;
    }
    bool glossy = (g.flags & F_FLOOR) != 0 && f.floorGlossy != 0 && n.y > 0.9f;
    bool metal = (g.flags & F_METAL) != 0;
    float rough = glossy ? f.floorRough : g.emis.w;
    float3 Ld = shadeHit(f, insts, scene, alb, hit.instance_id, P, n, d, trav * f.pixelAngle, f.shadow1, rough, srays);
    if (glossy || metal) {
      float Fs = metal ? schlick(0.7f, c) : schlick(0.04f, c); float3 R = reflect(d, n);
      L += w * (metal ? 1.f : (1 - Fs)) * Ld + w * Fs * envS(f, env, R, 4.0f * rough);
    } else L += w * Ld;
    break;
  }
  return L;
}
static float2 rnd2(uint2 p, uint s, uint seed) {
  float i1 = fract(52.9829189f * fract(0.06711056f * float(p.x) + 0.00583715f * float(p.y)));
  float i2 = fract(52.9829189f * fract(0.06711056f * float(p.x + 37u) + 0.00583715f * float(p.y + 113u)));
  float2 r2 = fract(float2(0.7548776662f, 0.5698402910f) * float(s + seed * 17u + 1u));
  return fract(float2(i1, i2) + r2);
}

// ---- b5: jitter modes (Frame.pad1 bits 1-3): 0 screen hash (seeded), 1 world cell 1 cm, 2 footprint-aware hashed
// (cell = pixel footprint along the view, two levels blended, Wyman & McGuire 2017 CDF fix), 3 same with the
// geometric-mean footprint.
static uint pcgh(uint v) { uint st = v * 747796405u + 2891336453u; uint w = ((st >> ((st >> 28u) + 4u)) ^ st) * 277803737u; return (w >> 22u) ^ w; }
static float2 hash2i(int3 qi, uint s, int lvl) {
  uint h = pcgh(uint(qi.x) * 73856093u ^ pcgh(uint(qi.y) * 19349663u ^ pcgh(uint(qi.z) * 83492791u ^ pcgh(s * 2654435761u + uint(lvl + 64)))));
  uint h2 = pcgh(h ^ 0x9E3779B9u);
  return float2(float(h >> 8) * (1.0f / 16777216.0f), float(h2 >> 8) * (1.0f / 16777216.0f));
}
static float wmCdf(float x, float t) {
  float a = min(t, 1.0f - t);
  if (a < 1e-4f) return x;
  float c = (x < a) ? x * x / (2.0f * a * (1.0f - a)) : ((x < 1.0f - a) ? (x - 0.5f * a) / (1.0f - a) : 1.0f - (1.0f - x) * (1.0f - x) / (2.0f * a * (1.0f - a)));
  return clamp(c, 1e-6f, 1.0f - 1e-6f);
}
static float2 jitterU(constant Frame& f, uint2 id, uint s, float3 P, float footprint) {
  uint jm = (f.pad1 >> 1) & 7u;
  if (jm == 0u) return rnd2(id, s, f.seed);
  uint ss = s + f.seed * 1000u;
  if (jm == 1u) return hash2i(int3(floor(P * 100.0f)), ss, 0);
  float lg = log2(max(footprint, 1e-4f)); float l0 = floor(lg); float t = lg - l0;
  float c0 = exp2(l0), c1 = exp2(l0 + 1.0f);
  float2 h0 = hash2i(int3(floor(P / c0)), ss, int(l0)), h1 = hash2i(int3(floor(P / c1)), ss, int(l0) + 1);
  float2 x = mix(h0, h1, t);
  return float2(wmCdf(x.x, t), wmCdf(x.y, t));
}
static float3 sampleVNDF(float3 Ve, float a, float2 u) {             // Heitz 2018, local frame z = normal
  float3 Vh = normalize(float3(a * Ve.x, a * Ve.y, Ve.z));
  float lensq = Vh.x * Vh.x + Vh.y * Vh.y;
  float3 T1 = lensq > 0 ? float3(-Vh.y, Vh.x, 0) * rsqrt(lensq) : float3(1, 0, 0);
  float3 T2 = cross(Vh, T1);
  float r = sqrt(u.x), phi = 2 * M_PI_F * u.y;
  float t1 = r * cos(phi), t2 = r * sin(phi);
  float s = 0.5f * (1 + Vh.z); t2 = (1 - s) * sqrt(1 - t1 * t1) + s * t2;
  float3 Nh = t1 * T1 + t2 * T2 + sqrt(max(0.f, 1 - t1 * t1 - t2 * t2)) * Vh;
  return normalize(float3(a * Nh.x, a * Nh.y, max(0.f, Nh.z)));
}
static float2 projPx(constant Frame& f, float3 X) {
  float3 v = X - f.camPos.xyz; float z = max(dot(v, f.camFwd.xyz), 1e-3f);
  float nx = dot(v, f.camRight.xyz) / (z * f.tanHalf * f.aspect);
  float ny = -dot(v, f.camUp.xyz) / (z * f.tanHalf);
  return (float2(nx, ny) * 0.5f + 0.5f) * float2(f.w, f.h) - 0.5f;
}
kernel void gbuffer(texture2d<float, access::write> gPos [[texture(0)]], texture2d<half, access::write> gNrm [[texture(1)]],
                    texture2d<half, access::write> base [[texture(2)]], texture2d_array<half> alb [[texture(3)]],
                    texturecube<half> env [[texture(4)]], instance_acceleration_structure scene [[buffer(0)]],
                    constant Frame& f [[buffer(1)]], constant GpuInst* insts [[buffer(2)]], uint2 id [[thread_position_in_grid]])
{
  if (id.x >= f.w || id.y >= f.h) return;
  float2 ndc = (float2(id) + .5f) / float2(f.w, f.h) * 2 - 1;
  float3 d = normalize(f.camFwd.xyz + f.camRight.xyz * ndc.x * f.tanHalf * f.aspect - f.camUp.xyz * ndc.y * f.tanHalf);
  ray r; r.origin = f.camPos.xyz; r.direction = d; r.min_distance = 0.01f; r.max_distance = 80.f;
  intersector<triangle_data, instancing> isect; isect.assume_geometry_type(geometry_type::triangle);
  float3 thr = 1, basec = 0, rP = 0, rN = 0; float rD = 0, trav = 0; uint cls = 0; uint dummy = 0;
  bool first = true;
  for (int layer = 0; layer < 8; layer++) {
    auto hit = isect.intersect(r, scene, f.rayMask);
    if (hit.type != intersection_type::triangle) { basec += thr * envS(f, env, r.direction, 0); break; }
    constant GpuInst& g = insts[hit.instance_id];
    float3 n = triNormal(g, hit.primitive_id); bool entering = dot(n, r.direction) < 0; if (!entering) n = -n;
    float3 P = r.origin + r.direction * hit.distance; trav += hit.distance;
    if ((g.flags & F_GLASS) && !entering) { r.origin = P + r.direction * 0.002f; continue; }   // back face of a thin pane
    bool wasFirst = first; first = false;
    if (g.flags & F_GLASS) {
      if (wasFirst) { cls = (f.receivers & 1u) ? 1u : 4u; rP = P; rN = n; rD = trav; }
      thr *= (1 - paneF(saturate(-dot(n, r.direction)))) * 0.9f; r.origin = P + r.direction * 0.002f; continue;
    }
    bool glossy = (g.flags & F_FLOOR) != 0 && f.floorGlossy != 0 && n.y > 0.9f;
    if (wasFirst && glossy) { cls = (f.receivers & 2u) ? 2u : 3u; rP = P; rN = n; rD = trav; }
    basec += thr * shadeHit(f, insts, scene, alb, hit.instance_id, P, n, r.direction, trav * f.pixelAngle, 8u,
                            glossy ? f.floorRough : g.emis.w, dummy);
    break;
  }
  gPos.write(float4(rP, rD), id); gNrm.write(half4(half3(rN), half(cls)), id); base.write(half4(half3(min(basec, 60000.f)), 1), id);
}
kernel void reflTrace(texture2d<half, access::write> out [[texture(0)]], texture2d<half, access::write> hitTex [[texture(1)]],
                      texture2d<float, access::read> gPos [[texture(2)]], texture2d<half, access::read> gNrm [[texture(3)]],
                      texture2d_array<half> alb [[texture(4)]], texturecube<half> env [[texture(5)]],
                      instance_acceleration_structure scene [[buffer(0)]], constant Frame& f [[buffer(1)]],
                      constant GpuInst* insts [[buffer(2)]], device atomic_uint* counters [[buffer(3)]],
                      uint2 id [[thread_position_in_grid]])
{
  if (id.x >= f.w || id.y >= f.h) return;
  float4 gn = float4(gNrm.read(id)); uint cls = uint(gn.w + 0.5f);
  uint rays = 0, srays = 0, maxDepth = 0; float3 res = 0; float hitT = 0; half outA = 1.0h;
  bool skip = cls == 2u && f.checker != 0u && ((id.x + id.y) & 1u) == 1u;
  if ((cls == 1u || cls == 2u) && !skip) {
    float4 gp = gPos.read(id);
    float3 P = gp.xyz, N = normalize(gn.xyz), V = normalize(f.camPos.xyz - P);
    if (cls == 1u) {
      float t; res = tracePath(f, insts, scene, alb, env, P + N * 0.002f, reflect(-V, N), paneF(dot(N, V)), gp.w, rays, srays, t, maxDepth); hitT = t;
    } else {
      uint ns = max(f.floorSpp, 1u); float sumT = 0; float3 acc = 0;
      float cosv = max(dot(N, V), 0.05f); float fpl = f.pixelAngle * gp.w / cosv, fps = f.pixelAngle * gp.w;
      float fpx = (((f.pad1 >> 1) & 7u) == 3u) ? sqrt(fpl * fps) : fpl;
      float3 T = normalize(cross(abs(N.x) < .9f ? float3(1, 0, 0) : float3(0, 1, 0), N)); float3 B = cross(N, T);
      float3 Vl = float3(dot(V, T), dot(V, B), dot(V, N));
      float a = max(f.floorRough * f.floorRough, 0.002f);
      for (uint s = 0; s < ns; s++) {
        float3 L; float w;
        if (f.floorMode == 1u) { L = reflect(-V, N); w = schlick(0.04f, dot(N, V)); }
        else {
          float3 Hl = sampleVNDF(Vl, a, jitterU(f, id, s, P, fpx)); float3 H = normalize(T * Hl.x + B * Hl.y + N * Hl.z);
          L = reflect(-V, H); float NL = dot(N, L);
          if (NL <= 0) continue;
          w = schlick(0.04f, dot(V, H)) * G1(NL, a);
        }
        float t; float3 Ls = tracePath(f, insts, scene, alb, env, P + N * 0.002f, L, w, gp.w, rays, srays, t, maxDepth);
        acc += w * Ls; sumT += t;
      }
      res = acc / float(ns); hitT = sumT / float(ns);
    }
  }
  if (skip) outA = 0.0h;
  out.write(half4(half3(min(res, 60000.f)), outA), id);
  hitTex.write(half4(half(hitT)), id);
  if (f.stats != 0u) {
    uint px = (cls == 1u || cls == 2u) ? 1u : 0u;
    uint a = simd_sum(px), b = simd_sum(rays), c = simd_sum(srays), g1 = simd_sum(cls == 1u ? 1u : 0u);
    uint rg = simd_sum((cls == 1u && maxDepth >= 2u) ? 1u : 0u), rf = simd_sum((cls == 2u && maxDepth >= 2u) ? 1u : 0u);
    uint r3 = simd_sum((px == 1u && maxDepth >= 3u) ? 1u : 0u);
    if (simd_is_first()) {
      atomic_fetch_add_explicit(&counters[0], a, memory_order_relaxed); atomic_fetch_add_explicit(&counters[1], b, memory_order_relaxed);
      atomic_fetch_add_explicit(&counters[2], c, memory_order_relaxed); atomic_fetch_add_explicit(&counters[3], g1, memory_order_relaxed);
      atomic_fetch_add_explicit(&counters[4], rg, memory_order_relaxed); atomic_fetch_add_explicit(&counters[5], rf, memory_order_relaxed);
      atomic_fetch_add_explicit(&counters[6], r3, memory_order_relaxed);
    }
  }
}

// Zone-cube capture from inside the bench scene (as G6 captures Refl_* from a cell): 1 bounce, flat-cube fallbacks.
kernel void captureCube(texturecube<half, access::write> outCube [[texture(0)]], texture2d_array<half> alb [[texture(1)]],
                        texturecube<half> env [[texture(2)]], instance_acceleration_structure scene [[buffer(0)]],
                        constant Frame& f [[buffer(1)]], constant GpuInst* insts [[buffer(2)]], uint3 id [[thread_position_in_grid]])
{
  uint N = outCube.get_width(); if (id.x >= N || id.y >= N || id.z >= 6u) return;
  float u = 2.0f * (float(id.x) + .5f) / float(N) - 1.0f, v = 2.0f * (float(id.y) + .5f) / float(N) - 1.0f; float3 d;
  switch (id.z) { case 0: d = float3(1, -v, -u); break; case 1: d = float3(-1, -v, u); break; case 2: d = float3(u, 1, v); break;
                  case 3: d = float3(u, -1, -v); break; case 4: d = float3(u, -v, 1); break; default: d = float3(-u, -v, -1); break; }
  d = normalize(d); uint rays = 0, srays = 0, md = 0; float t;
  float3 L = tracePath(f, insts, scene, alb, env, f.camPos.xyz, d, 1.0f, 0.0f, rays, srays, t, md);
  outCube.write(half4(half3(min(L, float3(60000.f))), 1.0h), uint2(id.xy), id.z);
}

// ---- b5: GGX-prefiltered cube mip (N = V = R), filtered importance sampling from the box-mipped source.
static float3 faceDir(uint face, float u, float v) {
  switch (face) { case 0: return float3(1, -v, -u); case 1: return float3(-1, -v, u); case 2: return float3(u, 1, v);
                  case 3: return float3(u, -1, -v); case 4: return float3(u, -v, 1); default: return float3(-u, -v, -1); }
}
kernel void prefilterGGX(texturecube<half> src [[texture(0)]], texture2d_array<half, access::write> dst [[texture(1)]],
                         constant float& rough [[buffer(0)]], uint3 id [[thread_position_in_grid]])
{
  uint NN = dst.get_width(); if (id.x >= NN || id.y >= NN || id.z >= 6u) return;
  float u = 2.0f * (float(id.x) + .5f) / float(NN) - 1.0f, v = 2.0f * (float(id.y) + .5f) / float(NN) - 1.0f;
  float3 Nn = normalize(faceDir(id.z, u, v));
  float a = max(rough * rough, 1e-3f);
  float3 T = normalize(cross(abs(Nn.y) < .99f ? float3(0, 1, 0) : float3(1, 0, 0), Nn)); float3 B = cross(Nn, T);
  constexpr sampler cs(filter::linear, mip_filter::linear);
  float srcW = float(src.get_width()); float omegaP = 4.0f * M_PI_F / (6.0f * srcW * srcW);
  float3 acc = 0; float wsum = 0;
  for (uint i = 0; i < 256u; i++) {
    float2 xi = float2((float(i) + 0.5f) / 256.0f, float(reverse_bits(i)) * 2.3283064365386963e-10f);
    float phi = 2 * M_PI_F * xi.y; float ct = sqrt((1 - xi.x) / (1 + (a * a - 1) * xi.x)); float st = sqrt(max(0.f, 1 - ct * ct));
    float3 H = T * (st * cos(phi)) + B * (st * sin(phi)) + Nn * ct;
    float3 L = 2 * dot(Nn, H) * H - Nn; float NL = dot(Nn, L);
    if (NL <= 0) continue;
    float d2 = ct * ct * (a * a - 1) + 1; float D = a * a / (M_PI_F * d2 * d2);
    float omegaS = 1.0f / (256.0f * D * 0.25f + 1e-4f);
    float lod = max(0.5f * log2(omegaS / omegaP) + 1.0f, 0.0f);
    acc += float3(src.sample(cs, L, level(lod)).rgb) * NL; wsum += NL;
  }
  dst.write(half4(half3(acc / max(wsum, 1e-4f)), 1.0h), uint2(id.xy), id.z);
}
kernel void latlong(texturecube<half> src [[texture(0)]], texture2d<half, access::write> dst [[texture(1)]],
                    constant float& lod [[buffer(0)]], uint2 id [[thread_position_in_grid]])
{
  uint W = dst.get_width(), H = dst.get_height(); if (id.x >= W || id.y >= H) return;
  float phi = (float(id.x) + .5f) / float(W) * 2 * M_PI_F - M_PI_F, th = (float(id.y) + .5f) / float(H) * M_PI_F;
  float3 d = float3(sin(th) * sin(phi), cos(th), sin(th) * cos(phi));   // x right, y up, z forward (phi 0 = +Z)
  constexpr sampler cs(filter::linear, mip_filter::linear);
  dst.write(src.sample(cs, d, level(lod)), id);
}
// Checkerboard fill: floor pixels that did not trace take the plane-aware mean of their 4 traced neighbours.
kernel void chkfill(texture2d<half, access::read> inTex [[texture(0)]], texture2d<half, access::write> outTex [[texture(1)]],
                    texture2d<half, access::read> hitIn [[texture(2)]], texture2d<half, access::write> hitOut [[texture(3)]],
                    texture2d<float, access::read> gPos [[texture(4)]], texture2d<half, access::read> gNrm [[texture(5)]],
                    constant Frame& f [[buffer(1)]], uint2 id [[thread_position_in_grid]])
{
  if (id.x >= f.w || id.y >= f.h) return;
  half4 c = inTex.read(id); half h = hitIn.read(id).x;
  float4 gn = float4(gNrm.read(id)); uint cls = uint(gn.w + .5f);
  if (cls != 2u || c.a > 0.5h) { outTex.write(half4(c.rgb, 1), id); hitOut.write(half4(h), id); return; }
  float4 gp = gPos.read(id); float3 N = gn.xyz;
  float3 sum = 0; float hs = 0, ws = 0;
  const int2 offs[4] = {int2(1, 0), int2(-1, 0), int2(0, 1), int2(0, -1)};
  for (int k = 0; k < 4; k++) {
    int2 qi = int2(id) + offs[k];
    if (qi.x < 0 || qi.y < 0 || qi.x >= int(f.w) || qi.y >= int(f.h)) continue;
    uint2 qq = uint2(qi);
    if (uint(float(gNrm.read(qq).w) + .5f) != 2u) continue;
    half4 qv = inTex.read(qq); if (qv.a < 0.5h) continue;
    float3 qp = gPos.read(qq).xyz;
    float w = exp(-abs(dot(N, qp - gp.xyz)) / (0.01f + 0.01f * gp.w));
    sum += w * float3(qv.rgb); hs += w * float(hitIn.read(qq).x); ws += w;
  }
  outTex.write(half4(half3(sum / max(ws, 1e-6f)), 1), id); hitOut.write(half4(half(hs / max(ws, 1e-6f))), id);
}
// Edge-aware a-trous pass (5x5 B3 taps) on glossy-floor receivers only; radius from the GGX lobe footprint.
// aniso: the long axis follows the screen image of the plane of incidence (through P and P + N), the short axis
// is x cos(theta_i): in-plane microfacet tilt turns the reflected ray by 2*delta, out-of-plane by 2*delta*cos(theta_i).
kernel void atrous(texture2d<half, access::read> inTex [[texture(0)]], texture2d<half, access::write> outTex [[texture(1)]],
                   texture2d<float, access::read> gPos [[texture(2)]], texture2d<half, access::read> gNrm [[texture(3)]],
                   texture2d<half, access::read> hitTex [[texture(4)]], constant Frame& f [[buffer(1)]],
                   constant uint& passIdx [[buffer(2)]], uint2 id [[thread_position_in_grid]])
{
  if (id.x >= f.w || id.y >= f.h) return;
  half4 c = inTex.read(id); float4 gn = float4(gNrm.read(id));
  if (uint(gn.w + .5f) != 2u) { outTex.write(c, id); return; }
  float4 gp = gPos.read(id); float3 N = gn.xyz; float hitT = float(hitTex.read(id).x);
  float a = f.floorRough * f.floorRough;
  float rpx = f.blurK * a * hitT / ((gp.w + hitT) * f.pixelAngle);
  float2 e = float2(1, 0), eo = float2(0, 1); float rin = clamp(rpx, 0.f, 48.f), rout = rin;
  if (f.aniso != 0u) {
    float3 V = normalize(f.camPos.xyz - gp.xyz); float cosi = saturate(dot(N, V));
    float2 p0 = projPx(f, gp.xyz), p1 = projPx(f, gp.xyz + N * 0.05f);
    float2 dv = p1 - p0; float dl = length(dv); e = dl > 1e-5f ? dv / dl : float2(0, -1); eo = float2(-e.y, e.x);
    rin = clamp(rpx, 0.f, 96.f); rout = clamp(rpx * max(cosi, 0.03f), 0.f, 96.f);
  }
  if (rin < 0.75f) { outTex.write(c, id); return; }
  float sIn, sOut;
  if (f.filterPasses == 1u) { sIn = rin / 4.f; sOut = rout / 4.f; }
  else { sIn = (passIdx == 0u ? 1.f : 3.f) * rin / 8.f; sOut = (passIdx == 0u ? 1.f : 3.f) * rout / 8.f; }
  if (f.aniso == 0u) { float s0 = max(1.f, floor(rin / 8.f + .5f)); sIn = sOut = (f.filterPasses == 1u) ? max(1.f, floor(rin / 4.f + .5f)) : (passIdx == 0u ? s0 : 3.f * s0); }
  const float h[3] = {0.375f, 0.25f, 0.0625f};
  float3 sum = 0; float wsum = 0;
  for (int dy = -2; dy <= 2; dy++) for (int dx = -2; dx <= 2; dx++) {
    float2 off = float(dx) * sIn * e + float(dy) * sOut * eo;
    int2 qi = int2(floor(float2(id) + off + 0.5f));
    if (qi.x < 0 || qi.y < 0 || qi.x >= int(f.w) || qi.y >= int(f.h)) continue;
    uint2 qq = uint2(qi);
    float4 qn = float4(gNrm.read(qq));
    if (uint(qn.w + .5f) != 2u) continue;
    float3 qp = gPos.read(qq).xyz;
    float wz = exp(-abs(dot(N, qp - gp.xyz)) / (0.01f + 0.01f * gp.w));
    float wn = pow(saturate(dot(N, qn.xyz)), 32.f);
    float w = h[abs(dx)] * h[abs(dy)] * wz * wn;
    sum += w * float3(inTex.read(qq).rgb); wsum += w;
  }
  outTex.write(half4(half3(sum / max(wsum, 1e-6f)), 1), id);
}
// Hardware-anisotropic lobe blur (aniso 2): the floor reflection is written premultiplied by a floor mask into a
// mipmapped texture (box mips), then each floor pixel takes 3 anisotropic, trilinear probes along the screen image of
// the plane of incidence, footprint 2*r_in x 2*r_out (r_out = r_in * cos(theta_i)); dividing by the mask keeps
// walls and props from bleeding in. Deterministic: no noise, no history.
kernel void premul(texture2d<half, access::read> inTex [[texture(0)]], texture2d<half, access::write> mipOut [[texture(1)]],
                   texture2d<half, access::read> gNrm [[texture(3)]], constant Frame& f [[buffer(1)]], uint2 id [[thread_position_in_grid]])
{
  if (id.x >= f.w || id.y >= f.h) return;
  uint cls = uint(float(gNrm.read(id).w) + .5f);
  half4 c = inTex.read(id);
  mipOut.write(cls == 2u ? half4(c.rgb, 1) : half4(0), id);
}
kernel void anisoGather(texture2d<half, access::sample> mipTex [[texture(0)]], texture2d<half, access::write> outTex [[texture(1)]],
                        texture2d<float, access::read> gPos [[texture(2)]], texture2d<half, access::read> gNrm [[texture(3)]],
                        texture2d<half, access::read> hitTex [[texture(4)]], texture2d<half, access::read> src [[texture(5)]],
                        constant Frame& f [[buffer(1)]], uint2 id [[thread_position_in_grid]])
{
  if (id.x >= f.w || id.y >= f.h) return;
  half4 c = src.read(id); float4 gn = float4(gNrm.read(id));
  if (uint(gn.w + .5f) != 2u) { outTex.write(c, id); return; }
  float4 gp = gPos.read(id); float3 N = gn.xyz; float hitT = float(hitTex.read(id).x);
  float a = f.floorRough * f.floorRough;
  float rpx = f.blurK * a * hitT / ((gp.w + hitT) * f.pixelAngle);
  float3 V = normalize(f.camPos.xyz - gp.xyz); float cosi = saturate(dot(N, V));
  float2 p0 = projPx(f, gp.xyz), p1 = projPx(f, gp.xyz + N * 0.05f);
  float2 dv = p1 - p0; float dl = length(dv); float2 e = dl > 1e-5f ? dv / dl : float2(0, -1); float2 eo = float2(-e.y, e.x);
  float rin = clamp(rpx, 0.f, 160.f), rout = clamp(rpx * max(cosi, 0.03f), 0.f, 160.f);
  if (rin < 0.75f) { outTex.write(c, id); return; }
  float2 sz = float2(f.w, f.h), uv = (float2(id) + 0.5f) / sz;
  float2 gx = e * rin / sz, gy = eo * (2.f * max(rout, 0.5f)) / sz;
  constexpr sampler sm(filter::linear, mip_filter::linear, max_anisotropy(16), address::clamp_to_edge);
  float4 acc = 0;
  for (int k = -1; k <= 1; k++) { float wk = k == 0 ? 0.5f : 0.25f; acc += wk * float4(mipTex.sample(sm, uv + e * (0.5f * rin * float(k)) / sz, gradient2d(gx, gy))); }
  float3 res = acc.a > 1e-3f ? acc.rgb / acc.a : float3(c.rgb);
  outTex.write(half4(half3(res), 1), id);
}
// Debug composite: compMode 0 = RT off (zone cube on every reflector), 1 = traced (where receivers)
kernel void composite(texture2d<half, access::read> base [[texture(0)]], texture2d<half, access::read> refl [[texture(1)]],
                      texture2d<float, access::read> gPos [[texture(2)]], texture2d<half, access::read> gNrm [[texture(3)]],
                      texturecube<half> env [[texture(4)]], texture2d<half, access::write> img [[texture(5)]],
                      constant Frame& f [[buffer(1)]], uint2 id [[thread_position_in_grid]])
{
  if (id.x >= f.w || id.y >= f.h) return;
  float3 b = float3(base.read(id).rgb); float4 gn = float4(gNrm.read(id)); uint cls = uint(gn.w + .5f);
  float3 col = b;
  if (cls != 0u) {
    float3 P = gPos.read(id).xyz, N = gn.xyz, V = normalize(f.camPos.xyz - P), R = reflect(-V, N);
    bool traced = f.compMode >= 1u && (cls == 1u || cls == 2u);
    float3 rr = traced ? float3(refl.read(id).rgb) : float3(0);
    if (cls == 1u || cls == 4u) col = b + paneF(dot(N, V)) * (traced ? rr : envS(f, env, R, 0));
    else { float Fs = schlick(0.04f, dot(N, V)); col = b * (1 - Fs) + (traced ? rr : Fs * envS(f, env, R, 4.0f * f.floorRough)); }
  }
  col *= f.exposure;
  col = saturate((col * (2.51f * col + 0.03f)) / (col * (2.43f * col + 0.59f) + 0.14f));   // ACES fit
  img.write(half4(half3(pow(col, 1.f / 2.2f)), 1), id);
}
)MSL";

        auto r = readHalf4(ti.refl, ti.W, ti.H); for (size_t i = 0; i < r.size(); i++) acc[i] += r[i]; if (fOut && k == 0) *fOut = f; }
      jitterMode = savedJ;
      std::vector<float> out(acc.size()); for (size_t i = 0; i < acc.size(); i++) out[i] = (float)(acc[i] / nPass); return out; };

    auto errStat = [&](const std::vector<uint8_t>& a, const std::vector<uint8_t>& b, const std::vector<float>& cls, int k) {
      std::vector<int> d; double sum = 0; size_t g2 = 0, g4 = 0, g8 = 0;
      for (size_t i = 0; i < (size_t)ti.W * ti.H; i++) { if ((int)(cls[i * 4 + 3] + .5f) != k) continue; int m = d8(a, b, i); d.push_back(m); sum += m; g2 += m >= 2; g4 += m >= 4; g8 += m >= 8; }
      EStat e{0, 0, 0, 0, 0, d.size()}; if (d.empty()) return e; std::sort(d.begin(), d.end());
      e.mean = sum / d.size(); e.p99 = d[(size_t)(0.99 * (d.size() - 1))]; e.ge2 = 100.0 * g2 / d.size(); e.ge4 = 100.0 * g4 / d.size(); e.ge8 = 100.0 * g8 / d.size(); return e; };

        }
      }

        }

      }

      printf("# --- dolly 1920x1080 Run S1 tripped, Ultra 3 rays + hw-aniso k1, cube mode %d  %s\n", cubeMode, loadStr().c_str());
      const int NF = 6;

          if (motion == 0 && getenv("DOLLYIMG")) {   // T heatmap of the frame pair 2->3, x8, for the log
            std::vector<uint8_t> img((size_t)ti.W * ti.H * 4, 0);
            for (size_t i = 0; i < (size_t)ti.W * ti.H; i++) { float e = fabs(E[3][i] - E[2][i]); uint8_t c = (uint8_t)std::min(255.f, 8.f * e); img[i * 4] = c; img[i * 4 + 1] = c; img[i * 4 + 2] = c; img[i * 4 + 3] = 255; }
            char fn[128]; snprintf(fn, sizeof fn, "dolly_%s_mode%u%s.png", tag.c_str(), jm.mode, jm.reseed ? "r" : ""); writePngG(fn, img, ti.W, ti.H); printf("wrote %s (screen-space |E3 - E2| x8, not reprojected)\n", fn); }
        }
      }
    }

    if (what == "img5") {
      // Off (flat cube, as 11) | Off (stated cube) | High | Ultra | 2048-ray reference; S1 tripped and the stream room
      struct Row { const char* key; const char* hi; const char* ul; };
      std::vector<std::vector<uint8_t>> pics; std::vector<std::string> labs;
      for (Row rw : {Row{"A20", "A_H", "A_U3"}, Row{"D20", "D_H", "D_U2"}}) {
        Cfg hi = findCfg(rw.hi), ul = findCfg(rw.ul); Cfg ref = hi; ref.spp = 64; ref.mode = 0; ref.filter = false; ref.id = "REF";
        auto R = refCached(ref, rw.key, 0, 32);
        int cmSaved = cubeMode; float esSaved = envScaleOverride; bool umSaved = urpMip;
        cubeMode = 0; envScaleOverride = -1.f; urpMip = false; auto offFlat = comp8(hi, 1, 0, nullptr);
        cubeMode = cmSaved; envScaleOverride = esSaved; urpMip = umSaved;
        auto off = comp8(hi, 1, 0, nullptr), H8 = comp8(hi, 1, 1, nullptr);
        uint32_t sj = jitterMode; jitterMode = 2; auto U8 = comp8(ul, 1, 1, nullptr); jitterMode = sj; auto R8 = comp8(ref, 1, 2, &R);
        const char* room = rw.key[0] == 'A' ? "S1 tripped" : "stream room";
        pics.push_back(offFlat); labs.push_back(std::string(room) + ": OFF, flat cube (as 11)");
        pics.push_back(off); labs.push_back(std::string(room) + ": OFF, main's Level 0 cube x0.5");
        pics.push_back(H8); labs.push_back(std::string(room) + ": HIGH");
        pics.push_back(U8); labs.push_back(std::string(room) + (rw.key[0] == 'A' ? ": ULTRA 3 rays" : ": ULTRA 2 rays"));
        pics.push_back(R8); labs.push_back(std::string(room) + ": REFERENCE 2048 rays");
        for (int k = 0; k < 5; k++) { char fn[96]; snprintf(fn, sizeof fn, "img5_%s_%s_%d.png", tag.c_str(), rw.key, k); writePngG(fn, pics[pics.size() - 5 + k], ti.W, ti.H); }
      }
      uint32_t cw = ti.W / 3, ch = ti.H / 3, SW = cw * 5, SH = ch * 2;
      std::vector<uint8_t> sheet((size_t)SW * SH * 4, 0);
      for (size_t k = 0; k < pics.size(); k++) { uint32_t ox = (k % 5) * cw, oy = (k / 5) * ch;
        for (uint32_t y = 0; y < ch; y++) for (uint32_t x = 0; x < cw; x++) for (int c = 0; c < 4; c++) { int acc = 0;
          for (int dy = 0; dy < 3; dy++) for (int dx = 0; dx < 3; dx++) acc += pics[k][((size_t)(3 * y + dy) * ti.W + 3 * x + dx) * 4 + c];
          sheet[((size_t)(oy + y) * SW + ox + x) * 4 + c] = (uint8_t)(acc / 9); } }
      CGColorSpaceRef cs = CGColorSpaceCreateWithName(kCGColorSpaceSRGB);
      CGContextRef ctx = CGBitmapContextCreate(sheet.data(), SW, SH, 8, SW * 4, cs, kCGImageAlphaPremultipliedLast);
      CTFontRef font = CTFontCreateWithName(CFSTR("Helvetica-Bold"), 17, NULL);
      for (size_t k = 0; k < labs.size(); k++) { uint32_t ox = (k % 5) * cw, oy = (k / 5) * ch;
        CGContextSetRGBFillColor(ctx, 0, 0, 0, 0.65); CGContextFillRect(ctx, CGRectMake(ox, SH - oy - 28, cw, 28));
        CFStringRef str = CFStringCreateWithCString(NULL, labs[k].c_str(), kCFStringEncodingUTF8);
        CFStringRef keys[] = {kCTFontAttributeName, kCTForegroundColorFromContextAttributeName}; CFTypeRef vals[] = {font, kCFBooleanTrue};
        CFDictionaryRef attr = CFDictionaryCreate(NULL, (const void**)keys, (const void**)vals, 2, &kCFTypeDictionaryKeyCallBacks, &kCFTypeDictionaryValueCallBacks);
        CFAttributedStringRef as = CFAttributedStringCreate(NULL, str, attr); CTLineRef line = CTLineCreateWithAttributedString(as);
        CGContextSetRGBFillColor(ctx, 1, 1, 1, 1); CGContextSetTextPosition(ctx, ox + 8, SH - oy - 20); CTLineDraw(line, ctx);
        CFRelease(line); CFRelease(as); CFRelease(attr); CFRelease(str); }
      CGImageRef im = CGBitmapContextCreateImage(ctx); std::vector<uint8_t> out((size_t)SW * SH * 4);
      CGContextRef c2 = CGBitmapContextCreate(out.data(), SW, SH, 8, SW * 4, cs, kCGImageAlphaPremultipliedLast); CGContextDrawImage(c2, CGRectMake(0, 0, SW, SH), im);
      std::string path = "img5_" + tag + "_sheet.png"; writePngG(path, out, SW, SH); printf("wrote %s\n", path.c_str());
      CGContextRelease(c2); CGImageRelease(im); CFRelease(font); CGContextRelease(ctx); CGColorSpaceRelease(cs);
    }

    if (what == "dump") {
      // Per case: RT off, references (two independent 1024-ray halves of the High tree and of the Ultra tree),
      // the 128/256/512/1024-ray blocks, High k1/k2/k3, Ultra 2/3/4 rays (k1, footprint jitter, seeds 1 and 8), masks.
      const char* only = argc > 3 ? argv[3] : nullptr;
      struct DC { const char* key; const char* hi; const char* ul; };
      for (DC dc : {DC{"A20", "A_H", "A_U3"}, DC{"B20", "B_H", "B_U2"}, DC{"D20", "D_H", "D_U2"}, DC{"F20", "F_H", "F_U3"}, DC{"G15", "G_H", "G_U2"}}) {
        if (only && strcmp(only, dc.key)) continue;
        Cfg hi = findCfg(dc.hi), ul = findCfg(dc.ul);
        Cfg ref = hi; ref.spp = 64; ref.mode = 0; ref.filter = false; ref.id = "REF";
        Cfg refU = ul; refU.spp = 64; refU.mode = 0; refU.filter = false; refU.bounces = 3; refU.layers = 2; refU.s1 = 8; refU.id = "REFU";
        char fn[160]; auto out = [&](const char* nm, std::vector<uint8_t> px) { snprintf(fn, sizeof fn, "d5_%s_c%d_sg%d_%s_%s.png", tag.c_str(), cubeMode, getenv("SOFTGATE") ? 1 : 0, dc.key, nm); writePngG(fn, px, ti.W, ti.H); };
        auto cls = clsOf(ref);
        std::vector<uint8_t> mk((size_t)ti.W * ti.H * 4, 0);
        for (size_t i = 0; i < (size_t)ti.W * ti.H; i++) { int k = (int)(cls[i * 4 + 3] + .5f); mk[i * 4] = k == 2 ? 255 : 0; mk[i * 4 + 1] = k == 1 ? 255 : 0; mk[i * 4 + 3] = 255; }
        out("mask", mk);
        out("off", comp8(hi, 1, 0, nullptr));
        auto Ra = refCached(ref, dc.key, 0, 16), Rb = refCached(ref, dc.key, 16, 16);
        std::vector<float> R(Ra.size()); for (size_t i = 0; i < R.size(); i++) R[i] = .5f * (Ra[i] + Rb[i]);
        out("ref", comp8(ref, 1, 2, &R)); out("refa", comp8(ref, 1, 2, &Ra)); out("refb", comp8(ref, 1, 2, &Rb));
        struct BK { const char* nm; int first, cnt; } bks[4] = {{"x128", 32, 2}, {"x256", 34, 4}, {"x512", 38, 8}, {"x1024", 46, 16}};
        for (auto& bk : bks) { auto X = refCached(ref, dc.key, bk.first, bk.cnt); out(bk.nm, comp8(ref, 1, 2, &X)); }
        for (float k : {1.f, 2.f, 3.f}) { Cfg c = hi; c.blurK = k; char nm[16]; snprintf(nm, sizeof nm, "high_k%d", (int)k); out(nm, comp8(c, 1, 1, nullptr)); }
        {
          auto RUa = refCached(refU, dc.key, 0, 16), RUb = refCached(refU, dc.key, 16, 16);
          std::vector<float> RU(RUa.size()); for (size_t i = 0; i < RU.size(); i++) RU[i] = .5f * (RUa[i] + RUb[i]);
          out("refu", comp8(refU, 1, 2, &RU)); out("refua", comp8(refU, 1, 2, &RUa)); out("refub", comp8(refU, 1, 2, &RUb));
        }
        uint32_t sj = jitterMode; jitterMode = 2;
        for (uint32_t n : {2u, 3u, 4u}) { Cfg c = ul; c.spp = n; c.mode = 0; c.aniso = 2; c.blurK = 1.f;
          if (!strcmp(dc.ul, dc.hi)) { c.bounces = 3; c.layers = 2; c.s1 = 8; c.cutoff = .004f; }
          char nm[16]; snprintf(nm, sizeof nm, "u%u", n); out(nm, comp8(c, 1, 1, nullptr));
          snprintf(nm, sizeof nm, "u%u_seed8", n); out(nm, comp8(c, 8, 1, nullptr)); }
        jitterMode = sj;
        printf("%s dump %s done | %s\n", tag.c_str(), dc.key, loadStr().c_str()); fflush(stdout);
      }
    }
    if (what == "stackab") {
      const char* ids[] = {"A_h1", "B_h1", "E_h1", "G_h1", "A_h2", "G_h2"};
      for (auto& rs : resList) {
        if (rs[0] != 2560) continue;
        Targets t = mkTargets(rs[0], rs[1]);
        for (const char* cid : ids) {
          Cfg c = findCfg(cid); FrameH f = frameFor(t, c, 1, 0); runG(t, c, f);
          std::vector<double> ts, tn;
          for (int rep = 0; rep < 31; rep++) { useNS = false; double a = runT(t, c, f); useNS = true; double b = runT(t, c, f); useNS = false; if (rep >= 3) { ts.push_back(a); tn.push_back(b); } }
          printf("%s stackab %ux%u %-6s depth %u: stack min %.3f med %.3f | stackless min %.3f med %.3f | ratio(min) %.3f ratio(med) %.3f | %s\n", tag.c_str(), rs[0], rs[1], cid, c.bounces,
                 pct(ts, 0), pct(ts, .5), pct(tn, 0), pct(tn, .5), pct(ts, 0) / pct(tn, 0), pct(ts, .5) / pct(tn, .5), loadStr().c_str());
          fflush(stdout);
        }
      }
      // sanity: stackless == stack at depth 1 (8-bit, floor+glass px)
      for (const char* cid : {"A_h1", "G_h1"}) {
        useNS = false; auto a = render(cid, 1, nullptr); useNS = true; std::vector<float> cls; auto b = render(cid, 1, &cls); useNS = false;
        int mx = 0; size_t n = 0; for (size_t i = 0; i < (size_t)ti.W * ti.H; i++) { int k = (int)(cls[i * 4 + 3] + .5f); if (k != 1 && k != 2) continue; mx = std::max(mx, d8(a, b, i)); n++; }
        printf("%s stackab sanity %s: stack vs stackless depth 1, max |d| %d/255 over %zu receiver px\n", tag.c_str(), cid, mx, n);
      }
    }
    if (what == "diff2") {
      printf("# --- diff2 1920x1080  %s\n", loadStr().c_str());
      struct DP { const char* a; uint32_t ma; const char* b; uint32_t mb; const char* label; };
      std::vector<DP> pairs = {
        {"A_h2", 0, "A_h2", 1, "S1 tripped: RT off vs floor receiver"}, {"B_h2", 0, "B_h2", 1, "near goal tripped: RT off vs floor receiver"},
        {"D_h2", 0, "D_h2", 1, "stream room: RT off vs floor receiver"}, {"F_h2", 0, "F_h2", 1, "S1 armed: RT off vs floor receiver"},
        {"E_h1", 1, "E_h2", 1, "near goal armed r0.20: 1 vs 2 bounces"}, {"G_h2", 0, "G_h2", 1, "2.5 m armed r0.15: RT off vs floor receiver"},
        {"G_h1", 1, "G_h2", 1, "2.5 m armed r0.15: 1 vs 2 bounces"}, {"G_h2", 1, "G_h3c", 1, "2.5 m armed r0.15: 2 vs 3 (cut-off)"}};
      for (auto& dp : pairs) {
        std::vector<float> cls; auto a = render(dp.a, dp.ma, nullptr); auto b = render(dp.b, dp.mb, &cls);
        for (int k = 1; k <= 2; k++) {
          std::vector<int> d; double sum = 0;
          for (size_t i = 0; i < (size_t)ti.W * ti.H; i++) { if ((int)(cls[i * 4 + 3] + .5f) != k) continue; int m = d8(a, b, i); d.push_back(m); sum += m; }
          if (d.empty()) continue;
          size_t n2 = 0, n8 = 0, n24 = 0; for (int v : d) { n2 += v >= 2; n8 += v >= 8; n24 += v >= 24; }
          std::sort(d.begin(), d.end());
          printf("diff2 %-5s(%u) vs %-5s(%u) %s px %6.2f%%: mean %.2f/255, p99 %d, max %d; >=8: %.2f%% (%zu px) >=24: %.2f%% | %s | %s\n",
                 dp.a, dp.ma, dp.b, dp.mb, k == 1 ? "glass" : "floor", 100.0 * d.size() / ((double)ti.W * ti.H), sum / d.size(), d[(size_t)(0.99 * (d.size() - 1))], d.back(),
                 100.0 * n8 / d.size(), n8, 100.0 * n24 / d.size(), dp.label, loadStr().c_str());
        }
        fflush(stdout);
      }
    }
    if (what == "img2") {
      for (const char* cid : {"A_h2", "D_h2"}) for (uint32_t m : {0u, 1u}) {
        auto px = render(cid, m, nullptr); std::string path = std::string("img2_") + tag + "_" + cid + (m ? "_rt" : "_off") + ".png"; writePng(path, px, ti.W, ti.H); printf("wrote %s\n", path.c_str()); }
    }
    printf("# done %s\n", loadStr().c_str());
  }
  return 0;
}
