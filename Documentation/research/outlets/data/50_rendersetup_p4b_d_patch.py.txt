"""P-4b + P-6 for the outlet palette, v3 (2026-10-21): rebased onto main's RenderSetup after exit_sign_fix1a/1b,
with Red's ivory D (research/outlets/40_final.md top line, 50_p4b_ivory_d.md).

v2 (2026-10-14, data/40_rendersetup_patch.py.txt) -> v3:
  - ivory A -> D: plate #A69F8C, device #A39C89 (Red, 2026-10-21, "all as recommended"; 0.95x the paper at 0.30 m);
  - Aged and Almond keep v2's per-channel ratios to the plate (Aged/A = .891 .864 .784, Almond/A = .987 .973 .979),
    applied to D in 8-bit sRGB: Aged #94896E, Almond #A49B89. PlasticOrange (the IG device) is unchanged (#D8642A);
  - EnsureSurfaces takes an optional name filter (null = every slot, exactly as before), so one slot can be
    regenerated without the full SetUp, which also rewrites the post profiles and the glass materials.
Everything else as v2: seven new SurfaceDefs (5 tint-only plastics + 2 outlet metals), instancing on 10 existing slots,
and the writer sets Material.enableInstancing from the def.

Text-anchored; writes nothing unless every anchor matches exactly once.
  python3 rendersetup_p4b_d_patch.py <project>/Assets/Editor/Rendering/FrontRoomsRenderSetup.cs
"""
import io, re, sys
p = sys.argv[1]; s = io.open(p, encoding="utf-8").read()
assert "Prop_ThermosetIvory" not in s, "already patched"
def sub(a, b):
    global s
    assert s.count(a) == 1, (s.count(a), a[:100]); s = s.replace(a, b)
def flag(name):
    global s
    pat = r'(        new SurfaceDef \{ name = "' + name + r'",[^\n]*?)( \},\n)'
    assert len(re.findall(pat, s)) == 1, name
    s = re.sub(pat, lambda m: m.group(1) + ", instancing = true" + m.group(2), s)
sub("        public Vector4 st = new Vector4(1, 1, 0, 0); // texture scale/offset after the metre tile\n",
    "        public Vector4 st = new Vector4(1, 1, 0, 0); // texture scale/offset after the metre tile\n"
    "        public bool instancing;      // GPU instancing on (the outlet palette: FrontRoomsWallFixtureRenderer draws it with RenderMeshInstanced)\n")
# P-6: the existing slots the outlet kits and their fallbacks use.
for n in ["Prop_LaminateBeige", "Prop_PlasticBeige", "Prop_PlasticBlack", "Prop_Aluminium", "Prop_Chrome", "Prop_Brass",
          "Prop_Ceramic", "Prop_PlasticPutty", "Prop_CeramicGlaze", "Prop_AluminiumAnodised"]:
    flag(n)
# P-4b: five new tint-only slots and two outlet metals, after Prop_Brass.
m = re.search(r'        new SurfaceDef \{ name = "Prop_Brass",[^\n]*\n', s)
assert m
s = s[:m.end()] + (
"        // Outlet palette (research/outlets/10_spec.md §1.5, P-4b): tint-only thermoset and nylon slots,\n"
"        // instanced (P-6). The plate colour is chosen per fixture by the renderer, so no extra FBX per colour.\n"
"        // The world-space macro wear gives every plate its own grime. Ivory = look-dev L-1 option D (Red, 2026-10-21):\n"
"        // the plate reads 0.95x the Level 0 paper beside it at 0.30 m (research/outlets/40_final.md §3.4). Aged and\n"
"        // Almond keep the first proposal's per-channel ratios to the ivory plate (.891 .864 .784 and .987 .973 .979).\n"
"        new SurfaceDef { name = \"Prop_ThermosetIvory\", tint = new Color(.651f, .624f, .549f), smooth = .75f, meshUV = true, macroTone = .05f, macroDirt = .08f, instancing = true },   // #A69F8C\n"
"        new SurfaceDef { name = \"Prop_ThermosetAged\", tint = new Color(.580f, .537f, .431f), smooth = .60f, meshUV = true, macroTone = .08f, macroDirt = .14f, instancing = true },   // #94896E\n"
"        new SurfaceDef { name = \"Prop_ThermosetAlmond\", tint = new Color(.643f, .608f, .537f), smooth = .70f, meshUV = true, macroTone = .05f, macroDirt = .08f, instancing = true },  // #A49B89\n"
"        new SurfaceDef { name = \"Prop_NylonIvory\", tint = new Color(.639f, .612f, .537f), smooth = .55f, meshUV = true, macroTone = .03f, macroDirt = .06f, instancing = true },       // #A39C89\n"
"        new SurfaceDef { name = \"Prop_PlasticOrange\", tint = new Color(.847f, .392f, .165f), smooth = .55f, meshUV = true, macroTone = .03f, macroDirt = .06f, instancing = true },    // #D8642A\n"
"        // Outlet metals (fix pass 2026-10-14, art critic issue 2): Prop_Aluminium (metallic 1, albedo .72, occlusion .8) takes\n"
"        // all its light from the dim zone reflection and reads gunmetal to black. A part-dielectric response keeps the lamps'\n"
"        // diffuse light: brushed stainless for wall plates, galvanized steel for handy boxes, EMT, straps and floor boxes.\n"
"        new SurfaceDef { name = \"Prop_StainlessBrushed\", tint = new Color(.78f, .78f, .76f), smooth = .55f, metallic = .6f, occlusion = 1f, meshUV = true, macroTone = 0f, macroDirt = .04f, instancing = true },   // #C7C7C2\n"
"        new SurfaceDef { name = \"Prop_Galvanized\", tint = new Color(.70f, .71f, .70f), smooth = .35f, metallic = .5f, occlusion = 1f, meshUV = true, macroTone = .05f, macroDirt = .10f, instancing = true },        // #B3B5B3\n"
) + s[m.end():]
# Optional name filter: SetUp() still calls EnsureSurfaces() and writes every slot.
sub("    static void EnsureSurfaces()\n    {\n",
    "    /// <summary>Writes the surface materials; <paramref name=\"only\"/> limits it to those slot names (null = all).</summary>\n"
    "    static void EnsureSurfaces(ICollection<string> only = null)\n    {\n")
sub("        foreach (var d in defs)\n        {\n            var path = SurfaceDir + \"/\" + d.name + \".mat\";\n",
    "        foreach (var d in defs)\n        {\n            if (only != null && !only.Contains(d.name)) continue;\n            var path = SurfaceDir + \"/\" + d.name + \".mat\";\n")
sub("            if (d.meshUV) m.EnableKeyword(\"_FR_MESH_UV\"); else m.DisableKeyword(\"_FR_MESH_UV\");\n",
    "            if (d.meshUV) m.EnableKeyword(\"_FR_MESH_UV\"); else m.DisableKeyword(\"_FR_MESH_UV\");\n"
    "            // Instanced draws need the flag (the shader has multi_compile_instancing in all four passes);\n"
    "            // renderers keep using the SRP Batcher either way.\n"
    "            m.enableInstancing = d.instancing;\n")
io.open(p, "w", encoding="utf-8").write(s); print("patched", p)
