"""O2 review renders (scratchpad tool, not a kit module).

Blender -b --factory-startup --python o2_review.py -- <module> <outdir> <proj_root> [--samples N] [--shots a,b]

* Rebuilds the module with outlet_plate_common.HAS_LODS forced True, so the
  hand-built LOD1/LOD2 part sets exist; counts triangles per level.
* Imports the EXPORTED FBX (<proj>/Assets/Resources/Props/Models/<NAME>.fbx)
  and renders LOD0 from it (what ships), LOD1/LOD2 from the forced build.
* Screws: O1's Kit_OutletScrew FBX at each screw_k anchor if it exists in the
  clone, else a preview-only #6-32 oval-head stand-in (spec §1.2 numbers),
  slot angle random per plate, like R3 at LOD0.
* Shots: front030 (0.30 m, FOV 62 V, 1080p crop), rake030 (45 deg raking key),
  obl030 (0.30 m, 35 deg off axis), d150 (LOD0 | LOD1), d400 (LOD1 | LOD2)
  at FOV 76 V, cropped to the plate at true 1080p pixel size.
"""

        FORCE_CPU = True; i += 1; continue

def s2l(c):
    return c / 12.92 if c <= 0.04045 else ((c + 0.055) / 1.055) ** 2.4


# Review only: the slot preview colour is the sRGB hex; render it as sRGB.
for slot_name, rgb in ((pc.TI, pc._hex("#E6DDC2")), (pc.NI, pc._hex("#E2D9BF")), (pc.CE, (0.353, 0.227, 0.141))):
    mat = bpy.data.materials.get(slot_name)
    if mat is not None and mat.node_tree:
        mat.node_tree.nodes["Principled BSDF"].inputs["Base Color"].default_value = tuple(s2l(x) for x in rgb) + (1.0,)

if os.path.exists(screw_fbx):

    new = [o for o in bpy.data.objects if o not in b2 and o.type == "MESH"]
    screw_src = new[0] if new else None
    screw_kind = "O1 Kit_OutletScrew"
if screw_src is None:
    screw_kind = "preview stand-in (spec 1.2 numbers)"
    bm = bmesh.new()
    R, C = pc.SCREW_HEAD_D / 2 * pc.MM, pc.SCREW_CROWN * pc.MM
    Rs = (R * R + C * C) / (2 * C)
    segs = 64
    rings = []
    for k in range(9):
        r = R * (1 - k / 8.0)
        h = math.sqrt(max(Rs * Rs - r * r, 0)) - (Rs - C)
        rings.append([bm.verts.new((r * math.cos(2 * math.pi * j / segs), -h, r * math.sin(2 * math.pi * j / segs))) for j in range(segs)] if r > 1e-9 else [bm.verts.new((0, -C, 0))])
    # underside cone (82 deg) to the shank
    cone_r = 1.75 * pc.MM
    cone_h = (R - cone_r) / math.tan(math.radians(41))
    under = [bm.verts.new((cone_r * math.cos(2 * math.pi * j / segs), cone_h, cone_r * math.sin(2 * math.pi * j / segs))) for j in range(segs)]
    for a, b in zip(rings, rings[1:]):
        if len(b) == 1:
            for j in range(segs):
                bm.faces.new((a[j], a[(j + 1) % segs], b[0]))
        else:
            for j in range(segs):
                bm.faces.new((a[j], a[(j + 1) % segs], b[(j + 1) % segs], b[j]))
    for j in range(segs):
        bm.faces.new((under[j], under[(j + 1) % segs], rings[0][(j + 1) % segs], rings[0][j]))
    bm.faces.new(list(reversed(under)))
    bmesh.ops.recalc_face_normals(bm, faces=bm.faces)
    me = bpy.data.meshes.new("screw")
    bm.to_mesh(me)
    bm.free()
    screw_src = bpy.data.objects.new("screw", me)
    bpy.context.scene.collection.objects.link(screw_src)
    cut = bpy.data.objects.new("slotcut", bpy.data.meshes.new("slotcut"))
    bmc = bmesh.new()

    bmesh.ops.scale(bmc, vec=Vector((8 * pc.MM, 2 * pc.MM, pc.SCREW_SLOT[0] * pc.MM)), verts=bmc.verts)
    bmesh.ops.translate(bmc, vec=Vector((0, -(C - pc.SCREW_SLOT[1] * pc.MM) - 1 * pc.MM, 0)), verts=bmc.verts)
    bmc.to_mesh(cut.data)
    bmc.free()
    bpy.context.scene.collection.objects.link(cut)
    bo = screw_src.modifiers.new("slot", "BOOLEAN")
    bo.operation = "DIFFERENCE"
    bo.solver = "EXACT"
    bo.object = cut

    bpy.ops.object.modifier_apply(modifier="slot")
    bpy.data.objects.remove(cut)
    screw_src.data.materials.append(bpy.data.materials.get(pc.TI) or kit._material(pc.TI))
    for p in screw_src.data.polygons:
        p.use_smooth = True

# Lights
def area(name, size, energy, loc, target, shape="RECTANGLE", size_y=None):
    L = bpy.data.lights.new(name, "AREA")
    L.energy = energy
    L.shape = shape
    L.size = size
    if size_y:
        L.size_y = size_y
    o = bpy.data.objects.new(name, L)
    o.location = loc
    d = Vector(target) - Vector(loc)
    o.rotation_euler = d.to_track_quat("-Z", "Y").to_euler()
    scene.collection.objects.link(o)
    return o

def set_visible(level):
    for o in shipped:
        o.hide_render = level != 0
    for s in screws:
        s.hide_render = level != 0
    for lv, objs in lod_sets.items():
        for o in objs:
            o.hide_render = True
    if level in (1, 2):
        for o in lod_sets[level]:
            o.hide_render = False


def look(pos, fov_v, target=(0, 0, 0)):
    cam.location = pos
    cam.rotation_euler = (Vector(target) - Vector(pos)).to_track_quat("-Z", "Y").to_euler()
    cam_data.angle_y = math.radians(fov_v)


def lights(mode):
    troffer.hide_render = mode == "rake"
    fill.hide_render = mode == "rake"
    rake.hide_render = mode != "rake"
    world.node_tree.nodes["Background"].inputs[1].default_value = 0.08 if mode == "rake" else 0.35


def crop_for(dist, fov_v, half_w_m, half_h_m):
    """Render border round the plate (asset centred in frame)."""
    vis_h = 2 * dist * math.tan(math.radians(fov_v) / 2)
    vis_w = vis_h * 16 / 9
    fx = min(0.5, half_w_m / vis_w)
    fy = min(0.5, half_h_m / vis_h)
    scene.render.use_border = True
    scene.render.use_crop_to_border = True
    scene.render.border_min_x, scene.render.border_max_x = 0.5 - fx, 0.5 + fx
    scene.render.border_min_y, scene.render.border_max_y = 0.5 - fy, 0.5 + fy


def shot(name, level, pos, fov_v, mode, half_w, half_h, dist, target=(0, 0, 0), tilt=0.0):
    set_visible(level)
    lights(mode)
    look(pos, fov_v, target)
    pivot.rotation_euler = (math.radians(tilt), 0, 0)
    crop_for(dist, fov_v, half_w, half_h)
    scene.render.filepath = os.path.join(OUT, "%s_%s.png" % (mod.NAME, name))
    bpy.ops.render.render(write_still=True)
    return scene.render.filepath
