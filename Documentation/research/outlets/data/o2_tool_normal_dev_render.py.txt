"""Render the shading-normal deviation of exported plate FBX files, front view.
Colour = |N_xz| (sin of tilt from the wall normal) x GAIN, so 1 deg reads 0.17 at GAIN 10.
Plus a Workbench matcap zebra (reflection_check_horizontal) render at 0.30 m.
-- <models_dir> <out_dir> <name> [<name> ...]"""
import bpy, sys, os, math
from mathutils import Vector
a = sys.argv[sys.argv.index("--") + 1:]
M, OUT, names = a[0], a[1], a[2:]
os.makedirs(OUT, exist_ok=True)
GAIN = 10.0
for n in names:
    bpy.ops.wm.read_factory_settings(use_empty=True)
    bpy.ops.import_scene.fbx(filepath=os.path.join(M, n + ".fbx"), axis_forward="-Z", axis_up="Y", bake_space_transform=True)
    objs = [o for o in bpy.data.objects if o.type == "MESH"]
    mat = bpy.data.materials.new("ndev"); mat.use_nodes = True
    nt = mat.node_tree; nt.nodes.clear()
    geo = nt.nodes.new("ShaderNodeNewGeometry")
    sep = nt.nodes.new("ShaderNodeSeparateXYZ"); nt.links.new(geo.outputs["Normal"], sep.inputs[0])
    comb = nt.nodes.new("ShaderNodeCombineXYZ")
    nt.links.new(sep.outputs["X"], comb.inputs["X"]); nt.links.new(sep.outputs["Z"], comb.inputs["Y"])
    ln = nt.nodes.new("ShaderNodeVectorMath"); ln.operation = "LENGTH"; nt.links.new(comb.outputs[0], ln.inputs[0])
    mul = nt.nodes.new("ShaderNodeMath"); mul.operation = "MULTIPLY"; mul.inputs[1].default_value = GAIN; mul.use_clamp = True
    nt.links.new(ln.outputs["Value"], mul.inputs[0])
    ramp = nt.nodes.new("ShaderNodeValToRGB")
    ramp.color_ramp.elements[0].color = (0.02, 0.02, 0.05, 1); ramp.color_ramp.elements[1].color = (1, 0.85, 0.1, 1)
    nt.links.new(mul.outputs[0], ramp.inputs[0])
    em = nt.nodes.new("ShaderNodeEmission"); nt.links.new(ramp.outputs[0], em.inputs[0])
    out = nt.nodes.new("ShaderNodeOutputMaterial"); nt.links.new(em.outputs[0], out.inputs[0])
    for o in objs:
        o.data.materials.clear(); o.data.materials.append(mat)
        for p in o.data.polygons: p.material_index = 0
    xs = [ (o.matrix_world @ v.co) for o in objs for v in o.data.vertices]
    W = max(v.x for v in xs) - min(v.x for v in xs); H = max(v.z for v in xs) - min(v.z for v in xs)
    scene = bpy.context.scene
    cam = bpy.data.objects.new("cam", bpy.data.cameras.new("cam")); scene.collection.objects.link(cam); scene.camera = cam
    cam.data.type = "ORTHO"; cam.data.ortho_scale = max(W, H) * 1.06
    cam.location = (0, -0.3, 0); cam.rotation_euler = (math.radians(90), 0, 0)
    scene.render.engine = "CYCLES"; scene.cycles.samples = 16; scene.cycles.use_denoising = False
    scene.cycles.device = "CPU"
    scene.render.resolution_x = round(1100 * W / max(W, H)); scene.render.resolution_y = round(1100 * H / max(W, H))
    scene.view_settings.view_transform = "Standard"
    scene.world = bpy.data.worlds.new("w"); scene.world.color = (0, 0, 0)
    scene.render.filepath = os.path.join(OUT, n + "_ndev.png"); bpy.ops.render.render(write_still=True)
    # zebra matcap, perspective 0.30 m, 25 deg off axis
    scene.render.engine = "BLENDER_WORKBENCH"
    sh = scene.display.shading; sh.light = "MATCAP"; sh.studio_light = "reflection_check_horizontal.exr"; sh.color_type = "SINGLE"
    scene.display.render_aa = "16"
    cam.data.type = "PERSP"; cam.data.lens = 50
    d = 0.30; az = math.radians(25)
    cam.location = (d * math.sin(az), -d * math.cos(az), 0.0)
    cam.rotation_euler = (Vector((0, 0, 0)) - cam.location).to_track_quat("-Z", "Y").to_euler()
    scene.render.resolution_x = 1200; scene.render.resolution_y = 1200
    scene.render.filepath = os.path.join(OUT, n + "_zebra.png"); bpy.ops.render.render(write_still=True)
    print("[nd]", n, "done")
