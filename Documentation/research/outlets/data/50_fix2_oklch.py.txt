import math, sys
def s2l(c):
    c/=255.0
    return c/12.92 if c<=0.04045 else ((c+0.055)/1.055)**2.4
def l2s(c):
    c = 12.92*c if c<=0.0031308 else 1.055*c**(1/2.4)-0.055
    return c
def hex2rgb(h):
    h=h.lstrip('#'); return [int(h[i:i+2],16) for i in (0,2,4)]
def to_oklab(h):
    r,g,b=[s2l(x) for x in hex2rgb(h)]
    l=0.4122214708*r+0.5363325363*g+0.0514459929*b
    m=0.2119034982*r+0.6806995451*g+0.1073969566*b
    s=0.0883024619*r+0.2817188376*g+0.6299787005*b
    l,m,s=[x**(1/3) for x in (l,m,s)]
    L=0.2104542553*l+0.7936177850*m-0.0040720468*s
    a=1.9779984951*l-2.4285922050*m+0.4505937099*s
    bb=0.0259040371*l+0.7827717662*m-0.8086757660*s
    return L,a,bb
def to_lch(h):
    L,a,b=to_oklab(h); return L, math.hypot(a,b), math.degrees(math.atan2(b,a))%360
def from_lch(L,C,H):
    a=C*math.cos(math.radians(H)); b=C*math.sin(math.radians(H))
    l=L+0.3963377774*a+0.2158037573*b
    m=L-0.1055613458*a-0.0638541728*b
    s=L-0.0894841775*a-1.2914855480*b
    l,m,s=l**3,m**3,s**3
    r=4.0767416621*l-3.3077115913*m+0.2309699292*s
    g=-1.2684380046*l+2.6097574011*m-0.3413193965*s
    bb=-0.0041960863*l-0.7034186147*m+1.7076147010*s
    out=[max(0,min(255,round(l2s(max(0,x))*255))) for x in (r,g,bb)]
    return '#%02X%02X%02X'%tuple(out)
def lum(h):
    r,g,b=[s2l(x) for x in hex2rgb(h)]; return 0.2126*r+0.7152*g+0.0722*b
A='#E6DDC2'; Aaged='#CDBF98'; D='#A69F8C'
for h in [A,Aaged,D,'#A39C89','#94896E','#9A866F','#9E8C70','#9D8471','#9C8570','#A49B89','#E3D7BE']:
    L,C,H=to_lch(h); print(h,'L %.4f C %.4f H %.1f lum %.4f'%(L,C,H,lum(h)))
LA,CA,HA=to_lch(A); LG,CG,HG=to_lch(Aaged)
print('A->aged: L x%.4f C x%.4f dH %.2f'%(LG/LA,CG/CA,HG-HA))
LD,CD,HD=to_lch(D)
for H in [HD+(HG-HA), HD, 78,72,70,65,62,60]:
    hx=from_lch(LD*LG/LA, CD*CG/CA, H)
    print('aged from D at hue %.1f -> %s  aged/ivory lum %.3f'%(H,hx,lum(hx)/lum(D)))
print('lum ratio v2 aged/A %.3f'%(lum(Aaged)/lum(A)))
print('---- almond')
Al='#E3D7BE'; LL,CL,HL=to_lch(Al)
rl, rc, dh = LL/LA, CL/CA, HL-HA
print('A->almond: L x%.4f C x%.4f dH %.2f'%(rl,rc,dh))
hx=from_lch(LD*rl, CD*rc, HD+dh); print('almond from D (OKLCH rule) ->', hx, 'vs built #A49B89', to_lch(hx), 'lum ratio to D %.3f'%(lum(hx)/lum(D)), 'built %.3f'%(lum('#A49B89')/lum(D)))
# nylon device from D: v2 device/plate
Nv='#E2D9BF'; LN,CN,HN=to_lch(Nv); print('v2 nylon/plate: L x%.4f C x%.4f dH %.2f'%(LN/LA,CN/CA,HN-HA))
