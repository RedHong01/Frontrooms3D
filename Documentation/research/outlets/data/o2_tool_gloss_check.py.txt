"""Unity-like glossy check of exported plate FBX files: ivory plastic at
roughness 0.22, a 0.6 m square troffer 1.2 m above and 0.4 m out, camera
0.55 m from the plate, 22 deg above and 18 deg to the side.
-- <models_dir> <out_dir> <tag> <name> [<name> ...]"""
import bpy, sys, os, math
from mathutils import Vector
a = sys.argv[sys.argv.index("--") + 1:]
M, OUT, TAG, names = a[0], a[1], a[2], a[3:]
os.makedirs(OUT, exist_ok=True)
def s2l(c): return c / 12.92 if c <= 0.04045 else ((c + 0.055) / 1.055) ** 2.4
for n in names:
    bpy.ops.wm.read_factory_settings(use_empty=True)
    bpy.ops.import_scene.fbx(filepath=os.path.join(M, n + ".fbx"), axis_forward="-Z", axis_up="Y", bake_space_transform=True)
    objs = [o for o in bpy.data.objects if o.type == "MESH"]
    for o in objs:
        for m in o.data.materials:
            if m is None: continue
            m.use_nodes = True
            b = m.node_tree.nodes.get("Principled BSDF")
            nm = m.name.split(".")[0]
            if nm in ("Prop_ThermosetIvory", "Prop_NylonIvory"):
                b.inputs["Base Color"].default_value = (s2l(0.902), s2l(0.867), s2l(0.761), 1)
                b.inputs["Roughness"].default_value = 0.22
            elif nm == "Prop_PlasticBlack":
                b.inputs["Base Color"].default_value = (0.02, 0.02, 0.02, 1); b.inputs["Roughness"].default_value = 0.4
    # wall
    bpy.ops.mesh.primitive_plane_add(size=1.2, location=(0, 0.0002, 0), rotation=(math.radians(90), 0, 0))
    wall = bpy.context.active_object
    wm = bpy.data.materials.new("wall"); wm.use_nodes = True
    wm.node_tree.nodes["Principled BSDF"].inputs["Base Color"].default_value = (0.32, 0.30, 0.22, 1)
    wm.node_tree.nodes["Principled BSDF"].inputs["Roughness"].default_value = 0.9
    wall.data.materials.append(wm)
    # troffer
    L = bpy.data.lights.new("troffer", "AREA"); L.shape = "SQUARE"; L.size = 0.6; L.energy = 260
    lo = bpy.data.objects.new("troffer", L); bpy.context.scene.collection.objects.link(lo)
    lo.location = (0.1, -0.4, 1.2); lo.rotation_euler = (0, 0, 0)
    w = bpy.context.scene.world = bpy.data.worlds.new("w"); w.use_nodes = True
    w.node_tree.nodes["Background"].inputs[1].default_value = 0.15
    cam = bpy.data.objects.new("cam", bpy.data.cameras.new("cam")); bpy.context.scene.collection.objects.link(cam)
    bpy.context.scene.camera = cam; cam.data.lens = 85
    d, el, az = 0.55, math.radians(22), math.radians(18)
    cam.location = (d * math.cos(el) * math.sin(az), -d * math.cos(el) * math.cos(az), d * math.sin(el))
    cam.rotation_euler = (Vector((0, -0.003, 0)) - cam.location).to_track_quat("-Z", "Y").to_euler()
    sc = bpy.context.scene
    sc.render.engine = "CYCLES"; sc.cycles.samples = 64; sc.cycles.use_denoising = True; sc.cycles.device = "CPU"
    sc.render.resolution_x = 900; sc.render.resolution_y = 1100
    sc.view_settings.view_transform = "AgX"
    sc.render.filepath = os.path.join(OUT, "%s_%s.png" % (n, TAG)); bpy.ops.render.render(write_still=True)
    print("[gc]", n, TAG)
