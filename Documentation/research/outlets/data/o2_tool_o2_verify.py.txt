def load_from(models, name):
    bpy.ops.wm.read_factory_settings(use_empty=True)
    bpy.ops.import_scene.fbx(filepath=os.path.join(models, name + ".fbx"), axis_forward="-Z", axis_up="Y", bake_space_transform=True)
    return [o for o in bpy.data.objects if o.type == "MESH"]


for n in names:
    objs = load(n)
    info = {"objects": [o.name for o in objs]}
    per_slot = {}
    hmin, lo, hi = 1e9, [1e9] * 3, [-1e9] * 3
    wear_ok, normals_ok = True, True
    for o in objs:
        me = o.data
        mats = [m.name.split(".")[0] if m else "" for m in me.materials]
        for p in me.polygons:
            s = mats[p.material_index]
            per_slot[s] = per_slot.get(s, 0) + len(p.vertices) - 2
        for v in me.vertices:
            w = o.matrix_world @ v.co
            p = (w.x * 1000, -w.y * 1000, w.z * 1000)
            hmin = min(hmin, p[1])
            for k in range(3):
                lo[k] = min(lo[k], p[k]); hi[k] = max(hi[k], p[k])
        ca = me.color_attributes
        wear_ok &= len(ca) > 0 and any(len(a.data) == len(me.loops) for a in ca)
        normals_ok &= me.has_custom_normals
    info.update({"trisPerSlot": per_slot, "tris": sum(per_slot.values()), "boundsMm": [lo, hi], "minH": hmin,
                 "wearOnEveryCorner": wear_ok, "customNormals": normals_ok})
    report["assets"][n] = info
    print("[verify]", n, info["tris"], per_slot, "minH %.4f" % hmin, "wear", wear_ok, "normals", normals_ok)

def plug(width, height=6.6, depth=5.0):
    bm = bmesh.new()
    bmesh.ops.create_cube(bm, size=1.0)
    bmesh.ops.scale(bm, vec=Vector((width, depth, height)) * 0.001, verts=bm.verts)
    bmesh.ops.translate(bm, vec=Vector((0, -(face_h + 1.0 - depth / 2) * 0.001, zc * 0.001)), verts=bm.verts)
    v = [x.co.copy() for x in bm.verts]
    f = [[x.index for x in fc.verts] for fc in bm.faces]
    bm.free()
    return BVHTree.FromPolygons(v, f)

# ------------------------------------- O1 vs O2 plate surface (ray scan)
# Front rays (+Y) at scan lines that are plain plate on both kits: z = 0
# from x = 4 to the edge (O1's duplex has only its centre screw there),
# x = +-25 over the full height (clear of every opening and screw on both),
# and the four corner diagonals. Hit height h (mm) must agree to 0.05 mm;
# a miss on one kit and a hit on the other is an outline mismatch.
def bvh_of(name, slots=None, models=None):
    objs = load(name) if models is None else load_from(models, name)
    vv, pp = [], []
    for o in objs:
        me = o.data
        mats = [m.name.split(".")[0] if m else "" for m in me.materials]
        base = len(vv)
        vv.extend(o.matrix_world @ v.co for v in me.vertices)
        for p in me.polygons:
            if slots is None or mats[p.material_index] in slots:
                pp.append([base + i for i in p.vertices])
    return BVHTree.FromPolygons(vv, pp)


def scan(bvh, pts):
    out = []
    for x, z in pts:
        hit = bvh.ray_cast(Vector((x / 1000, -0.05, z / 1000)), Vector((0, 1, 0)), 0.06)
        out.append(None if hit[0] is None else -hit[0].y * 1000)
    return out
