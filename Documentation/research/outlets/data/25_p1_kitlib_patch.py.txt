def apply(path, edits):
    text = open(path).read()
    for old, new in edits:
        n = text.count(old)
        if n != 1:
            sys.exit("anchor matched %d times in %s:\n%s" % (n, path, old[:200]))
        text = text.replace(old, new)
    return text
