def crop(p, box):
    im = Image.open(p).convert('RGB')
    w, h = im.size
    return im.crop((int(w * box[0]), int(h * box[1]), int(w * box[2]), int(h * box[3])))


def sheet(cells, cols, tw, th, title, path):
    rows = (len(cells) + cols - 1) // cols
    lab = 56
    S = Image.new('RGB', (cols * tw, 48 + rows * (th + lab)), (24, 24, 24))
    d = ImageDraw.Draw(S)
    d.text((12, 12), title, fill=(240, 240, 240), font=font)
    for i, (p, box, l1, l2) in enumerate(cells):
        x, y = (i % cols) * tw, 48 + (i // cols) * (th + lab)
        if p is not None and os.path.exists(p):
            S.paste(crop(p, box).resize((tw - 6, th), Image.LANCZOS), (x + 3, y + lab))
        elif p is not None:
            raise SystemExit('missing ' + p)
        d.text((x + 8, y + 4), l1, fill=(255, 255, 255), font=small)
        d.text((x + 8, y + 28), l2, fill=(190, 190, 190), font=small)
    S.save(path, quality=85)
    print(path, S.size)
