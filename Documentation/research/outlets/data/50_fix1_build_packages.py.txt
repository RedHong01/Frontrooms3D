def sha(p):
    h = hashlib.sha1()
    with open(p, 'rb') as f:
        h.update(f.read())
    return h.hexdigest()


def files(root):
    out = []
    for d, _, fs in os.walk(root):
        for f in fs:
            out.append(os.path.relpath(os.path.join(d, f), root))
    return sorted(out)


def order(paths):
    """.meta first (so an open Unity never invents a GUID), then assets, then C# last."""
    def k(p):
        if p.endswith('.meta'):
            return (0, p)
        if p.endswith('.cs'):
            return (2, p)
        return (1, p)
    return sorted(paths, key=k)

def build(track, payload_dir, base_of, keep):
    pay = files(payload_dir)
    assert pay, track
    for f in os.listdir(APPLY):
        if f == track + '.sh' or f.startswith(track + '_') or f == track + '.diff':
            if not (f.startswith(track + '_fbxmeta') and track != 'outlets_p4b_d_fbxmeta'):
                raise SystemExit(f'REFUSED: {APPLY}/{f} exists (move it to _partial first)')
    dst_pay = os.path.join(APPLY, track + '_payload')
    dst_base = os.path.join(APPLY, track + '_base')
    exp, bases, diffs = [], [], []
    for p in order(pay):
        src = os.path.join(payload_dir, p)
        os.makedirs(os.path.dirname(os.path.join(dst_pay, p)), exist_ok=True)
        shutil.copyfile(src, os.path.join(dst_pay, p))
        exp.append(f'{sha(src)}  {p}')
        b = base_of(p)
        if b is None:
            bases.append(f'ABSENT  {p}')
            if p.endswith(('.cs', '.mat', '.meta', '.txt', '.md')):
                diffs.append(subprocess.run(['diff', '-u', '/dev/null', src, '--label', 'a/' + p, '--label', 'b/' + p], capture_output=True, text=True).stdout)
            else:
                diffs.append(f'Binary file b/{p} (new, sha1 {sha(src)})\n')
        else:
            bases.append(f'{sha(b)}  {p}')
            os.makedirs(os.path.dirname(os.path.join(dst_base, p)), exist_ok=True)
            shutil.copyfile(b, os.path.join(dst_base, p))
            diffs.append(subprocess.run(['diff', '-u', b, src, '--label', 'a/' + p, '--label', 'b/' + p], capture_output=True, text=True).stdout)
    os.makedirs(dst_base, exist_ok=True)
    stamp = subprocess.run(['date', '+%F %T'], capture_output=True, text=True).stdout.strip()
    head = subprocess.run(['git', '-C', M, 'rev-parse', '--short', 'HEAD'], capture_output=True, text=True, env=dict(os.environ, GIT_OPTIONAL_LOCKS='0')).stdout.strip()
    with open(os.path.join(APPLY, track + '_bases.txt'), 'w') as f:
        f.write(f'# {track} bases: sha1 of every path the script writes, recorded {stamp} on main {head} + working tree\n' + '\n'.join(bases) + '\n')
    with open(os.path.join(APPLY, track + '_expected.txt'), 'w') as f:
        f.write(f'# {track} payload (= the target after --apply), {stamp}\n' + '\n'.join(exp) + '\n')
    with open(os.path.join(APPLY, track + '_keep.txt'), 'w') as f:
        f.write(f'# {track} kept, never written: GUID .meta files (must stay as verified) and notes\n')
        for kp, ksrc in keep:
            f.write(f'{sha(ksrc)}  {kp}\n')
    with open(os.path.join(APPLY, track + '.diff'), 'w') as f:
        f.write(''.join(diffs))
    # the script: outlets_lods.sh's v2 body with this track's names and header
    t = open(TEMPLATE).read()
    body = t[t.index('#   outlets_lods.sh            DRY RUN'):]
    body = body.replace('outlets_lods', track)
    with open(os.path.join(APPLY, track + '.sh'), 'w') as f:
        f.write('#!/bin/bash\n' + HEADERS[track] + '\n#\n' + body)
    os.chmod(os.path.join(APPLY, track + '.sh'), 0o755)
    print(f'{track}: {len(pay)} paths ({sum(1 for b in bases if b.startswith("ABSENT"))} new), keep {len(keep)}')


src = os.path.join(STAGE, 'pkg_src')
p4b = os.path.join(src, 'outlets_p4b_d', 'payload')
p6 = os.path.join(src, 'outlets_p6', 'payload')
fbx = os.path.join(src, 'outlets_p4b_d_fbxmeta', 'payload')


def main_base(p):
    q = os.path.join(M, p)
    return q if os.path.exists(q) else None


def stacked_base(p):
    q = os.path.join(p4b, p)
    return q if os.path.exists(q) else main_base(p)
