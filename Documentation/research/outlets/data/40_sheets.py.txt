def tile(path, crop=None, h=None, w=None, label=None, dark_chip=True):
    im = Image.open(path).convert('RGB')
    if crop: im = im.crop(crop)
    if h: im = im.resize((round(im.width * h / im.height), h), Image.LANCZOS)
    elif w: im = im.resize((w, round(im.height * w / im.width)), Image.LANCZOS)
    if label:
        d = ImageDraw.Draw(im)
        tw = d.textlength(label, font=FONT)
        d.rectangle((12, 12, 12 + tw + 16, 12 + 32), fill=(10, 10, 10) if dark_chip else (244, 223, 59))
        d.text((20, 15), label, font=FONT, fill=(255, 255, 255) if dark_chip else (10, 10, 10))
    return im

def grid(rows, out, gap=8, bg=(24, 24, 24), maxw=1920):
    widths = [sum(t.width for t in r) + gap * (len(r) - 1) for r in rows]
    W_ = max(widths); H_ = sum(max(t.height for t in r) for r in rows) + gap * (len(rows) - 1)
    s = Image.new('RGB', (W_, H_), bg); y = 0
    for r in rows:
        x = 0
        for t in r: s.paste(t, (x, y)); x += t.width + gap
        y += max(t.height for t in r) + gap
    if s.width > maxw: s = s.resize((maxw, round(s.height * maxw / s.width)), Image.LANCZOS)
    s.save(D + out, quality=85)
    print(out, s.size)
