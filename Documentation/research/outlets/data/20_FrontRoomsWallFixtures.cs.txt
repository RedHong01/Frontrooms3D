// Outlets, switches and jacks: a sub-logic of the room and wall logic
// (research/outlets/10_spec.md, rules version 1).
//
// Stage 1 runs once per map chunk, after the map has built the chunk's walls
// and before furniture. The map collects one WallFace for every wall face it
// built (both sides of every built edge, BuildEdge) and calls Install once;
// Install reads the chunk data and calls Plan, a pure function. The title
// stream (FrontRoomsRoomStream) calls InstallLocal with its own faces.
//
// Nothing here makes a GameObject, a Collider or a Mesh. The plan is kept on a
// FrontRoomsWallFixtureSet component on the chunk root (freed with the chunk),
// and FrontRoomsWallFixtureRenderer draws it with instanced, shadowless,
// culled draws. The E ray, the Relay's sight ray and nav never meet an outlet.
using System;
using System.Collections.Generic;
using FrontRooms.Map;
using UnityEngine;

public static class FrontRoomsWallFixtures
{
    public const int RulesVersion = 1;

    /// <summary>
    /// One wall face the map actually built (both sides of every built edge).
    /// Chunk-local metres, Y up. Filled by FrontRoomsMapWorld.BuildEdge.
    /// </summary>
    [Serializable]
    public struct WallFace
    {
        public Vector3 start;      // on the face plane (cell line + normal * WallHalf), floor level, at the edge's low end
        public Vector3 direction;  // +X or +Z, unit
        public float length;       // 3.0 on the map
        public Vector3 normal;     // into the cell this face serves
        public EdgeKind kind;      // Wall, Arch, Door, Window (Open is never listed)
        public float openFrom, openTo;    // opening span along direction, metres from start (0, 0 for a Wall)
        public float openBottom, openTop; // sill (window 0.35) and head of the opening
        public long edgeId;        // FrontRoomsMapWorld.EdgeId(a, b): world, the same from both sides
        public byte side;          // 0: faces into a (the west/south cell), 1: faces into b (the east/north cell)
        public GridCoord cell;     // world cell this face serves
        public bool foreign;       // that cell belongs to the next chunk (east/north border): planned from seed-pure inputs only
        public int room;           // index into MapChunk.rooms (topmost rect holding the cell); -1 = corridor or foreign
        public GridCoord zone;     // zone id of the served cell
        public ZoneTheme theme;
        public ZoneHeight height;
        public float ceiling;      // ceiling height of the served cell
        public bool zoneBorder;    // the cell behind is another zone
        public float baseTop;      // top of a base trim on this face (map 0; title stream 0.26)
    }

    /// <summary>One planned fixture. Chunk-local metres (the set's root space).</summary>
    [Serializable]
    public struct Fixture
    {
        public FixtureKit kit;
        public Vector3 position;   // the kit origin: plate back centre on the face plane (floor contact for floor boxes)
        public Vector3 normal;     // the kit front (Unity +Z, Blender -Y) points along this
        public float roll;         // degrees about normal: crooked, ground-up (180), horizontal (90)
        public float tilt;         // degrees about the plate's local X through its bottom edge: top edge off the wall
        public float screwRoll;    // degrees: the slot angle of every screw on the plate
        public Finish plate, device;
        public FixtureFlags flags;
        public short face;         // index into the face list; -1 for column, floor and extra fixtures
        public FixtureClass cls;
        public float along;        // metres from the face start (wall fixtures)
    }

    /// <summary>A volume no fixture may enter (module props, key spots, Relay entries). Chunk-local.</summary>
    [Serializable]
    public struct KeepOut
    {
        public Rect xz;
        public float y0, y1;
    }

    public const float Step = .15f;              // along-wall grid
    public const float EndMargin = .30f;         // plate centre from a face end (cell corner)
    public const float ArchClear = .20f;         // plate centre from an arch edge
    public const float BackToBack = .61f;        // opposite faces of one wall (UL 24 in)
    public const float SameFaceGap = .30f;       // two fixtures on one face, centre to centre
    public const float CompanionOffset = .30f;   // phone/data plate beside its receptacle

    public const float BaseClear = .03f;         // plate bottom above a base trim top
    public const float SwitchHeight = 1.22f, SwitchFromOpening = .20f;
    public const float RoomSideMin = 4.5f;       // a room side with this much solid wall gets at least one plate
    public const int ChunkCap = 160;

    struct ClassRule
    {
        public float spacing;            // metres of solid wall per receptacle (p = solid / spacing, max 1 per face)
        public float h, hAlt, pAlt;      // centre heights per wall line
        public int jitter;               // +- steps of 0.15 m per face
        public float pOff;               // off-height receptacle (Level 0 uncanny)
        public float pPhone, pData;      // companions (data only next to a phone)
        public float pArchSwitch;        // switch beside an arch (room faces)
        public float p20, pIG;           // 5-20R device; IG (orange) device
        public float pSteel;             // stainless plate
        public int cap;                  // per room
        public bool wear, office;
        public bool fixedFinish;         // every plate and device in this class is `finish` (title Run: white)
        public Finish finish;
    }

    static readonly ClassRule[] Rules = BuildRules();

        r[(int)FixtureClass.OfficeCorridor] = new ClassRule { spacing = 9.0f, h = .41f, hAlt = .46f, pAlt = .25f, jitter = 1, p20 = .20f, pSteel = .25f, wear = true, office = true, cap = 99 };
        r[(int)FixtureClass.OfficeRoom] = new ClassRule { spacing = 3.3f, h = .41f, hAlt = .46f, pAlt = .25f, jitter = 1, pPhone = .50f, pData = .33f, pArchSwitch = .50f, p20 = .45f, pIG = .33f, wear = true, office = true, cap = 24 };
        r[(int)FixtureClass.Tall] = new ClassRule { spacing = 7.5f, h = .31f, hAlt = .46f, pAlt = .40f, jitter = 1, pSteel = .30f, wear = true, cap = 99 };
        r[(int)FixtureClass.StreamLobby] = new ClassRule { spacing = 6.0f, h = .38f, hAlt = .38f, jitter = 0, pPhone = .25f, pSteel = 1f, cap = 99 };
        r[(int)FixtureClass.StreamShift] = new ClassRule { spacing = 4.0f, h = .38f, hAlt = .38f, jitter = 0, wear = true, cap = 99 };

    /// <summary>The planner class of each title-stream RoomRule (10_spec.md §1.6).</summary>
    public static FixtureClass StreamClass(RoomRule rule)
    {
        switch (rule)
        {
            case RoomRule.Shift: return FixtureClass.StreamShift;
            case RoomRule.Office: return FixtureClass.StreamOffice;
            case RoomRule.Run: return FixtureClass.StreamRun;
            case RoomRule.Exit: return FixtureClass.StreamExit;
            default: return FixtureClass.StreamLobby;
        }
    }

    // ---------- Hashes (salts 409-600; MapHash uses 11-73, the map 89, 97, 211, 223, 307) ----------

    const int SaltBase = 409;
    enum Field { Face, Line, Wear, Device, Companion, Switch, Column, Floor, Chunk, Forced, Move }

    static uint FaceHash(int seed, in WallFace f, Field field) =>
        MapHash.Hash(seed, unchecked((int)f.edgeId), unchecked((int)(f.edgeId >> 32)), SaltBase + 16 * (int)field + f.side);

    /// <summary>One wall line (world cell line, orientation, side) per 24 m segment: one electrician's run.</summary>
    static uint LineHash(int seed, in WallFace f)
    {
        var alongX = f.direction.x > .5f;
        var line = alongX ? (f.side == 0 ? f.cell.y + 1 : f.cell.y) : (f.side == 0 ? f.cell.x + 1 : f.cell.x);
        var segment = (alongX ? f.cell.x : f.cell.y) >> 3;
        return MapHash.Hash(seed, line * 2 + (alongX ? 1 : 0), segment, SaltBase + 16 * (int)Field.Line + f.side);
    }

    static uint Sub(uint h, int k) => MapHash.Hash(unchecked((int)h), k, 0, 1);
    static float U(uint h) => MapHash.Unit(h);
    static float U(uint h, int k) => MapHash.Unit(Sub(h, k));

    public static FixtureClass Classify(in WallFace f)
    {
        if (f.height == ZoneHeight.Tall) return FixtureClass.Tall;
        var room = f.room >= 0 && !f.foreign;
        if (f.theme == ZoneTheme.Office) return room ? FixtureClass.OfficeRoom : FixtureClass.OfficeCorridor;
        return room ? FixtureClass.Level0Room : FixtureClass.Level0Corridor;
    }

    struct Gaps
    {
        public float a0, a1, b0, b1;
        public int n;
        public float Total => (n > 0 ? a1 - a0 : 0f) + (n > 1 ? b1 - b0 : 0f);
        public bool Inside(float u) => (n > 0 && u >= a0 - 1e-4f && u <= a1 + 1e-4f) || (n > 1 && u >= b0 - 1e-4f && u <= b1 + 1e-4f);
    }

    /// <summary>Where a plate centre may go along the face: inside the end margins and clear of an arch.</summary>
    static Gaps PlateGaps(in WallFace f)
    {
        var g = new Gaps();
        float lo = EndMargin, hi = f.length - EndMargin;
        if (hi < lo) return g;
        if (f.kind == EdgeKind.Wall) { g.a0 = lo; g.a1 = hi; g.n = 1; return g; }
        float x1 = Mathf.Min(hi, f.openFrom - ArchClear), y0 = Mathf.Max(lo, f.openTo + ArchClear);
        if (x1 >= lo) { g.a0 = lo; g.a1 = x1; g.n = 1; }
        if (hi >= y0)
        {
            if (g.n == 0) { g.a0 = y0; g.a1 = hi; } else { g.b0 = y0; g.b1 = hi; }
            g.n++;
        }
        return g;
    }

    static float SolidLength(in WallFace f) => f.kind == EdgeKind.Wall ? f.length : Mathf.Max(0f, f.length - (f.openTo - f.openFrom));

    // ---------- Scratch: reused buffers, cleared on every call (no state survives a call) ----------

    sealed class Scratch
    {
        public int[] roomCount = new int[32];
        public int[] twin = new int[256];
        public readonly List<SideRecord> sides = new List<SideRecord>(32);
        public readonly ZoneTheme[] themes = new ZoneTheme[MapGrid.CellsPerChunk];
        public bool[] dressed = new bool[32];
        public readonly ReadOnlySlice<bool> dressedView = new ReadOnlySlice<bool>();
    }

    [ThreadStatic] static Scratch scratch;
    static Scratch S => scratch ??= new Scratch();

    /// <summary>A fixed-length read-only view of the front of a reused array (no copy, no allocation per call).</summary>
    sealed class ReadOnlySlice<T> : IReadOnlyList<T>
    {
        public T[] array;
        public int count;
        public T this[int index] => (uint)index < (uint)count ? array[index] : throw new ArgumentOutOfRangeException(nameof(index));
        public int Count => count;
        public IEnumerator<T> GetEnumerator() { for (var i = 0; i < count; i++) yield return array[i]; }
        System.Collections.IEnumerator System.Collections.IEnumerable.GetEnumerator() => GetEnumerator();
    }

    struct SideRecord
    {
        public long key;
        public float solid, bestGap;
        public int best;
        public bool has;
    }

    sealed class SideOrder : IComparer<SideRecord>
    {
        public static readonly SideOrder Instance = new SideOrder();
        public int Compare(SideRecord a, SideRecord b) => a.key.CompareTo(b.key);

    static int[] Grow(int[] a, int n) => a.Length >= n ? a : new int[Mathf.NextPowerOfTwo(n)];

    public static int StatMoved, StatDropped, StatForced, StatCapped, StatKeptOut;

    // ---------- Plan ----------

    /// <summary>
    /// Pure and deterministic: the same inputs give the same list, in the same
    /// order. No Unity objects, no UnityEngine.Random, no state between calls.
    /// Hashes: seed + EdgeId + face side (+ the wall line), never the revision.
    /// </summary>
    public static void Plan(int seed, IReadOnlyList<WallFace> faces, MapChunk data, RectInt startArea,
        IReadOnlyList<ZoneTheme> cellTheme, IReadOnlyList<bool> roomDressed, IReadOnlyList<KeepOut> keepOut, List<Fixture> into)
    {
        into.Clear();
        var s = S;
        var rooms = data != null ? data.rooms.Length : 0;
        s.roomCount = Grow(s.roomCount, rooms);
        Array.Clear(s.roomCount, 0, rooms);
        // 1. Receptacles, their companions, and arch switches, face by face. Foreign faces first and
        //    outside the cap: their plan must not depend on this chunk's revision.
        for (var i = 0; i < faces.Count; i++)
            if (faces[i].foreign) PlanFace(seed, faces, i, Classify(faces[i]), into, s.roomCount, rooms, null);
        for (var i = 0; i < faces.Count && into.Count < ChunkCap; i++)
            if (!faces[i].foreign) PlanFace(seed, faces, i, Classify(faces[i]), into, s.roomCount, rooms, keepOut);
        // 2. Every room side with enough solid wall gets at least one receptacle.
        if (data != null) ForceRoomSides(seed, faces, into, s.roomCount, rooms, keepOut);
        // 3. One lone switch per chunk at most (a switch with no door near it).
        if (data != null) PlanLoneSwitch(seed, faces, data, into);
        // 4. Opposite faces of one wall: side 0 yields to side 1 (side 1 may be a foreign, seed-pure face).
        ResolveBackToBack(faces, into);
        // 5. Columns and floor boxes from the chunk data.
        if (data != null)
        {
            PlanColumns(seed, data, startArea, cellTheme, into);
            PlanFloorBoxes(seed, data, startArea, cellTheme, roomDressed, into);
        }
    }

    /// <summary>The title stream (RoomStream): one class for every face, no chunk data.</summary>
    public static void PlanLocal(int seed, IReadOnlyList<WallFace> faces, FixtureClass cls, List<Fixture> into)
    {
        into.Clear();
        var s = S;
        for (var i = 0; i < faces.Count && into.Count < ChunkCap; i++) PlanFace(seed, faces, i, cls, into, s.roomCount, 0, null);
        ResolveBackToBack(faces, into);
    }

    static void PlanFace(int seed, IReadOnlyList<WallFace> faces, int index, FixtureClass cls, List<Fixture> into, int[] roomCount, int rooms, IReadOnlyList<KeepOut> keepOut)
    {
        var f = faces[index];
        if (f.kind != EdgeKind.Wall && f.kind != EdgeKind.Arch) return;   // never on Door or Window faces (binding)
        var rule = Rules[(int)cls];
        var gaps = PlateGaps(f);
        if (gaps.n == 0) return;
        var hF = FaceHash(seed, f, Field.Face);
        var roomed = f.room >= 0 && !f.foreign && f.room < rooms;
        var p = Mathf.Min(1f, SolidLength(f) / rule.spacing);
        if (U(hF) < p && (!roomed || roomCount[f.room] < rule.cap))
            AddReceptacle(seed, f, index, cls, rule, gaps, into, roomed, roomCount, keepOut, FixtureFlags.None);
        if (rule.pArchSwitch > 0f && f.kind == EdgeKind.Arch && roomed && U(FaceHash(seed, f, Field.Switch)) < rule.pArchSwitch && roomCount[f.room] < rule.cap)
        {
            var hS = FaceHash(seed, f, Field.Switch);
            var first = (Sub(hS, 1) & 1) == 0;
            for (var t = 0; t < 2; t++)
            {
                var left = first == (t == 0);
                var u = left ? f.openFrom - SwitchFromOpening : f.openTo + SwitchFromOpening;
                if (u < EndMargin || u > f.length - EndMargin || !ClearAtHeight(into, index, u, SwitchHeight)) continue;
                var kit = rule.office && U(hS, 2) < .3f ? FixtureKit.Toggle2 : FixtureKit.Toggle1;
                var x = Make(f, index, cls, kit, u, SwitchHeight, PlateFinish(seed, f, rule), hS);
                if (Blocked(x, f, keepOut)) continue;
                x.device = x.plate == Finish.Stainless ? Finish.Ivory : DeviceFor(x.plate);
                into.Add(x);
                roomCount[f.room]++;
                break;
            }
        }
    }

    static void AddReceptacle(int seed, in WallFace f, int index, FixtureClass cls, in ClassRule rule, in Gaps gaps, List<Fixture> into,
        bool countRoom, int[] roomCount, IReadOnlyList<KeepOut> keepOut, FixtureFlags extra)
    {
        var hF = FaceHash(seed, f, Field.Face);
        var hL = LineHash(seed, f);
        var anchorLow = (hL & 1u) == 0;
        var u0 = EndMargin + Step * (Sub(hL, 1) % 3u);
        var j = rule.jitter > 0 ? (int)(Sub(hF, 1) % (uint)(2 * rule.jitter + 1)) - rule.jitter : 0;
        var u = anchorLow ? u0 + j * Step : f.length - (u0 + j * Step);
        var start = into.Count;
        if (!SnapSameFace(gaps, u, .60f, into, index, out u)) return;
        var y = U(hL, 2) < rule.pAlt ? rule.hAlt : rule.h;
        var flags = extra;
        if (rule.pOff > 0f && U(hF, 3) < rule.pOff)
        {
            flags |= FixtureFlags.OffHeight;
            if (U(hF, 4) < .6f) y = .46f;
            else { y = 1.07f; flags |= FixtureFlags.Horizontal; }
        }
        var plate = PlateFinish(seed, f, rule);
        var hD = FaceHash(seed, f, Field.Device);
        var kit = plate == Finish.Stainless ? FixtureKit.DuplexSteel : (U(hD) < rule.p20 ? FixtureKit.Duplex20 : FixtureKit.Duplex);
        var device = rule.fixedFinish ? rule.finish : DeviceFor(plate);
        if (rule.pIG > 0f && U(hD, 1) < rule.pIG) { kit = plate == Finish.Stainless ? FixtureKit.DuplexSteel : FixtureKit.Duplex20; device = Finish.Orange; }
        var x = Make(f, index, cls, kit, u, y, plate, hF);
        x.device = device;
        x.flags |= flags;
        if ((flags & FixtureFlags.Horizontal) != 0) x.roll += 90f;
        if (rule.wear) Wear(seed, f, rule, ref x);
        else if (U(hF, 9) < .15f && IsDuplex(x.kit)) { x.roll += 180f; x.flags |= FixtureFlags.GroundUp; }
        x.position.y = Mathf.Max(x.position.y, f.baseTop + BaseClear + (x.kit == FixtureKit.DuplexJumbo ? JumboHalfH : PlateHalfH));
        if (Blocked(x, f, keepOut)) { StatKeptOut++; return; }
        into.Add(x);
        // Companions: a phone plate beside about half the Office desk receptacles, a data plate beside a third of those.
        // Only while the room stays within its cap (the receptacle counts; a forced one may sit on the cap).
        var hC = FaceHash(seed, f, Field.Companion);
        var roomLeft = countRoom ? rule.cap - roomCount[f.room] - (into.Count - start) : int.MaxValue;
        if (roomLeft > 0 && (flags & FixtureFlags.OffHeight) == 0 && rule.pPhone > 0f && U(hC) < rule.pPhone)
        {
            var dir = u < f.length * .5f ? 1f : -1f;
            if (AddCompanion(f, index, cls, FixtureKit.JackPhone, u, dir, x.position.y, plate, hC, into, keepOut) && roomLeft > 1 && rule.pData > 0f && U(hC, 1) < rule.pData)
            {
                var zoneNet = (MapHash.Hash(seed, f.zone.x, f.zone.y, SaltBase + 16 * (int)Field.Companion + 7) & 1u) == 0 ? FixtureKit.JackData : FixtureKit.JackBNC;
                AddCompanion(f, index, cls, zoneNet, into[into.Count - 1].along, dir, x.position.y, plate, Sub(hC, 2), into, keepOut);
            }
        }
        if (countRoom) roomCount[f.room] += into.Count - start;
    }

    static bool AddCompanion(in WallFace f, int index, FixtureClass cls, FixtureKit kit, float from, float dir, float y, Finish plate, uint h,
        List<Fixture> into, IReadOnlyList<KeepOut> keepOut)
    {
        var gaps = PlateGaps(f);
        for (var t = 0; t < 2; t++)
        {
            var d = t == 0 ? dir : -dir;
            var u = from + d * CompanionOffset;
            if (!gaps.Inside(u) || !FarFromSameFace(into, index, u)) continue;
            var x = Make(f, index, cls, kit, u, y, plate == Finish.Stainless ? Finish.Stainless : plate, h);
            x.device = Finish.Ivory;
            x.flags |= FixtureFlags.Companion;
            if (Blocked(x, f, keepOut)) continue;
            into.Add(x);
            return true;
        }
        return false;
    }

    /// <summary>The nearest point of the 0.15 m grid inside the gaps, within maxShift of u, clear of the face's other fixtures (ties go to the lower point).</summary>
    static bool SnapSameFace(in Gaps g, float u, float maxShift, List<Fixture> into, int face, out float result)
    {
        var k0 = Mathf.RoundToInt(u / Step);
        var steps = Mathf.FloorToInt(maxShift / Step + 1e-4f);
        for (var d = 0; d <= steps; d++)
        for (var s = -1; s <= 1; s += 2)
        {
            if (d == 0 && s > 0) continue;
            var c = (k0 + s * d) * Step;
            if (Mathf.Abs(c - u) > maxShift + 1e-4f || !g.Inside(c) || !FarFromSameFace(into, face, c)) continue;
            result = c;
            return true;
        }
        result = u;
        return false;
    }

    /// <summary>As SnapSameFace, for a side-0 fixture that must also clear its twin face by BackToBack.</summary>
    static bool SnapBackToBack(in Gaps g, float u, float maxShift, List<Fixture> into, int face, int self, int twin, out float result)
    {
        var k0 = Mathf.RoundToInt(u / Step);
        var steps = Mathf.FloorToInt(maxShift / Step + 1e-4f);
        for (var d = 0; d <= steps; d++)
        for (var s = -1; s <= 1; s += 2)
        {
            if (d == 0 && s > 0) continue;
            var c = (k0 + s * d) * Step;
            if (Mathf.Abs(c - u) > maxShift + 1e-4f || !g.Inside(c) || !BackToBackOk(into, face, self, twin, c)) continue;
            result = c;
            return true;
        }
        result = u;
        return false;
    }

    static bool TwinClear(List<Fixture> into, int twin, float u)
    {
        for (var m = 0; m < into.Count; m++)
            if (into[m].face == twin && Mathf.Abs(into[m].along - u) < BackToBack - 1e-4f) return false;
        return true;
    }

    static bool BackToBackOk(List<Fixture> into, int face, int self, int twin, float u)
    {
        if (!TwinClear(into, twin, u)) return false;
        for (var m = 0; m < into.Count; m++) if (m != self && into[m].face == face && Mathf.Abs(into[m].along - u) < SameFaceGap - 1e-4f) return false;
        return true;
    }

    static bool IsDuplex(FixtureKit k) => k == FixtureKit.Duplex || k == FixtureKit.Duplex20 || k == FixtureKit.DuplexSteel || k == FixtureKit.DuplexCracked
        || k == FixtureKit.DuplexJumbo || k == FixtureKit.DuplexScorched || k == FixtureKit.Bare;

    static bool FarFromSameFace(List<Fixture> into, int face, float u)
    {
        for (var k = into.Count - 1; k >= 0; k--)
            if (into[k].face == face && Mathf.Abs(into[k].along - u) < SameFaceGap - 1e-4f) return false;
        return true;
    }

    static Fixture Make(in WallFace f, int index, FixtureClass cls, FixtureKit kit, float u, float y, Finish plate, uint h)
    {
        return new Fixture
        {
            kit = kit, position = f.start + f.direction * u + Vector3.up * y, normal = f.normal,
            screwRoll = 180f * U(h, 11), plate = plate, device = DeviceFor(plate), face = (short)index, cls = cls, along = u,
        };
    }

    static Finish DeviceFor(Finish plate)
    {
        switch (plate)
        {
            case Finish.White: return Finish.White;
            case Finish.Brown: return Finish.Brown;
            case Finish.Grey: return Finish.Grey;
            default: return Finish.Ivory;
        }
    }

    /// <summary>One colour per wall line (one electrician); stainless by class; a fixed finish for the title Run.</summary>
    static Finish PlateFinish(int seed, in WallFace f, in ClassRule rule)
    {
        if (rule.fixedFinish) return rule.finish;
        var hL = LineHash(seed, f);
        var pSteel = rule.pSteel * (f.zoneBorder && rule.office ? 2f : 1f);
        if (pSteel > 0f && U(hL, 5) < pSteel) return Finish.Stainless;
        var c = U(hL, 6);
        if (f.height == ZoneHeight.Tall) return c < .40f ? Finish.Ivory : c < .70f ? Finish.Grey : Finish.Almond;
        if (rule.office) return c < .55f ? Finish.Ivory : c < .90f ? Finish.Almond : Finish.White;
        return c < .70f ? Finish.Ivory : c < .80f ? Finish.Almond : c < .90f ? Finish.White : Finish.Brown;
    }

    /// <summary>Independent wear flags (01 §8 rates: Level 0 / Office). One geometry variant at most.</summary>
    static void Wear(int seed, in WallFace f, in ClassRule rule, ref Fixture x)
    {
        var h = FaceHash(seed, f, Field.Wear);
        var o = rule.office;
        if (U(h, 1) < (o ? .08f : .10f)) x.roll += (U(h, 2) < .5f ? -1f : 1f) * (2f + 4f * U(h, 3));             // crooked 2-6 deg
        if (U(h, 4) < (o ? .05f : .08f)) x.tilt = .5f + 1.0f * U(h, 5);                                          // not flush: top edge 1-3 mm out
        if (U(h, 6) < (o ? .04f : .06f)) { x.flags |= FixtureFlags.NoScrew; x.tilt = Mathf.Max(x.tilt, .4f); }  // missing screw
        if (IsDuplex(x.kit) && U(h, 7) < .15f) { x.roll += 180f; x.flags |= FixtureFlags.GroundUp; }           // another electrician
        if (x.plate == Finish.Ivory && U(h, 8) < (o ? .15f : .30f)) x.plate = Finish.IvoryAged;                  // grime / yellowing
        if (x.plate != Finish.Stainless && U(h, 9) < (o ? .08f : .03f)) x.plate = Finish.WallMatch;              // papered or painted over
        if (U(h, 10) < .05f) x.device = x.device == Finish.White ? Finish.Ivory : Finish.White;                 // mismatched device
        var thermoset = x.plate != Finish.Stainless;
        var g = U(h, 12);
        // Geometry variants, one at most, in this order of precedence.
        if (g < (o ? .01f : .03f)) x.kit = FixtureKit.Bare;
        else if ((g -= (o ? .01f : .03f)) < (o ? .04f : .03f)) x.kit = FixtureKit.Blank;
        else if ((g -= (o ? .04f : .03f)) < .02f && thermoset) x.kit = FixtureKit.DuplexScorched;
        else if ((g -= .02f) < (o ? .03f : .05f) && thermoset && x.kit == FixtureKit.Duplex) x.kit = FixtureKit.DuplexJumbo;
        else if ((g -= (o ? .03f : .05f)) < (o ? .03f : .05f) && thermoset && x.kit == FixtureKit.Duplex) x.kit = FixtureKit.DuplexCracked;
        if (x.kit == FixtureKit.Bare) x.flags |= FixtureFlags.NoScrew;
    }

    static bool Blocked(in Fixture x, in WallFace f, IReadOnlyList<KeepOut> keepOut)
    {
        if (keepOut == null || keepOut.Count == 0) return false;
        var a = x.position - f.direction * (PlateHalfW + .02f);
        var b = x.position + f.direction * (PlateHalfW + .02f) + f.normal * .012f;
        var r = Rect.MinMaxRect(Mathf.Min(a.x, b.x), Mathf.Min(a.z, b.z), Mathf.Max(a.x, b.x), Mathf.Max(a.z, b.z));
        for (var i = 0; i < keepOut.Count; i++)
        {
            var k = keepOut[i];
            if (k.xz.Overlaps(r) && x.position.y + PlateHalfH > k.y0 && x.position.y - PlateHalfH < k.y1) return true;
        }
        return false;
    }

    static int DirIndex(Vector3 n) => n.x > .5f ? 0 : n.z > .5f ? 1 : n.x < -.5f ? 2 : 3;

    static void ForceRoomSides(int seed, IReadOnlyList<WallFace> faces, List<Fixture> into, int[] roomCount, int rooms, IReadOnlyList<KeepOut> keepOut)
    {
        // A room side = own faces of one room with the same normal on the same cell line.
        var sides = S.sides;
        sides.Clear();
        for (var i = 0; i < faces.Count; i++)
        {
            var f = faces[i];
            if (f.foreign || f.room < 0 || (f.kind != EdgeKind.Wall && f.kind != EdgeKind.Arch)) continue;
            var key = SideKey(f);
            var k = FindSide(sides, key);
            if (k < 0) { sides.Add(new SideRecord { key = key, best = -1 }); k = sides.Count - 1; }
            var s = sides[k];
            s.solid += SolidLength(f);
            var gap = PlateGaps(f).Total;
            if (gap > s.bestGap + 1e-4f) { s.bestGap = gap; s.best = i; }
            sides[k] = s;
        }
        for (var m = 0; m < into.Count; m++)
        {
            var x = into[m];
            if (x.face < 0 || (x.flags & FixtureFlags.Companion) != 0 || !IsReceptacle(x.kit)) continue;
            var f = faces[x.face];
            if (f.foreign || f.room < 0) continue;
            var k = FindSide(sides, SideKey(f));
            if (k < 0) continue;
            var s = sides[k];
            s.has = true;
            sides[k] = s;
        }
        sides.Sort(SideOrder.Instance);
        for (var k = 0; k < sides.Count; k++)
        {
            var s = sides[k];
            if (s.has || s.solid < RoomSideMin || s.best < 0 || into.Count >= ChunkCap) continue;
            var f = faces[s.best];
            var cls = Classify(f);
            var before = into.Count;
            AddReceptacle(seed, f, s.best, cls, Rules[(int)cls], PlateGaps(f), into, f.room < rooms, roomCount, keepOut, FixtureFlags.Forced);
            if (into.Count > before) StatForced++;
        }
    }

    static int FindSide(List<SideRecord> sides, long key)
    {
        for (var k = 0; k < sides.Count; k++) if (sides[k].key == key) return k;
        return -1;
    }

    static long SideKey(in WallFace f)
    {
        var alongX = f.direction.x > .5f;
        var line = alongX ? (f.side == 0 ? f.cell.y + 1 : f.cell.y) : (f.side == 0 ? f.cell.x + 1 : f.cell.x);
        return ((long)f.room << 40) ^ ((long)DirIndex(f.normal) << 36) ^ (uint)(line + (1 << 20));
    }

    static bool IsReceptacle(FixtureKit k) => IsDuplex(k) || k == FixtureKit.Blank || k == FixtureKit.HandyBox;

    static void ResolveBackToBack(IReadOnlyList<WallFace> faces, List<Fixture> into)
    {
        // The twin of a side-0 face is the side-1 face of the same edge (the map lists the two faces next to each other).
        var s = S;
        s.twin = Grow(s.twin, faces.Count);
        var twinOf = s.twin;
        for (var i = 0; i < faces.Count; i++)
        {
            twinOf[i] = -1;
            if (faces[i].side != 0) continue;
            var id = faces[i].edgeId;
            if (i + 1 < faces.Count && faces[i + 1].side == 1 && faces[i + 1].edgeId == id) { twinOf[i] = i + 1; continue; }
            for (var k = faces.Count - 1; k >= 0; k--)
                if (faces[k].side == 1 && faces[k].edgeId == id) { twinOf[i] = k; break; }
        }
        for (var k = 0; k < into.Count; k++)
        {
            var x = into[k];
            if (x.face < 0 || faces[x.face].side != 0) continue;
            var twin = twinOf[x.face];
            if (twin < 0 || TwinClear(into, twin, x.along)) continue;
            var f = faces[x.face];
            if (SnapBackToBack(PlateGaps(f), x.along, .90f, into, x.face, k, twin, out var moved))
            {
                x.position += f.direction * (moved - x.along);
                x.along = moved;
                x.flags |= FixtureFlags.Moved;
                into[k] = x;
                StatMoved++;
            }
            else
            {
                into.RemoveAt(k);
                k--;
                StatDropped++;
            }
        }
    }

    static void PlanColumns(int seed, MapChunk data, RectInt startArea, IReadOnlyList<ZoneTheme> cellTheme, List<Fixture> into)
    {
        const int n = MapGrid.ChunkCells;
        var o = data.Origin;
        for (var j = 0; j < n && into.Count < ChunkCap; j++)
        for (var i = 0; i < n && into.Count < ChunkCap; i++)
        {
            if (!data.pillar[i + j * (n + 1)]) continue;
            if (InRect(startArea, o.x + i - 1, o.y + j - 1, 2)) continue;
            var k = MapGrid.LocalIndex(i, j);
            var theme = cellTheme != null && cellTheme.Count > k ? cellTheme[k] : ZoneTheme.Level0;
            var tall = data.height[k] == ZoneHeight.Tall;
            var h = MapHash.Hash(seed, o.x + i, o.y + j, SaltBase + 16 * (int)Field.Column);
            var p = tall || theme == ZoneTheme.Office ? .33f : .12f;
            if (U(h) >= p) continue;
            var w = (data.pillarStyle[i + j * (n + 1)] & MapChunk.ColumnLarge) != 0 ? ModuleUnits.ColumnLarge : ModuleUnits.ColumnSmall;
            var dir = (int)(Sub(h, 1) % 4u);
            var normal = dir == 0 ? Vector3.right : dir == 1 ? Vector3.forward : dir == 2 ? Vector3.left : Vector3.back;
            var y = tall ? .46f : theme == ZoneTheme.Office ? .41f : .31f;
            var plate = tall ? (U(h, 2) < .5f ? Finish.Stainless : Finish.Grey) : theme == ZoneTheme.Office ? Finish.Almond : Finish.Ivory;
            into.Add(new Fixture
            {
                kit = plate == Finish.Stainless ? FixtureKit.DuplexSteel : FixtureKit.Duplex,
                position = new Vector3(i * MapGrid.CellSize, y, j * MapGrid.CellSize) + normal * (w * .5f),
                normal = normal, screwRoll = 180f * U(h, 3), plate = plate, device = DeviceFor(plate), face = -1, cls = FixtureClass.Column,
            });
        }
    }

    static void PlanFloorBoxes(int seed, MapChunk data, RectInt startArea, IReadOnlyList<ZoneTheme> cellTheme, IReadOnlyList<bool> roomDressed, List<Fixture> into)
    {
        const int n = MapGrid.ChunkCells;
        var o = data.Origin;
        for (var j = 1; j < n && into.Count < ChunkCap; j++)
        for (var i = 1; i < n && into.Count < ChunkCap; i++)
        {
            // Bay centres of the 6 m column grid (odd world corners), in open floor of 2 x 2 cells that no kit
            // will furnish (corridor cells, or a room the dress step skips): "the desks have gone".
            if (((o.x + i) & 1) == 0 || ((o.y + j) & 1) == 0) continue;
            if (InRect(startArea, o.x + i - 1, o.y + j - 1, 2)) continue;
            if (data.east[MapGrid.LocalIndex(i - 1, j - 1)] != EdgeKind.Open || data.east[MapGrid.LocalIndex(i - 1, j)] != EdgeKind.Open
                || data.north[MapGrid.LocalIndex(i - 1, j - 1)] != EdgeKind.Open || data.north[MapGrid.LocalIndex(i, j - 1)] != EdgeKind.Open) continue;
            var k = MapGrid.LocalIndex(i, j);
            var theme = cellTheme != null && cellTheme.Count > k ? cellTheme[k] : ZoneTheme.Level0;
            var height = data.height[k];
            var ok = true;
            for (var dj = -1; dj <= 0 && ok; dj++)
            for (var di = -1; di <= 0 && ok; di++)
            {
                var c = MapGrid.LocalIndex(i + di, j + dj);
                var room = RoomOf(data, i + di, j + dj);
                var furnished = room >= 0 && (roomDressed == null || room >= roomDressed.Count || roomDressed[room]);
                if (furnished || data.height[c] != height || (cellTheme != null && cellTheme[c] != theme)) ok = false;
            }
            if (!ok) continue;
            var h = MapHash.Hash(seed, o.x + i, o.y + j, SaltBase + 16 * (int)Field.Floor);
            var p = theme == ZoneTheme.Office ? .30f : height == ZoneHeight.Tall ? .10f : .04f;
            if (U(h) >= p) continue;
            var dx = ((int)(Sub(h, 1) % 3u) - 1) * ModuleUnits.Fine;
            var dz = ((int)(Sub(h, 2) % 3u) - 1) * ModuleUnits.Fine;
            var yaw = 90f * (Sub(h, 3) % 4u);
            into.Add(new Fixture
            {
                kit = U(h, 4) < .2f ? FixtureKit.FloorBox2 : FixtureKit.FloorBox,
                position = new Vector3(i * MapGrid.CellSize + dx, 0f, j * MapGrid.CellSize + dz),
                normal = Quaternion.Euler(0f, yaw, 0f) * Vector3.forward, plate = Finish.Stainless, device = Finish.Ivory,
                screwRoll = 180f * U(h, 5), face = -1, cls = FixtureClass.Floor,
            });
        }
    }

    static void PlanLoneSwitch(int seed, IReadOnlyList<WallFace> faces, MapChunk data, List<Fixture> into)
    {
        var h = MapHash.Hash(seed, data.coord.x, data.coord.y, SaltBase + 16 * (int)Field.Chunk);
        if (U(h) >= .25f || into.Count >= ChunkCap) return;
        int best = -1;
        uint bestKey = uint.MaxValue;
        for (var i = 0; i < faces.Count; i++)
        {
            var f = faces[i];
            if (f.foreign || f.kind != EdgeKind.Wall || Classify(f) != FixtureClass.Level0Corridor) continue;
            var key = FaceHash(seed, f, Field.Switch);
            if (key < bestKey) { bestKey = key; best = i; }
        }
        if (best < 0) return;
        var g = faces[best];
        var k0 = (int)(Sub(h, 1) % 3u);
        for (var t = 0; t < 3; t++)
        {
            var u = g.length * .5f + Step * 3 * ((k0 + t) % 3 - 1);
            if (!ClearAtHeight(into, best, u, SwitchHeight)) continue;
            into.Add(Make(g, best, FixtureClass.Level0Corridor, FixtureKit.Toggle1, u, SwitchHeight, Finish.Ivory, h));
            return;
        }
    }

    /// <summary>Spec §2.4.5 for switches: no fixture of the same face within SameFaceGap, centre to centre, at a similar height (0.30 m).</summary>
    static bool ClearAtHeight(List<Fixture> into, int face, float u, float y)
    {
        for (var k = into.Count - 1; k >= 0; k--)
        {
            var o = into[k];
            if (o.face != face || Mathf.Abs(o.position.y - y) >= SameFaceGap) continue;
            if (Mathf.Abs(o.along - u) < SameFaceGap - 1e-4f) return false;
        }
        return true;
    }

    static int RoomOf(MapChunk data, int i, int j)
    {
        for (var r = data.rooms.Length - 1; r >= 0; r--) if (data.rooms[r].Contains(i, j)) return r;
        return -1;
    }

    static bool InRect(RectInt r, int x, int y, int size) =>
        r.width > 0 && r.height > 0 && x < r.xMax && x + size > r.xMin && y < r.yMax && y + size > r.yMin;

    /// <summary>
    /// The map's one call (CONTRACT, 10_spec.md §4.1): once per chunk, after
    /// AddZoneGrades and before Furnish. Adds a FrontRoomsWallFixtureSet to the
    /// chunk root (freed with it by Unregister) and plans into it. Allocates no
    /// mesh, so <paramref name="ownedMeshes"/> stays untouched (it is there for
    /// a combined-mesh fallback). Returns null while Suppressed.
    /// </summary>
    public static FrontRoomsWallFixtureSet Install(Transform chunkRoot, List<WallFace> faces, int seed, MapChunk data, RectInt startArea, List<Mesh> ownedMeshes)
    {
        if (Suppressed || chunkRoot == null || data == null || faces == null) return null;
        var set = chunkRoot.gameObject.AddComponent<FrontRoomsWallFixtureSet>();
        set.coord = data.coord;
        set.isLocal = false;
        set.seed = seed;
        set.faces.Clear();
        set.faces.AddRange(faces);
        var s = S;
        var world = chunkRoot.GetComponentInParent<FrontRoomsMapWorld>();
        var cache = world != null ? world.Cache : null;
        for (var j = 0; j < MapGrid.ChunkCells; j++)
        for (var i = 0; i < MapGrid.ChunkCells; i++)
            s.themes[MapGrid.LocalIndex(i, j)] = cache != null ? cache.ZoneOf(data.Cell(i, j)).theme : ZoneTheme.Level0;
        set.keepOut.Clear();
        KeepOuts(data, set.keepOut);
        // A room the dress step may furnish (the map's Furnish test): no floor boxes there in stage 1.
        var rooms = data.rooms.Length;
        if (s.dressed.Length < rooms) s.dressed = new bool[Mathf.NextPowerOfTwo(rooms)];
        for (var r = 0; r < rooms; r++)
        {
            var module = data.ModuleOf(r);
            var fillNone = module != null && module.fill == ModuleFill.None;
            s.dressed[r] = !fillNone && data.RoomIntact(r) && (cache == null || cache.Generator.Uniform(data, data.rooms[r]));
        }
        s.dressedView.array = s.dressed;
        s.dressedView.count = rooms;
        Plan(seed, set.faces, data, startArea, s.themes, s.dressedView, set.keepOut, set.fixtures);
        s.dressedView.array = null;
        set.wallMatchLevel0 = FrontRoomsSurfaces.Room(RoomRule.Lobby, FrontRoomsSurfaces.Slot.Wall);
        set.wallMatchOffice = FrontRoomsSurfaces.Room(RoomRule.Office, FrontRoomsSurfaces.Slot.Wall);
        set.MarkChanged();
        return set;
    }

    /// <summary>
    /// The title stream (visual-owned): faces in the variant's local space, one
    /// class for every face. Adds (or reuses) the set on <paramref name="root"/>,
    /// so toggling the variant hides its outlets too. Returns null while Suppressed.
    /// </summary>
    public static FrontRoomsWallFixtureSet InstallLocal(Transform root, List<WallFace> faces, int seed, FixtureClass cls, Material wallMatch)
    {
        if (Suppressed || root == null || faces == null) return null;
        if (!root.TryGetComponent<FrontRoomsWallFixtureSet>(out var set)) set = root.gameObject.AddComponent<FrontRoomsWallFixtureSet>();
        set.isLocal = true;
        set.streamClass = cls;
        set.seed = seed;
        set.extraCount = 0;   // a re-plan drops earlier extras with the rest of the list
        if (set.occluders.Count > 0) { set.occluders.Clear(); FrontRoomsWallFixtureSet.OccluderVersion++; }   // the room is dressed again after this
        set.faces.Clear();
        set.faces.AddRange(faces);
        set.keepOut.Clear();
        PlanLocal(seed, set.faces, cls, set.fixtures);
        set.wallMatchLevel0 = set.wallMatchOffice = wallMatch;
        set.MarkChanged();
        return set;
    }

    /// <summary>Stage 2, rule 2: a desk this far (xz) from a drawn receptacle or floor box counts as powered.</summary>
    public const float StationReach = 2f;

    static bool occlusion = true;

    /// <summary>Tools and tests (before/after captures): false draws plates behind wall units again. Default true.</summary>
    public static bool OcclusionEnabled
    {
        get => occlusion;
        set
        {
            if (occlusion == value) return;
            occlusion = value;

    /// <summary>Plate half width and half height of a wall kit (the across and up sizes before any roll).</summary>
    public static Vector2 PlateHalf(FixtureKit k) => k == FixtureKit.Toggle2 ? new Vector2(.05795f, PlateHalfH)
        : k == FixtureKit.DuplexJumbo ? new Vector2(.04445f, JumboHalfH) : new Vector2(PlateHalfW, PlateHalfH);

    /// <summary>
    /// The keep-out volume of a wall fixture, in its set's root space (10_spec.md §2.9, as stage 1's
    /// keep-outs use it): the plate rect plus 0.02 m each side along the wall, 0.012 m out from the face.
    /// </summary>
    public static Bounds PlateBounds(in Fixture x)
    {
        var half = PlateHalf(x.kit);
        if ((x.flags & FixtureFlags.Horizontal) != 0) half = new Vector2(half.y, half.x);
        half.x += .02f;
        var n = x.normal;
        var across = new Vector3(Mathf.Abs(n.z), 0f, Mathf.Abs(n.x));
        const float proud = .012f;
        return new Bounds(x.position + n * (proud * .5f), (across * half.x + Vector3.up * half.y + new Vector3(Mathf.Abs(n.x), 0f, Mathf.Abs(n.z)) * (proud * .5f)) * 2f);
    }

    /// <summary>
    /// Stage 2, rule 1: true when a wall unit's box (its own set's occluders, or any enabled set's)
    /// meets the plate volume of <paramref name="x"/>, a fixture of <paramref name="set"/>. Floor boxes never are.
    /// </summary>
    public static bool IsOccluded(FrontRoomsWallFixtureSet set, in Fixture x) => IsOccludedBy(set, x, FrontRoomsWallFixtureSet.Registry);

    /// <summary>As IsOccluded, testing only the occluders of <paramref name="set"/> itself and of <paramref name="others"/>.</summary>
    public static bool IsOccludedBy(FrontRoomsWallFixtureSet set, in Fixture x, List<FrontRoomsWallFixtureSet> others)
    {
        if (!occlusion || set == null || x.cls == FixtureClass.Floor) return false;
        var plate = PlateBounds(x);
        if (Meets(set.occluders, plate)) return true;
        if (others == null || others.Count == 0) return false;
        Bounds world = default;
        var haveWorld = false;
        for (var s = 0; s < others.Count; s++)
        {
            var other = others[s];
            if (other == null || other == set || other.occluders.Count == 0) continue;
            if (!haveWorld) { world = XformBounds(set.transform.localToWorldMatrix, plate); haveWorld = true; }
            if (Meets(other.occluders, XformBounds(other.transform.worldToLocalMatrix, world))) return true;
        }
        return false;
    }

    /// <summary>The world box round every occluder of <paramref name="set"/> (empty bounds when it has none).</summary>
    public static Bounds OccluderWorldBounds(FrontRoomsWallFixtureSet set)
    {
        var m = set.transform.localToWorldMatrix;
        var b = new Bounds();
        for (var i = 0; i < set.occluders.Count; i++)
        {
            var w = XformBounds(m, set.occluders[i]);
            if (i == 0) b = w; else b.Encapsulate(w);
        }
        return b;
    }

    static bool Meets(List<Bounds> boxes, in Bounds b)
    {
        for (var i = 0; i < boxes.Count; i++) if (boxes[i].Intersects(b)) return true;
        return false;
    }

    static Bounds XformBounds(in Matrix4x4 m, in Bounds b)
    {
        var c = m.MultiplyPoint3x4(b.center);
        var e = b.extents;
        var x = new Vector3(Mathf.Abs(m.m00), Mathf.Abs(m.m10), Mathf.Abs(m.m20)) * e.x;
        var y = new Vector3(Mathf.Abs(m.m01), Mathf.Abs(m.m11), Mathf.Abs(m.m21)) * e.y;
        var z = new Vector3(Mathf.Abs(m.m02), Mathf.Abs(m.m12), Mathf.Abs(m.m22)) * e.z;
        return new Bounds(c, (x + y + z) * 2f);
    }

    static bool PowersPoint(in Fixture x) => (IsReceptacle(x.kit) && x.kit != FixtureKit.Blank) || x.kit == FixtureKit.FloorBox || x.kit == FixtureKit.FloorBox2;

    /// <summary>
    /// Stage 2, rule 2 (10_spec.md §2.9; ON since R-4): make sure a drawn receptacle or floor box lies
    /// within <see cref="StationReach"/> (xz) of <paramref name="deskWorld"/>, a desk of a wall row the
    /// Office kit just placed. If none does, add one receptacle on the nearest own Wall or Arch face of
    /// <paramref name="set"/> that keeps every stage-1 rule: 0.15 m grid, 0.30 m from face ends,
    /// 0.20 m from an arch, 0.30 m from the face's other fixtures, 0.61 m from the other face of the wall,
    /// clear of keep-outs and wall units; the face's line height and finish, like the plan. Never on
    /// Door or Window faces, never on a foreign face (the neighbour plans those). Deterministic.
    /// Returns true when the desk is powered afterwards.
    /// </summary>
    public static bool EnsureReceptacleNear(FrontRoomsWallFixtureSet set, Vector3 deskWorld)
    {
        if (set == null || Suppressed) return false;
        var p = set.transform.InverseTransformPoint(deskWorld);
        var reach2 = StationReach * StationReach;
        var fixtures = set.fixtures;
        for (var i = 0; i < fixtures.Count; i++)
        {
            var x = fixtures[i];
            if (!PowersPoint(x)) continue;
            float dx = x.position.x - p.x, dz = x.position.z - p.z;
            if (dx * dx + dz * dz <= reach2 && !IsOccluded(set, x)) return true;
        }
        var faces = set.faces;
        int bestFace = -1;
        float bestU = 0f, bestD2 = float.MaxValue;
        for (var fi = 0; fi < faces.Count; fi++)
        {
            var f = faces[fi];
            if (f.foreign || (f.kind != EdgeKind.Wall && f.kind != EdgeKind.Arch)) continue;
            var off = Vector3.Dot(p - f.start, f.normal);
            if (off < 0f || off > StationReach) continue;   // the desk stands in front of the face it would use
            var gaps = PlateGaps(f);
            if (gaps.n == 0) continue;
            var uStar = Vector3.Dot(p - f.start, f.direction);
            var k0 = Mathf.RoundToInt(uStar / Step);
            var steps = Mathf.CeilToInt(StationReach / Step);
            var twin = TwinOf(faces, fi);
            for (var d = 0; d <= steps; d++)
            for (var s = -1; s <= 1; s += 2)
            {
                if (d == 0 && s > 0) continue;
                var u = (k0 + s * d) * Step;
                if (!gaps.Inside(u)) continue;
                var q = f.start + f.direction * u;
                float dx = q.x - p.x, dz = q.z - p.z;
                var d2 = dx * dx + dz * dz;
                if (d2 > reach2 || d2 >= bestD2 - 1e-6f) continue;
                if (!FarFromSameFace(fixtures, fi, u) || (twin >= 0 && !TwinClear(fixtures, twin, u))) continue;
                if (ThroughWall(faces, p, q + f.normal * .05f)) continue;   // same room only: never through a wall
                var x = Stage2Receptacle(set, f, fi, u);
                if (Blocked(x, f, set.keepOut) || IsOccluded(set, x)) continue;
                bestFace = fi; bestU = u; bestD2 = d2;
            }
        }
        if (bestFace < 0) return false;
        set.AddStage2(Stage2Receptacle(set, faces[bestFace], bestFace, bestU));
        StatStage2++;
        return true;
    }

    /// <summary>Stage-2 receptacles added by EnsureReceptacleNear since the counter was reset (tools and tests).</summary>
    public static int StatStage2;

    /// <summary>
    /// True when the floor-plan segment a -> b crosses a wall of the face list (its cell line, 3 m),
    /// outside an arch opening. Walls built by the west and south neighbours are not in the list, but
    /// a desk and an own face of this chunk never lie on opposite sides of one of those.
    /// </summary>
    static bool ThroughWall(List<WallFace> faces, Vector3 a, Vector3 b)
    {
        float ax = a.x, az = a.z, dx = b.x - a.x, dz = b.z - a.z;
        for (var i = 0; i < faces.Count; i++)
        {
            var g = faces[i];
            if (g.side != 0 && i > 0 && faces[i - 1].edgeId == g.edgeId) continue;   // one test per edge
            var w0 = g.start - g.normal * ModuleUnits.WallHalf;
            float ex = g.direction.x * g.length, ez = g.direction.z * g.length;
            var den = dx * ez - dz * ex;
            if (Mathf.Abs(den) < 1e-6f) continue;   // parallel
            float qx = w0.x - ax, qz = w0.z - az;
            var s = (qx * ez - qz * ex) / den;      // along a -> b
            var t = (qx * dz - qz * dx) / den;      // along the wall, 0..1
            if (s <= 1e-4f || s >= 1f - 1e-4f || t < 0f || t > 1f) continue;
            var u = t * g.length;
            if (g.kind == EdgeKind.Arch && u > g.openFrom && u < g.openTo) continue;   // through the opening
            return true;
        }
        return false;
    }

    static int TwinOf(List<WallFace> faces, int fi)
    {
        var f = faces[fi];
        for (var k = 0; k < faces.Count; k++)
            if (k != fi && faces[k].edgeId == f.edgeId && faces[k].side != f.side) return k;
        return -1;
    }

    /// <summary>A forced receptacle at <paramref name="u"/> on face <paramref name="fi"/>: the face's class rule, wall-line height and finish, no wear.</summary>
    static Fixture Stage2Receptacle(FrontRoomsWallFixtureSet set, in WallFace f, int fi, float u)
    {
        var cls = set.isLocal ? set.streamClass : Classify(f);
        var rule = Rules[(int)cls];
        var seed = set.seed;
        var hL = LineHash(seed, f);
        var y = U(hL, 2) < rule.pAlt ? rule.hAlt : rule.h;
        var plate = PlateFinish(seed, f, rule);
        var hD = FaceHash(seed, f, Field.Forced);
        var kit = plate == Finish.Stainless ? FixtureKit.DuplexSteel : (U(hD) < rule.p20 ? FixtureKit.Duplex20 : FixtureKit.Duplex);
        var device = rule.fixedFinish ? rule.finish : DeviceFor(plate);
        if (rule.pIG > 0f && U(hD, 1) < rule.pIG) { kit = plate == Finish.Stainless ? FixtureKit.DuplexSteel : FixtureKit.Duplex20; device = Finish.Orange; }
        var x = Make(f, fi, cls, kit, u, y, plate, hD);
        x.device = device;
        x.flags |= FixtureFlags.Forced;
        x.position.y = Mathf.Max(x.position.y, f.baseTop + BaseClear + PlateHalfH);
        return x;
    }

    // ---------- Keep-outs (module props, key spot, Relay entries) ----------

    static readonly Dictionary<string, float[]> FootprintCache = new Dictionary<string, float[]>();

    /// <summary>Forget cached kit footprints (the editor calls this when kits are re-imported).</summary>
    public static void ClearCaches()
    {
        FootprintCache.Clear();
        KitPresence.Clear();
    }

    static float[] Footprint(string kit)
    {
        if (FootprintCache.TryGetValue(kit, out var f)) return f;
        f = FrontRoomsMapWorld.KitFootprint(kit);
        FootprintCache[kit] = f;
        return f;
    }

    /// <summary>Module props (any height up to 2.1 m), the key spot and the Relay entries: chunk-local volumes.</summary>
    static void KeepOuts(MapChunk data, List<KeepOut> list)
    {
        var cs = MapGrid.CellSize;
        for (var r = 0; r < data.rooms.Length; r++)
        {
            var module = data.ModuleOf(r);
            if (module == null) continue;
            var room = data.rooms[r];
            float ox = room.x * cs, oz = room.y * cs;
            if (module.props != null)
                for (var pi = 0; pi < module.props.Length; pi++)
                {
                    var p = module.props[pi];
                    if (string.IsNullOrEmpty(p.kit) || p.y > 2.1f) continue;
                    var f = Footprint(p.kit);
                    if (f == null) continue;
                    PropBounds(p, f, out var x0, out var z0, out var x1, out var z1);
                    list.Add(new KeepOut { xz = Rect.MinMaxRect(ox + x0, oz + z0, ox + x1, oz + z1), y0 = p.y, y1 = p.y + (f.Length > 4 ? f[4] : 1f) });
                }
            if (module.markers != null)
                for (var mi = 0; mi < module.markers.Length; mi++)
                {
                    var mk = module.markers[mi];
                    if (mk.kind == ModuleMarkerKind.RelayEntry)
                        list.Add(new KeepOut { xz = new Rect(ox + mk.x - .8f, oz + mk.z - .8f, 1.6f, 1.6f), y0 = 0f, y1 = 2.1f });
                }
        }
        if (data.hasKey && data.keySpot)
            list.Add(new KeepOut { xz = new Rect(data.keyX - .6f, data.keyZ - .6f, 1.2f, 1.2f), y0 = 0f, y1 = 2.1f });
    }

    /// <summary>RoomModuleData.Bounds without its two temporary arrays: the same corners in the same order.</summary>
    static void PropBounds(in ModuleProp p, float[] f, out float x0, out float z0, out float x1, out float z1)
    {
        var rad = p.yaw * (Math.PI / 180.0);
        float c = (float)Math.Cos(rad), s = (float)Math.Sin(rad);
        x0 = z0 = float.MaxValue;
        x1 = z1 = float.MinValue;
        for (var a = 0; a < 2; a++)
        for (var b = 0; b < 2; b++)
        {
            var cx = a == 0 ? f[0] : f[2];
            var cz = b == 0 ? f[1] : f[3];
            var wx = p.x + cx * c + cz * s;
            var wz = p.z - cx * s + cz * c;
            if (wx < x0) x0 = wx; if (wx > x1) x1 = wx;
            if (wz < z0) z0 = wz; if (wz > z1) z1 = wz;
        }
    }

    // ---------- Kit presence (quiet: no "missing model" warning for kits not built yet) ----------

    static readonly Dictionary<string, bool> KitPresence = new Dictionary<string, bool>();

    /// <summary>True when Resources/Props/Models/&lt;name&gt; exists. Cached; never logs.</summary>
    public static bool KitExists(string name)
    {
        if (string.IsNullOrEmpty(name)) return false;
        if (KitPresence.TryGetValue(name, out var has)) return has;
        has = Resources.Load<GameObject>(FrontRoomsKitLibrary.ModelFolder + name) != null;
        KitPresence[name] = has;
        return has;
    }
