"""Q16 placard render stage: post-processing (sheets, curves, JPG copies).
Usage: W/venv/bin/python post.py <render root> <doc images dir>
<render root> holds main_P1/, a9_P2/, a9_P3/ from FrontRoomsPlacardRender (clone tool)."""
import csv

FONT_PATHS = ["/System/Library/Fonts/Supplemental/Arial.ttf", "/Library/Fonts/Arial.ttf", "/System/Library/Fonts/Helvetica.ttc"]
MONO_PATHS = ["/System/Library/Fonts/Menlo.ttc", "/System/Library/Fonts/Monaco.ttf"]


def font(size, mono=False):
    for p in (MONO_PATHS if mono else FONT_PATHS):
        if os.path.exists(p):
            try:
                return ImageFont.truetype(p, size)
            except Exception:
                pass
    return ImageFont.load_default()


def jpg(src, name, maxw=1920):
    im = Image.open(src).convert("RGB")
    if im.width > maxw:
        im = im.resize((maxw, round(im.height * maxw / im.width)), Image.LANCZOS)
    dst = os.path.join(OUT, name + ".jpg")
    im.save(dst, quality=85, optimize=True)
    return dst


def frames_meta(d):
    meta = {}
    p = os.path.join(d, "frames.txt")
    if not os.path.exists(p):
        return meta
    for line in open(p):
        name = line.split(" | ")[0].strip()
        g = None
        if " G " in line:
            try:
                g = float(line.split(" G ")[1].split()[0])
            except Exception:
                pass
        meta[name] = (g, line.strip())
    return meta


def crop_centre(im, w, h):
    cx, cy = im.width // 2, im.height // 2
    return im.crop((cx - w // 2, cy - h // 2, cx + w // 2, cy + h // 2))


def label(img, text, pos=(10, 8), size=22, fill=(255, 255, 255), bg=(10, 10, 10)):
    d = ImageDraw.Draw(img)
    f = font(size)
    l, t, r, b = d.textbbox((0, 0), text, font=f)
    x, y = pos
    d.rectangle((x - 6, y - 4, x + (r - l) + 6, y + (b - t) + 8), fill=bg)
    d.text((x, y - t), text, font=f, fill=fill)

# ---------------------------------------------------------------- curves from series.csv
data = defaultdict(list)
for sub in ("main_P1", "a9_P2", "a9_P3"):
    p = os.path.join(ROOT, sub, "series.csv")
    if os.path.exists(p):
        for row in csv.DictReader(open(p)):
            data[row["tag"]].append({k: (float(v) if k not in ("tag", "mode") else v) for k, v in row.items()})


def panel(draw, box, title, curves, tmax, ymax=1.05, hlines=()):
    x0, y0, x1, y1 = box
    f = font(18)
    fs = font(14)
    draw.text((x0, y0 - 28), title, font=f, fill=(10, 10, 10))
    draw.rectangle(box, outline=(160, 160, 160))
    for yv in [0, .25, .5, .75, 1.0]:
        if yv > ymax:
            continue
        yy = y1 - (y1 - y0) * yv / ymax
        draw.line((x0, yy, x1, yy), fill=(232, 232, 232))
        draw.text((x0 - 34, yy - 8), "%.2f" % yv, font=fs, fill=(90, 90, 90))
    step = 1 if tmax <= 6 else (2 if tmax <= 14 else 5)
    t = 0
    while t <= tmax + 1e-6:
        xx = x0 + (x1 - x0) * t / tmax
        draw.line((xx, y1, xx, y1 + 5), fill=(90, 90, 90))
        draw.text((xx - 8, y1 + 8), "%g" % t, font=fs, fill=(90, 90, 90))
        t += step
    for yv, col in hlines:
        yy = y1 - (y1 - y0) * yv / ymax
        draw.line((x0, yy, x1, yy), fill=col, width=1)
    lx, ly = x0, y1 + 30
    for name, pts, col, w in curves:
        if not pts:
            continue
        xy = [(x0 + (x1 - x0) * min(t, tmax) / tmax, y1 - (y1 - y0) * max(0.0, min(v, ymax)) / ymax) for t, v in pts if t <= tmax]
        if len(xy) > 1:
            draw.line(xy, fill=col, width=w)
        tw = draw.textbbox((0, 0), name, font=fs)[2]
        if lx + 20 + tw > x1:
            lx, ly = x0, ly + 20
        draw.rectangle((lx, ly + 4, lx + 14, ly + 14), fill=col)
        draw.text((lx + 20, ly), name, font=fs, fill=(10, 10, 10))
        lx += 20 + tw + 18


def series(tag, key, shift=0.0):
    return [(r["t"] + shift, r[key]) for r in data.get(tag, [])]
