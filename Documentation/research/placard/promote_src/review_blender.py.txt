artwork on the sheet, and renders named views. Views are Unity part space.

def U(X, Y, Z):

def mat_alu():
    m = bpy.data.materials.new("rv_alu")
    m.use_nodes = True
    b = m.node_tree.nodes["Principled BSDF"]
    b.inputs["Base Color"].default_value = (0.91, 0.92, 0.925, 1)
    b.inputs["Metallic"].default_value = 1.0
    b.inputs["Roughness"].default_value = 0.40
    return m


def mat_print(glow=0.0):
    m = bpy.data.materials.new("rv_print")
    m.use_nodes = True
    nt = m.node_tree
    b = nt.nodes["Principled BSDF"]
    img = nt.nodes.new("ShaderNodeTexImage")
    img.image = bpy.data.images.load(os.path.join(TEX, "Prop_EvacPlan_A.png"))
    img.interpolation = "Cubic"
    nt.links.new(img.outputs["Color"], b.inputs["Base Color"])
    b.inputs["Roughness"].default_value = 0.85
    if glow > 0:
        e = nt.nodes.new("ShaderNodeTexImage")
        e.image = bpy.data.images.load(os.path.join(TEX, "Prop_EvacPlan_E.png"))
        e.image.colorspace_settings.name = "Non-Color"
        mul = nt.nodes.new("ShaderNodeMix")
        mul.data_type = "RGBA"
        mul.blend_type = "MULTIPLY"
        mul.inputs["Factor"].default_value = 1.0
        nt.links.new(e.outputs["Color"], mul.inputs[6])
        mul.inputs[7].default_value = (0.5705, 0.8900, 0.2957, 1)
        nt.links.new(mul.outputs[2], b.inputs["Emission Color"])
        b.inputs["Emission Strength"].default_value = glow
    return m


def mat_lens():
    m = bpy.data.materials.new("rv_lens")
    m.use_nodes = True
    nt = m.node_tree
    for n in list(nt.nodes):
        if n.type != "OUTPUT_MATERIAL":
            nt.nodes.remove(n)
    out = [n for n in nt.nodes if n.type == "OUTPUT_MATERIAL"][0]
    tr = nt.nodes.new("ShaderNodeBsdfTransparent")
    gl = nt.nodes.new("ShaderNodeBsdfGlossy")
    gl.inputs["Roughness"].default_value = 0.45
    fr = nt.nodes.new("ShaderNodeFresnel")
    fr.inputs["IOR"].default_value = 1.49
    mix = nt.nodes.new("ShaderNodeMixShader")
    nt.links.new(fr.outputs[0], mix.inputs[0])
    nt.links.new(tr.outputs[0], mix.inputs[1])
    nt.links.new(gl.outputs[0], mix.inputs[2])
    nt.links.new(mix.outputs[0], out.inputs["Surface"])
    return m


def flat(name, rgb, emit=1.0):
    m = bpy.data.materials.new(name)
    m.use_nodes = True
    nt = m.node_tree
    for n in list(nt.nodes):
        if n.type != "OUTPUT_MATERIAL":
            nt.nodes.remove(n)
    out = [n for n in nt.nodes if n.type == "OUTPUT_MATERIAL"][0]
    em = nt.nodes.new("ShaderNodeEmission")
    em.inputs["Color"].default_value = (*rgb, 1)
    em.inputs["Strength"].default_value = emit
    nt.links.new(em.outputs[0], out.inputs["Surface"])
    return m

if FBX:
    # The exported asset itself (the FBX round trip): it replaces the LOD0 parts in every LOD0 view.
    before = set(bpy.data.objects)
    bpy.ops.import_scene.fbx(filepath=FBX)
    imp = [o for o in bpy.data.objects if o not in before and o.type == "MESH"]
    assert len(imp) == 1, imp
    fo = imp[0]
    for i, m in enumerate(fo.data.materials):
        fo.data.materials[i] = {pc.ALU: ALU, pc.PRINT: PRINT, pc.LENS: LENS}[m.name.split(".")[0]]
    for o in LOD[0]:
        o.hide_render = True
    kit.parts.append(fo)
    LOD[0] = [fo]
    # Round-trip check: every loop on the four rails' FACE arc must carry the arc's true normal.
    import mathutils
    R, c, ang = pc.face_arc()
    me = fo.data
    mw = fo.matrix_world
    rot = mw.to_3x3()
    worst, count = 0.0, 0
    for poly in me.polygons:
        for li in poly.loop_indices:
            co = mw @ me.vertices[me.loops[li].vertex_index].co
            X, Y, Z = -co.x / pc.MM, co.z / pc.MM, -co.y / pc.MM
            for side in ("top", "bottom", "left", "right"):
                nn, t, E, L = pc.SIDES[side]
                u = E - (nn[0] * X + nn[1] * Y)
                along = t[0] * X + t[1] * Y
                if not (7.0 < u < 23.0 and abs(along) < L - 30.0):
                    continue
                w = Z
                r = math.hypot(u - c[0], w - c[1])
                if abs(r - R) > 0.02:
                    continue
                want_uw = ((u - c[0]) / r, (w - c[1]) / r)
                want = mathutils.Vector(pc.side_normal(side, want_uw))
                ln = rot @ me.corner_normals[li].vector
                got = mathutils.Vector((-ln.x, ln.z, -ln.y)).normalized()
                d = math.degrees(want.angle(got))
                # Only loops of faces lying on this side's face arc.
                if (rot @ poly.normal).normalized().dot(mathutils.Vector(pc.bl_dir(pc.side_normal(side, want_uw)))) < 0.9:
                    continue
                worst = max(worst, d)
                count += 1
    print("[review] FBX round trip: %d face-arc loops, worst normal deviation %.3f deg" % (count, worst))

def add_plane(name, size, loc_u, mat, facing="+Z"):
    bm = bmesh.new()
    bmesh.ops.create_grid(bm, x_segments=1, y_segments=1, size=size / 2)
    me = bpy.data.meshes.new(name)
    bm.to_mesh(me)
    bm.free()
    ob = bpy.data.objects.new(name, me)
    scene.collection.objects.link(ob)
    ob.rotation_euler = (math.radians(90), 0, 0)      # grid normal +z -> Blender -y = Unity +Z
    ob.location = U(*loc_u)
    me.materials.append(mat)
    return ob

def camera(name, eye, target, fov=None, ortho=None, res=(1920, 1080), clip=None, lens_shift=(0, 0)):
    cd = bpy.data.cameras.new(name)
    if ortho:
        cd.type = "ORTHO"
        cd.ortho_scale = ortho
    else:
        cd.sensor_fit = "VERTICAL"
        cd.angle_y = math.radians(fov)
    if clip:
        cd.clip_start = clip
    cd.clip_end = 50
    if clip is None:
        cd.clip_start = 0.002
    cd.shift_x, cd.shift_y = lens_shift
    cam = bpy.data.objects.new(name, cd)
    scene.collection.objects.link(cam)
    cam.location = U(*eye)
    d = U(*target) - U(*eye)
    cam.rotation_euler = d.to_track_quat("-Z", "Y").to_euler()
    scene.camera = cam
    scene.render.resolution_x, scene.render.resolution_y = res
    scene.render.resolution_percentage = 100
    return cam


def show_lod(lv, with_lens=True):

def render(name):
    scene.render.filepath = os.path.join(OUT, (TAG + "_" if TAG else "") + name + ".png")
    bpy.ops.render.render(write_still=True)
    print("[review] wrote", scene.render.filepath)


def section_polys():
    """Flat section fills at the top rail's mid-length (Unity X = 0)."""
    objs = []
    def poly(name, pts_uw, rgb, xoff):
        bm = bmesh.new()
        vs = [bm.verts.new(U(xoff, (pc.B_HALF - u) * pc.MM, w * pc.MM)) for u, w in pts_uw]
        bm.faces.new(vs)
        me = bpy.data.meshes.new(name)
        bm.to_mesh(me)
        bm.free()
        ob = bpy.data.objects.new(name, me)
        scene.collection.objects.link(ob)
        me.materials.append(flat("m_" + name, rgb))
        objs.append(ob)
    x = -0.0001
    poly("sec rail", pc.profile_lod0(), (0.75, 0.10, 0.08), x)
    poly("sec base", pc.BASE_PROFILE, (0.85, 0.45, 0.10), x)
    sh = pc.SHEET_TO_OUTSIDE
    poly("sec sheet", [(sh, 4.40), (60, 4.40), (60, pc.SHEET_W), (sh, pc.SHEET_W)], (0.95, 0.93, 0.85), x)
    poly("sec lens", [(sh, pc.SHEET_W), (60, pc.SHEET_W), (60, pc.LENS_TOP), (sh, pc.LENS_TOP)], (0.35, 0.65, 0.85), x)
    poly("sec board", [(1.6, 1.20), (60, 1.20), (60, 4.40), (1.6, 4.40)], (0.30, 0.22, 0.15), x)
    return objs
