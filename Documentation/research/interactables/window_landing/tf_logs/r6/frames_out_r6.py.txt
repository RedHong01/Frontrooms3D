#!/usr/bin/env python3
"""Window landing r6 frames -> JPG (q85) into a staging folder. before = proj_win kitoff (the map's look before the kit:
trims + 30 mm pane cube), after = proj_win kit (main 279c144 + contract r6 R1 R2 T + facade r6 + Prop_Putty).
One BEFORE|AFTER pair per camera, the single AFTER frame for the hero views, and the Play frames.
Also compares each r6 frame with its r5 twin (mean |diff| luma) when the r5 JPG exists."""
import os, sys
from PIL import Image, ImageDraw, ImageFont, ImageStat
import numpy as np
W = "/Users/redwang/FrontRoomsVisualWork"
BASE = W + "/proj_win/Verification/window_tf4/frames_kitoff"
KIT = W + "/proj_win/Verification/window_tf4/frames_kit"
OUT = sys.argv[1]
R5 = sys.argv[2] if len(sys.argv) > 2 else None
os.makedirs(OUT, exist_ok=True)
font = ImageFont.truetype("/System/Library/Fonts/Supplemental/Courier New Bold.ttf", 26)
lum = lambda im: ImageStat.Stat(im.convert("L")).mean[0]
HERO_SINGLE = {"L0_A_front_1.5m", "L0_B_front_1.5m", "OF_A_front_1.5m", "OF_B_front_1.5m", "L0_A_close_stop_0.3m_insp", "OF_B_close_head_0.3m"}
names = sorted(n for n in set(os.listdir(BASE)) | set(os.listdir(KIT)) if n.endswith(".png"))
rows = []
for n in names:
    stem = n[:-4]
    ib = Image.open(os.path.join(BASE, n)).convert("RGB") if os.path.exists(os.path.join(BASE, n)) else None
    ik = Image.open(os.path.join(KIT, n)).convert("RGB") if os.path.exists(os.path.join(KIT, n)) else None
    if ik and (stem in HERO_SINGLE or ib is None): ik.save(os.path.join(OUT, f"r6_{stem}_after.jpg"), quality=85)
    d5 = ""
    if ib and ik:
        w, h = 960, 540
        sheet = Image.new("RGB", (w * 2 + 8, h), (20, 20, 18))
        sheet.paste(ib.resize((w, h), Image.LANCZOS), (0, 0)); sheet.paste(ik.resize((w, h), Image.LANCZOS), (w + 8, 0))
        d = ImageDraw.Draw(sheet)
        for x, t in ((12, "BEFORE"), (w + 20, "AFTER")):
            d.rectangle([x - 6, 8, x + 16 * len(t) + 8, 44], fill=(0, 0, 0)); d.text((x, 12), t, fill=(235, 235, 225), font=font)
        p6 = os.path.join(OUT, f"r6_{stem}_pair.jpg")
        sheet.save(p6, quality=85)
        if R5 and os.path.exists(os.path.join(R5, f"r5_{stem}_pair.jpg")):
            a = np.asarray(Image.open(os.path.join(R5, f"r5_{stem}_pair.jpg")).convert("L"), dtype=np.float32)
            b = np.asarray(Image.open(p6).convert("L"), dtype=np.float32)
            if a.shape == b.shape:
                dd = np.abs(a - b)
                d5 = f"  vs r5 pair: mean |dL| {dd.mean():5.2f}, px > 16: {(dd > 16).mean() * 100:5.2f} %"
    rows.append(f"{stem:44s} luma before {lum(ib) if ib else -1:6.1f}  after {lum(ik) if ik else -1:6.1f}{d5}")
open(os.path.join(OUT, "frames_r6.txt"), "w").write("\n".join(rows) + "\n")
print("\n".join(rows)); print(len(names), "cameras")
