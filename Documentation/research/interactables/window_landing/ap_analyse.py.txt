import re, json, statistics as st
S = "/private/tmp/claude-501/-Users-redwang-Desktop-ArtCenter-Fall26T7-EGAM-401A-01-Individual-Game-Project/5656cffd-bc90-45f6-86a3-09b26549df8d/scratchpad/win_r5/autopilot/"
out = []
def q(v, p): v = sorted(v); return v[min(len(v) - 1, int(p * len(v)))] if v else float('nan')
for t in ["ap_kit_2554", "ap_off_2554", "ap_kit_20388", "ap_off_20388"]:
    rep = json.load(open(S + t + "/report.json"))
    lines = open(S + t + "/" + t + "_work.txt").read().splitlines()
    head = lines[0]
    chunks, rooms, spikes, spikes_work = [], [], 0, 0
    over33 = over50 = 0; over50_work = []
    for l in lines[1:]:
        p = l.split("\t")
        if len(p) < 3: continue
        ms = float(p[2]); work = p[3] if len(p) > 3 else ""
        t_ = float(p[1])
        for m in re.finditer(r"chunk \((-?\d+), (-?\d+)\) ([\d.]+) ms", work): chunks.append(float(m.group(3)))
        for m in re.finditer(r"room \d+ of \((-?\d+), (-?\d+)\) ([\d.]+) ms", work): rooms.append(float(m.group(3)))
        if ms > 33: over33 += 1
        if ms > 50:
            over50 += 1
            if work: over50_work.append(f"{ms:.0f} ms [{work}]")
    s = (f"{t}: avg fps {rep['averageFps']:.1f}, p95 {rep['p95FrameMs']:.1f} ms, p99 {rep['p99FrameMs']:.1f} ms, worst {rep['worstFrameMs']:.0f} ms; "
         f"frames >33 ms {over33}, >50 ms {over50} (with map work: {len(over50_work)}); chunk builds {len(chunks)}: median {st.median(chunks):.1f} ms, p90 {q(chunks,.9):.1f}, max {max(chunks):.1f}; "
         f"room dresses {len(rooms)}: median {st.median(rooms):.1f}, max {max(rooms):.1f}; spikes listed {len(rep['frameSpikes'])}; errors {rep['errors']}; tier {rep['tier']}, zones {rep['zonesVisited']}, cells {rep['cellsVisited']}, office rooms {rep['officeRoomsDressed']}; {head.split(' · ')[2]} / {head.split(' · ')[3]}")
    out.append(s); print(s)
    for w in over50_work[:4]: print("     ", w)
open(S + "../ap_summary.txt", "w").write("\n".join(out) + "\n")
