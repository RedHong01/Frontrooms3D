def font(px):
    try:
        return ImageFont.truetype(FONT, px)
    except Exception:
        return ImageFont.load_default()


def load(name):
    if name.startswith("BEFORE:"):
        p = os.path.join(BEFORE, name[7:] + ".png")
        return Image.open(p).convert("RGB") if BEFORE and os.path.exists(p) else None
    p = os.path.join(SRC, name + ".png")
    return Image.open(p).convert("RGB") if os.path.exists(p) else None


def lin(a):
    a = a / 255.0
    return np.where(a <= 0.04045, a / 12.92, ((a + 0.055) / 1.055) ** 2.4)


def lum(img, rect):
    x0, y0, x1, y1 = [int(round(v)) for v in rect]
    w, h = img.size
    x0, x1 = max(0, min(x0, x1)), min(w, max(x0, x1))
    y0, y1 = max(0, min(y0, y1)), min(h, max(y0, y1))
    if x1 - x0 < 2 or y1 - y0 < 2:
        return None
    a = lin(np.asarray(img.crop((x0, y0, x1, y1)), dtype=np.float64))
    return float((0.2126 * a[..., 0] + 0.7152 * a[..., 1] + 0.0722 * a[..., 2]).mean())


SHEET_JOBS = []


def sheet(name, rows, cell_w=640, label_px=22, title=None, notes=None):
    """Deferred: recorded here, rendered in parallel at the end (the host runs ~800 load; one process gets ~2 % of a core)."""
    SHEET_JOBS.append((name, rows, cell_w, label_px, title, notes))
    return name


def _sheet_now(job):
    name, rows, cell_w, label_px, title, notes = job
    """rows: list of lists of (frame name or Image, caption)."""
    cols = max(len(r) for r in rows)
    cell_h = int(cell_w * 1080 / 1920)
    pad, cap = 8, label_px + 10
    top = (label_px * 2 + 16) if title else 0
    bottom = (len(notes) * (label_px + 6) + 12) if notes else 0
    W = cols * (cell_w + pad) + pad
    H = top + len(rows) * (cell_h + cap + pad) + pad + bottom
    out = Image.new("RGB", (W, H), (238, 236, 230))
    d = ImageDraw.Draw(out)
    f = font(label_px)
    if title:
        d.text((pad, 8), title, fill=(10, 10, 10), font=font(int(label_px * 1.3)))
    for r, row in enumerate(rows):
        for c, item in enumerate(row):
            if item is None:
                continue
            src, caption = item
            if isinstance(src, str) and src.startswith("MASKX6:"):
                im = load(src[7:])
                if im is not None:
                    im = Image.fromarray(np.clip(np.asarray(im, dtype=np.float32) * 6.0, 0, 255).astype(np.uint8))   # amplify sub-pixel slits
            else:
                im = load(src) if isinstance(src, str) else src
            x = pad + c * (cell_w + pad)
            y = top + pad + r * (cell_h + cap + pad)
            if im is None:
                d.rectangle([x, y + cap, x + cell_w, y + cap + cell_h], fill=(200, 60, 50))
                d.text((x + 6, y + cap + 6), "missing " + str(src), fill=(255, 255, 255), font=f)
            else:
                out.paste(im.resize((cell_w, cell_h), Image.BILINEAR, reducing_gap=2.0), (x, y + cap))
            d.text((x + 2, y + 2), caption, fill=(10, 10, 10), font=f)
    if notes:
        y = H - bottom + 6
        for n in notes:
            d.text((pad, y), n, fill=(10, 10, 10), font=f)
            y += label_px + 6
    out.save(os.path.join(SHEETS, name + ".png"))
    return name

def to_jpg(src, dst):
    im = Image.open(src).convert("RGB")
    if im.width > 1920:
        im = im.resize((1920, int(im.height * 1920 / im.width)), Image.BILINEAR, reducing_gap=2.0)
    im.save(dst, "JPEG", quality=85)

def _copy_now(pth):
    base = os.path.basename(pth)[:-4]
    if base.endswith("_mask"):
        Image.open(pth).convert("L").save(os.path.join(DST, RDIR, "masks", PFX + base + ".png"), optimize=True)
    else:
        to_jpg(pth, os.path.join(DST, RDIR, PFX + base + ".jpg"))

PHASES = os.environ.get("INT_POST_PHASES", "sheets,copy").split(",")

    if "sheets" in PHASES:
