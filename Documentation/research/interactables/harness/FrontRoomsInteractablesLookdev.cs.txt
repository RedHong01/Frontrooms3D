/// <summary>
/// PRIVATE CLONE ONLY (proj_int, interactables-kit render stage, 2026-10-07). Never merge into Red's project.
///
/// In-situ, in-engine renders of the interactables kit (Documentation/research/interactables/10_spec.md):
/// the kit FBX + sidecar anchors composed with the raw recipe of §6.2 (door root D, A2 hinge axis,
/// Leaf rig, face rules of §1.2 incl. handing S_sign = ±1), in rooms built by the KitLookdev copy
/// (FrontRoomsKitLookdevAccess.Room: the levels' own surfaces, troffer grid, game post, zone grade).
/// Every frame goes through the game's camera setup (FrontRoomsPostStack.ConfigureCamera, 4x MSAA,
/// 1920x1080); player frames use the game's 76° vertical FOV at the 1.62 m eye, the head dip uses
/// FrontRoomsShotTimings.Unlock (eye −0.25, 0.45 m off the lock, 38° down, FOV 62).
///
/// Groups (-intOnly a,b,...; default all):
///   probe       materials Unity binds per kit slot (materials_probe.json) + locked leaf material compare
///   lineup      every new asset on a neutral floor with a 1.80 m reference (labels.json for annotation)
///   family      Lobby / Office / Run / Exit, FREE | LOCKED side by side in matching rooms, 3 m and 8 m
///   readability free/locked alternating down an L0 corridor (lit, dim), an Office, a Run and an Exit corridor, 6/12/20 m (rects.json)
///   room        Level 0 room: L0-F and L0-K in a 1.0 x 2.1 opening at 1.5 m and 0.6 m; key push-in at 0.35/0.25 m
///   dip         head-dip key shot, 5 frames (standing 1.2 m -> pose P -> 0.30 m, key turning), L0-K and OF-K
///   lock        pose P lock sequence: key 0/50/100 % insertion, plug 0/45/90°, deadbolt thrown / retracted (head-on + profile)
///   Every frame sets the zone reflection cube the game would use in the camera's room (FrontRoomsZoneReflection.SetImmediate).
///   keys        zone keys on board / cabinet / hook / desk / cabinet top, 2 m and 0.6 m, L0 and Office; old cube
///   windows     window family W-L0/W-OF/W-RN/W-EX intact and broken, 1.5 m and oblique, matching rooms
///   fracture    crack stages 1-3, the shatter (pieces mid-air), floor glass after (STAND-IN fracture, see below)
///   gap         door gap test: old primitive vs kit door between a dim near and a bright far room, 7 cameras,
///               both faces; every frame also gets a leak mask (far room white, all else black) and a pixel count
///   swing       single-acting swing 0/5/30/60/95°, both handings: oblique, plan over the hinge jamb, stop-side latch jamb (0/30/95°); old door at 95°
///
/// The fracture set is NOT the glass-destruction track's (not built yet): the pieces come from that track's own
/// crack-graph prototype (glass/destruction/04_proto/web_proto_crack_graph.py), exported by
/// W/int_work/fracture_data.py to Verification/interactables_data/*_u.json, with teeth kept in the approved tooth band.
///
/// Batch: Unity -batchmode -projectPath proj_int -buildTarget OSXUniversal -executeMethod
///        FrontRoomsInteractablesLookdev.RunBatch [-intOut DIR] [-intOnly g1,g2] [-intLockedLeaf Kit_...] -quit
/// Output: Verification/interactables/*.png, frames.tsv, leaks.tsv, rects.json, labels.json, int_log.txt
/// </summary>

    public static void RunBatch()
    {
        var args = Environment.GetCommandLineArgs();
        string onlyArg = null;
        for (var i = 0; i < args.Length - 1; i++)
        {
            if (args[i] == "-intOut") outDir = args[i + 1];
            if (args[i] == "-intOnly") onlyArg = args[i + 1];
            if (args[i] == "-intLockedLeaf") lockedLeaf = args[i + 1];
        }
        var project = Directory.GetParent(Application.dataPath).FullName;
        if (string.IsNullOrEmpty(outDir)) outDir = Path.Combine(project, "Verification", "interactables");
        dataDir = Path.Combine(project, "Verification", "interactables_data");
        Directory.CreateDirectory(outDir);
        only = string.IsNullOrEmpty(onlyArg) ? null : new HashSet<string>(onlyArg.Split(','));

        EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
        world = new GameObject("INTERACTABLES LOOKDEV").transform;
        FrontRoomsLook.ApplyAmbient();
        FrontRoomsPostStack.Ensure(world);
        var co = new GameObject("INT camera");
        cam = co.AddComponent<Camera>();
        cam.nearClipPlane = .03f;
        cam.farClipPlane = 90f;
        cam.clearFlags = CameraClearFlags.SolidColor;
        cam.backgroundColor = Color.black;
        FrontRoomsPostStack.ConfigureCamera(cam);
        rt = new RenderTexture(Wpx, Hpx, 24, RenderTextureFormat.ARGB32) { antiAliasing = 4 };
        cam.targetTexture = rt;
        Log("locked leaf asset: " + lockedLeaf + "; urp " + (GraphicsSettings.currentRenderPipeline != null ? GraphicsSettings.currentRenderPipeline.name : "none"));

        Group("probe", Probe);
        Group("lineup", Lineup);
        Group("family", Family);
        Group("readability", Readability);
        Group("room", RoomInSitu);
        Group("dip", HeadDip);
        Group("lock", LockSequence);
        Group("keys", Keys);
        Group("windows", Windows);
        Group("fracture", Fracture);
        Group("gap", GapTest);
        Group("swing", Swing);

        cam.targetTexture = null;
        rt.Release();
        File.WriteAllText(Path.Combine(outDir, "frames.tsv"), framesTsv.ToString());
        File.WriteAllText(Path.Combine(outDir, "leaks.tsv"), leakTsv.ToString());
        File.WriteAllText(Path.Combine(outDir, "rects.json"), rectJson.Append("\n]\n").ToString());
        File.WriteAllText(Path.Combine(outDir, "labels.json"), labelJson.Append("\n]\n").ToString());
        File.WriteAllText(Path.Combine(outDir, "int_log.txt"), string.Join("\n", log) + "\n");
        Debug.Log("[IntLookdev] DONE " + outDir + " (" + log.Count + " log lines)");
    }

    static void Group(string name, Action a)
    {
        if (only != null && !only.Contains(name)) return;
        group = name;
        var t0 = EditorApplication.timeSinceStartup;
        try { a(); Log("group " + name + " ok in " + (EditorApplication.timeSinceStartup - t0).ToString("F1", Inv) + " s"); }
        catch (Exception e) { Log("FAILED group " + name + ": " + e); Debug.LogException(e); }
        // Each group builds under its own root; anything left behind is cleared before the next.
        zoneRooms.Clear();
        for (var i = world.childCount - 1; i >= 0; i--)
        {
            var c = world.GetChild(i);
            if (!c.name.StartsWith("Post /")) Object.DestroyImmediate(c.gameObject);
        }
    }

    static void Log(string s) { log.Add(s); Debug.Log("[IntLookdev] " + s); }

    static Transform G(string name)
    {
        var g = new GameObject(name).transform;
        g.SetParent(world, false);
        return g;
    }

    // ======================================================================================== rendering

    /// <summary>Every lookdev room's world bounds and the reflection cube the game would use there (FrontRooms3DGame.UpdateReflectionZone).</summary>
    static readonly List<(Bounds b, FrontRoomsLook.ReflectionZone z)> zoneRooms = new List<(Bounds, FrontRoomsLook.ReflectionZone)>();

    /// <summary>The game's zone reflection for the camera's room: Office rooms the Office cube, Tall rooms Tall, the rest Level 0.</summary>
    static FrontRoomsLook.ReflectionZone ZoneFor(Vector3 pos)
    {
        foreach (var (b, z) in zoneRooms) if (b.Contains(pos)) return z;
        return FrontRoomsLook.ReflectionZone.Level0;
    }

    static void Shot(string file, Vector3 pos, Quaternion rot, float fov, string note = "")
    {
        var zone = ZoneFor(pos);
        FrontRoomsZoneReflection.SetImmediate(zone);
        note = (note + " · refl " + zone).Trim();
        cam.fieldOfView = fov;
        cam.transform.SetPositionAndRotation(pos, rot);
        // Local volumes blend by camera position: render twice so the stack and shadows settle for this pose.
        cam.Render();
        cam.Render();
        FrontRoomsKitLookdevAccess.Save(rt, Path.Combine(outDir, file + ".png"));
        framesTsv.Append(file).Append('\t').Append(group).Append('\t').Append(V3(pos)).Append('\t').Append(V3(rot.eulerAngles)).Append('\t')
            .Append(fov.ToString("F1", Inv)).Append('\t').Append(note).Append('\n');
    }

    static void ShotAt(string file, Vector3 pos, Vector3 target, float fov, string note = "")
    {
        if (shotLight != null && shotLight.enabled)
        {
            // ShotLightIntensity is not wired in the game yet (FrontRoomsShotTimings.cs:56 only). Read here as 0.6 of
            // illuminance AT THE SUBJECT: a point light on the camera scaled by distance², so it is a fill, not a flash.
            var dd = Vector3.Distance(pos, target);
            shotLight.intensity = .6f * dd * dd;
            shotLight.range = Mathf.Max(1f, dd * 4f);
            note += " · shot light 0.6 at the subject (" + dd.ToString("0.00", Inv) + " m)";
        }
        Shot(file, pos, Quaternion.LookRotation(target - pos, Vector3.up), fov, note);
    }

    static Light shotLight;

    /// <summary>The Unlock shot's optional camera light (FrontRoomsShotTimings.Unlock.ShotLightIntensity 0.6, unshadowed), on or off.</summary>
    static void ShotLight(bool on)
    {
        if (on && shotLight == null)
        {
            shotLight = new GameObject("Shot light (Unlock.ShotLightIntensity)").AddComponent<Light>();
            shotLight.transform.SetParent(cam.transform, false);
            shotLight.type = LightType.Point;
            shotLight.range = 3f;
            shotLight.intensity = .6f;
            shotLight.color = new Color(1f, .96f, .9f);
            shotLight.shadows = LightShadows.None;
        }
        if (shotLight != null) shotLight.enabled = on;
    }

    /// <summary>A camera on the head-dip line, yawed about the keyhole's vertical axis (degrees, + toward door +Z of the opener face).</summary>
    static Vector3 Yawed(Vector3 keyhole, Vector3 camPos, float deg) => keyhole + Quaternion.Euler(0f, deg, 0f) * (camPos - keyhole);

    static string V3(Vector3 v) => "(" + v.x.ToString("F3", Inv) + ", " + v.y.ToString("F3", Inv) + ", " + v.z.ToString("F3", Inv) + ")";
    static string F(float v) => v.ToString("0.0", Inv);

    static Material maskBlack, maskWhite;

    /// <summary>
    /// The leak mask: every renderer black and unlit except the bright room behind the door, which is white and unlit.
    /// Any white pixel is the far room seen through the door's perimeter. Post off, 4x MSAA resolved (sub-pixel
    /// slits still show as partial grey). Returns the number of pixels above 8/255.
    /// </summary>
    static int LeakMask(string file, Transform bright, out float max, out double sum, bool save = true)
    {
        if (maskBlack == null)
        {
            var sh = Shader.Find("Universal Render Pipeline/Unlit");
            maskBlack = new Material(sh) { name = "INT mask black" };
            maskBlack.SetColor("_BaseColor", Color.black);
            maskWhite = new Material(sh) { name = "INT mask white" };
            maskWhite.SetColor("_BaseColor", Color.white);
        }
        var renderers = Object.FindObjectsByType<Renderer>(FindObjectsInactive.Exclude, FindObjectsSortMode.None);
        var saved = new Material[renderers.Length][];
        for (var i = 0; i < renderers.Length; i++)
        {
            saved[i] = renderers[i].sharedMaterials;
            var white = renderers[i].transform.IsChildOf(bright);
            var arr = new Material[saved[i].Length];
            for (var k = 0; k < arr.Length; k++) arr[k] = white ? maskWhite : maskBlack;
            renderers[i].sharedMaterials = arr;
        }
        var data = cam.GetUniversalAdditionalCameraData();
        var pp = data.renderPostProcessing;
        data.renderPostProcessing = false;
        cam.Render();
        var prev = RenderTexture.active;
        RenderTexture.active = rt;
        var img = new Texture2D(Wpx, Hpx, TextureFormat.RGB24, false);
        img.ReadPixels(new Rect(0, 0, Wpx, Hpx), 0, 0);
        img.Apply();
        RenderTexture.active = prev;
        var px = img.GetPixels32();
        var count = 0;
        max = 0f;
        sum = 0;
        foreach (var p in px)
        {
            var v = Mathf.Max(p.r, Mathf.Max(p.g, p.b)) / 255f;
            if (v > 8f / 255f) { count++; sum += v; if (v > max) max = v; }
        }
        if (save || count > 0) File.WriteAllBytes(Path.Combine(outDir, file + "_mask.png"), img.EncodeToPNG());
        Object.DestroyImmediate(img);
        data.renderPostProcessing = pp;
        for (var i = 0; i < renderers.Length; i++) renderers[i].sharedMaterials = saved[i];
        return count;
    }

    // ======================================================================================== rooms

    static FrontRoomsKitLookdevAccess.Opening Op(int side, float centre, float width) =>
        new FrontRoomsKitLookdevAccess.Opening { side = side, centre = centre, width = width };

    static Transform Room(Transform parent, string name, Vector3 origin, float w, float d, RoomRule rule, float h, params FrontRoomsKitLookdevAccess.Opening[] openings)
    {
        var room = FrontRoomsKitLookdevAccess.Room(parent, name, origin, w, d, rule, openings, h);
        var wo = parent.TransformPoint(origin);
        zoneRooms.Add((new Bounds(wo + new Vector3(0f, h * .5f, d * .5f), new Vector3(w, h, d)),
            rule == RoomRule.Office ? FrontRoomsLook.ReflectionZone.Office : h > 4f ? FrontRoomsLook.ReflectionZone.Tall : FrontRoomsLook.ReflectionZone.Level0));
        if (rule == RoomRule.Office)
            FrontRoomsPostStack.EnsureZoneVolume(room, "Office", new Bounds(new Vector3(0f, h * .5f, d * .5f), new Vector3(w + 2f, h + .4f, d + 2f)), 1.0f);
        return room;
    }

    /// <summary>Remove a room's south wall (and its cove), so a neighbour's north wall is the only wall there.</summary>
    static void DropSouthWall(Transform room)
    {
        var kill = new List<GameObject>();
        foreach (Transform c in room)
            if ((c.name == "wall" || c.name == "cove") && Mathf.Abs(c.localPosition.z) < .01f) kill.Add(c.gameObject);
        foreach (var k in kill) Object.DestroyImmediate(k);
    }

    /// <summary>Clip the floor and ceiling slabs to z0..z1 (room-local) so two rooms never overlap coplanar.</summary>
    static void ClipSlabs(Transform room, float z0, float z1)
    {
        foreach (Transform c in room)
            if (c.name == "floor" || c.name == "ceiling")
            {
                var p = c.localPosition; var s = c.localScale;
                c.localPosition = new Vector3(p.x, p.y, (z0 + z1) * .5f);
                c.localScale = new Vector3(s.x, s.y, z1 - z0);
            }
    }

    static List<Light> Lights(Transform room)
    {
        var list = new List<Light>();
        foreach (var l in room.GetComponentsInChildren<Light>(true)) list.Add(l);
        return list;
    }

    static void ScaleLights(Transform room, float k) { foreach (var l in Lights(room)) l.intensity *= k; }

    static void KillLamps(Transform room, Func<Vector3, bool> where)
    {
        var kill = new List<GameObject>();
        foreach (Transform c in room) if (c.name.StartsWith("troffer") && where(c.localPosition)) kill.Add(c.gameObject);
        foreach (var k in kill) Object.DestroyImmediate(k);
    }

    /// <summary>Fill the rest of a 1.4-wide opening in a Room wall to make the map's window cut: sill 0-0.35, head 2.0-2.1.</summary>
    static void WindowCut(Transform room, RoomRule rule, float centreX, float wallZ)
    {
        var wall = FrontRoomsSurfaces.Room(rule, FrontRoomsSurfaces.Slot.Wall);
        Box(room, "wall", new Vector3(centreX, .175f, wallZ), new Vector3(1.4f, .35f, .16f), wall);
        Box(room, "wall", new Vector3(centreX, 2.05f, wallZ), new Vector3(1.4f, .1f, .16f), wall);
        Box(room, "cove", new Vector3(centreX, .05f, wallZ), new Vector3(1.4f, .1f, .18f), FrontRoomsSurfaces.CoveBase);
    }

    static GameObject Box(Transform p, string n, Vector3 pos, Vector3 size, Material m) => FrontRoomsKitLookdevAccess.Box(p, n, pos, size, m);

    // ======================================================================================== kit helpers

    static Vector3 A(string kit, string anchor)
    {
        var info = FrontRoomsKitLibrary.GetInfo(kit);
        if (info != null && info.TryAnchor(anchor, out var p)) return p;
        throw new Exception(kit + " has no anchor " + anchor);
    }

    static Transform S(string kit, Transform parent, Vector3 pos, Quaternion rot, Vector3? scale = null, string label = null)
    {
        var go = FrontRoomsKitLibrary.Spawn(kit, parent, pos, rot, scale ?? Vector3.one, false, label ?? kit);
        if (go == null) throw new Exception("no model " + kit);
        foreach (var c in go.GetComponentsInChildren<Collider>(true)) Object.DestroyImmediate(c);
        return go.transform;
    }

    static readonly Quaternion Rs = Quaternion.Euler(0f, 90f, 0f), Rp = Quaternion.Euler(0f, -90f, 0f);
    static readonly Vector3 Mp = new Vector3(-1f, 1f, 1f);
    static Vector3 On(Vector3 at, Quaternion r, Vector3 m, Vector3 local) => at + r * Vector3.Scale(m, local);

    // ======================================================================================== doors

    sealed class DoorSpec
    {
        public string code, leaf, frame, finish = "", sign, plate, exitSign;
        public bool locked, free, closer, crossbar;
        public RoomRule rule;
    }

    static DoorSpec Spec(string code)
    {
        switch (code)
        {
            case "L0-F": return new DoorSpec { code = code, leaf = "Kit_DoorLeaf_Veneer", frame = "Kit_DoorFrame_Wood", finish = "_Brass", free = true, rule = RoomRule.Lobby };
            case "L0-K": return new DoorSpec { code = code, leaf = lockedLeaf, frame = "Kit_DoorFrame_Steel", locked = true, sign = "Kit_DoorSign", plate = "Kit_DoorNumberPlate", rule = RoomRule.Lobby };
            case "OF-F": return new DoorSpec { code = code, leaf = "Kit_DoorLeaf_Veneer_Oak", frame = "Kit_DoorFrame_Steel", free = true, closer = true, rule = RoomRule.Office };
            case "OF-K": return new DoorSpec { code = code, leaf = lockedLeaf, frame = "Kit_DoorFrame_Steel", locked = true, closer = true, sign = "Kit_DoorSign", plate = "Kit_DoorNumberPlate_Blue", rule = RoomRule.Office };
            case "RN-F": return new DoorSpec { code = code, leaf = "Kit_DoorLeaf_Ward", frame = "Kit_DoorFrame_Steel", closer = true, rule = RoomRule.Run };
            case "RN-K": return new DoorSpec { code = code, leaf = lockedLeaf, frame = "Kit_DoorFrame_Steel", locked = true, closer = true, sign = "Kit_DoorSign", plate = "Kit_DoorNumberPlate_White", rule = RoomRule.Run };
            case "EX-F": return new DoorSpec { code = code, leaf = "Kit_DoorLeaf_Veneer_Oak", frame = "Kit_DoorFrame_Steel_Alu", free = true, closer = true, crossbar = true, exitSign = "Kit_ExitSign", rule = RoomRule.Exit };
            case "EX-K": return new DoorSpec { code = code, leaf = lockedLeaf, frame = "Kit_DoorFrame_Steel_Alu", locked = true, closer = true, plate = "Kit_DoorNumberPlate", exitSign = "Kit_ExitSign_Dead", rule = RoomRule.Exit };
        }
        throw new Exception("no door " + code);
    }

    sealed class Door
    {
        public DoorSpec spec;
        public bool old;
        public int s;                       // S_sign: +1 = the swing room is on door-root +X
        public Transform root, frame, hinge, rig;
        public Transform kfS, kfP, plugS, plugP, keyS, keyP;
        public Transform knobS, knobP, leverS, leverP;
        public Quaternion knobSR, knobPR, leverSR, leverPR;
        public Transform deadbolt, latch;
        public Vector3 deadboltHome, latchHome;
        public float latchTravel;
        public Transform arm, forearm;
        public Vector3 spindle0, shoe;
        public float armA, foreB, latchT, angle;
    }

    static readonly Vector3 Axis = new Vector3(.0295f, 0f, .0098f);   // A2 hinge axis, door root (S = +X)

    /// <summary>The map's render-only trims (MapWorld BuildEdge): 0.07 face x 0.20 deep jambs and head, outside the opening.</summary>
    static void MapTrims(Transform root)
    {
        var trim = FrontRoomsSurfaces.CoveBase;
        Box(root, "Map trim jamb hinge", new Vector3(0f, 1.05f, -.035f), new Vector3(.20f, 2.1f, .07f), trim);
        Box(root, "Map trim jamb latch", new Vector3(0f, 1.05f, 1.035f), new Vector3(.20f, 2.1f, .07f), trim);
        Box(root, "Map trim head", new Vector3(0f, 2.135f, .5f), new Vector3(.20f, .07f, 1.14f), trim);
    }

    /// <summary>
    /// The new door (10_spec §6.2 raw recipe): frame at the door root with M = (S_sign,1,1); the hinge on the A2 axis;
    /// the Leaf rig under it, mirrored by M; face hardware per §1.2 (plug and text parts with (S_sign,1,1));
    /// bolts at the leaf anchors; strike on the frame; closer linkage solved per angle; the Leaf shadow proxy.
    /// </summary>
    static Door KitDoor(DoorSpec d, Transform parent, Vector3 pos, Quaternion rot, int s, bool trims = true)
    {
        var door = new Door { spec = d, s = s };
        var root = new GameObject("DoorSet " + d.code + (s > 0 ? " S+" : " S-")).transform;
        root.SetParent(parent, false);
        root.localPosition = pos;
        root.localRotation = rot;
        door.root = root;
        if (trims) MapTrims(root);
        var m = new Vector3(s, 1f, 1f);
        door.frame = S(d.frame, root, Vector3.zero, Quaternion.identity, m, "Door frame (kit)");
        door.hinge = new GameObject("Door hinge (A2)").transform;
        door.hinge.SetParent(root, false);
        door.hinge.localPosition = new Vector3(s * Axis.x, 0f, Axis.z);
        door.rig = new GameObject("Leaf rig").transform;
        door.rig.SetParent(door.hinge, false);
        door.rig.localPosition = new Vector3(-s * Axis.x, 0f, -Axis.z);
        door.rig.localScale = m;
        var rig = door.rig;
        S(d.leaf, rig, Vector3.zero, Quaternion.identity, null, "Leaf (kit)");
        var shadow = GameObject.CreatePrimitive(PrimitiveType.Cube);
        Object.DestroyImmediate(shadow.GetComponent<Collider>());
        shadow.name = "Leaf shadow";
        shadow.transform.SetParent(rig, false);
        shadow.transform.localPosition = new Vector3(0f, 1.055f, .5f);
        shadow.transform.localScale = new Vector3(.10f, 2.08f, .96f);
        var sr = shadow.GetComponent<MeshRenderer>();
        sr.sharedMaterial = FrontRoomsSurfaces.DoorVeneer;
        sr.shadowCastingMode = ShadowCastingMode.ShadowsOnly;
        var txt = new Vector3(s, 1f, 1f);

        if (d.free)
        {
            foreach (var f in new[] { "s", "p" })
            {
                if (d.crossbar && f == "p") continue;   // EX-F: the crossbar replaces the P-face trim
                var r = f == "s" ? Rs : Rp;
                var mm = f == "s" ? Vector3.one : Mp;
                var at = A(d.leaf, "rose_" + f);
                var rose = "Kit_Lock_Rose" + d.finish;
                S(rose, rig, at, r, mm, "Lock rose " + f);
                var lever = S("Kit_Lock_Lever" + d.finish, rig, On(at, r, mm, A(rose, "spindle")), r, mm, "Lock lever " + f);
                if (f == "s") { door.leverS = lever; door.leverSR = r; } else { door.leverP = lever; door.leverPR = r; }
            }
            door.latch = S("Kit_Lock_Latchbolt_Bored" + d.finish, rig, A(d.leaf, "latchbolt"), Quaternion.identity, null, "Lock latchbolt");
            door.latchTravel = .013f;
            var strikeAt = d.frame == "Kit_DoorFrame_Wood" ? A(d.frame, "strike") : A(d.frame, "strike_bored");
            S("Kit_Lock_StrikeBored" + d.finish, door.frame, strikeAt, Quaternion.Euler(0f, 180f, 0f), null, "Lock strike");
        }
        if (d.locked)
        {
            foreach (var f in new[] { "s", "p" })
            {
                var r = f == "s" ? Rs : Rp;

        }

    static void SetAngle(Door door, float deg)
    {
        door.angle = deg;
        door.hinge.localRotation = Quaternion.Euler(0f, door.s * deg, 0f);
        if (door.old) return;
        // Facade rule (G2): the spring latch is held back within ~8° of shut, so it never rubs the strike or lining.
        if (door.latch != null) SetLatch(door, deg > .01f && deg <= 8f ? 1f : door.latchT);
        if (door.arm != null) SolveCloser(door);
    }

    static void SetLatch(Door door, float t)
    {
        if (door.latch == null) return;
        door.latch.localPosition = door.latchHome - new Vector3(0f, 0f, door.latchTravel * t);
    }

    static void SetDeadbolt(Door door, float retracted)
    {
        if (door.deadbolt == null) return;
        door.deadbolt.localPosition = door.deadboltHome - new Vector3(0f, 0f, .025f * retracted);
    }

    /// <summary>Plug and key on one face: the plug turns about its part −Z; +angle sends the cuts toward the hinge (§3.4).</summary>
    static void SetPlug(Door door, string face, float deg, float insert, bool withKey)
    {
        var kf = face == "s" ? door.kfS : door.kfP;
        var plug = face == "s" ? door.plugS : door.plugP;
        if (kf == null) return;
        var sign = (face == "s" ? 1f : -1f) * door.s;
        var turn = Quaternion.AngleAxis(sign * deg, new Vector3(0f, 0f, -1f));
        plug.localRotation = turn;
        var key = face == "s" ? door.keyS : door.keyP;
        if (withKey && key == null)
        {
            key = S("Kit_Key_Zone", kf, Vector3.zero, Quaternion.identity, null, "Key");
            if (face == "s") door.keyS = key; else door.keyP = key;
        }
        if (key == null) return;
        key.gameObject.SetActive(withKey);
        // Key frame: origin at the shoulder, +Z toward the tip (into the lock), +Y = cuts up. Part +Z points out of the door.
        var outMetres = Unlock.InsertDepth * (1f - Mathf.Clamp01(insert));
        key.localPosition = new Vector3(0f, 0f, outMetres);
        key.localRotation = turn * Quaternion.LookRotation(Vector3.back, Vector3.up);
    }

    static class Unlock { public const float InsertDepth = .025f; }

    static void SetKnob(Door door, string face, float deg)
    {
        var k = face == "s" ? door.knobS : door.knobP;
        if (k == null) return;
        k.localRotation = (face == "s" ? door.knobSR : door.knobPR) * Quaternion.AngleAxis(deg, Vector3.forward);
    }

    /// <summary>Closer elbow (10_spec §2.4): circle(spindle(θ), a) ∩ circle(shoe pivot, b), the root farther from the leaf's S face.</summary>
    static void SolveCloser(Door door)
    {
        var q = Quaternion.Euler(0f, door.angle, 0f);   // frame space is authored (S = +X): the leaf turns +θ
        var sp = Axis + q * (door.spindle0 - Axis);
        var n3 = q * Vector3.right;
        var p0 = new Vector2(sp.x, sp.z);
        var p1 = new Vector2(door.shoe.x, door.shoe.z);
        float a = door.armA, b = door.foreB;
        var dist = Vector2.Distance(p0, p1);
        var along = (a * a - b * b + dist * dist) / (2f * dist);
        var h = Mathf.Sqrt(Mathf.Max(0f, a * a - along * along));
        var dir = (p1 - p0) / dist;
        var mid = p0 + dir * along;
        var perp = new Vector2(-dir.y, dir.x);
        var e1 = mid + perp * h;
        var e2 = mid - perp * h;
        var n = new Vector2(n3.x, n3.z);
        var e = Vector2.Dot(e1 - p0, n) >= Vector2.Dot(e2 - p0, n) ? e1 : e2;
        float Yaw(Vector2 from, Vector2 to) => Mathf.Atan2(-(to.y - from.y), to.x - from.x) * Mathf.Rad2Deg;
        door.arm.localPosition = sp;
        door.arm.localRotation = Quaternion.Euler(0f, Yaw(p0, e), 0f);
        door.forearm.localPosition = new Vector3(e.x, sp.y + A("Kit_DoorCloser_Arm", "elbow").y, e.y);
        door.forearm.localRotation = Quaternion.Euler(0f, Yaw(e, p1), 0f);
    }

    /// <summary>World keyhole, out-of-door direction and up for one face (through the rig: valid at any angle and handing).</summary>
    static void Keyhole(Door door, string face, out Vector3 point, out Vector3 outDir)
    {
        var kf = face == "s" ? door.kfS : door.kfP;
        point = kf.position;
        outDir = kf.TransformVector(Vector3.forward).normalized;
    }

    // Door placement on a Room wall. side 0 S, 1 N (opening centre along x), 2 W, 3 E (centre along z).
    // Door root on the hinge jamb (low coordinate), door +Z along the opening. Returns whether door +X points into the room.
    static bool DoorOnWall(int side, float centre, float w, float d, out Vector3 pos, out Quaternion rot)
    {
        switch (side)
        {
            case 0: pos = new Vector3(centre - .5f, 0f, 0f); rot = Quaternion.LookRotation(Vector3.right); return false;
            case 1: pos = new Vector3(centre - .5f, 0f, d); rot = Quaternion.LookRotation(Vector3.right); return true;
            case 2: pos = new Vector3(-w * .5f, 0f, centre - .5f); rot = Quaternion.LookRotation(Vector3.forward); return true;
            default: pos = new Vector3(w * .5f, 0f, centre - .5f); rot = Quaternion.LookRotation(Vector3.forward); return false;
        }
    }

    /// <summary>S_sign that puts the requested face toward the room the wall belongs to.</summary>
    static int SFor(bool plusXIntoRoom, bool sFaceToRoom) => plusXIntoRoom == sFaceToRoom ? 1 : -1;

    static Door PlaceDoor(Transform room, string code, int side, float centre, float w, float d, bool sFaceToRoom)
    {
        var intoRoom = DoorOnWall(side, centre, w, d, out var pos, out var rot);
        return KitDoor(Spec(code), room, pos, rot, SFor(intoRoom, sFaceToRoom));
    }

    {
        var g = G("probe");

    static void RectRec(string file, string condition, float dist, Transform room, IList<(string name, Transform doorRoot, float z0, float z1, float y0, float y1)> regions)
    {
        rectJson.Append(rectFirst ? "" : ",\n");
        rectFirst = false;
        rectJson.Append("{\"file\":\"" + file + "\",\"condition\":\"" + condition + "\",\"distance\":" + dist.ToString(Inv) + ",\"rects\":{");
        for (var i = 0; i < regions.Count; i++)
        {
            var r = regions[i];
            float x0 = 1e9f, y0 = 1e9f, x1 = -1e9f, y1 = -1e9f;
            foreach (var z in new[] { r.z0, r.z1 })
                foreach (var y in new[] { r.y0, r.y1 })
                {
                    var p = cam.WorldToScreenPoint(r.doorRoot.TransformPoint(new Vector3(0f, y, z)));
                    x0 = Mathf.Min(x0, p.x); x1 = Mathf.Max(x1, p.x);
                    y0 = Mathf.Min(y0, Hpx - p.y); y1 = Mathf.Max(y1, Hpx - p.y);
                }
            rectJson.Append((i > 0 ? "," : "") + "\"" + r.name + "\":[" + F(x0) + "," + F(y0) + "," + F(x1) + "," + F(y1) + "]");
        }
        rectJson.Append("}}");
    }

    {

    /// <summary>Hung pose (10_spec §4.1; same as FrontRoomsInteractableHeroes.Hang).</summary>
    static Vector3 Hang(Transform parent, Vector3 hook, string tag, string keyKit, float ringYaw, float lift, float keyTwist, float tagTwist)
    {
        var ringRot = Quaternion.Euler(0f, ringYaw, 0f);
        var ringPos = hook + Vector3.up * lift - ringRot * A("Kit_KeyRing", "hook_contact");
        S("Kit_KeyRing", parent, ringPos, ringRot, null, "Key ring");
        var tX = Quaternion.Euler(0f, ringYaw + keyTwist, 0f) * Vector3.right;
        var keyRot = Quaternion.LookRotation(Vector3.down, Vector3.Cross(Vector3.down, tX));
        var keyPos = ringPos + ringRot * A("Kit_KeyRing", "key_contact") - keyRot * A(keyKit, "ring_hole");
        S(keyKit, parent, keyPos, keyRot, null, "Key");
        var gX = Quaternion.Euler(0f, ringYaw + tagTwist, 0f) * Vector3.right;
        var tagRot = Quaternion.LookRotation(gX, Vector3.up);
        var tagPos = ringPos + ringRot * A("Kit_KeyRing", "tag_contact") - tagRot * A(tag, "hole");
        S(tag, parent, tagPos, tagRot, null, "Key tag");
        return keyPos + keyRot * new Vector3(0f, 0f, .0f);
    }

    /// <summary>Flat pose on a surface: key flats down, ring tilted through the bow hole, tag flat beside (approximate; G3's search is not ported).</summary>
    static Vector3 LayFlat(Transform parent, Vector3 top, float yaw, string tag)
    {
        var keyRot = Quaternion.Euler(0f, yaw, 0f) * Quaternion.LookRotation(Vector3.right, Vector3.forward);   // key X (flat normal) = up
        var keyPos = top + Vector3.up * .0011f;
        S("Kit_Key_Zone", parent, keyPos, keyRot, null, "Key");
        var hole = keyPos + keyRot * A("Kit_Key_Zone", "ring_hole");
        // Ring tilted 28° about the key's long axis, its key_contact in the bow hole.
        var ringRot = Quaternion.Euler(0f, yaw, 0f) * Quaternion.Euler(0f, 90f, 0f) * Quaternion.Euler(-90f + 28f, 0f, 0f);
        var ringPos = hole - ringRot * A("Kit_KeyRing", "key_contact");
        if (ringPos.y < top.y + .0012f) ringPos.y = top.y + .0012f;
        S("Kit_KeyRing", parent, ringPos, ringRot, null, "Key ring");
        var tagContact = ringPos + ringRot * A("Kit_KeyRing", "tag_contact");
        var tagRot = Quaternion.Euler(0f, yaw - 140f, 0f) * Quaternion.Euler(-90f, 0f, 0f);   // tag face (+Z) up, body lying away from the key
        var tagPos = tagContact - tagRot * A(tag, "hole");
        tagPos.y = top.y + .0021f;
        S(tag, parent, tagPos, tagRot, null, "Key tag");
        return keyPos;
    }

    static float SupportTop(string kit, out Vector3 centre)
    {
        var info = FrontRoomsKitLibrary.GetInfo(kit);
        if (info != null && info.TrySupport(null, out centre, out _)) return centre.y;
        centre = info != null ? (info.Min + info.Max) * .5f : Vector3.zero;
        return info != null ? info.Max.y : .75f;
    }

    /// <summary>Ear-clipping triangulation of a simple CCW polygon.</summary>
    static List<int> Triangulate(List<Vector2> p)
    {
        var tris = new List<int>();
        var idx = new List<int>();
        for (var i = 0; i < p.Count; i++) idx.Add(i);
        float Cross(Vector2 a, Vector2 b, Vector2 c) => (b.x - a.x) * (c.y - a.y) - (b.y - a.y) * (c.x - a.x);
        var guard = 0;
        while (idx.Count > 3 && guard++ < 500)
        {
            var clipped = false;
            for (var i = 0; i < idx.Count; i++)
            {
                int i0 = idx[(i + idx.Count - 1) % idx.Count], i1 = idx[i], i2 = idx[(i + 1) % idx.Count];
                if (Cross(p[i0], p[i1], p[i2]) <= 1e-12f) continue;
                var inside = false;
                foreach (var j in idx)
                {
                    if (j == i0 || j == i1 || j == i2) continue;
                    if (Cross(p[i0], p[i1], p[j]) > 0 && Cross(p[i1], p[i2], p[j]) > 0 && Cross(p[i2], p[i0], p[j]) > 0) { inside = true; break; }
                }
                if (inside) continue;
                tris.Add(i0); tris.Add(i1); tris.Add(i2);
                idx.RemoveAt(i);
                clipped = true;
                break;
            }
            if (!clipped) break;
        }
        if (idx.Count == 3) { tris.Add(idx[0]); tris.Add(idx[1]); tris.Add(idx[2]); }
        else if (idx.Count > 3) for (var i = 1; i < idx.Count - 1; i++) { tris.Add(idx[0]); tris.Add(idx[i]); tris.Add(idx[i + 1]); }   // fallback fan
        return tris;
    }

    /// <summary>A 6 mm glass piece: the polygon (piece-local, XY) extruded ±0.003 in Z. Submesh 0 faces (glass), 1 the edge band.</summary>
    static Mesh PieceMesh(float[] flat, float halfT = .003f)
    {
        var p = new List<Vector2>();
        for (var i = 0; i + 1 < flat.Length; i += 2) p.Add(new Vector2(flat[i], flat[i + 1]));
        var area = 0f;
        for (var i = 0; i < p.Count; i++) { var a = p[i]; var b = p[(i + 1) % p.Count]; area += a.x * b.y - b.x * a.y; }
        if (area < 0f) p.Reverse();
        var tri = Triangulate(p);
        var v = new List<Vector3>(); var n = new List<Vector3>(); var uv = new List<Vector2>();
        var faces = new List<int>(); var edges = new List<int>();
        // front (+Z) and back (-Z)
        var baseF = v.Count;
        foreach (var q in p) { v.Add(new Vector3(q.x, q.y, halfT)); n.Add(Vector3.forward); uv.Add(q); }
        for (var i = 0; i < tri.Count; i += 3) { faces.Add(baseF + tri[i]); faces.Add(baseF + tri[i + 1]); faces.Add(baseF + tri[i + 2]); }   // CCW in XY faces +Z
        var baseB = v.Count;
        foreach (var q in p) { v.Add(new Vector3(q.x, q.y, -halfT)); n.Add(Vector3.back); uv.Add(q); }
        for (var i = 0; i < tri.Count; i += 3) { faces.Add(baseB + tri[i]); faces.Add(baseB + tri[i + 2]); faces.Add(baseB + tri[i + 1]); }
        for (var i = 0; i < p.Count; i++)
        {
            var a = p[i]; var b = p[(i + 1) % p.Count];
            var e = (b - a).normalized;
            var nn = new Vector3(e.y, -e.x, 0f);
            var k = v.Count;
            v.Add(new Vector3(a.x, a.y, halfT)); v.Add(new Vector3(b.x, b.y, halfT)); v.Add(new Vector3(b.x, b.y, -halfT)); v.Add(new Vector3(a.x, a.y, -halfT));
            for (var j = 0; j < 4; j++) { n.Add(nn); uv.Add(new Vector2(j < 2 ? 0f : 1f, j % 3 == 0 ? 0f : 1f)); }
            edges.Add(k); edges.Add(k + 2); edges.Add(k + 1);
            edges.Add(k); edges.Add(k + 3); edges.Add(k + 2);
        }
        var mesh = new Mesh { name = "glass piece" };
        mesh.SetVertices(v); mesh.SetNormals(n); mesh.SetUVs(0, uv);
        mesh.subMeshCount = 2;
        mesh.SetTriangles(faces, 0);
        mesh.SetTriangles(edges, 1);
        mesh.RecalculateBounds();
        return mesh;
    }

    static GameObject Piece(Transform parent, string name, float[] poly, Vector3 pos, Quaternion rot)
    {
        GlassMats();
        var go = new GameObject(name);
        go.transform.SetParent(parent, false);
        go.transform.localPosition = pos;
        go.transform.localRotation = rot;
        go.AddComponent<MeshFilter>().sharedMesh = PieceMesh(poly);
        var r = go.AddComponent<MeshRenderer>();
        r.sharedMaterials = new[] { shardFace, shardEdge };
        r.shadowCastingMode = ShadowCastingMode.Off;
        return go;
    }

    /// <summary>Crack lines on both glass faces for a stage (1, 2 or 3): radial cracks first, then the concentric chords.</summary>
    static void Cracks(Transform root, FracData f, int stage)
    {
        GlassMats();
        var cy = f.centreY;
        float ix = f.impact[0], iy = f.impact[1];
        var v = new List<Vector3>(); var tris = new List<int>();
        const float halfW = .0011f, zf = .0032f;
        for (var i = 0; i + 3 < f.edges.Length; i += 4)
        {
            var a = new Vector2(f.edges[i], f.edges[i + 1]);
            var b = new Vector2(f.edges[i + 2], f.edges[i + 3]);
            var mid = (a + b) * .5f;
            var r = Vector2.Distance(mid, new Vector2(ix, iy));
            var dir = (b - a).normalized;
            var radial = Mathf.Abs(Vector2.Dot(dir, (mid - new Vector2(ix, iy)).normalized)) > .7f;
            var keep = stage >= 3 || (stage == 2 ? (radial ? r < .55f : r < .22f) : (radial ? r < .2f : r < .045f));
            if (!keep) continue;
            var nrm = new Vector2(-dir.y, dir.x) * halfW;
            foreach (var z in new[] { zf, -zf })
            {
                var k = v.Count;
                v.Add(new Vector3(a.x + nrm.x, cy + a.y + nrm.y, z)); v.Add(new Vector3(b.x + nrm.x, cy + b.y + nrm.y, z));
                v.Add(new Vector3(b.x - nrm.x, cy + b.y - nrm.y, z)); v.Add(new Vector3(a.x - nrm.x, cy + a.y - nrm.y, z));
                if (z > 0) { tris.Add(k); tris.Add(k + 2); tris.Add(k + 1); tris.Add(k); tris.Add(k + 3); tris.Add(k + 2); }
                else { tris.Add(k); tris.Add(k + 1); tris.Add(k + 2); tris.Add(k); tris.Add(k + 2); tris.Add(k + 3); }
            }
        }
        var mesh = new Mesh { name = "cracks stage " + stage, indexFormat = IndexFormat.UInt32 };
        mesh.SetVertices(v);
        mesh.SetTriangles(tris, 0);
        mesh.RecalculateNormals();
        mesh.RecalculateBounds();
        var go = new GameObject("Glass cracks (stand-in) stage " + stage);
        go.transform.SetParent(root, false);
        go.AddComponent<MeshFilter>().sharedMesh = mesh;
        var mr = go.AddComponent<MeshRenderer>();
        mr.sharedMaterial = crackMat;
        mr.shadowCastingMode = ShadowCastingMode.Off;
        // The crushed core: frosted, from stage 1 on.
        foreach (var p in f.pieces)
            if (p.kind == "crush")
            {
                var c = Piece(root, "Glass crush (stand-in)", p.poly, new Vector3(p.cx, cy + p.cy, 0f), Quaternion.identity);
                c.transform.localScale = new Vector3(1f, 1f, 1.15f);
                c.GetComponent<MeshRenderer>().sharedMaterials = new[] { crushMat, crushMat };
            }
    }

    enum Pane { Intact, Crack1, Crack2, Crack3, Shatter, After }

    /// <summary>A window opening in the near room's north wall at x = cx; the frame member, glass state, blind. Window root: +Z = face A = the room.</summary>
    static Transform Window(Transform room, string frameKit, float cx, float d, Pane state, FracData f, bool blind)
    {
        var root = new GameObject("Window " + frameKit + " " + state).transform;
        root.SetParent(room, false);
        root.localPosition = new Vector3(cx, 0f, d);
        root.localRotation = Quaternion.Euler(0f, 180f, 0f);   // +Z (face A) toward the room (-Z)
        S(frameKit, root, Vector3.zero, Quaternion.identity, null, "Window frame (kit)");
        if (blind) S("Kit_MiniBlind_Raised", root, A(frameKit, "blind_rail"), Quaternion.identity, null, "Window blind (kit)");
        var slabC = A(frameKit, "glass_slab");
        if (state <= Pane.Crack3)
        {
            var glass = Resources.Load<Material>("Surfaces/Glass_Window") ?? FrontRoomsSurfaces.TryGet("Prop_Glass");
            var slab = Box(root, "Window glass (interim slab)", slabC, new Vector3(1.391f, 1.642f, .006f), glass);
            slab.GetComponent<MeshRenderer>().shadowCastingMode = ShadowCastingMode.Off;
            if (state != Pane.Intact) Cracks(root, f, (int)state);
            return root;
        }
        var cy = f.centreY;
        var rng = new System.Random(7);
        foreach (var p in f.pieces)
        {
            if (p.kind == "tooth") { Piece(root, "Glass tooth (stand-in)", p.poly, new Vector3(p.cx, cy + p.cy, 0f), Quaternion.identity); continue; }
            if (state == Pane.Shatter)
            {
                if (p.air == null || p.air.Length < 6) continue;
                Piece(root, "Glass shard mid-air (stand-in)", p.poly, new Vector3(p.cx + p.air[0], cy + p.cy + p.air[1], p.air[2]), Quaternion.Euler(p.air[3], p.air[4], p.air[5]));
            }
            else if (p.floor != null)
            {
                foreach (var fl in p.floor)
                {
                    var rot = Quaternion.Euler(0f, fl.yaw, 0f) * Quaternion.Euler(90f + (fl.flip ? 180f : 0f) + fl.tilt, 0f, 0f);
                    Piece(root, "Glass on the floor (stand-in)", fl.poly, new Vector3(fl.x, .0032f + fl.tilt * .0006f, fl.z), rot);
                }
            }
        }
        if (state == Pane.After && f.glitter != null)
            foreach (var gl in f.glitter)
                Piece(root, "Glass chip (stand-in)", gl.poly, new Vector3(gl.x, .0031f, gl.z), Quaternion.Euler(0f, gl.yaw, 0f) * Quaternion.Euler(90f, 0f, 0f));
        return root;
    }

    /// <summary>Near room (the window's room) and a Tall Level 0 hall behind, sharing the window wall.</summary>
    static Transform WindowRooms(Transform g, string name, Vector3 origin, RoomRule rule, float w, float d, out Transform hall)
    {
        var room = Room(g, name, origin, w, d, rule, Standard, Op(1, 0f, 1.4f));

        hall = Room(g, name + " hall", origin + new Vector3(0f, 0f, d), w + 2f, 7f, RoomRule.Lobby, Tall, Op(0, 0f, 1.4f));

        ClipSlabs(room, -.2f, d);
        ClipSlabs(hall, 0f, 7.2f);
        // The hall's south wall is gone: close the extra width either side of the room's north wall.

        Box(hall, "wall", new Vector3(0f, (Standard + Tall) * .5f, 0f), new Vector3(w, Tall - Standard, .16f), hw);
        return room;
    }

    static void GapTest()
    {
        var g = G("gap");
        const float w = 5f, d = 5f;
        var near = Room(g, "gap near (dim)", Vector3.zero, w, d, RoomRule.Lobby, Standard, Op(1, 0f, 1f));
        var far = Room(g, "gap far (bright)", new Vector3(0f, 0f, d), w, d, RoomRule.Lobby, Standard, Op(0, 0f, 1f));
        DropSouthWall(far);
        ClipSlabs(near, -.2f, d);
        ClipSlabs(far, 0f, d + .2f);
        ScaleLights(near, .3f);
        ScaleLights(far, 3f);
        DoorOnWall(1, 0f, w, d, out var dpos, out var drot);
        var c = near.TransformPoint(new Vector3(0f, 0f, d));
        var n = Vector3.back;
        var t = near.TransformDirection(drot * Vector3.forward);
        var views = GapViews(c, n, t);
        // Old primitive door (symmetric), then the kit door with the S face toward the camera (s = +1) and with the P face (s = -1).
        foreach (var (label, code, s) in new[] { ("old", "", 0), ("L0F_Sface", "L0-F", 1), ("L0F_Pface", "L0-F", -1), ("OFK_Sface", "OF-K", 1), ("OFK_Pface", "OF-K", -1) })
        {
            var door = s == 0 ? OldDoor(near, dpos, drot, 1) : KitDoor(Spec(code), near, dpos, drot, s);
            var doorName = s == 0 ? "OLD primitive" : code;
            var faceName = s == 0 ? "-" : (s > 0 ? "S (pull)" : "P (push, stops)");
            foreach (var v in views)
            {
                var file = "10_gap_" + label + "_" + v.name;
                ShotAt(file, v.pos, v.look, GameFov, "dim near room (lamps x0.3), bright far room (lamps x3); door shut");
                var px = LeakMask(file, far, out var mx, out var sum);
                leakTsv.Append(file).Append('\t').Append(doorName).Append('\t').Append(faceName).Append('\t')
                    .Append(v.name).Append('\t').Append(px).Append('\t').Append(mx.ToString("F3", Inv)).Append('\t').Append(sum.ToString("F1", Inv)).Append('\n');
            }
            // Mask-only sweep: inline cameras across every jamb gap position (1 mm steps) at 1.2 and 2.5 m, and the head from 3-5 m.
            var sweepPx = 0; var sweepWorst = ""; var sweepN = 0;
            foreach (var side in new[] { -1f, 1f })
                for (var k = 0; k <= 12; k++)
                    foreach (var dist in new[] { 1.2f, 2.5f })
                    {
                        var along = side * (.488f + k * .001f);
                        var file = "10_gapsweep_" + label + "_t" + (along >= 0 ? "+" : "") + along.ToString("0.000", Inv) + "_n" + dist.ToString("0.0", Inv);
                        cam.fieldOfView = GameFov;
                        cam.transform.SetPositionAndRotation(c + t * along + n * dist + Vector3.up * 1.2f, Quaternion.LookRotation(-n, Vector3.up));
                        var px = LeakMask(file, far, out var mx, out _, false);
                        sweepN++;
                        if (px > sweepPx) { sweepPx = px; sweepWorst = file; }
                    }
            foreach (var dist in new[] { 3f, 4f, 4.8f })
                foreach (var lat in new[] { -.3f, 0f, .3f })
                {
                    var file = "10_gapsweep_" + label + "_head_n" + dist.ToString("0.0", Inv) + "_t" + lat.ToString("0.0", Inv);
                    cam.fieldOfView = GameFov;
                    var pos = c + t * lat + n * dist + Vector3.up * Eye;
                    cam.transform.SetPositionAndRotation(pos, Quaternion.LookRotation(c + t * lat + Vector3.up * 2.09f - pos, Vector3.up));
                    var px = LeakMask(file, far, out var mx, out _, false);
                    sweepN++;
                    if (px > sweepPx) { sweepPx = px; sweepWorst = file; }
                }
            leakTsv.Append("SWEEP " + label).Append('\t').Append(doorName).Append('\t').Append(faceName).Append('\t')
                .Append(sweepN + " inline masks; worst " + (sweepWorst == "" ? "none" : sweepWorst)).Append('\t').Append(sweepPx).Append("\t-\t-\n");
            // The reverse light (bright near room, dim far room): the perimeter should read as a dark line, never a lit one.
            ScaleLights(near, 5f);
            ScaleLights(far, 1f / 60f);
            ShotAt("10_gap_" + label + "_r_front_reverse_light", views[0].pos, views[0].look, GameFov, "reverse light: near room lamps x1.5, far room x0.05");
            ShotAt("10_gap_" + label + "_r_latch_inline_reverse_light", views[8].pos, views[8].look, GameFov, "reverse light, latch inline");
            ScaleLights(near, .2f);
            ScaleLights(far, 60f);
            Object.DestroyImmediate(door.root.gameObject);
        }
    }

    static void Swing()
    {
        var g = G("swing");
        const float w = 5f, d = 5f;
        var near = Room(g, "swing near", Vector3.zero, w, d, RoomRule.Lobby, Standard, Op(1, 0f, 1f));
        var far = Room(g, "swing far", new Vector3(0f, 0f, d), w, d, RoomRule.Lobby, Standard, Op(0, 0f, 1f));
        DropSouthWall(far);
        ClipSlabs(near, -.2f, d);
        ClipSlabs(far, 0f, d + .2f);
        ScaleLights(far, 1.6f);
        DoorOnWall(1, 0f, w, d, out var dpos, out var drot);
        var c = near.TransformPoint(new Vector3(0f, 0f, d));
        var n = Vector3.back;                                  // toward the near (camera) room
        var t = near.TransformDirection(drot * Vector3.forward); // toward the latch jamb
        var obl = c + (n * Mathf.Cos(40f * Mathf.Deg2Rad) + t * Mathf.Sin(40f * Mathf.Deg2Rad)) * 2.1f + Vector3.up * Eye;
        var oblLook = c + Vector3.up * 1.1f - t * .15f;
        foreach (var code in new[] { "L0-F", "OF-K" })
            foreach (var s in new[] { 1, -1 })
            {
                var door = KitDoor(Spec(code), near, dpos, drot, s);
                var dirName = s > 0 ? "toward_camera_pull" : "away_from_camera_push";
                var sw = s > 0 ? n : -n;   // the swing room (the S / pull side); the stops are on the other side
                // Plan: high in the swing room over the hinge jamb, looking down at knuckles, leaf edge, casing, closer.
                var planCam = c - t * .18f + sw * .80f + Vector3.up * 2.62f;
                var planLook = c - t * .46f + sw * .12f + Vector3.up * 1.55f;
                // Stop: on the push (stop) side, looking at the latch jamb: stop strip, strike, the leaf landing on the stop.
                // 0.55 m off the latch jamb at ~40° to the wall: casing, lining, stop and leaf edge read as a stepped section.
                var stopCam = c + t * .13f - sw * .42f + Vector3.up * 1.52f;
                var stopLook = c + t * .50f - sw * .01f + Vector3.up * 1.22f;
                foreach (var deg in new[] { 0f, 5f, 30f, 60f, 95f })
                {
                    SetAngle(door, deg);
                    var tag = deg.ToString("00", Inv);
                    ShotAt("11_swing_" + code + "_" + dirName + "_" + tag + "deg_oblique", obl, oblLook, GameFov, code + " swing " + tag + "°, leaf moves " + (s > 0 ? "toward the camera (camera on the S/pull side)" : "away (camera on the P/stop side)"));
                    ShotAt("11_swing_" + code + "_" + dirName + "_" + tag + "deg_plan", planCam, planLook, 70f, "plan: 2.62 m up in the swing room over the hinge jamb (knuckles, leaf edge, casing" + (Spec(code).closer ? ", closer" : "") + ")");
                    if (deg == 0f || deg == 30f || deg == 95f)
                        ShotAt("11_swing_" + code + "_" + dirName + "_" + tag + "deg_stop", stopCam, stopLook, 50f, "push (stop) side, latch jamb: the 16 mm stop and strike, the leaf landing on the stop");
                }
                Object.DestroyImmediate(door.root.gameObject);
            }
        // The old map door at 95° both ways (hinge on the centre line, no stop): before/after reference.
        foreach (var s in new[] { 1, -1 })
        {
            var door = OldDoor(near, dpos, drot, s);
            SetAngle(door, 95f);
            var dirName = s > 0 ? "toward_camera_pull" : "away_from_camera_push";
            var sw = s > 0 ? n : -n;
            ShotAt("11_swing_OLD_" + dirName + "_95deg_oblique", obl, oblLook, GameFov, "old primitive door at 95°");
            ShotAt("11_swing_OLD_" + dirName + "_95deg_plan", c - t * .18f + sw * .80f + Vector3.up * 2.62f, c - t * .46f + sw * .12f + Vector3.up * 1.55f, 70f, "old primitive door at 95°, plan over the hinge jamb");
            Object.DestroyImmediate(door.root.gameObject);
        }
    }
