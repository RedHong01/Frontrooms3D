"""Before | after three-view sheets for the interactables fix pass (2026-10-08).

  W/venv/bin/python fix_sheets.py <before png dir> <after png dir> <out dir>

before = Documentation/research/interactables/threeview/png (the 2026-10-03 hand-off, rendered from main's FBX),
after  = proj_int/Verification/threeview_fix (the same FrontRoomsThreeView tool on the fix-pass builds).
Transparent PNGs are laid on the hero-panel colour #EEECE6, aspect kept, BEFORE left / AFTER right per pair.
Writes PNG (exact) and JPG q85 (<= 1920 wide is NOT applied: these are evidence sheets, width capped at 3840).
"""
import os
import sys

from PIL import Image, ImageDraw, ImageFont

B, A, OUT = sys.argv[1], sys.argv[2], sys.argv[3]
os.makedirs(OUT, exist_ok=True)
BG = (238, 236, 230)
FONT = os.path.expanduser("~/Library/Fonts/IBMPlexMono-Medium.otf")


def font(px):
    try:
        return ImageFont.truetype(FONT, px)
    except Exception:
        return ImageFont.load_default()


def load(d, kit, view):
    p = os.path.join(d, "%s_%s.png" % (kit, view))
    if not os.path.exists(p):
        return None
    im = Image.open(p).convert("RGBA")
    bg = Image.new("RGBA", im.size, BG + (255,))
    bg.alpha_composite(im)
    return bg.convert("RGB")


def crop_frac(im, box):
    if im is None or box is None:
        return im
    w, h = im.size
    return im.crop((int(box[0] * w), int(box[1] * h), int(box[2] * w), int(box[3] * h)))


def sheet(name, title, items, cell_h=520, cols=2, label_px=22, max_w=3840):
    """items: (label, kit, view, crop box in fractions or None)."""
    cells = []
    for label, kit, view, box in items:
        b = crop_frac(load(B, kit, view), box)
        a = crop_frac(load(A, kit, view), box)
        pair = []
        for im, tag in ((b, "BEFORE"), (a, "AFTER")):
            if im is None:
                im = Image.new("RGB", (cell_h, cell_h), (200, 60, 50))
                tag += " (missing)"
            s = cell_h / im.height
            pair.append((im.resize((max(1, int(im.width * s)), cell_h), Image.LANCZOS), tag))
        cells.append((label, pair))
    pad, cap = 14, label_px + 12
    rows = [cells[i:i + cols] for i in range(0, len(cells), cols)]
    widths = [sum(sum(p[0].width for p in c[1]) + pad * 3 for c in r) + pad for r in rows]
    W = min(max(widths), 10 ** 6)
    H = (label_px * 2 + 20) + len(rows) * (cell_h + cap * 2 + pad) + pad
    out = Image.new("RGB", (W, H), (250, 249, 246))
    d = ImageDraw.Draw(out)
    d.text((pad, 8), title, fill=(10, 10, 10), font=font(int(label_px * 1.3)))
    y = label_px * 2 + 20
    f = font(label_px)
    for r in rows:
        x = pad
        for label, pair in r:
            d.text((x, y), label, fill=(10, 10, 10), font=f)
            xx = x
            for im, tag in pair:
                out.paste(im, (xx, y + cap))
                d.text((xx + 4, y + cap + cell_h + 4), tag, fill=(10, 10, 10), font=f)
                xx += im.width + pad
            x = xx + pad * 2
        y += cell_h + cap * 2 + pad
    out.save(os.path.join(OUT, name + ".png"))
    if out.width > max_w:
        out = out.resize((max_w, int(out.height * max_w / out.width)), Image.LANCZOS)
    out.save(os.path.join(OUT, name + ".jpg"), "JPEG", quality=85)
    print("sheet", name, out.size)
