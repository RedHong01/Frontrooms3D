def U(X, Y, Z):
    return Vector((-X, -Z, Y))


def reset():
    bpy.ops.wm.read_factory_settings(use_empty=True)


def import_asset(name, lod=0):
    before = set(bpy.data.objects)
    bpy.ops.import_scene.fbx(filepath=os.path.join(MODELS, name + ".fbx"))
    new = [o for o in bpy.data.objects if o not in before]
    meshes = [o for o in new if o.type == "MESH"]
    want = [o for o in meshes if ("_LOD1" in o.name) == (lod == 1)]
    keep = want[0] if want else meshes[0]
    mw = keep.matrix_world.copy()
    keep.parent = None
    keep.data.transform(mw)
    keep.matrix_world = Matrix.Identity(4)
    for o in new:
        if o is not keep:
            bpy.data.objects.remove(o, do_unlink=True)
    keep.name = name + ("_LOD1" if lod == 1 else "")
    return keep


def set_materials(obj):
    for i, m in enumerate(obj.data.materials):
        if m is None:
            continue
        slot = m.name.split(".")[0]
        if slot not in kitlib.SLOTS:
            continue
        col, rough, metal = kitlib.SLOTS[slot]
        mat = bpy.data.materials.get("pv_" + slot)
        if mat is None:
            mat = bpy.data.materials.new("pv_" + slot)
            mat.use_nodes = True
            b = mat.node_tree.nodes["Principled BSDF"]
            b.inputs["Base Color"].default_value = (*col, 1)
            b.inputs["Roughness"].default_value = rough
            b.inputs["Metallic"].default_value = metal
            if slot == "Prop_Glass":
                b.inputs["Transmission Weight"].default_value = 0.9
        obj.data.materials[i] = mat


def rot_about_axis(obj, deg, sign=1):
    """Leaf rig rotation: Unity Euler(0, deg, 0) about the A2 axis (S_sign = sign)."""
    ax = U(AXIS[0] * sign, 0, AXIS[1])
    R = Matrix.Rotation(math.radians(-deg * sign), 4, "Z")
    obj.matrix_world = Matrix.Translation(ax) @ R @ Matrix.Translation(-ax)


def box(name, lo, hi, col=(0.5, 0.48, 0.44), emit=0.0):
    """Axis box from Unity corners."""
    a, b = U(*lo), U(*hi)
    mn = Vector((min(a.x, b.x), min(a.y, b.y), min(a.z, b.z)))
    mx = Vector((max(a.x, b.x), max(a.y, b.y), max(a.z, b.z)))
    bm = bmesh.new()
    bmesh.ops.create_cube(bm, size=1.0)
    for v in bm.verts:
        v.co = Vector(((v.co.x + 0.5) * (mx.x - mn.x) + mn.x, (v.co.y + 0.5) * (mx.y - mn.y) + mn.y, (v.co.z + 0.5) * (mx.z - mn.z) + mn.z))
    me = bpy.data.meshes.new(name)
    bm.to_mesh(me)
    bm.free()
    ob = bpy.data.objects.new(name, me)
    bpy.context.scene.collection.objects.link(ob)
    mat = bpy.data.materials.new(name + "_m")
    mat.use_nodes = True
    bs = mat.node_tree.nodes["Principled BSDF"]
    bs.inputs["Base Color"].default_value = (*col, 1)
    bs.inputs["Roughness"].default_value = 0.8
    if emit:
        bs.inputs["Emission Color"].default_value = (*col, 1)
        bs.inputs["Emission Strength"].default_value = emit
    me.materials.append(mat)
    return ob


def wall_standin(col=(0.42, 0.38, 0.27), trims=True, height=2.4, extent=1.6):
    """The map's wall around a 1.0 x 2.1 opening: |X| <= 0.08, plus the
    map trims (0.07 face, 0.20 deep) as the map builds them."""
    out = [box("wall hinge side", (-0.08, 0, -extent), (0.08, height, 0.0), col),
           box("wall latch side", (-0.08, 0, 1.0), (0.08, height, 1.0 + extent), col),
           box("wall head", (-0.08, 2.1, 0.0), (0.08, height, 1.0), col)]
    if trims:
        tc = (0.08, 0.06, 0.05)
        out += [box("trim hinge", (-0.10, 0, -0.07), (0.10, 2.17, 0.0), tc),
                box("trim latch", (-0.10, 0, 1.0), (0.10, 2.17, 1.07), tc),
                box("trim head", (-0.10, 2.10, -0.07), (0.10, 2.17, 1.07), tc)]
    return out


def floor(col=(0.30, 0.28, 0.25), size=8.0):
    return box("floor", (-size, -0.02, -size), (size, 0.0, size), col)


def camera(eye, target, lens=None, fov=None, name="cam", res=(1200, 900)):
    cd = bpy.data.cameras.new(name)
    if fov:
        cd.sensor_fit = "VERTICAL"
        cd.angle_y = math.radians(fov)
    elif lens:
        cd.lens = lens
    cd.clip_start = 0.01
    cam = bpy.data.objects.new(name, cd)
    bpy.context.scene.collection.objects.link(cam)
    e, t = U(*eye), U(*target)
    cam.location = e
    cam.rotation_euler = (t - e).to_track_quat("-Z", "Y").to_euler()
    bpy.context.scene.camera = cam
    sc = bpy.context.scene
    sc.render.resolution_x, sc.render.resolution_y = res
    return cam


def area_light(pos, target, energy, size=1.0, col=(1, 0.97, 0.9), name="light"):
    ld = bpy.data.lights.new(name, "AREA")
    ld.energy = energy
    ld.size = size
    ld.color = col
    lo = bpy.data.objects.new(name, ld)
    bpy.context.scene.collection.objects.link(lo)
    p, t = U(*pos), U(*target)
    lo.location = p
    lo.rotation_euler = (t - p).to_track_quat("-Z", "Y").to_euler()
    return lo


def render_setup(samples=64, world=0.25):
    sc = bpy.context.scene
    sc.render.engine = "CYCLES"
    sc.cycles.samples = samples
    sc.cycles.use_denoising = True
    sc.cycles.device = "CPU"
    sc.view_settings.view_transform = "AgX"
    w = bpy.data.worlds.new("w")
    w.use_nodes = True
    w.node_tree.nodes["Background"].inputs[0].default_value = (0.30, 0.29, 0.27, 1)
    w.node_tree.nodes["Background"].inputs[1].default_value = world
    sc.world = w


def render(path):
    bpy.context.scene.render.filepath = path
    bpy.ops.render.render(write_still=True)
    print("[render]", path)
