#!/usr/bin/env python3
"""Clone-only: main's FrontRoomsMapWorld.DressNext is now DressNext(bool all = false) (MapWorld.cs:2050), so the
window harness's reflective dress.Invoke(w, null) throws TargetParameterCountException. Pass all=true when the
method takes a parameter. Usage: tf_dressnext.py <FrontRoomsWindowTF.cs> (in place; idempotent)."""
import sys
p = sys.argv[1]
s = open(p, encoding="utf-8").read()
old = "while ((bool)dress.Invoke(w, null)) { }"
new = "while ((bool)dress.Invoke(w, dress.GetParameters().Length == 0 ? null : new object[] { true })) { }"
if new in s and old not in s:
    print("already patched"); sys.exit(0)
n = s.count(old)
assert n == 2, "expected 2 call sites, found %d" % n
open(p, "w", encoding="utf-8").write(s.replace(old, new))
print("patched %d call sites" % n)
