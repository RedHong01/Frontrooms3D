"""D1.6 atlases (research/interactables/d16/10_design.md §9), written into a clone's Resources/Surfaces/Textures.
Prop_KeyTagNo_A/_S: 4096^2, 10 x 10 cells "00".."99" (cell n = row * 10 + col, row 0 at the TOP: the kits' quads
  ship on cell 00 at UV (0..0.1, 0.9..1.0) and the facade offsets by (n%10/10, -(n//10)/10)). Courier Prime Bold,
  ink #1E1D1A on Prop_Paper's ground sRGB (220, 216, 204); digit ink height 0.24 of a cell (G12), centred.
  _S: R smoothness paper 0.24 / ink 0.18, G cavity 1.
Prop_SignEngraved_A/_N/_S: 2048 x 1024 (the kit's cells: 1024 x 512 px = 0.254 x 0.127 m, 4031 px/m, square
  texels; 10_design said 2048^2, the module's UV math wants 2:1), 2 x 2 cells: 0 EMPLOYEES ONLY, 1 STAFF ONLY,
  2 STORAGE, 3 blank. TeX Gyre Heros Bold, all caps, cap height 17 mm, tracking +40/1000 em, centred on the face
  (0.252 x 0.074 m inside the bevel, centred in the cell). Face dark brown sRGB (48, 36, 30); routed letters show the
  ivory core (#E9E1C9, a proposal for 平面视觉). _N: a 0.3 mm recess from the letter mask; _S: R face 0.55 / letters
  0.35, G cavity face 1 / letters 0.75.
Usage: python make_atlases.py <fonts dir (Assets/Fonts/Period1990)> <out dir> [--proof <png>]
"""
import sys, os, json
import numpy as np
from PIL import Image, ImageDraw, ImageFont, ImageFilter

fonts, out = sys.argv[1], sys.argv[2]
proof = sys.argv[sys.argv.index("--proof") + 1] if "--proof" in sys.argv else None
report = {}

# ------------------------------------------------------------------ key tag numbers
N = 4096
cell = N / 10.0
paper = (220, 216, 204)
ink = (0x1E, 0x1D, 0x1A)
cp = os.path.join(fonts, "CourierPrime", "CourierPrime-Bold.ttf")
target_h = 0.24 * cell
size = 200
for _ in range(6):
    f = ImageFont.truetype(cp, size)
    b = f.getbbox("0")
    h = b[3] - b[1]
    size = max(10, int(round(size * target_h / h)))
f = ImageFont.truetype(cp, size)
img = Image.new("RGB", (N, N), paper)
mask = Image.new("L", (N, N), 0)
d = ImageDraw.Draw(img)
dm = ImageDraw.Draw(mask)
widths, heights = [], []
for n in range(100):
    r, c = divmod(n, 10)
    s = "%02d" % n
    bb = f.getbbox(s)
    w, hh = bb[2] - bb[0], bb[3] - bb[1]
    widths.append(w / cell); heights.append(hh / cell)
    cx, cy = (c + .5) * cell, (r + .5) * cell
    # centre the ink box: the digits' measured box, not the font's line box
    x = cx - (bb[0] + bb[2]) / 2.0
    y = cy - (bb[1] + bb[3]) / 2.0
    d.text((x, y), s, font=f, fill=ink)
    dm.text((x, y), s, font=f, fill=255)
img.save(os.path.join(out, "Prop_KeyTagNo_A.png"))
m = np.asarray(mask).astype(np.float32) / 255.0
S = np.zeros((N, N, 4), np.uint8)
S[..., 0] = np.round((0.24 + (0.18 - 0.24) * m) * 255).astype(np.uint8)
S[..., 1] = 255
S[..., 3] = 255
Image.fromarray(S, "RGBA").save(os.path.join(out, "Prop_KeyTagNo_S.png"))
report["keyTagNo"] = {"size": N, "fontPx": size, "digitInkHeightOfCell": [round(min(heights), 4), round(max(heights), 4)],
                      "inkWidthOfCell": [round(min(widths), 4), round(max(widths), 4)], "font": "CourierPrime-Bold.ttf", "ink": "#1E1D1A", "paper_sRGB": paper}

# ------------------------------------------------------------------ engraved door signs
SW, SH = 2048, 1024
cw, ch = SW // 2, SH // 2
pxm = cw / 0.254
face = (48, 36, 30)
ivory = (0xE9, 0xE1, 0xC9)
hb = os.path.join(fonts, "TeXGyre", "texgyreheros-bold.otf")
cap_px = 0.017 * pxm
size = 60
for _ in range(6):
    f2 = ImageFont.truetype(hb, size)
    b = f2.getbbox("H")
    size = max(8, int(round(size * cap_px / (b[3] - b[1]))))
f2 = ImageFont.truetype(hb, size)
track = 0.040 * size
texts = ["EMPLOYEES ONLY", "STAFF ONLY", "STORAGE", ""]
smask = Image.new("L", (SW, SH), 0)
dm2 = ImageDraw.Draw(smask)
widths_mm = []
for n, txt in enumerate(texts):
    r, c = divmod(n, 2)
    cx, cy = (c + .5) * cw, (r + .5) * ch
    if not txt:
        widths_mm.append(0.0); continue
    # tracked width
    adv = [f2.getlength(ch_) for ch_ in txt]
    total = sum(adv) + track * (len(txt) - 1)
    hbox = f2.getbbox("H")
    x = cx - total / 2.0
    y = cy - (hbox[1] + hbox[3]) / 2.0
    for ch_, a in zip(txt, adv):
        dm2.text((x, y), ch_, font=f2, fill=255)
        x += a + track
    widths_mm.append(round(total / pxm * 1000, 1))
sm = np.asarray(smask.filter(ImageFilter.GaussianBlur(0.6))).astype(np.float32) / 255.0
A = np.zeros((SH, SW, 3), np.float32)
for i in range(3):
    A[..., i] = face[i] + (ivory[i] - face[i]) * sm
Image.fromarray(np.round(A).astype(np.uint8), "RGB").save(os.path.join(out, "Prop_SignEngraved_A.png"))
# normal: height = -0.3 mm in the letters, with a soft 1-2 px wall
hgt = -0.0003 * np.asarray(smask.filter(ImageFilter.GaussianBlur(1.2))).astype(np.float32) / 255.0
gy, gx = np.gradient(hgt * pxm)          # metres per pixel -> slope
nx, ny, nz = -gx, gy, np.ones_like(gx)   # image y down -> texture v up: dh/dv = -dh/dy_img
ln = np.sqrt(nx * nx + ny * ny + nz * nz)
Nn = np.stack([nx / ln, ny / ln, nz / ln], -1)
Image.fromarray(np.round((Nn * .5 + .5) * 255).astype(np.uint8), "RGB").save(os.path.join(out, "Prop_SignEngraved_N.png"))
SS = np.zeros((SH, SW, 4), np.uint8)
SS[..., 0] = np.round((0.55 + (0.35 - 0.55) * sm) * 255).astype(np.uint8)
SS[..., 1] = np.round((1.0 + (0.75 - 1.0) * sm) * 255).astype(np.uint8)
SS[..., 3] = 255
Image.fromarray(SS, "RGBA").save(os.path.join(out, "Prop_SignEngraved_S.png"))
capb = f2.getbbox("H")
report["signEngraved"] = {"size": [SW, SH], "fontPx": size, "capHeightMm": round((capb[3] - capb[1]) / pxm * 1000, 2),
                          "trackingEm": 0.040, "textWidthsMm": widths_mm, "faceWidthMm": 252, "face_sRGB": face, "letters_sRGB": ivory,
                          "font": "texgyreheros-bold.otf"}
print(json.dumps(report, indent=1))
if proof:
    p = Image.new("RGB", (2048, 1400), (40, 40, 40))
    p.paste(img.crop((0, 0, 2048, 820)).resize((1024, 410)), (0, 0))
    p.paste(Image.open(os.path.join(out, "Prop_SignEngraved_A.png")).resize((1024, 512)), (1024, 0))
    p.save(proof)
