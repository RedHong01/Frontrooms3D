# ---- VL323 close-ups: hardware (with a luminance measure) and type / materials
import math
import numpy as np
from PIL import Image as _I


def _lum(path):
    a = np.asarray(_I.open(path).convert('RGB')).astype(np.float32) / 255.0
    lin = np.where(a <= 0.04045, a / 12.92, ((a + 0.055) / 1.055) ** 2.4)
    return 0.2126 * lin[..., 0] + 0.7152 * lin[..., 1] + 0.0722 * lin[..., 2]


def _box(l, b): return float(l[b[1]:b[3], b[0]:b[2]].mean())
