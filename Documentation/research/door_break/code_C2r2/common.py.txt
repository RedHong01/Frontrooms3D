"""Direction C rework (C2): shared constants and fields, imported by the generator (drywall.py, the
local venv) and by the Blender scene (scene2.py). numpy only.

Door root frame (10_design.md 1.2): Z along the opening (hinge 0 -> latch 1.0), Y up, X toward S
(the swing side, the player's room in BURST). Blender = (-X, -Z, Y).

The wall build is 21_C_realism_research.md 1.1-1.6, default (a):
  1/2 in gypsum board (12.7 mm) on 2x6 kiln-dried studs (38.1 x 139.7 mm) at 16 in (406.4 mm),
  the board hung horizontally (long edges horizontal, joint at Y 1.68 on both faces), notched
  round the opening (layout L1), screwed at 406 mm on the framing, paper-backed vinyl over it.
The map wall is 0.16 m, so the boards' inner faces are 134.6 mm apart; the 139.7 mm studs are
drawn 2.55 mm into each board's back (hidden), not shortened.
"""

# r2 (critic "construction" #6): the 16 in layout runs from a WALL END, not from the kings. The kings land
# off-layout, so one side of the door usually gets a partial bay; a layout stud that would collide with a
# king is left out (the king takes its place); the cripples over the header stay on layout. The offset of
# the layout from the door is per door: seeded by DB_SEED (in the game: the edge hash). Seeds used in the
# round-2 frames (layout offset o = position of a layout line in door-root Z, mod 406.4 mm):
#   1  o 0.160  hinge bay 126 mm (narrow), latch bay 259 mm          (L0-F main frames)
#   2  o 0.300  hinge bay 393 mm, latch bay 399 mm (both near studs fell on the kings and were left out)
#   3  o 0.360  hinge bay 333 mm, latch bay  53 mm (too narrow to break out: crack + crush only)
#   4  o 0.200  hinge bay  86 mm (too narrow), latch bay 299 mm      (L0-K main frames)
import os as _os
SEED = int(_os.environ.get("DB_SEED", "1"))
LAYOUT_O = {1: 0.160, 2: 0.300, 3: 0.360, 4: 0.200}.get(SEED, (SEED * 0.1373) % OC)
NEAR_GAP = 0.020          # a layout stud closer than 20 mm to a king's face is left out (the king serves)
MIN_BAY = 0.090           # a bay narrower than this cannot break out as a plate (crack + crush only)


def stud_centres():
    """Common studs on the 406.4 mm layout from the wall end (offset LAYOUT_O): the 3 nearest each side of
    the door outside the kings, and the cripples over the header on the same layout."""
    h = STUD_T / 2
    zs = [LAYOUT_O + OC * n for n in range(-8, 12)]
    left = sorted([z for z in zs if z + h <= KING["L"][0] - NEAR_GAP], reverse=True)[:3]
    right = sorted([z for z in zs if z - h >= KING["R"][1] + NEAR_GAP])[:3]
    cripples = [z for z in zs if z - h >= KING["L"][1] + 0.005 and z + h <= KING["R"][0] - 0.005]
    return left, right, cripples


STUDS_L, STUDS_R, CRIPPLES = stud_centres()


def members():
    """Framing members as (name, z0, z1, y0, y1) in the wall plane (X spans the stud depth)."""
    m = []
    h = STUD_T / 2
    for s in ("L", "R"):
        m.append(("jack_" + s, JACK[s][0], JACK[s][1], BOTPLATE[1], HEADER[0]))
        m.append(("king_" + s, KING[s][0], KING[s][1], BOTPLATE[1], TOPPLATE[0]))
    for i, z in enumerate(STUDS_L + STUDS_R):
        m.append(("stud_%d" % i, z - h, z + h, BOTPLATE[1], TOPPLATE[0]))
    for i, z in enumerate(CRIPPLES):
        m.append(("cripple_%d" % i, z - h, z + h, HEADER[1], TOPPLATE[0]))
    return m


def stud_lines():
    """Centre Z of every vertical framing member the board is screwed to, with its Y span."""
    out = []
    for nm, z0, z1, y0, y1 in members():
        out.append((nm, 0.5 * (z0 + z1), y0, y1))
    return out


def sp(stage, steel=False):
    """Stage parameters (ESTIMATES, cumulative), kept from the C pre-render (scene.py SP_) for the
    frame and the leaf, so the leaf stays inside its validated collider band; casing pry raised to
    the research's FINAL 10-30 mm (21 2.8)."""
    t = {
        "S0": dict(B=0.0, rack=0.0, lift=0.0, pry=0.0, bow=0.0, dish=0.0, crease=0.0, twist=0.0),
        "M": dict(B=0.0008, rack=0.0, lift=0.0, pry=0.0, bow=0.0, dish=0.0, crease=0.0, twist=0.0),
        "D1": dict(B=0.004, rack=0.003, lift=0.002, pry=0.0015, bow=0.008, dish=1.0, crease=0.0, twist=0.004),
        "D2": dict(B=0.014, rack=0.008, lift=0.005, pry=0.006, bow=0.030, dish=1.35, crease=0.0, twist=0.014),
        "FINAL": dict(B=0.020, rack=0.010, lift=0.007, pry=0.016, bow=0.036, dish=1.6, crease=0.0, twist=0.018),
        "S4": dict(B=0.018, rack=0.010, lift=0.007, pry=0.020, bow=0.026, dish=1.6, crease=1.0, twist=0.009),
    }[stage]
    d = dict(t)
    if steel and stage == "FINAL":
        d["bow"] = 0.031
    return d


def smooth(a, b, x):
    t = np.clip((np.asarray(x, float) - a) / (b - a), 0, 1)
    return t * t * (3 - 2 * t)


def d_cont(Z, Y, B):
    """The continuous push toward S of the frame and the leaf (scene.py, unchanged: the leaf band
    was validated with it)."""
    Z = np.asarray(Z, float)
    Y = np.asarray(Y, float)
    g = np.exp(-(np.maximum(0.0, np.abs(Z - 0.5) - 0.62) / 0.42) ** 2)
    g = g * smooth(-0.95, -0.70, Z) * (1.0 - smooth(2.0, 2.251, Z))
    h = smooth(0.9, 2.15, Y) * np.where(Y > 2.25, np.exp(-((Y - 2.25) / 0.55) ** 2), 1.0)
    return B * g * h


DOM = (-0.95, 2.05, 0.30, 2.90)       # damage domain on both faces (Z0, Z1, Y0, Y1); r2: from 0.30 (the strike-height bay)
MAPR = (-0.80, 1.90, 0.40, 2.90)      # wallpaper map region (1 mm texels); r2: from 0.40
MPX = 0.001

# ---------------------------------------------------------------- r2: the frame-driven load path
# Round 2 (critic "construction" #1): ONE load path, stated. Each blow drives the leaf toward S; the leaf
# bears on the frame through its top pivot (head jamb, hinge end -> header + king L top), its bottom
# pivot (floor) and the latch line (L0-K: the thrown deadbolt in the steel strike jamb, Y 1.0; L0-F: the
# latch in the strike, held there by the game rule, the double bevel cannot hold a push). The frame
# members are carried toward S (shims crush, 16d nails withdraw, 21 3.4). The board rides on the
# framing: over a loaded member it moves with it; across a bay it LAGS, ramping to 0 at the next
# common stud (the bay hinges on the outer stud), and above the header it ramps to 0 at the top plate.
# Amplitudes (m, toward S, ESTIMATES): A_R at the strike, A_T at the top pivot.
LOADS = {
    "S0": (0.0, 0.0), "M": (0.0005, 0.0004), "D1": (0.002, 0.0015), "D2": (0.007, 0.006),
    "FINAL": (0.026, 0.022), "S4": (0.020, 0.017),
}
LOADS_STEEL = {
    "S0": (0.0, 0.0), "M": (0.0006, 0.0004), "D1": (0.0025, 0.0015), "D2": (0.008, 0.006),
    "FINAL": (0.030, 0.022), "S4": (0.024, 0.017),
}
STRIKE_Y = 1.0
S3I = STUDS_R[0] - STUD_T / 2      # stud_3 (the latch side's near common stud): its face toward the opening
S0I = STUDS_L[0] + STUD_T / 2      # stud_0 (the hinge side's near common stud): its face toward the opening
BAY_L = KING["L"][0] - S0I          # the hinge-side bay (king L outer face -> stud_0)
BAY_R = S3I - KING["R"][1]          # the latch-side bay (king R outer face -> stud_3)


def _h_bay():
    """The cripple bay over the head at the hinge end (where the top pivot's push is largest): the first
    bay from king L that is at least 120 mm wide, as (z0, z1) between member faces."""
    faces = [KING["L"][1]] + [f for c in CRIPPLES for f in (c - STUD_T / 2, c + STUD_T / 2)] + [KING["R"][0]]
    for i in range(0, len(faces) - 1, 2):
        if faces[i + 1] - faces[i] >= 0.12:
            return faces[i], faces[i + 1]
    return faces[0], faces[1]


H_BAY = _h_bay()


def data_tag(member):
    """Data / frame-name suffix: '' for the member's main seed (L0-F 1, L0-K 4), else '_s<seed>'."""
    return "" if SEED == (4 if member == "L0K" else 1) else "_s%d" % SEED


def loads(stage, steel=False):
    return (LOADS_STEEL if steel else LOADS)[stage]


def _hR(Y, ys=STRIKE_Y):
    Y = np.asarray(Y, float)
    return np.exp(-((Y - ys) / 0.62) ** 2) * smooth(0.04, 0.35, Y)


def _hT(Y):
    """Along the hinge king / header: peaks at the head (2.10-2.31), 0 at the floor and at the top plate."""
    Y = np.asarray(Y, float)
    return smooth(0.9, 2.12, Y) * (1.0 - smooth(2.3142, 2.8238, Y))


def frame_field(Z, Y, stage, steel=False, ys=STRIKE_Y):
    """Displacement toward S of the frame (jambs, casings, saddle ends) and of the members behind it."""
    aR, aT = loads(stage, steel)
    Z = np.asarray(Z, float)
    Y = np.asarray(Y, float)
    fR = smooth(0.45, 0.98, Z)
    fT = 1.0 - 0.65 * smooth(-0.1, 1.1, Z)
    hT = smooth(1.2, 2.12, Y)
    return aR * fR * _hR(Y, ys) + aT * fT * hT


def wall_field(Z, Y, stage, steel=False, ys=STRIKE_Y):
    """r2: out-of-plane move of the boards toward S (both faces ride on the framing): the LOADED FRAMING
    LEADS and the bays lag (the inverse of round 1's bay_profile bulge, which held the board back at the
    framing and pushed it out mid-bay). 0 at the domain edges."""
    aR, aT = loads(stage, steel)
    Z = np.asarray(Z, float)
    Y = np.asarray(Y, float)
    kl0, kr1 = KING["L"][0], KING["R"][1]
    # latch side: full over jack R + king R, ramping to 0 at stud_3; the strike hump in Y
    rR = np.where(Z >= kr1, np.clip((S3I - Z) / (S3I - kr1), 0, 1), smooth(0.80, 1.025, Z))
    gR = rR * _hR(Y, ys)
    # top pivot: the hinge king (ramp to stud_0), the header span (hinge end full, latch end 0.35),
    # above the header ramping to 0 at the top plate (the cripple bays hinge on it)
    hf = 1.0 - 0.65 * smooth(-0.1, 1.1, Z)
    rL = np.where(Z <= kl0, np.clip((Z - S0I) / (kl0 - S0I), 0, 1), 1.0)
    rL = np.where(Z >= kr1, np.clip((S3I - Z) / (S3I - kr1), 0, 1) * 0.35, rL * np.where(Z > kl0, hf, 1.0))
    over_head = (Z > kl0) & (Z < kr1)
    yk = _hT(Y)
    yh = np.where(Y <= HEADER[1], 1.0, 1.0 - smooth(HEADER[1], TOPPLATE[0], Y))
    gT = rL * np.where(over_head, yh * smooth(2.0, 2.13, Y), yk)
    side = smooth(DOM[0], DOM[0] + 0.12, Z) * (1 - smooth(DOM[1] - 0.12, DOM[1], Z)) * smooth(DOM[2], DOM[2] + 0.1, Y)
    return (aR * gR + aT * gT) * side


def srgb_to_lin(c):
    c = np.asarray(c, float) / 255.0
    return np.where(c <= 0.04045, c / 12.92, ((c + 0.055) / 1.055) ** 2.4)
