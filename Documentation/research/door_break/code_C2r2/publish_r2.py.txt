def crop(src, box, out, scale=1):
    im = Image.open(src).convert("RGB").crop(box)
    if scale != 1:
        im = im.resize((im.width * scale, im.height * scale), Image.NEAREST)
    im.save(out)
    return out

def build(name):
    vl, title, sentence, status, cols, tiles = SHEETS[name]
    miss = [t[0] for t in tiles if not os.path.exists(t[0])]
    if miss:
        print("skip", name, "missing", [os.path.basename(x) for x in miss])
        return None
    img = S.grid([(p, lab, dat) for p, lab, dat in tiles], cols=cols, width=1920)
    out = S.frame(img, title=title, sentence=sentence, meta=META(vl), status=status)
    path = DOC + "db_C2r2_sheet_%s.jpg" % name
    S.save(out, path, quality=85)
    print("wrote", path, out.size)
    return path


if __name__ == "__main__":
    for n in (sys.argv[1:] or list(SHEETS)):
        build(n)
