"""C2 re-check (21_C_diagnosis.md 5): the same measures as look/diag2 on the rework's V1 finals.

  ../py measure2.py <final dir> [member]  -> rework/look/measure_<member>.json

1 crack lines on intact paper > 0.3 m from the head corners: FWHM and dip of a perpendicular
  luminance profile (target: dip < 10 %, or no line found)
2 torn zones vs the print beside them (target +0.3 .. +1.0 stop, lighter)
3 hole interiors vs the wallpaper (target about -2.5 .. -4 stops, not black)
4 debris: mass share within 0.3 / 0.5 m of the wall face, size classes (target >= 60 % within 0.3 m)
5 floor powder: S4 / S0 linear luminance ratio by distance from the wall (target highest at 0-0.15 m)
"""
import json
import math
import os
import sys

import numpy as np
from PIL import Image
from scipy import ndimage

HERE = os.path.dirname(os.path.abspath(__file__))
sys.path.insert(0, HERE)
sys.path.insert(0, "/Users/redwang/FrontRoomsVisualWork/door_break/C/look/diag2")
import common as C  # noqa
import drywall as DW  # noqa
import geom2d as g2  # noqa
from cam import Cam  # noqa

RW = os.path.dirname(HERE)
FIN = sys.argv[1]
MEM = sys.argv[2] if len(sys.argv) > 2 else "L0F"
DATA = RW + "/data/%s_burst" % MEM
CORN = [(C.CASING[0], C.CASING[2]), (C.CASING[1], C.CASING[2])]


def lin(a):
    a = a / 255.0
    return np.where(a <= 0.04045, a / 12.92, ((a + 0.055) / 1.055) ** 2.4)


def lum(im):
    return 0.2126 * im[..., 0] + 0.7152 * im[..., 1] + 0.0722 * im[..., 2]


def load(stage, view="V1", light="LA"):
    fn = os.path.join(FIN, "db_C2_%s_burst_%s_%s_%s.png" % (MEM, stage, view, light))
    return np.asarray(Image.open(fn).convert("RGB"), np.float32)


cam = Cam("V1", 76)
res = {}
S, P = DW.build_events(MEM, "burst")
lines = [(np.asarray(p), b) for p, b in S.hairlines] + [(np.asarray(p), b) for t, p, b in S.tears if t == "crack_line"]
for st in ("D1", "D2", "S4"):
    im = load(st)
    L = lum(im)
    sj = json.load(open(DATA + "/stage_S_%s.json" % st))
    holes = [np.asarray(h["rows"]["pf"]) for h in sj["holes"]]
    rows = []
    for pts, born in lines:
        if C.SI[born] > C.SI[st]:
            continue
        cum = np.concatenate([[0], np.cumsum(np.linalg.norm(np.diff(pts, axis=0), axis=1))])
        for s in np.arange(0.01, cum[-1] - 0.01, 0.01):
            zq, yq = np.interp(s, cum, pts[:, 0]), np.interp(s, cum, pts[:, 1])
            if -0.08 < zq < 1.08 and yq < 2.19:
                continue
            if any(g2.point_in_poly(np.array([[zq, yq]]), h)[0] or DW.dist_to_poly((zq, yq), h) < 0.03 for h in holes):
                continue
            dc = min(math.dist((zq, yq), c) for c in CORN)
            za, ya = np.interp(s - .01, cum, pts[:, 0]), np.interp(s - .01, cum, pts[:, 1])
            zb, yb = np.interp(s + .01, cum, pts[:, 0]), np.interp(s + .01, cum, pts[:, 1])
            P0, _ = cam.project(np.array((0.08, yq, zq)))
            Pa, _ = cam.project(np.array((0.08, ya, za)))
            Pb, _ = cam.project(np.array((0.08, yb, zb)))
            t = Pb - Pa
            t /= np.linalg.norm(t) + 1e-9
            n = np.array([-t[1], t[0]])
            mm_px = 20.0 / max(np.linalg.norm(Pb - Pa), 1e-6)
            offs = np.arange(-10, 10.01, 0.25)
            prof = ndimage.map_coordinates(L, [P0[1] + offs * n[1], P0[0] + offs * n[0]], order=1)
            bg = np.median(np.concatenate([prof[:12], prof[-12:]]))
            k = np.abs(offs) <= 3
            mn = prof[k].min()
            j = np.where(k)[0][np.argmin(prof[k])]
            dip = (bg - mn) / max(bg, 1)
            half = bg - 0.5 * (bg - mn)
            a = j
            while a > 0 and prof[a - 1] < half:
                a -= 1
            b = j
            while b < len(prof) - 1 and prof[b + 1] < half:
                b += 1
            rows.append((dc, dip, (b - a + 1) * 0.25 * mm_px))
    a = np.array(rows) if rows else np.zeros((0, 3))
    out = {}
    for lo, hi in ((0, .3), (.3, 9)):
        k = (a[:, 0] >= lo) & (a[:, 0] < hi)
        if k.sum() == 0:
            continue
        found = k & (a[:, 1] > 0.10)
        out["d%.1f-%.1f" % (lo, hi)] = dict(samples=int(k.sum()), dip_p50=round(float(np.median(a[k, 1])), 3),
                                            dip_p90=round(float(np.percentile(a[k, 1], 90)), 3),
                                            frac_dip_over_10pc=round(float(found.sum() / k.sum()), 3),
                                            fwhm_mm_p50_where_found=round(float(np.median(a[found, 2])), 1) if found.any() else None)
    res["cracks_" + st] = out
    print(st, out)

# ---------------------------------------------------------------- torn zones, holes (S4 V1)
im = load("S4")
mp = np.load(DATA + "/wp_S_S4.npy").astype(np.float32)
sj = json.load(open(DATA + "/stage_S_S4.json"))
z0, z1, y0, y1 = C.MAPR
px = C.MPX
dep = mp[..., 0]
hole_mask = np.zeros(dep.shape, bool)
for h in sj["holes"]:
    hole_mask |= g2.polygon_mask(np.asarray(h["rows"]["cf"]), dep.shape, (z0, y1), px)
for q in sj["pieces"]:
    if q["state"] == "hang":
        hole_mask |= g2.polygon_mask(np.asarray(q["rows"]["pf"]), dep.shape, (z0, y1), px)
torn = (dep > 0.6) & (dep < 3.4) & (mp[..., 1] < 0.08) & ~ndimage.binary_dilation(hole_mask, iterations=6)
torn = ndimage.binary_erosion(torn, iterations=1)
rr, cc = np.nonzero(torn[::2, ::2])
rr *= 2
cc *= 2
Z = z0 + (cc + .5) * px
Y = y1 - (rr + .5) * px
k = ~((Z > -0.09) & (Z < 1.09) & (Y < 2.21))
Pp, _ = cam.project(np.stack([np.full(k.sum(), 0.085), Y[k], Z[k]], 1))
pxs = np.unique(np.round(Pp).astype(int), axis=0)
vt = im[pxs[:, 1], pxs[:, 0]]
# print beside the torn zones: 10-30 mm outside them, no powder
ring = ndimage.binary_dilation(torn, iterations=30) & ~ndimage.binary_dilation(torn, iterations=10) & (dep < 0.3) & (mp[..., 1] < 0.05) \
    & ~ndimage.binary_dilation(hole_mask, iterations=10)
rr, cc = np.nonzero(ring[::2, ::2])
rr *= 2
cc *= 2
Z2 = z0 + (cc + .5) * px
Y2 = y1 - (rr + .5) * px
k2 = ~((Z2 > -0.09) & (Z2 < 1.09) & (Y2 < 2.21))
Pq, _ = cam.project(np.stack([np.full(k2.sum(), 0.085), Y2[k2], Z2[k2]], 1))
pxq = np.unique(np.round(Pq).astype(int), axis=0)
vp = im[pxq[:, 1], pxq[:, 0]]
lt, lp = float(np.median(lum(lin(vt)))), float(np.median(lum(lin(vp))))
res["torn_vs_print_S4_V1"] = dict(torn_px=int(len(pxs)), print_px=int(len(pxq)), torn_sRGB=np.median(vt, 0).round(0).tolist(),
                                  print_sRGB=np.median(vp, 0).round(0).tolist(), stops=round(math.log2(lt / lp), 2))
# hole interiors: inside each hole's core outline, 12 mm in, minus hanging plates
inner = np.zeros(dep.shape, bool)
for h in sj["holes"]:
    inner |= g2.polygon_mask(np.asarray(h["rows"]["cf"]), dep.shape, (z0, y1), px)
for q in sj["pieces"]:
    if q["state"] in ("hang", "step"):
        inner &= ~ndimage.binary_dilation(g2.polygon_mask(np.asarray(q["rows"]["pf"]), dep.shape, (z0, y1), px), iterations=8)
inner = ndimage.binary_erosion(inner, iterations=12)
rr, cc = np.nonzero(inner[::2, ::2])
rr *= 2
cc *= 2
Z = z0 + (cc + .5) * px
Y = y1 - (rr + .5) * px
Ph, _ = cam.project(np.stack([np.full(len(Z), 0.08), Y, Z], 1))
pxh = np.unique(np.round(Ph).astype(int), axis=0)
vh = im[pxh[:, 1], pxh[:, 0]]
wall = []
for zq in np.linspace(1.6, 2.1, 30):
    for yq in np.linspace(1.3, 1.6, 20):
        wall.append((0.08, yq, zq))
Pw, _ = cam.project(np.array(wall))
pw = np.unique(np.round(Pw).astype(int), axis=0)
lw = float(np.median(lum(lin(im[pw[:, 1], pw[:, 0]]))))
lh = lum(lin(vh))
res["holes_S4_V1"] = dict(pixels=int(len(pxh)), sRGB_median=np.median(vh, 0).round(1).tolist(), lin_lum_median=round(float(np.median(lh)), 4),
                          wallpaper_lin_lum=round(lw, 4), stops_vs_wallpaper=round(math.log2(float(np.median(lh)) / lw), 2),
                          frac_sRGB_luma_below_20=round(float((lum(vh) < 20).mean()), 3))
# ---------------------------------------------------------------- debris (S4 model data)
vols = []
for q in sj["pieces"]:
    if q["state"] != "floor":
        continue
    poly = np.asarray(q["rows"]["cf"])
    A = abs(g2.area(poly))
    T = np.asarray(q["T"])
    c = g2.centroid(poly)
    p = np.array([0.08 - C.BOARD / 2, c[1], c[0]]) @ T[:3, :3].T + T[:3, 3]
    vols.append((A * C.BOARD, p[0] - C.WALL, float(np.max(np.ptp(poly, axis=0)))))
v = np.array(vols)
tot = v[:, 0].sum()
cr = np.array([c for c in sj["debris"]["crumbs"]])
res["debris_S4"] = dict(pieces=int(len(v)), within_0p3m_mass=round(float(v[v[:, 1] < 0.3, 0].sum() / tot), 3),
                        within_0p5m_mass=round(float(v[v[:, 1] < 0.5, 0].sum() / tot), 3),
                        plates_100_400mm=int((v[:, 2] >= 0.1).sum()), chips_15_100mm=int(((v[:, 2] >= 0.015) & (v[:, 2] < 0.1)).sum()),
                        plate_mass_share=round(float(v[v[:, 2] >= 0.1, 0].sum() / tot), 3),
                        dist_m_p10_50_90=np.percentile(v[:, 1], [10, 50, 90]).round(3).tolist(),
                        grains=int(len(cr)), grain_size_mm_p10_50_90=(np.percentile(cr[:, 3], [10, 50, 90]) * 1000).round(1).tolist(),
                        grains_within_0p15m=round(float(((cr[:, 0] - C.WALL) < 0.15).mean()), 3))
# ---------------------------------------------------------------- floor powder profile V1, S4 / S0
im0 = load("S0")
prof = []
for X in (0.03, 0.08, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0):
    for Zc in (-0.33, 1.33):
        zz = np.linspace(Zc - 0.06, Zc + 0.06, 9)
        Pf, _ = cam.project(np.stack([np.full(9, 0.08 + X), np.zeros(9), zz], 1))
        p_ = np.round(Pf).astype(int)
        a = lum(lin(im[p_[:, 1], p_[:, 0]])).mean()
        b = lum(lin(im0[p_[:, 1], p_[:, 0]])).mean()
        prof.append(dict(X_from_face_m=X, Z=Zc, ratio=round(float(a / b), 3)))
res["floor_powder_V1"] = prof
json.dump(res, open(RW + "/look/measure_%s.json" % MEM, "w"), indent=1)
print(json.dumps({k: v for k, v in res.items() if not k.startswith("cracks")}, indent=1))
